@@tipp
Suche zuerst den <b>Begleitsatz</b>: Wer spricht? Steht er <b>vorn</b>, kommt danach ein Doppelpunkt. Steht er <b>hinten</b>, kommt nach den Anführungszeichen oben ein Komma. Steht er <b>in der Mitte</b>, braucht er zwei Kommas, und beide Teile der Rede bekommen eigene Anführungszeichen.

/* ---------- Satz zerlegen und einfärben (in allen sechs Spielen gleich) ---------- */
/* model(satz) liefert: words = die Wörter (r: gehört zur wörtlichen Rede),
   sig[i] = die Zeichen „ “ : , die vor Wort i stehen (sig[letztes + 1] = am Satzende),
   kind = wo der Begleitsatz steht: vorn, hinten oder mitte */
function model(s) {
  const words = [], sig = [''];
  let inR = false, w = '';
  const flush = () => { if (w) { words.push({ w, r: inR }); w = ''; sig[words.length] = ''; } };
  for (const ch of s) {
    if (ch === '„') { flush(); sig[words.length] += ch; inR = true; }
    else if (ch === '“') { flush(); sig[words.length] += ch; inR = false; }
    else if (!inR && (ch === ':' || ch === ',')) { flush(); sig[words.length] += ch; }
    else if (ch === ' ') flush();
    else w += ch;
  }
  flush();
  const kind = sig[0] !== '„' ? 'vorn' : sig.filter(x => x.includes('„')).length > 1 ? 'mitte' : 'hinten';
  return { words, sig, kind };
}
const KIND = { vorn: 'Begleitsatz vorn', hinten: 'Begleitsatz hinten', mitte: 'Begleitsatz in der Mitte' };
const Z = ch => `<b class="z">${ch}</b>`;
const W = x => `<span class="t ${x.r ? 'rede' : 'begl'}">${x.w}</span>`;
/* Setzt den Satz wieder zusammen. Zeichen und Wort, die zusammengehören, bleiben in einer Zeile. */
function layout(m, zf, wf) {
  return m.words.map((x, i) => {
    const open = m.sig[i].includes('„') ? zf('„') : '';
    const close = [...m.sig[i + 1].replace('„', '')].map(ch => zf(ch)).join('');
    return `<span class="nb">${open}${wf(x)}${close}</span>`;
  }).join(' ');
}
const colored = m => `<span class="sol">${layout(m, Z, W)}</span>`;
const reviewBy = (list, kinds) => `<div class="review ${kinds.length === 2 ? 'c2' : ''}">${kinds.map(k =>
  `<div class="col neutral"><h3>${KIND[k]}</h3><ul>${list.filter(s => s.m.kind === k).map(s => `<li>${layout(s.m, Z, W)}</li>`).join('')}</ul></div>`).join('')}</div>`;

/* ---------- Spiel: Satzzeichen-Setzer ---------- */
const NAME = { '„': 'Anführungszeichen unten', '“': 'Anführungszeichen oben', ':': 'Doppelpunkt', ',': 'Komma' };
const TIP = {
  vorn: 'Der Begleitsatz steht <b>vorn</b>. Danach kommt der Doppelpunkt. Die wörtliche Rede steht zwischen „ und “.',
  hinten: 'Der Begleitsatz steht <b>hinten</b>. Die wörtliche Rede steht zwischen „ und “. Danach kommt ein Komma.',
  mitte: 'Der Begleitsatz steht <b>in der Mitte</b>. Vor ihm und nach ihm steht ein Komma. Beide Teile der Rede stehen zwischen „ und “.'
};
let all = [], queue = [], cur = null, vals = [], sel = -1;
const slotEl = i => $(`.slot[data-i="${i}"]`);

function startGame() {
  all = SENTENCES.map(text => ({ text, m: model(text), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Tippe eine Stelle im Satz an. Wähle dann unten das Zeichen. Wo steht der Begleitsatz?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; sel = -1;
  vals = cur.m.sig.map(() => '');
  const box = $('#sentence');
  box.classList.remove('solved');
  const slot = i => `<button class="slot" data-i="${i}"><i></i></button>`;
  box.innerHTML = cur.m.words.map((x, i) => `<span class="wd">${i ? '' : slot(0)}<span>${x.w}</span>${slot(i + 1)}</span>`).join('');
  paint();
}

function paint() {
  vals.forEach((v, i) => {
    const el = slotEl(i);
    el.classList.toggle('set', !!v);
    el.classList.toggle('cur', i === sel);
    el.firstChild.textContent = v;
    el.setAttribute('aria-pressed', String(i === sel));
    el.setAttribute('aria-label', `Stelle ${i ? 'nach „' + cur.m.words[i - 1].w + '“' : 'am Satzanfang'}: ${v ? [...v].map(ch => NAME[ch]).join(' und ') : 'leer'}`);
  });
}

function select(i) {
  if (state.busy || !cur || i < 0 || i >= vals.length) return;
  sel = i; sfx.pick(); paint();
}
function put(ch) {
  if (state.busy || !cur) return;
  if (sel < 0) { say('Tippe zuerst eine Stelle im Satz an. Dann wähle das Zeichen.'); return; }
  if (vals[sel].length >= 2) { say('An dieser Stelle stehen schon zwei Zeichen. Mit „leeren“ machst du sie wieder frei.'); return; }
  vals[sel] += ch;
  slotEl(sel).classList.remove('hint');
  sfx.pick(); paint();
}
function clear() {
  if (state.busy || !cur || sel < 0) return;
  vals[sel] = ''; paint();
}

function check() {
  if (state.busy || !cur) return;
  if (vals.every(v => !v)) { say('Setze zuerst die Zeichen in den Satz. Tippe dafür eine Stelle an.'); return; }
  const sig = cur.m.sig, wrong = vals.map((v, i) => i).filter(i => vals[i] !== sig[i]);
  $$('.slot').forEach(s => s.classList.remove('hint'));
  if (wrong.length) {
    const filled = wrong.filter(i => vals[i]);
    miss(cur, filled.length ? slotEl(filled[0]) : $('#sentence'));
    filled.slice(1).forEach(i => wobble(slotEl(i)));
    if (cur.misses >= 2) wrong.forEach(i => slotEl(i).classList.add('hint'));
    say(cur.misses === 1 ? `Fast! ${wrong.length === 1 ? 'Eine Stelle stimmt' : wrong.length + ' Stellen stimmen'} noch nicht. Überlege: Wo steht der Begleitsatz?`
      : cur.misses === 2 ? `Tipp: ${TIP[cur.m.kind]} Die gelben Stellen stimmen noch nicht.`
      : `So ist es richtig: ${colored(cur.m)} Setze die Zeichen genau so.`, 'think');
    return;
  }
  state.busy = true; sel = -1;
  const box = $('#sentence');
  solve(cur, box, 'sun');
  box.classList.add('solved');
  box.innerHTML = layout(cur.m, Z, W);
  say(`${pick(PRAISE, 'praise')} ${TIP[cur.m.kind].replace(/<\/?b>/g, '')}`, 'happy');
  later(next, 3000);
}

$('#sentence').addEventListener('click', e => {
  const s = e.target.closest('.slot');
  if (s) select(+s.dataset.i);
});
$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.key');
  if (!k) return;
  if (k.id === 'clear') clear(); else put(k.dataset.ch);
});
$('#check').addEventListener('click', check);
document.addEventListener('keydown', e => {
  if (!gameOn()) return;
  if (/^[1-4]$/.test(e.key)) put($$('.key')[+e.key - 1].dataset.ch);
  else if (e.key === 'Backspace' || e.key === 'Delete') clear();
  else if (e.key === 'ArrowRight') select(Math.min(vals.length - 1, sel + 1));
  else if (e.key === 'ArrowLeft') select(Math.max(0, sel - 1));
});
