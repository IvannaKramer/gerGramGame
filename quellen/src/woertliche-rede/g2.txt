/* ---------- Satz zerlegen und einfärben (in allen sechs Spielen gleich) ---------- */
/* model(satz) liefert: words = die Wörter (r: gehört zur wörtlichen Rede),
   sig[i] = die Zeichen „ “ : , die vor Wort i stehen (sig[letztes + 1] = am Satzende),
   kind = wo der Begleitsatz steht: vorn, hinten oder mitte */
function model(s) {
  const words = [], sig = [''];
  let inR = false, w = '';
  const flush = () => { if (w) { words.push({ w, r: inR }); w = ''; sig[words.length] = ''; } };
  for (const ch of s) {
    if (ch === '„') { flush(); sig[words.length] += ch; inR = true; }
    else if (ch === '“') { flush(); sig[words.length] += ch; inR = false; }
    else if (!inR && (ch === ':' || ch === ',')) { flush(); sig[words.length] += ch; }
    else if (ch === ' ') flush();
    else w += ch;
  }
  flush();
  const kind = sig[0] !== '„' ? 'vorn' : sig.filter(x => x.includes('„')).length > 1 ? 'mitte' : 'hinten';
  return { words, sig, kind };
}
const KIND = { vorn: 'Begleitsatz vorn', hinten: 'Begleitsatz hinten', mitte: 'Begleitsatz in der Mitte' };
const Z = ch => `<b class="z">${ch}</b>`;
const W = x => `<span class="t ${x.r ? 'rede' : 'begl'}">${x.w}</span>`;
/* Setzt den Satz wieder zusammen. Zeichen und Wort, die zusammengehören, bleiben in einer Zeile. */
function layout(m, zf, wf) {
  return m.words.map((x, i) => {
    const open = m.sig[i].includes('„') ? zf('„') : '';
    const close = [...m.sig[i + 1].replace('„', '')].map(ch => zf(ch)).join('');
    return `<span class="nb">${open}${wf(x)}${close}</span>`;
  }).join(' ');
}
const colored = m => `<span class="sol">${layout(m, Z, W)}</span>`;
const reviewBy = (list, kinds) => `<div class="review ${kinds.length === 2 ? 'c2' : ''}">${kinds.map(k =>
  `<div class="col neutral"><h3>${KIND[k]}</h3><ul>${list.filter(s => s.m.kind === k).map(s => `<li>${layout(s.m, Z, W)}</li>`).join('')}</ul></div>`).join('')}</div>`;

/* ---------- Spiel: Zeichen-Helfer ---------- */
const ASK = {
  '„': 'Hier beginnt die pinke wörtliche Rede. Welches Zeichen steht am Anfang?',
  '“': 'Hier ist die pinke wörtliche Rede zu Ende. Welches Zeichen steht am Ende?',
  ':': 'Der blaue Begleitsatz steht vorn. Welches Zeichen kommt danach?',
  ',': 'Der blaue Begleitsatz steht hinten. Welches Zeichen trennt ihn von der wörtlichen Rede?'
};
const RULE = {
  '„': 'Am Anfang der wörtlichen Rede stehen die Anführungszeichen <b>unten</b>.',
  '“': 'Am Ende der wörtlichen Rede stehen die Anführungszeichen <b>oben</b>.',
  ':': 'Steht der Begleitsatz vorn, kommt danach ein <b>Doppelpunkt</b>.',
  ',': 'Steht der Begleitsatz hinten, kommt nach den Anführungszeichen oben ein <b>Komma</b>.'
};
let all = [], queue = [], cur = null, signs = [], step = 0, part = '';
const keyEl = ch => $(`.key[data-ch="${ch}"]`);
const slotEl = n => $(`.slot[data-n="${n}"]`);

function startGame() {
  all = SENTENCES.map(text => ({ text, m: model(text), misses: 0 }));
  queue = shuffle(all.filter(s => s.m.kind === 'vorn')).concat(shuffle(all.filter(s => s.m.kind !== 'vorn')));
  resetState(all.length);
  part = '';
  say('Schau auf das gelbe Feld. Welches Zeichen fehlt dort?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; step = 0; signs = [];
  if (cur.m.kind !== part) {
    if (part) say('Jetzt wird es neu: Der Begleitsatz steht <b>hinten</b>. Dann brauchst du ein Komma.');
    part = cur.m.kind;
  }
  $('#round').textContent = part === 'vorn' ? 'Teil 1 von 2: Begleitsatz vorn' : 'Teil 2 von 2: Begleitsatz hinten';
  $('#sentence').innerHTML = layout(cur.m, ch => { signs.push(ch); return `<span class="slot" data-n="${signs.length - 1}"></span>`; }, W);
  mark();
}

function mark() {
  $$('.key').forEach(k => k.classList.remove('hint'));
  $$('.slot').forEach(s => s.classList.toggle('cur', +s.dataset.n === step));
  $('#sentence').setAttribute('aria-label', `Feld ${step + 1} von ${signs.length} ist dran.`);
}

function choose(ch) {
  if (state.busy || !cur) return;
  const want = signs[step], slot = slotEl(step);
  if (ch !== want) {
    const hint = miss(cur, keyEl(ch));
    wobble(slot);
    if (hint) keyEl(want).classList.add('hint');
    say(hint ? `Tipp: ${RULE[want]}` : `Fast! ${ASK[want]}`, 'think');
    return;
  }
  slot.textContent = ch;
  slot.classList.remove('cur'); slot.classList.add('set');
  step++;
  if (step < signs.length) {
    sfx.pick(); mark();
    say(`Genau! ${RULE[ch]} Weiter mit dem nächsten Feld.`, 'happy');
    return;
  }
  state.busy = true;
  $$('.key').forEach(k => k.classList.remove('hint'));
  solve(cur, slot, 'sun');
  const end = cur.m.words.filter(x => x.r).pop().w.slice(-1);
  say(`${pick(PRAISE, 'praise')} ${cur.m.kind === 'vorn' ? 'Begleitsatz vorn: erst der Doppelpunkt, dann die Rede in Anführungszeichen.'
    : end === '?' ? 'Begleitsatz hinten: Komma nach den Anführungszeichen. Das Fragezeichen bleibt stehen.'
    : end === '!' ? 'Begleitsatz hinten: Komma nach den Anführungszeichen. Das Ausrufezeichen bleibt stehen.'
    : 'Begleitsatz hinten: Komma nach den Anführungszeichen. Der Punkt der Rede fällt weg.'} ${colored(cur.m)}`, 'happy');
  later(next, 2800);
}

$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.key');
  if (k) choose(k.dataset.ch);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-4]$/.test(e.key)) return;
  choose($$('.key')[+e.key - 1].dataset.ch);
});
