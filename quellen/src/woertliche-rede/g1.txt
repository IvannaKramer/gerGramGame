/* ---------- Satz zerlegen und einfärben (in allen sechs Spielen gleich) ---------- */
/* model(satz) liefert: words = die Wörter (r: gehört zur wörtlichen Rede),
   sig[i] = die Zeichen „ “ : , die vor Wort i stehen (sig[letztes + 1] = am Satzende),
   kind = wo der Begleitsatz steht: vorn, hinten oder mitte */
function model(s) {
  const words = [], sig = [''];
  let inR = false, w = '';
  const flush = () => { if (w) { words.push({ w, r: inR }); w = ''; sig[words.length] = ''; } };
  for (const ch of s) {
    if (ch === '„') { flush(); sig[words.length] += ch; inR = true; }
    else if (ch === '“') { flush(); sig[words.length] += ch; inR = false; }
    else if (!inR && (ch === ':' || ch === ',')) { flush(); sig[words.length] += ch; }
    else if (ch === ' ') flush();
    else w += ch;
  }
  flush();
  const kind = sig[0] !== '„' ? 'vorn' : sig.filter(x => x.includes('„')).length > 1 ? 'mitte' : 'hinten';
  return { words, sig, kind };
}
const KIND = { vorn: 'Begleitsatz vorn', hinten: 'Begleitsatz hinten', mitte: 'Begleitsatz in der Mitte' };
const Z = ch => `<b class="z">${ch}</b>`;
const W = x => `<span class="t ${x.r ? 'rede' : 'begl'}">${x.w}</span>`;
/* Setzt den Satz wieder zusammen. Zeichen und Wort, die zusammengehören, bleiben in einer Zeile. */
function layout(m, zf, wf) {
  return m.words.map((x, i) => {
    const open = m.sig[i].includes('„') ? zf('„') : '';
    const close = [...m.sig[i + 1].replace('„', '')].map(ch => zf(ch)).join('');
    return `<span class="nb">${open}${wf(x)}${close}</span>`;
  }).join(' ');
}
const colored = m => `<span class="sol">${layout(m, Z, W)}</span>`;
const reviewBy = (list, kinds) => `<div class="review ${kinds.length === 2 ? 'c2' : ''}">${kinds.map(k =>
  `<div class="col neutral"><h3>${KIND[k]}</h3><ul>${list.filter(s => s.m.kind === k).map(s => `<li>${layout(s.m, Z, W)}</li>`).join('')}</ul></div>`).join('')}</div>`;

/* ---------- Spiel: Rede-Finder ---------- */
let all = [], queue = [], cur = null;

/* Teilt den Satz in zwei Teile zum Antippen: wörtliche Rede (mit Anführungszeichen) und Begleitsatz */
function chunks(m) {
  const out = [];
  m.words.forEach((x, i) => {
    const s = m.sig[i], last = out[out.length - 1];
    if (s.includes(':')) last.text += ':';
    if (s.includes('“')) last.text += '“';
    if (s.includes(',')) out.push({ t: 'cm', text: ',' });
    if (s || !i) out.push({ t: x.r ? 'rede' : 'begl', text: (s.includes('„') ? '„' : '') + x.w });
    else last.text += ' ' + x.w;
  });
  if (m.sig[m.words.length]) out[out.length - 1].text += m.sig[m.words.length];
  return out;
}

function startGame() {
  all = SENTENCES.map(s => ({ text: s[0], emoji: s[1], want: s[2] === 'R' ? 'rede' : 'begl', m: model(s[0]), misses: 0 }));
  queue = shuffle(all.filter(s => s.m.kind === 'vorn')).concat(shuffle(all.filter(s => s.m.kind !== 'vorn')));
  resetState(all.length);
  say('Schau dir die Sprechblase an. Was dort steht, wird gesagt. Das ist die wörtliche Rede.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#who').textContent = cur.emoji;
  $('#talk').textContent = cur.m.words.filter(x => x.r).map(x => x.w).join(' ');
  $('#ask').innerHTML = cur.want === 'rede'
    ? 'Was wird gesagt? Tippe die <span class="pill rede">wörtliche Rede</span> an.'
    : 'Wer spricht? Tippe den <span class="pill begl">Begleitsatz</span> an.';
  const box = $('#sentence');
  box.classList.remove('solved');
  let n = 0;
  box.innerHTML = chunks(cur.m).map(c => c.t === 'cm' ? '<span class="cm">,</span>'
    : `<button class="chunk ${c.t}" data-t="${c.t}">${c.text}<span class="kbd" aria-hidden="true">${++n}</span></button>`).join('');
}

function choose(el) {
  if (state.busy || !cur || !el) return;
  if (el.dataset.t !== cur.want) {
    const hint = miss(cur, el);
    if (hint) $(`.chunk[data-t="${cur.want}"]`).classList.add('hint');
    say(cur.want === 'rede'
      ? (hint ? 'Tipp: Die wörtliche Rede steht in Anführungszeichen: unten „ und oben “. Dort steht dasselbe wie in der Sprechblase.'
              : 'Fast! Das ist der Begleitsatz. Er sagt nur, wer spricht. Welcher Teil steht in der Sprechblase?')
      : (hint ? 'Tipp: Der Begleitsatz steht außerhalb der Anführungszeichen. Er sagt, wer spricht.'
              : 'Fast! Das ist die wörtliche Rede. Sie steht in Anführungszeichen. Welcher Teil sagt, wer spricht?'), 'think');
    return;
  }
  state.busy = true;
  $$('.chunk').forEach(c => c.classList.remove('hint'));
  $('#sentence').classList.add('solved');
  solve(cur, el, cur.want === 'rede' ? 'die' : 'der');
  say(`${pick(PRAISE, 'praise')} ${cur.want === 'rede'
    ? 'Das ist die <span class="pill rede">wörtliche Rede</span>. Sie steht in Anführungszeichen.'
    : 'Das ist der <span class="pill begl">Begleitsatz</span>. Er sagt, wer spricht.'} ${colored(cur.m)}`, 'happy');
  later(next, 2600);
}

$('#sentence').addEventListener('click', e => choose(e.target.closest('.chunk')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[12]$/.test(e.key)) return;
  choose($$('.chunk')[+e.key - 1]);
});
