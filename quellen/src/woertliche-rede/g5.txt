@@tipp
Begleitsatz <b>vorn</b>: Doppelpunkt, dann geht es groß weiter. Begleitsatz <b>hinten</b>: Komma nach den Anführungszeichen oben, dann geht es klein weiter. Begleitsatz <b>in der Mitte</b>: Er steht zwischen zwei Kommas, und beide Teile der Rede haben eigene Anführungszeichen.

/* ---------- Satz zerlegen und einfärben (in allen sechs Spielen gleich) ---------- */
/* model(satz) liefert: words = die Wörter (r: gehört zur wörtlichen Rede),
   sig[i] = die Zeichen „ “ : , die vor Wort i stehen (sig[letztes + 1] = am Satzende),
   kind = wo der Begleitsatz steht: vorn, hinten oder mitte */
function model(s) {
  const words = [], sig = [''];
  let inR = false, w = '';
  const flush = () => { if (w) { words.push({ w, r: inR }); w = ''; sig[words.length] = ''; } };
  for (const ch of s) {
    if (ch === '„') { flush(); sig[words.length] += ch; inR = true; }
    else if (ch === '“') { flush(); sig[words.length] += ch; inR = false; }
    else if (!inR && (ch === ':' || ch === ',')) { flush(); sig[words.length] += ch; }
    else if (ch === ' ') flush();
    else w += ch;
  }
  flush();
  const kind = sig[0] !== '„' ? 'vorn' : sig.filter(x => x.includes('„')).length > 1 ? 'mitte' : 'hinten';
  return { words, sig, kind };
}
const KIND = { vorn: 'Begleitsatz vorn', hinten: 'Begleitsatz hinten', mitte: 'Begleitsatz in der Mitte' };
const Z = ch => `<b class="z">${ch}</b>`;
const W = x => `<span class="t ${x.r ? 'rede' : 'begl'}">${x.w}</span>`;
/* Setzt den Satz wieder zusammen. Zeichen und Wort, die zusammengehören, bleiben in einer Zeile. */
function layout(m, zf, wf) {
  return m.words.map((x, i) => {
    const open = m.sig[i].includes('„') ? zf('„') : '';
    const close = [...m.sig[i + 1].replace('„', '')].map(ch => zf(ch)).join('');
    return `<span class="nb">${open}${wf(x)}${close}</span>`;
  }).join(' ');
}
const colored = m => `<span class="sol">${layout(m, Z, W)}</span>`;
const reviewBy = (list, kinds) => `<div class="review ${kinds.length === 2 ? 'c2' : ''}">${kinds.map(k =>
  `<div class="col neutral"><h3>${KIND[k]}</h3><ul>${list.filter(s => s.m.kind === k).map(s => `<li>${layout(s.m, Z, W)}</li>`).join('')}</ul></div>`).join('')}</div>`;

/* ---------- Spiel: Satz-Detektiv ---------- */
let all = [], queue = [], cur = null;

function startGame() {
  all = TASKS.map(t => ({ text: t[0], wrong: t.slice(1), m: model(t[0]), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies alle drei Sätze genau. Nur einer ist richtig geschrieben. Tippe ihn an.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  const opts = shuffle([{ text: cur.text, ok: true }].concat(cur.wrong.map(w => ({ text: w[0], why: w[1] }))));
  cur.opts = opts;
  $('#opts').innerHTML = opts.map((o, i) => `<button class="opt" data-i="${i}"><span>${o.text}</span><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function choose(el) {
  if (state.busy || !cur || !el || el.classList.contains('off')) return;
  const o = cur.opts[+el.dataset.i];
  if (!o.ok) {
    const hint = miss(cur, el);
    el.classList.add('off'); el.setAttribute('aria-disabled', 'true');
    if (hint) $(`.opt[data-i="${cur.opts.findIndex(x => x.ok)}"]`).classList.add('hint');
    say(hint ? `Tipp: ${o.why} Vergleiche jetzt genau mit dem Satz, der leuchtet.` : `Fast! Schau genau hin: ${o.why}`, 'think');
    return;
  }
  state.busy = true;
  el.classList.remove('hint'); el.classList.add('good');
  el.firstChild.innerHTML = layout(cur.m, Z, W);
  solve(cur, el, 'sun');
  say(`${pick(PRAISE, 'praise')} ${KIND[cur.m.kind]}: ${cur.m.kind === 'vorn' ? 'Doppelpunkt, dann die Rede in Anführungszeichen.'
    : cur.m.kind === 'hinten' ? 'Komma nach den Anführungszeichen oben, dann geht es klein weiter.'
    : 'zwei Kommas und zweimal Anführungszeichen.'} ${colored(cur.m)}`, 'happy');
  later(next, 3000);
}

$('#opts').addEventListener('click', e => choose(e.target.closest('.opt')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[123]$/.test(e.key)) return;
  choose($$('.opt')[+e.key - 1]);
