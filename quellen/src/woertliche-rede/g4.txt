@@tipp
<b>Fragezeichen</b> und <b>Ausrufezeichen</b> bleiben in der wörtlichen Rede immer stehen. Der <b>Punkt</b> bleibt nur, wenn der Satz mit der wörtlichen Rede zu Ende ist. Geht der Satz danach noch weiter, fällt der Punkt weg: „Ich komme gleich“, sagt Tom.

/* ---------- Satz zerlegen und einfärben (in allen sechs Spielen gleich) ---------- */
/* model(satz) liefert: words = die Wörter (r: gehört zur wörtlichen Rede),
   sig[i] = die Zeichen „ “ : , die vor Wort i stehen (sig[letztes + 1] = am Satzende),
   kind = wo der Begleitsatz steht: vorn, hinten oder mitte */
function model(s) {
  const words = [], sig = [''];
  let inR = false, w = '';
  const flush = () => { if (w) { words.push({ w, r: inR }); w = ''; sig[words.length] = ''; } };
  for (const ch of s) {
    if (ch === '„') { flush(); sig[words.length] += ch; inR = true; }
    else if (ch === '“') { flush(); sig[words.length] += ch; inR = false; }
    else if (!inR && (ch === ':' || ch === ',')) { flush(); sig[words.length] += ch; }
    else if (ch === ' ') flush();
    else w += ch;
  }
  flush();
  const kind = sig[0] !== '„' ? 'vorn' : sig.filter(x => x.includes('„')).length > 1 ? 'mitte' : 'hinten';
  return { words, sig, kind };
}
const KIND = { vorn: 'Begleitsatz vorn', hinten: 'Begleitsatz hinten', mitte: 'Begleitsatz in der Mitte' };
const Z = ch => `<b class="z">${ch}</b>`;
const W = x => `<span class="t ${x.r ? 'rede' : 'begl'}">${x.w}</span>`;
/* Setzt den Satz wieder zusammen. Zeichen und Wort, die zusammengehören, bleiben in einer Zeile. */
function layout(m, zf, wf) {
  return m.words.map((x, i) => {
    const open = m.sig[i].includes('„') ? zf('„') : '';
    const close = [...m.sig[i + 1].replace('„', '')].map(ch => zf(ch)).join('');
    return `<span class="nb">${open}${wf(x)}${close}</span>`;
  }).join(' ');
}
const colored = m => `<span class="sol">${layout(m, Z, W)}</span>`;
const reviewBy = (list, kinds) => `<div class="review ${kinds.length === 2 ? 'c2' : ''}">${kinds.map(k =>
  `<div class="col neutral"><h3>${KIND[k]}</h3><ul>${list.filter(s => s.m.kind === k).map(s => `<li>${layout(s.m, Z, W)}</li>`).join('')}</ul></div>`).join('')}</div>`;

/* ---------- Spiel: Zeichen-Körbe ---------- */
const SIGN = { punkt: '.', frage: '?', ausruf: '!', kein: '' };
const COLOR = { punkt: 'teal', frage: 'bonus', ausruf: 'sun', kein: 'das' };
const BLANK = '<span class="blank" role="img" aria-label="Lücke"></span>';
let all = [], rounds = [], round = 0, left = 0, selCard = null, selKind = null;
const byId = id => all.find(s => s.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = k => $(`.basket[data-kind="${k}"]`);

function startGame() {
  all = SENTENCES.map((raw, i) => {
    const at = raw.indexOf('|'), prev = raw[at - 1];
    const k = KINDS.find(x => SIGN[x] === prev) || 'kein';
    const full = raw.replace('|', '');
    return { id: 's' + i, k, m: model(full), card: raw.replace(/(\S+)\|“(,?)/, (x, w, c) => `<span class="nb">${k === 'kein' ? w : w.slice(0, -1)}${BLANK}“${c}</span>`),
      goesOn: raw.slice(at).includes('„'), misses: 0, done: false };
  });
  /* In jeder Runde liegen Karten für verschiedene Körbe */
  const n = Math.ceil(all.length / CARDS_PER_ROUND);
  rounds = Array.from({ length: n }, () => []);
  let r = 0;
  KINDS.forEach(k => shuffle(all.filter(s => s.k === k)).forEach(s => { rounds[r % n].push(s); r++; }));
  rounds.forEach(shuffle);
  resetState(all.length);
  round = 0;
  startRound();
  say('Tippe eine Karte an. Was gehört in die Lücke?');
  show('game');
}

function startRound() {
  selCard = selKind = null; left = rounds[round].length;
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#words').innerHTML = rounds[round].map(s =>
    `<button class="word-card" data-id="${s.id}" aria-pressed="false" style="--tilt:${(Math.random() * 1.4 - .7).toFixed(2)}deg"><span class="word">${s.card}</span></button>`).join('');
  renderSelection();
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selCard;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.kind === selKind;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const s = selCard && byId(selCard);
  if (s && s.misses >= 2) basketEl(s.k).classList.add('hint');
}

function tapCard(id) {
  const s = byId(id);
  if (!s || s.done || state.busy) return;
  if (selKind) { attempt(id, selKind); return; }
  selCard = selCard === id ? null : id;
  sfx.pick(); renderSelection();
  if (selCard) say('Was gehört in die Lücke: ein Punkt, ein Fragezeichen, ein Ausrufezeichen oder kein Zeichen?');
}
function tapBasket(k) {
  if (state.busy) return;
  if (selCard) { attempt(selCard, k); return; }
  selKind = selKind === k ? null : k;
  sfx.pick(); renderSelection();
  if (selKind) say(`Welcher Satz gehört in den Korb <span class="pill ${k}">${LABEL[k]}</span>?`);
}

function tip(s) {
  if (s.k === 'frage') return 'Tipp: Hier wird etwas gefragt. Das <b>Fragezeichen</b> bleibt in der wörtlichen Rede immer stehen.';
  if (s.k === 'ausruf') return 'Tipp: Hier wird gerufen. Das <b>Ausrufezeichen</b> bleibt in der wörtlichen Rede immer stehen.';
  if (s.k === 'punkt') return 'Tipp: Hier wird etwas erzählt. Nach der wörtlichen Rede ist der Satz zu Ende. Dann bleibt der <b>Punkt</b> stehen.';
  return s.goesOn ? 'Tipp: Die wörtliche Rede geht nach dem Begleitsatz noch weiter. An dieser Stelle steht <b>kein Zeichen</b>.'
    : 'Tipp: Hier wird etwas erzählt und der Begleitsatz steht hinten. Dann fällt der Punkt weg: <b>kein Zeichen</b>.';
}
function why(s) {
  if (s.k === 'frage') return 'Das <span class="pill frage">Fragezeichen</span> bleibt immer stehen.';
  if (s.k === 'ausruf') return 'Das <span class="pill ausruf">Ausrufezeichen</span> bleibt immer stehen.';
  if (s.k === 'punkt') return 'Der Satz endet mit der wörtlichen Rede. Der <span class="pill punkt">Punkt</span> bleibt.';
  return s.goesOn ? 'Die wörtliche Rede geht noch weiter. Hier steht <span class="pill kein">kein Zeichen</span>.'
    : 'Der Begleitsatz steht hinten. Der Punkt fällt weg: <span class="pill kein">kein Zeichen</span>.';
}

function attempt(id, k) {
  const s = byId(id);
  if (!s || s.done) return;
  if (s.k !== k) {
    miss(s, cardEl(id));
    wobble(basketEl(k));
    selCard = id; selKind = null;
    renderSelection();
    say(s.misses < 2 ? 'Fast! Überlege: Wird gefragt, gerufen oder etwas erzählt? Und geht der Satz nach der Lücke noch weiter?' : tip(s), 'think');
    return;
  }
  const card = cardEl(id);
  selCard = selKind = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(s, basketEl(k), COLOR[k]);
  say(`${pick(PRAISE, 'praise')} ${why(s)} ${colored(s.m)}`, 'happy');
  if (--left === 0) {
    state.busy = true;
    if (round < rounds.length - 1) later(() => { round++; startRound(); state.busy = false; say('Runde geschafft! Hier kommen neue Karten.', 'happy'); }, 2400);
    else later(finish, 2200);
  }
}

  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapCard(c.dataset.id);
});
$$('.basket').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapBasket(b.dataset.kind); }));
enableDrag('.word-card:not(.done)', '.basket', (card, basket) => attempt(card.dataset.id, basket.dataset.kind));

  if (!gameOn() || !/^[1-4]$/.test(e.key)) return;
  tapBasket(KINDS[+e.key - 1]);
