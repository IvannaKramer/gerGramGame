/* ---------- Satz zerlegen und einfärben (in allen sechs Spielen gleich) ---------- */
/* model(satz) liefert: words = die Wörter (r: gehört zur wörtlichen Rede),
   sig[i] = die Zeichen „ “ : , die vor Wort i stehen (sig[letztes + 1] = am Satzende),
   kind = wo der Begleitsatz steht: vorn, hinten oder mitte */
function model(s) {
  const words = [], sig = [''];
  let inR = false, w = '';
  const flush = () => { if (w) { words.push({ w, r: inR }); w = ''; sig[words.length] = ''; } };
  for (const ch of s) {
    if (ch === '„') { flush(); sig[words.length] += ch; inR = true; }
    else if (ch === '“') { flush(); sig[words.length] += ch; inR = false; }
    else if (!inR && (ch === ':' || ch === ',')) { flush(); sig[words.length] += ch; }
    else if (ch === ' ') flush();
    else w += ch;
  }
  flush();
  const kind = sig[0] !== '„' ? 'vorn' : sig.filter(x => x.includes('„')).length > 1 ? 'mitte' : 'hinten';
  return { words, sig, kind };
}
const KIND = { vorn: 'Begleitsatz vorn', hinten: 'Begleitsatz hinten', mitte: 'Begleitsatz in der Mitte' };
const Z = ch => `<b class="z">${ch}</b>`;
const W = x => `<span class="t ${x.r ? 'rede' : 'begl'}">${x.w}</span>`;
/* Setzt den Satz wieder zusammen. Zeichen und Wort, die zusammengehören, bleiben in einer Zeile. */
function layout(m, zf, wf) {
  return m.words.map((x, i) => {
    const open = m.sig[i].includes('„') ? zf('„') : '';
    const close = [...m.sig[i + 1].replace('„', '')].map(ch => zf(ch)).join('');
    return `<span class="nb">${open}${wf(x)}${close}</span>`;
  }).join(' ');
}
const colored = m => `<span class="sol">${layout(m, Z, W)}</span>`;
const reviewBy = (list, kinds) => `<div class="review ${kinds.length === 2 ? 'c2' : ''}">${kinds.map(k =>
  `<div class="col neutral"><h3>${KIND[k]}</h3><ul>${list.filter(s => s.m.kind === k).map(s => `<li>${layout(s.m, Z, W)}</li>`).join('')}</ul></div>`).join('')}</div>`;

/* ---------- Spiel: Satz-Puzzle ---------- */
const LABEL = { '„': 'unten', '“': 'oben', ':': 'Doppelpunkt', ',': 'Komma' };
const NAME = { '„': 'Anführungszeichen unten', '“': 'Anführungszeichen oben', ':': 'Doppelpunkt', ',': 'Komma' };
let all = [], queue = [], cur = null, order = [], step = 0;

/* Zerlegt den Satz in Teile: Text (rede/begl) und einzelne Zeichen (zei) */
function pieces(m) {
  const out = []; let buf = [], r = false;
  const flush = () => { if (buf.length) { out.push({ t: r ? 'rede' : 'begl', text: buf.join(' ') }); buf = []; } };
  m.words.forEach((x, i) => {
    if (m.sig[i]) { flush(); [...m.sig[i]].forEach(ch => out.push({ t: 'zei', text: ch })); }
    buf.push(x.w); r = x.r;
  });
  flush();
  [...m.sig[m.words.length]].forEach(ch => out.push({ t: 'zei', text: ch }));
  return out;
}
const pieceEl = i => $(`.piece[data-i="${i}"]`);

function startGame() {
  all = SENTENCES.map(text => ({ text, m: model(text), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Schau zuerst auf die gelben Zeichen: Gibt es einen Doppelpunkt oder ein Komma?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; step = 0;
  order = pieces(cur.m).map((p, i) => ({ ...p, i }));
  let mixed;
  do { mixed = shuffle(order.slice()); } while (mixed.every((p, i) => p.i === i));
  $('#pieces').innerHTML = mixed.map((p, n) => p.t === 'zei'
    ? `<button class="piece zei" data-i="${p.i}" aria-label="${NAME[p.text]}"><b aria-hidden="true">${p.text}</b><small>${LABEL[p.text]}</small><span class="kbd" aria-hidden="true">${n + 1}</span></button>`
    : `<button class="piece ${p.t}" data-i="${p.i}">${p.text}<span class="kbd" aria-hidden="true">${n + 1}</span></button>`).join('');
  drawLine();
}

function drawLine() {
  const box = $('#line');
  box.classList.toggle('solved', step === order.length);
  box.innerHTML = step === 0 ? '<span class="ph">Hier entsteht dein Satz.</span>'
    : '<span>' + order.slice(0, step).map((p, i) => (i && (p.t === 'begl' || p.text === '„') ? ' ' : '') +
        (p.t === 'zei' ? Z(p.text) : `<span class="t ${p.t}">${p.text}</span>`)).join('') + '</span>';
}

/* Hilfe-Sätze: [beim ersten Fehlversuch, als Tipp] */
function help(p) {
  if (p.t === 'begl' && step === 0) return ['Schau auf die gelben Zeichen: Es gibt einen Doppelpunkt. Wo steht dann der Begleitsatz?', 'Zum Doppelpunkt gehört ein Begleitsatz vorn. Beginne mit dem blauen Teil.'];
  if (p.t === 'begl') return ['Es fehlt nur noch ein Teil.', 'Der Begleitsatz steht hinten. Er ist kleingeschrieben und endet mit einem Punkt.'];
  if (p.t === 'rede') return ['Die Anführungszeichen unten sind schon da. Was wird gesagt?', 'Nach den Anführungszeichen unten kommt die pinke wörtliche Rede.'];
  if (p.text === ':') return ['Was kommt direkt nach dem Begleitsatz vorn?', 'Nach dem Begleitsatz vorn steht der <b>Doppelpunkt</b>.'];
  if (p.text === ',') return ['Was steht zwischen der wörtlichen Rede und dem Begleitsatz hinten?', 'Nach den Anführungszeichen oben steht das <b>Komma</b>.'];
  if (p.text === '“') return ['Die wörtliche Rede ist zu Ende. Welches Zeichen fehlt jetzt?', 'Die wörtliche Rede endet mit den Anführungszeichen <b>oben</b>.'];
  return step === 0
    ? ['Schau auf die gelben Zeichen: Es gibt ein Komma. Dann steht der Begleitsatz hinten. Womit beginnt die wörtliche Rede?', 'Der Satz beginnt mit der wörtlichen Rede. Davor stehen die Anführungszeichen <b>unten</b>.']
    : ['Jetzt beginnt die wörtliche Rede. Welches Zeichen steht davor?', 'Die wörtliche Rede beginnt mit den Anführungszeichen <b>unten</b>.'];
}

function choose(el) {
  if (state.busy || !cur || !el || el.classList.contains('used')) return;
  const want = order[step];
  if (+el.dataset.i !== want.i) {
    const hint = miss(cur, el), h = help(want);
    if (hint) pieceEl(want.i).classList.add('hint');
    say(hint ? `Tipp: ${h[1]}` : `Fast! ${h[0]}`, 'think');
    return;
  }
  el.classList.remove('hint'); el.classList.add('used'); el.disabled = true;
  step++;
  drawLine();
  if (step < order.length) { sfx.pick(); return; }
  state.busy = true;
  solve(cur, $('#line'), 'sun');
  say(`${pick(PRAISE, 'praise')} ${cur.m.kind === 'vorn' ? 'Der Begleitsatz steht vorn. Danach kommt der Doppelpunkt.'
    : 'Der Begleitsatz steht hinten. Nach den Anführungszeichen oben kommt das Komma.'} ${colored(cur.m)}`, 'happy');
  later(next, 2800);
}

$('#pieces').addEventListener('click', e => choose(e.target.closest('.piece')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-5]$/.test(e.key)) return;
  choose($$('.piece')[+e.key - 1]);
});
