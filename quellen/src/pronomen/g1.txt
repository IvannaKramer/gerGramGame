   Pronomen: 'er', 'sie' (eine), 'es' oder 'pl' (sie für viele). Immer 5 Karten bilden eine Runde. */
const WORDS = [
  ['der Hund',   'er',  '🐕'],
  ['der Ball',   'er',  '⚽'],
  ['der Opa',    'er',  '👴'],
  ['der Apfel',  'er',  '🍎'],
  ['die Katze',  'sie', '🐈'],
  ['die Blume',  'sie', '🌸'],
  ['die Oma',    'sie', '👵'],
  ['die Uhr',    'sie', '⏰'],
  ['das Pferd',  'es',  '🐎'],
  ['das Buch',   'es',  '📖'],
  ['das Auto',   'es',  '🚗'],
  ['das Baby',   'es',  '👶'],
  ['die Kinder', 'pl',  '🧒🧒'],
  ['die Schuhe', 'pl',  '👟👟'],
  ['die Vögel',  'pl',  '🐦🐦']
];
const CARDS_PER_ROUND = 5;

const GAME = {
  id: 'koerbe',
  next: { href: '2-pronomen-luecke.html', title: 'Pronomen-Lücke' },
  review: () => `<div class="review c4">${KINDS.map(k =>
    `<div class="col ${k}"><h3>${PRON[k]}${k === 'pl' ? ' (viele)' : ''}</h3><ul>${all.filter(w => w.k === k).map(w => `<li>${w.text} → <b>${PRON[k]}</b></li>`).join('')}</ul></div>`).join('')}</div>`
};

/* ---------- Spiel: Pronomen-Körbe ---------- */
const KINDS = ['er', 'sie', 'es', 'pl'];
const PRON = { er: 'er', sie: 'sie', es: 'es', pl: 'sie' };
const COLOR = { er: 'der', sie: 'die', es: 'das', pl: 'bonus' };
let all = [], rounds = [], round = 0, left = 0, selCard = null, selKind = null;
const byId = id => all.find(w => w.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = k => $(`.basket[data-kind="${k}"]`);
const pair = w => `<span class="pill ${w.k}">${w.text}</span> → <span class="pill ${w.k}">${PRON[w.k]}</span>`;

function startGame() {
  all = WORDS.map((w, i) => ({ id: 'w' + i, text: w[0], k: w[1], emoji: w[2], misses: 0, done: false }));
  /* In jeder Runde liegen Karten für verschiedene Körbe */
  const n = Math.ceil(all.length / CARDS_PER_ROUND);
  rounds = Array.from({ length: n }, () => []);
  let r = 0;
  KINDS.forEach(k => shuffle(all.filter(w => w.k === k)).forEach(w => { rounds[r % n].push(w); r++; }));
  rounds.forEach(shuffle);
  resetState(all.length);
  round = 0;
  startRound();
  say('Tippe eine Karte an und dann den passenden Korb. Der Artikel hilft dir!');
  show('game');
}

function startRound() {
  selCard = selKind = null; left = rounds[round].length;
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#words').innerHTML = rounds[round].map(w =>
    `<button class="word-card" data-id="${w.id}" aria-pressed="false" style="--tilt:${(Math.random() * 3 - 1.5).toFixed(2)}deg">
      <span class="pic" aria-hidden="true">${w.emoji}</span><span class="word">${w.text}</span></button>`).join('');
  renderSelection();
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selCard;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.kind === selKind;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const w = selCard && byId(selCard);
  if (w && w.misses >= 2) basketEl(w.k).classList.add('hint');
}

function tapCard(id) {
  const w = byId(id);
  if (!w || w.done || state.busy) return;
  if (selKind) { attempt(id, selKind); return; }
  selCard = selCard === id ? null : id;
  sfx.pick(); renderSelection();
  if (selCard) say(`<b>${w.text}</b> – welches Pronomen passt?`);
}
function tapBasket(k) {
  if (state.busy) return;
  if (selCard) { attempt(selCard, k); return; }
  selKind = selKind === k ? null : k;
  sfx.pick(); renderSelection();
  if (selKind) say(`Welches Nomen gehört in den Korb <span class="pill ${k}">${PRON[k]}${k === 'pl' ? ' (viele)' : k === 'sie' ? ' (eine)' : ''}</span>?`);
}

function hintText(w) {
  const [art, noun] = w.text.split(' ');
  if (w.k === 'pl') return `Tipp: <b>${w.text}</b> sind viele. Für viele sagst du immer <b>sie</b>. Nimm den lila Korb.`;
  if (w.k === 'sie') return `Tipp: <b>${w.text}</b> ist nur eine. <b>die</b> ${noun} → <b>sie</b>. Nimm den pinken Korb.`;
  return `Tipp: Schau auf den Artikel. <b>${art}</b> ${noun} → <b>${PRON[w.k]}</b>.`;
}

function attempt(id, k) {
  const w = byId(id);
  if (!w || w.done) return;
  if (w.k !== k) {
    const tip = miss(w, cardEl(id));
    wobble(basketEl(k));
    selCard = id; selKind = null;
    renderSelection();
    const twoSie = (w.k === 'sie' && k === 'pl') || (w.k === 'pl' && k === 'sie');
    say(tip ? hintText(w)
      : twoSie ? `Fast! <b>sie</b> stimmt schon. Aber ist <b>${w.text}</b> nur eine oder sind es viele?`
      : `Fast! Schau noch einmal auf den Artikel von <b>${w.text}</b>.`, 'think');
    return;
  }
  const card = cardEl(id);
  selCard = selKind = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(w, basketEl(k), COLOR[k]);
  say(`${pick(PRAISE, 'praise')} ${pair(w)}`, 'happy');
  if (--left === 0) {
    state.busy = true;
    if (round < rounds.length - 1) later(() => { round++; startRound(); state.busy = false; say('Runde geschafft! Hier kommen neue Karten.', 'happy'); }, 2000);
    else later(finish, 1800);
  }
}

$('#words').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapCard(c.dataset.id);
});
$$('.basket').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapBasket(b.dataset.kind); }));
enableDrag('.word-card:not(.done)', '.basket', (card, basket) => attempt(card.dataset.id, basket.dataset.kind));
