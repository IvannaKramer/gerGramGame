@@tipp
Das Pronomen steht für das Nomen aus dem ersten Satz. Schau auf den Artikel: <b>der</b> → <b>er</b>, <b>die</b> → <b>sie</b>, <b>das</b> → <b>es</b>. Sind es viele? Dann passt <b>sie</b> – und das Verb ändert sich: sie flieg<b>en</b>.

/* ---------- Spiel: Pronomen-Lücke ---------- */
const PRON = { er: 'er', sie: 'sie', es: 'es', pl: 'sie' };
const COLOR = { er: 'der', sie: 'die', es: 'das', pl: 'bonus' };
const cap = s => s[0].toUpperCase() + s.slice(1);
let all = [], queue = [], cur = null;
const tileEl = k => $(`.tile[data-k="${k}"]`);
const full = s => `${s.s1} ${s.s2.replace('___', `<b class="w ${s.k}">${cap(PRON[s.k])}</b>`)}`;

function startGame() {
  all = SENTENCES.map(s => {
    const [art, noun] = s[0].split(' ');
    return { s1: s[0], s2: s[1], k: s[2], emoji: s[3], art: art.toLowerCase(), noun, misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welches Pronomen steht für das unterstrichene Nomen? Tippe unten darauf.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  $$('.tile').forEach(t => t.classList.remove('hint'));
  if (!cur) { finish(); return; }
  if (state.solved) say('Weiter geht es! Welches Pronomen passt hier?');
  state.busy = false;
  $('#pic').textContent = cur.emoji;
  $('#s1').innerHTML = cur.s1.replace(/^(\S+ \S+)/, '<u>$1</u>');
  $('#s2').innerHTML = cur.s2.replace('___', '<span class="gap" id="gap" role="img" aria-label="Lücke">?</span>');
}

function hintText(s) {
  if (s.k === 'pl') return `Tipp: <b>die ${s.noun}</b> sind viele. Für viele sagst du immer <b>sie</b>. Nimm das lila sie.`;
  if (s.k === 'sie') return `Tipp: <b>die ${s.noun}</b> ist nur eine. <b>die</b> → <b>sie</b>. Nimm das pinke sie.`;
  return `Tipp: Schau auf den Artikel. <b>${s.art}</b> ${s.noun} → <b>${PRON[s.k]}</b>.`;
}

function choose(k) {
  if (state.busy || !cur || !gameOn()) return;
  const el = tileEl(k);
  if (k !== cur.k) {
    /* Das Wort „sie“ stimmt, nur die Farbe nicht: kein Fehlversuch, aber ein Hinweis */
    if ((k === 'sie' && cur.k === 'pl') || (k === 'pl' && cur.k === 'sie')) {
      wobble(el);
      say(cur.k === 'pl'
        ? `Das Wort <b>sie</b> stimmt schon! Aber <b>die ${cur.noun}</b> sind viele. Nimm das lila sie.`
        : `Das Wort <b>sie</b> stimmt schon! Aber <b>die ${cur.noun}</b> ist nur eine. Nimm das pinke sie.`, 'think');
      return;
    }
    const tip = miss(cur, el);
    if (tip) tileEl(cur.k).classList.add('hint');
    say(tip ? hintText(cur) : `Fast! Schau auf den Artikel: <b>${cur.art} ${cur.noun}</b>.`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const gap = $('#gap');
  gap.textContent = cap(PRON[k]);
  gap.classList.add('filled', k);
  gap.setAttribute('aria-label', cap(PRON[k]));
  solve(cur, gap, COLOR[k]);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${k}">${cur.art} ${cur.noun}</span> → <span class="pill ${k}">${PRON[k]}</span>`, 'happy');
  later(next, 2400);
}

$('#tiles').addEventListener('click', e => { const t = e.target.closest('.tile'); if (t) choose(t.dataset.k); });
document.addEventListener('keydown', e => {
  const k = ['er', 'sie', 'es', 'pl'][+e.key - 1];
  if (k && gameOn() && !e.ctrlKey && !e.metaKey && !e.altKey) choose(k);
});
