@@tipp
Es zählt immer der <b>Artikel</b>: <b>das</b> Mädchen → <b>es</b>, <b>der</b> Tisch → <b>er</b>, <b>die</b> Tür → <b>sie</b>. Manchmal ändert sich das Pronomen. Frage <b>Wen?</b> → <b>ihn</b>. Frage <b>Wem?</b> → <b>ihm</b>, <b>ihr</b> oder (bei vielen) <b>ihnen</b>.

/* ---------- Spiel: Pronomen-Profi ---------- */
const GAP = /\[(\w+)\]/;
const COLOR = { er: 'der', sie: 'die', es: 'das', pl: 'bonus' };
let all = [], queue = [], cur = null;
const full = (s, cls) => s.text.replace(GAP, cls === 'w' ? `<b class="w ${s.k}">${s.ans}</b>` : `<span class="pill ${s.k}">${s.ans}</span>`);
const input = () => $('#sentence input');

function startGame() {
  all = SENTENCES.map(s => {
    const ans = s[0].match(GAP)[1];
    return { text: s[0], k: s[1], tip: s[2], ans, start: /^[A-ZÄÖÜ]/.test(ans), obj: /^ih/.test(ans), misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Schreibe das Pronomen in die Lücke. Für welches Nomen steht es?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  if (state.solved) say('Nächster Satz! Für welches Nomen steht das Pronomen?');
  state.busy = false;
  $('#sentence').innerHTML = cur.text.replace(GAP,
    '<input type="text" autocapitalize="off" autocorrect="off" autocomplete="off" spellcheck="false" maxlength="8" aria-label="Lücke für das Pronomen">');
  input().focus({ preventScroll: true });
}

function check() {
  if (state.busy || !cur) return;
  const inp = input(), v = inp.value.trim();
  if (!v) { say('Schreibe zuerst das Pronomen in die Lücke.'); inp.focus(); return; }
  inp.classList.remove('hint');
  if (v !== cur.ans && v.toLowerCase() === cur.ans.toLowerCase()) {
    wobble(inp);
    say(cur.start ? 'Fast richtig! Am Satzanfang schreibt man groß.' : 'Fast richtig! Mitten im Satz schreibt man das Pronomen klein.', 'think');
    inp.focus();
    return;
  }
  if (v !== cur.ans) {
    miss(cur, inp);
    if (cur.misses === 2) inp.classList.add('hint');
    say(cur.misses === 1 ? (cur.obj ? 'Fast! Frage nach: <b>Wen?</b> oder <b>Wem?</b> Und für welches Nomen steht das Pronomen?'
                                    : 'Fast! Schau auf das Nomen im ersten Satz. Welchen Artikel hat es? Ist es eins oder sind es viele?')
      : cur.misses === 2 ? `Tipp: ${cur.tip}`
      : `So ist es richtig: ${full(cur, 'pill')} Schreibe das Pronomen genau so ab.`, 'think');
    inp.focus();
    return;
  }
  state.busy = true;
  inp.readOnly = true; inp.classList.add('good', cur.k);
  solve(cur, inp, COLOR[cur.k]);
  say(`${pick(PRAISE, 'praise')} ${full(cur, 'pill')}`, 'happy');
  later(next, 2600);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
