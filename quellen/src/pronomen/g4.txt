@@tipp
<b>er</b> steht für ein Nomen mit <b>der</b>, <b>es</b> für ein Nomen mit <b>das</b>. Bei <b>sie</b> musst du auf das Verb schauen: <b>Sie schwimmt</b> ist nur eine. <b>Sie schwimmen</b> sind viele.

/* ---------- Spiel: Wer ist gemeint? ---------- */
const KINDS = ['er', 'sie', 'es', 'pl'];
const PRON = { er: 'er', sie: 'sie', es: 'es', pl: 'sie' };
const COLOR = { er: 'der', sie: 'die', es: 'das', pl: 'bonus' };
const cap = s => s[0].toUpperCase() + s.slice(1);
let all = [], queue = [], cur = null, opts = [];
const rest = s => s.text.slice(s.text.indexOf(' ') + 1);

function startGame() {
  all = SENTENCES.map(s => ({ set: SETS[s[0]], text: s[1], k: s[2], verb: s[3], noun: SETS[s[0]][KINDS.indexOf(s[2])], misses: 0 }));
  /* Zwei Sätze mit derselben Auswahl sollen nicht direkt nacheinander kommen */
  do { queue = shuffle(all.slice()); } while (queue.some((s, i) => i && s.set === queue[i - 1].set));
  resetState(all.length);
  say('Für welches Nomen steht das Pronomen? Schau auch auf das Verb!');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  if (state.solved) say('Und hier? Für welches Nomen steht das Pronomen?');
  state.busy = false;
  $('#sentence').innerHTML = `<u>${cur.text.split(' ')[0]}</u> ${rest(cur)}`;
  opts = shuffle(KINDS.map((k, i) => ({ k, noun: cur.set[i] })));
  $('#opts').innerHTML = opts.map((o, i) =>
    `<button class="opt" data-i="${i}">${o.noun}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function hintText(s) {
  if (s.k === 'pl') return `Tipp: Das Verb heißt <b>${s.verb}</b>. Also sind es viele. Such das Nomen in der Mehrzahl!`;
  if (s.k === 'sie') return `Tipp: Das Verb heißt <b>${s.verb}</b>. Also ist es nur eine. Such das Nomen mit <b>die</b> in der Einzahl!`;
  return `Tipp: <b>${cap(PRON[s.k])}</b> steht für ein Nomen mit dem Artikel <b>${s.k === 'er' ? 'der' : 'das'}</b>.`;
}

function choose(i) {
  if (state.busy || !cur || !gameOn() || !opts[i]) return;
  const o = opts[i], el = $(`.opt[data-i="${i}"]`);
  if (o.k !== cur.k) {
    const tip = miss(cur, el);
    if (tip) $(`.opt[data-i="${opts.findIndex(x => x.k === cur.k)}"]`).classList.add('hint');
    const twoSie = (o.k === 'sie' && cur.k === 'pl') || (o.k === 'pl' && cur.k === 'sie');
    say(tip ? hintText(cur)
      : twoSie ? `Fast! Schau auf das Verb: <b>${cur.verb}</b>. Ist es nur eine oder sind es viele?`
      : `Fast! Für <b>${o.noun}</b> sagst du <b>${PRON[o.k]}</b>. Im Satz steht aber <b>${cur.text.split(' ')[0]}</b>.`, 'think');
    return;
  }
  state.busy = true;
  $$('.opt').forEach(b => b.classList.remove('hint'));
  el.classList.add('good', cur.k);
  solve(cur, el, COLOR[cur.k]);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${cur.k}">${cur.text.split(' ')[0]}</span> steht hier für <span class="pill ${cur.k}">${cur.noun}</span>. ${cap(cur.noun)} ${rest(cur)}`, 'happy');
  later(next, 2800);
}

$('#opts').addEventListener('click', e => { const b = e.target.closest('.opt'); if (b) choose(+b.dataset.i); });
document.addEventListener('keydown', e => {
  if (/^[1-4]$/.test(e.key) && gameOn() && !e.ctrlKey && !e.metaKey && !e.altKey) choose(+e.key - 1);
});
