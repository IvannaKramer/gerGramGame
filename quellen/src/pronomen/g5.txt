@@tipp
Beim <b>ersten Mal</b> brauchst du das Nomen. Sonst weiß niemand, wer gemeint ist. Kommt dasselbe Nomen <b>noch einmal</b>, nimmst du ein Pronomen: <b>er</b>, <b>sie</b> oder <b>es</b>. Sind es viele, heißt es <b>sie</b>.

   {Nomen} = dieses Nomen muss stehen bleiben (es steht zum ersten Mal da oder kommt nur einmal vor).
   Jeder Text hat vier Wiederholungen. */

   '{Lina} hat {einen Kater}. [Der Kater|er] heißt Felix. Am Morgen hat [der Kater|er] großen Hunger. [Lina|sie] füllt {den Napf}. Dann geht [Lina|sie] in {die Schule}.'],
  ['Das Zelt',
   '{Jonas} bekommt {ein Zelt}. [Das Zelt|es] ist grün. Im Garten steht [das Zelt|es] unter {dem Baum}. [Jonas|er] holt {eine Lampe}. Am Abend schläft [Jonas|er] draußen.'],

/* ---------- Spiel: Ersetz-Spiel ---------- */
const PRON = { er: 'er', sie: 'sie', es: 'es', pl: 'sie' };
const COLOR = { er: 'der', sie: 'die', es: 'das', pl: 'bonus' };
const MARK = /\[([^\]|]+)\|(\w+)\]|\{([^}]+)\}/g;
const cap = s => s[0].toUpperCase() + s.slice(1);
let stories = [], si = 0, sel = null;
const tileEl = k => $(`.tile[data-k="${k}"]`);
const pron = p => p.start ? cap(PRON[p.k]) : PRON[p.k];
/* „Der Kater“ → „der Kater“, Namen bleiben groß */
const label = p => /^(Der|Die|Das) /.test(p.text) ? p.text[0].toLowerCase() + p.text.slice(1) : p.text;

function parse(title, raw) {
  const parts = []; let last = 0, plain = '';
  for (const m of raw.matchAll(MARK)) {
    const before = raw.slice(last, m.index);
    if (before) parts.push({ t: 'text', text: before });
    plain += before;
    if (m[1]) parts.push({ t: 'target', text: m[1], k: m[2], start: plain === '' || /[.!?] $/.test(plain), misses: 0, tm: 0, done: false });
    else parts.push({ t: 'keep', text: m[3] });
    plain += m[1] || m[3];
    last = m.index + m[0].length;
  }
  if (last < raw.length) parts.push({ t: 'text', text: raw.slice(last) });
  return { title, parts, targets: parts.filter(p => p.t === 'target') };
}

function startGame() {
  stories = STORIES.map(s => parse(s[0], s[1]));
  resetState(stories.reduce((n, s) => n + s.targets.length, 0));
  si = 0;
  show('game');
  startStory();
  say('Tippe auf ein Nomen, das sich wiederholt. Dann wähle unten das Pronomen.');
}

function startStory() {
  sel = null; state.busy = false;
  $('#round').textContent = `Text ${si + 1} von ${stories.length}`;
  $('#story-title').textContent = stories[si].title;
  $('#next-box').hidden = true; $('#tiles').hidden = false;
  render();
}

function render() {
  const s = stories[si];
  $('#story').innerHTML = s.parts.map((p, i) =>
    p.t === 'text' ? p.text
    : p.t === 'keep' ? `<button class="np" data-i="${i}">${p.text}</button>`
    : p.done ? `<span class="pron ${p.k}" data-i="${i}">${pron(p)}</span>`
    : `<button class="np${sel === p ? ' selected' : ''}${!sel && p.misses >= 2 && p === nextTarget() ? ' hint' : ''}" data-i="${i}" aria-pressed="${sel === p}">${p.text}</button>`).join('');
  $('#tiles').classList.toggle('off', !sel);
  $$('.tile').forEach(t => t.classList.toggle('hint', !!sel && sel.tm >= 2 && t.dataset.k === sel.k));
}
const nextTarget = () => stories[si].targets.find(p => !p.done);

function hintText(p) {
  const l = label(p), art = l.split(' ')[0];
  if (p.k === 'pl') return `Tipp: <b>${l}</b> sind viele. Für viele sagst du immer <b>sie</b>. Nimm das lila sie.`;
  if (l === p.text && !/ /.test(l)) return `Tipp: Für <b>${l}</b> sagst du <b>${PRON[p.k]}</b>.`;
  if (p.k === 'sie') return `Tipp: <b>${l}</b> ist nur eine. <b>die</b> → <b>sie</b>. Nimm das pinke sie.`;
  return `Tipp: Schau auf den Artikel. <b>${art}</b> → <b>${PRON[p.k]}</b>.`;
}

function tapPhrase(i) {
  if (state.busy) return;
  const p = stories[si].parts[i];
  if (!p || p.done) return;
  if (p.t === 'keep') {
    const t = nextTarget(), el = $(`.np[data-i="${i}"]`);
    const tip = miss(t, el);
    sel = null; render();
    say(tip ? 'Tipp: Das leuchtende Nomen stand schon einmal im Text. Tippe es an!'
      : `<b>${p.text}</b> muss stehen bleiben. Such ein Nomen, das schon einmal im Text stand!`, 'think');
    return;
  }
  sfx.pick();
  sel = sel === p ? null : p;
  render();
  if (sel) say(`<b>${label(p)}</b> stand schon einmal da. Welches Pronomen passt?`);
}

function choose(k) {
  if (state.busy || !gameOn() || $('#tiles').hidden) return;
  if (!sel) { wobble($('#tiles')); say('Tippe zuerst im Text auf ein Nomen, das sich wiederholt.', 'think'); return; }
  const p = sel, el = tileEl(k);
  if (k !== p.k) {
    if ((k === 'sie' && p.k === 'pl') || (k === 'pl' && p.k === 'sie')) {
      wobble(el);
      say(p.k === 'pl'
        ? `Das Wort <b>sie</b> stimmt schon! Aber <b>${label(p)}</b> sind viele. Nimm das lila sie.`
        : `Das Wort <b>sie</b> stimmt schon! Aber <b>${label(p)}</b> ist nur eine. Nimm das pinke sie.`, 'think');
      return;
    }
    miss(p, el); p.tm++;
    render();
    say(p.tm >= 2 ? hintText(p) : `Fast! Passt <b>${PRON[k]}</b> wirklich zu <b>${label(p)}</b>?`, 'think');
    return;
  }
  p.done = true; sel = null;
  const i = stories[si].parts.indexOf(p);
  render();
  solve(p, $(`.pron[data-i="${i}"]`), COLOR[k]);
  if (nextTarget()) { say(`${pick(PRAISE, 'praise')} <span class="pill ${k}">${label(p)}</span> → <span class="pill ${k}">${PRON[k]}</span>`, 'happy'); return; }
  /* Text fertig: Zeit zum Nachlesen, weiter geht es mit dem Knopf */
  const last = si === stories.length - 1;
  $('#tiles').hidden = true; $('#next-box').hidden = false;
  $('#next-btn').textContent = last ? 'Fertig!' : 'Nächster Text';
  say('Geschafft! Lies den Text noch einmal. Mit Pronomen klingt er viel besser!', 'happy');
  $('#next-btn').focus({ preventScroll: true });
}

$('#story').addEventListener('click', e => { const b = e.target.closest('.np'); if (b) tapPhrase(+b.dataset.i); });
$('#tiles').addEventListener('click', e => { const t = e.target.closest('.tile'); if (t) choose(t.dataset.k); });
$('#next-btn').addEventListener('click', () => {
  if (state.busy) return;
  if (si < stories.length - 1) { si++; startStory(); say('Neuer Text! Welche Nomen wiederholen sich hier?'); window.scrollTo(0, 0); }
  else { state.busy = true; finish(); }
});
document.addEventListener('keydown', e => {
  const k = ['er', 'sie', 'es', 'pl'][+e.key - 1];
  if (k && gameOn() && !e.ctrlKey && !e.metaKey && !e.altKey) choose(k);
});
