/* Aufbau: Jede Runde übt ein Verb. Darunter stehen fünf Paare: [Pronomen, Rest des Satzes].
   In einer Runde darf jede Verbform nur zu einem Pronomen passen. */
const ROUNDS = [
  { verb: 'sein', pairs: [
    ['ich', 'bin müde.'],
    ['du',  'bist mutig.'],
    ['er',  'ist stark.'],
    ['wir', 'sind Freunde.'],
    ['ihr', 'seid schnell.'] ] },
  { verb: 'haben', pairs: [
    ['ich', 'habe Durst.'],
    ['du',  'hast Zeit.'],
    ['er',  'hat Hunger.'],
    ['wir', 'haben Ferien.'],
    ['ihr', 'habt Glück.'] ] },
  { verb: 'sehen', pairs: [
    ['ich', 'sehe einen Vogel.'],
    ['du',  'siehst müde aus.'],
    ['er',  'sieht einen Film.'],
    ['wir', 'sehen das Meer.'],
    ['ihr', 'seht alles.'] ] }
];

const GAME = {
  id: 'memory',
  unit: 'Paare',
  next: { href: '4-wer-ist-gemeint.html', title: 'Wer ist gemeint?' },
  /* Sterne nach Versuchen: Jedes Paar braucht mindestens einen Versuch. */
  stars: () => { const extra = tries - state.total; return extra <= 9 ? 3 : extra <= 18 ? 2 : 1; },
  note: stars => stars === 3
    ? `Du hast nur ${tries} Versuche gebraucht. Starkes Gedächtnis!`
    : `Du hast ${tries} Versuche gebraucht. Spiel nochmal, dann findest du die Paare noch schneller!`,
  review: () => `<div class="review">${rounds.map(r =>
    `<div class="col sp"><h3>${r.verb}</h3><ul>${r.items.map(w => `<li><b>${cap(w.p)}</b> ${w.rest}</li>`).join('')}</ul></div>`).join('')}</div>`
};

/* ---------- Spiel: Pronomen-Memory ---------- */
const KIND = { p: 'Pronomen', v: 'Satz' };
const cap = s => s[0].toUpperCase() + s.slice(1);
let rounds = [], round = 0, cards = [], open = [], tries = 0, left = 0;
const closedLabel = c => `${KIND[c.kind]}-Karte, verdeckt`;
const text = c => c.kind === 'p' ? c.w.p : '… ' + c.w.rest;

function startGame() {
  rounds = ROUNDS.map(r => ({ verb: r.verb, items: r.pairs.map(p => ({ p: p[0], rest: p[1], misses: 0 })) }));
  resetState(rounds.reduce((n, r) => n + r.items.length, 0));
  round = 0; tries = 0;
  startRound();
  say('Dreh zwei Karten um: ein Pronomen und einen Satz. Passen sie zusammen?');
  show('game');
}

function startRound() {
  open = []; left = rounds[round].items.length;
  cards = shuffle(rounds[round].items.flatMap(w => [{ w, kind: 'p' }, { w, kind: 'v' }]));
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}: das Verb „${rounds[round].verb}“`;
  $('#memo').innerHTML = cards.map((c, i) =>
    `<button class="mcard ${c.kind === 'p' ? 'sp' : 'vb'}" data-i="${i}" aria-label="${closedLabel(c)}">
      <span class="inner">
        <span class="back" aria-hidden="true"><b>${c.kind === 'p' ? '?' : '…'}</b><small>${KIND[c.kind]}</small></span>
        <span class="face" aria-hidden="true"><span class="word">${text(c)}</span></span>
      </span></button>`).join('');
}

$('#memo').addEventListener('click', e => {
  const el = e.target.closest('.mcard');
  if (!el || state.busy) return;
  const c = cards[+el.dataset.i];
  if (c.open || c.found) return;
  sfx.pick();
  c.open = true;
  el.classList.add('open');
  el.setAttribute('aria-label', `${KIND[c.kind]}: ${text(c)}`);
  open.push({ c, el });
  if (open.length < 2) return;

  tries++;
  const [a, b] = open; open = [];
  state.busy = true;
  if (a.c.w === b.c.w) {
    a.c.found = b.c.found = true;
    later(() => {
      a.el.classList.add('found'); b.el.classList.add('found');
      solve(a.c.w, b.el, 'teal');
      say(`${pick(PRAISE, 'praise')} <span class="pill sp">${cap(a.c.w.p)}</span> ${a.c.w.rest}`, 'happy');
      state.busy = false;
      if (--left === 0) roundDone();
    }, 450);
  } else {
    const same = a.c.kind === b.c.kind;
    const p = a.c.kind === 'p' ? a.c.w : b.c.w, v = a.c.kind === 'v' ? a.c.w : b.c.w;
    say(same
      ? `Das sind zwei ${KIND[a.c.kind]}-Karten. Dreh eine grünblaue und eine gelbe Karte um!`
      : `„${cap(p.p)} ${v.rest}“ Das klingt noch nicht richtig. Merk dir, wo die Karten liegen!`, 'think');
    later(() => {
      [a, b].forEach(o => { o.c.open = false; o.el.classList.remove('open'); o.el.setAttribute('aria-label', closedLabel(o.c)); });
      state.busy = false;
    }, 1900);
  }
});

function roundDone() {
  state.busy = true;
  if (round < rounds.length - 1) {
    later(() => { round++; startRound(); say(`Runde geschafft! Jetzt kommt das Verb „${rounds[round].verb}“.`, 'happy'); state.busy = false; }, 1600);
  } else later(finish, 1300);
}
