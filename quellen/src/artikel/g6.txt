@@tipp
Bist du beim Artikel unsicher? Dann schlag im <b>Wörterbuch</b> nach. Dort steht der Artikel hinter dem Nomen: <b>Rätsel, das</b>. Manche Endungen helfen dir: Nomen mit <b style="white-space:nowrap">-ung</b> und <b style="white-space:nowrap">-heit</b> haben fast immer <b>die</b>, Nomen mit <b style="white-space:nowrap">-chen</b> haben <b>das</b>.

const WORDS = [
  ['Ärger',         'der', ''],
  ['Hunger',        'der', ''],
  ['Urlaub',        'der', ''],
  ['Unterricht',    'der', ''],
  ['Fehler',        'der', ''],
  ['Streit',        'der', ''],
  ['Gedanke',       'der', ''],
  ['Angst',         'die', ''],
  ['Idee',          'die', ''],
  ['Antwort',       'die', ''],
  ['Reise',         'die', ''],
  ['Gefahr',        'die', ''],
  ['Überraschung',  'die', 'ung'],
  ['Gesundheit',    'die', 'heit'],
  ['Abenteuer',     'das', ''],
  ['Geheimnis',     'das', ''],
  ['Gefühl',        'das', ''],
  ['Wetter',        'das', ''],
  ['Rätsel',        'das', ''],
  ['Märchen',       'das', 'chen']
];
/* Endungs-Hilfen: Sie stimmen fast immer, aber nicht bei jedem Wort. */
const ENDINGS = {
  ung:  'Nomen mit der Endung <b>-ung</b> haben fast immer den Artikel <b>die</b>.',
  heit: 'Nomen mit der Endung <b>-heit</b> haben fast immer den Artikel <b>die</b>.',
  chen: 'Nomen mit der Endung <b>-chen</b> haben den Artikel <b>das</b>.'
};

const GAME = {
  id: 'profi',
  next: null,
  note: () => {
    const m = state.mistakes, n = all.filter(w => w.looked).length;
    const a = m === 0 ? 'Kein einziger Fehlversuch.' : m === 1 ? 'Nur ein Fehlversuch.' : `Du hattest ${m} Fehlversuche.`;
    return `${a} ${n === 0 ? 'Und du musstest kein Wort nachschlagen. Stark!' : n === 1 ? 'Ein Wort hast du nachgeschlagen. Das machen Profis so!' : `${n} Wörter hast du nachgeschlagen. Das machen Profis so!`}`;
  },
  review: () => `<div class="review">${['der', 'die', 'das'].map(k =>
    `<div class="col ${k}"><h3>${k}</h3><ul>${all.filter(w => w.k === k).map(w => `<li><b>${w.k}</b> ${w.noun}</li>`).join('')}</ul></div>`).join('')}</div>`
};

/* ---------- Spiel: Wörterbuch-Profi ---------- */
let all = [], queue = [], cur = null;
const input = $('#answer'), lookupBtn = $('#lookup');
const entry = w => `<small>Im Wörterbuch steht:</small>${w.noun}, <i>${w.k}</i>`;

function startGame() {
  all = WORDS.map(w => ({ noun: w[0], k: w[1], end: w[2], misses: 0, looked: false }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Schreibe das Nomen mit seinem Artikel in das Feld. Bist du unsicher? Dann schlag nach!');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#from').textContent = cur.noun;
  $('#dict').hidden = true;
  lookupBtn.disabled = false;
  input.value = ''; input.readOnly = false; input.className = '';
  input.focus({ preventScroll: true });
}

function showEntry() {
  $('#dict').innerHTML = entry(cur);
  $('#dict').hidden = false;
  lookupBtn.disabled = true;
}

/* Nachschlagen ist kein Fehlversuch. Es gibt dann aber 5 statt 10 Punkte. */
function lookup() {
  if (state.busy || !cur || lookupBtn.disabled) return;
  cur.looked = true;
  if (cur.misses === 0) cur.misses = 1;
  sfx.pick();
  showEntry();
  say(`Gut nachgeschlagen! Im Wörterbuch steht der Artikel hinter dem Nomen: <b>${cur.noun}, ${cur.k}</b>. Jetzt schreibst du ihn davor.`);
  input.focus({ preventScroll: true });
}

function check() {
  if (state.busy || !cur) return;
  const v = input.value.trim().replace(/\s+/g, ' ');
  const parts = v.split(' ');
  input.focus({ preventScroll: true });
  if (!v) { say('Schreibe zuerst den Artikel und das Nomen in das Feld.'); return; }
  if (parts.length !== 2) {
    wobble(input);
    say(`Schreibe zwei Wörter: zuerst den Artikel, dann das Nomen <b>${cur.noun}</b>.`, 'think');
    return;
  }
  const a = parts[0].toLowerCase(), n = parts[1];
  if (n.toLowerCase() !== cur.noun.toLowerCase()) {
    wobble(input);
    say(`Schau genau hin: Das Nomen heißt <b>${cur.noun}</b>. Schreibe es genau so ab.`, 'think');
    return;
  }
  if (/^(ein|eine)$/.test(a)) {
    wobble(input);
    say('Hier brauchst du den bestimmten Artikel: <b>der</b>, <b>die</b> oder <b>das</b>.', 'think');
    return;
  }
  if (!/^(der|die|das)$/.test(a)) {
    wobble(input);
    say('Vor dem Nomen fehlt der Artikel: <b>der</b>, <b>die</b> oder <b>das</b>.', 'think');
    return;
  }
  if (a !== cur.k) {
    miss(cur, input);
    if (cur.misses >= 2 && !cur.looked) {
      showEntry();
      say(`Tipp: ${cur.end ? ENDINGS[cur.end] + ' ' : ''}Ich habe das Wörterbuch für dich aufgeschlagen. Der Artikel steht hinter dem Nomen.`, 'think');
    } else if (cur.looked) {
      say(`Schau noch einmal ins Wörterbuch: <b>${cur.noun}, ${cur.k}</b>. Richtig heißt es <span class="pill ${cur.k}">${cur.k} ${cur.noun}</span>`, 'think');
    } else {
      say('Das passt noch nicht. Sprich das Nomen leise mit den anderen Artikeln. Oder schlag im Wörterbuch nach!', 'think');
    }
    return;
  }
  if (n !== cur.noun) {
    wobble(input);
    say('Fast richtig! Denk daran: Nomen schreibt man groß.', 'think');
    return;
  }
  state.busy = true;
  input.value = `${cur.k} ${cur.noun}`; input.readOnly = true; input.className = 'good ' + cur.k;
  lookupBtn.disabled = true;
  solve(cur, input, cur.k);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${cur.k}">${cur.k} ${cur.noun}</span>` + (cur.end ? ` ${ENDINGS[cur.end]}` : ''), 'happy');
  later(next, cur.end ? 2600 : 1700);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
lookupBtn.addEventListener('click', lookup);
$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.keycap');
  if (!k || input.readOnly) return;
  const a = input.selectionStart ?? input.value.length, b = input.selectionEnd ?? a;
  input.value = input.value.slice(0, a) + k.textContent + input.value.slice(b);
  input.focus();
  input.setSelectionRange(a + 1, a + 1);
});
