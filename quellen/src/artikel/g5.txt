/* ---------- Spiel: Neu oder bekannt? ---------- */
let all = [], queue = [], cur = null, step = 0;
const cap = t => t[0].toUpperCase() + t.slice(1);
const full = s => `${s.s1.replace('___', `<span class="u">${UNB[s.k]}</span>`)} ${s.s2.replace('___', `<b>${cap(s.k)}</b>`)}`;
const tileEl = k => $(`.tile[data-k="${k}"]`);
const gapEl = i => $(`.lgap[data-i="${i}"]`);
const want = () => step === 0 ? UNB[cur.k] : cur.k;

function startGame() {
  all = STORIES.map(s => ({ noun: s[0], k: s[1], s1: s[2], s2: s[3], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies beide Sätze. Welcher Artikel fehlt im ersten Satz? Dort ist das Nomen noch neu.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; step = 0;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const gap = i => `<span class="lgap${i === 0 ? ' cur' : ''}" data-i="${i}" role="img" aria-label="Lücke ${i + 1}">?</span>`;
  $('#story').innerHTML = `<span class="s">${cur.s1.replace('___', gap(0))}</span><span class="s">${cur.s2.replace('___', gap(1))}</span>`;
}

function choose(t) {
  if (state.busy || !cur) return;
  const ok = want(), isUnb = x => x === 'ein' || x === 'eine';
  if (t !== ok) {
    const hint = miss(cur, tileEl(t));
    $$('.tile').forEach(x => x.classList.remove('hint'));
    if (hint) {
      tileEl(ok).classList.add('hint');
      say(step === 0
        ? `Tipp: Hier ist das Nomen neu. Dann steht <b>ein</b> oder <b>eine</b>. Es heißt <b>${cur.k} ${cur.noun}</b>, also …?`
        : `Tipp: Jetzt kennst du das Nomen schon. Dann steht <b>der</b>, <b>die</b> oder <b>das</b>. Aus <b>${UNB[cur.k]} ${cur.noun}</b> wird …?`, 'think');
    } else if (isUnb(t) !== isUnb(ok)) {
      say(step === 0
        ? `Hier kommt <b>${cur.noun}</b> zum ersten Mal vor. Passt dann ein bestimmter oder ein unbestimmter Artikel?`
        : `<b>${cur.noun}</b> kennst du schon aus dem ersten Satz. Passt dann ein bestimmter oder ein unbestimmter Artikel?`, 'think');
    } else say(`Fast! Sprich es leise: Welcher Artikel klingt bei <b>${cur.noun}</b> richtig?`, 'think');
    return;
  }
  $$('.tile').forEach(x => x.classList.remove('hint'));
  const gap = gapEl(step);
  gap.textContent = step === 0 ? ok : cap(ok);
  gap.className = 'lgap filled ' + (step === 0 ? 'unb' : cur.k);
  gap.removeAttribute('role'); gap.removeAttribute('aria-label');
  if (step === 0) {
    step = 1; sfx.pick();
    gapEl(1).classList.add('cur');
    say(`Genau! <span class="pill unb">${ok} ${cur.noun}</span> ist neu. Und was fehlt im zweiten Satz?`, 'happy');
    return;
  }
  state.busy = true;
  solve(cur, gap, cur.k);
  say(`${pick(PRAISE, 'praise')} Erst <span class="pill unb">${UNB[cur.k]} ${cur.noun}</span>, dann <span class="pill ${cur.k}">${cur.k} ${cur.noun}</span>`, 'happy');
  later(next, 2000);
}

$('#tiles').addEventListener('click', e => {
  const t = e.target.closest('.tile');
  if (t) choose(t.dataset.k);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-5]$/.test(e.key)) return;
  choose(TILES[+e.key - 1]);
});
