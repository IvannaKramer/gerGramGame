const PAIRS_PER_ROUND = 5;
/* Aus der wird ein, aus die wird eine, aus das wird ein */
const UNB = { der: 'ein', die: 'eine', das: 'ein' };

const GAME = {
  id: 'zwillinge',
  next: { href: '3-ein-oder-eine.html', title: 'Ein oder eine?' },
  /* Sterne nach Versuchen: Jedes Paar braucht mindestens einen Versuch. */
  stars: () => { const extra = tries - state.total; return extra <= 9 ? 3 : extra <= 18 ? 2 : 1; },
  note: stars => stars === 3
    ? `Du hast nur ${tries} Versuche gebraucht. Starkes Gedächtnis!`
    : `Du hast ${tries} Versuche gebraucht. Spiel nochmal, dann findest du die Paare noch schneller!`,
  review: () => `<div class="review">${['der', 'die', 'das'].map(k =>
    `<div class="col ${k}"><h3>${k} → ${UNB[k]}</h3><ul>${rounds.flat().filter(w => w.k === k).map(w =>

};

/* ---------- Spiel: Artikel-Zwillinge ---------- */
const KIND = { best: 'bestimmt', unb: 'unbestimmt' };
const BACK = { best: '<span>der</span><span>die</span><span>das</span>', unb: '<span>ein</span><span>eine</span>' };
const COLOR = { best: 'lila', unb: 'türkise' };
let rounds = [], round = 0, cards = [], open = [], tries = 0, left = 0;
const closedLabel = c => `Karte ${KIND[c.kind]}, verdeckt`;
const art = c => c.kind === 'best' ? c.w.k : UNB[c.w.k];
const both = w => `<span class="pill ${w.k}">${w.k} ${w.noun}</span> und <span class="pill ${w.k}">${UNB[w.k]} ${w.noun}</span>`;

function startGame() {
  const all = WORDS.map(w => ({ noun: w[0], k: w[1], emoji: w[2], misses: 0 }));
  /* In jeder Runde kommen der, die und das vor */
  const n = Math.ceil(all.length / PAIRS_PER_ROUND);
  rounds = Array.from({ length: n }, () => []);
  let r = 0;
  ['der', 'die', 'das'].forEach(k => shuffle(all.filter(w => w.k === k)).forEach(w => { rounds[r % n].push(w); r++; }));
  resetState(all.length);
  round = 0; tries = 0;
  startRound();
  say('Dreh zwei Karten um: eine lila und eine türkise. Zeigen beide dasselbe Nomen?');
  show('game');
}

function startRound() {
  open = []; left = rounds[round].length;
  cards = shuffle(rounds[round].flatMap(w => [{ w, kind: 'best' }, { w, kind: 'unb' }]));
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#memo').innerHTML = cards.map((c, i) =>
    `<button class="mcard ${c.kind}" data-i="${i}" aria-label="${closedLabel(c)}">
      <span class="inner">
        <span class="back" aria-hidden="true"><b>${BACK[c.kind]}</b><small>${KIND[c.kind]}</small></span>
        <span class="face" aria-hidden="true"><span class="pic">${c.w.emoji}</span><span class="word"><span class="w ${c.w.k}">${art(c)}</span>${c.w.noun}</span></span>
      </span></button>`).join('');
}

$('#memo').addEventListener('click', e => {
  const el = e.target.closest('.mcard');
  if (!el || state.busy) return;
  const c = cards[+el.dataset.i];
  if (c.open || c.found) return;
  sfx.pick();
  c.open = true;
  el.classList.add('open');
  el.setAttribute('aria-label', `${KIND[c.kind]}: ${art(c)} ${c.w.noun}`);
  open.push({ c, el });
  if (open.length < 2) return;

  tries++;
  const [a, b] = open; open = [];
  state.busy = true;
  if (a.c.w === b.c.w) {
    a.c.found = b.c.found = true;
    later(() => {
      a.el.classList.add('found'); b.el.classList.add('found');
      solve(a.c.w, b.el, a.c.w.k);
      say(`${pick(PRAISE, 'praise')} ${both(a.c.w)} gehören zusammen. Aus <b>${a.c.w.k}</b> wird <b>${UNB[a.c.w.k]}</b>.`, 'happy');
      state.busy = false;
      if (--left === 0) roundDone();
    }, 450);
  } else {
    const same = a.c.kind === b.c.kind;
    say(same
      ? `Das sind zwei ${COLOR[a.c.kind]} Karten. Dreh eine lila und eine türkise Karte um!`
      : 'Das sind verschiedene Nomen. Merk dir, wo die Karten liegen!', 'think');
    later(() => {
      [a, b].forEach(o => { o.c.open = false; o.el.classList.remove('open'); o.el.setAttribute('aria-label', closedLabel(o.c)); });
      state.busy = false;
    }, 1700);
  }
});

function roundDone() {
  state.busy = true;
  if (round < rounds.length - 1) {
    later(() => { say('Runde geschafft! Hier kommen neue Karten.', 'happy'); round++; startRound(); state.busy = false; }, 1600);
  } else later(finish, 1300);
}
