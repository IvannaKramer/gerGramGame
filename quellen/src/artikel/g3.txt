@@tipp
Schau auf den bestimmten Artikel: Aus <b>der</b> wird <b>ein</b>. Aus <b>die</b> wird <b>eine</b>. Aus <b>das</b> wird <b>ein</b>. Nur bei <b>die</b> hängt also ein <b>e</b> am Ende.

/* ---------- Spiel: Ein oder eine? ---------- */
let all = [], queue = [], cur = null;
const tileEl = k => $(`.tile[data-k="${k}"]`);

function startGame() {
  all = WORDS.map(w => ({ noun: w[0], k: w[1], emoji: w[2], ans: UNB[w[1]], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies das Nomen mit seinem Artikel. Heißt es dann <b>ein</b> oder <b>eine</b>?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#pic').textContent = cur.emoji;
  $('#from').innerHTML = `<span class="w ${cur.k}">${cur.k}</span> ${cur.noun}`;
  $('#build').innerHTML = `<span class="gap" id="gap" role="img" aria-label="Lücke">?</span> ${cur.noun}`;
}

function choose(k) {
  if (state.busy || !cur) return;
  if (k !== cur.ans) {
    const hint = miss(cur, tileEl(k));
    if (hint) tileEl(cur.ans).classList.add('hint');
    say(hint ? `Tipp: Aus <b>${cur.k}</b> wird immer <b>${cur.ans}</b>. Nur bei <b>die</b> hängt ein e am Ende.`
      : `Fast! Schau auf den Artikel: Es heißt <b>${cur.k}</b> ${cur.noun}. Was wird aus <b>${cur.k}</b>?`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const gap = $('#gap');
  gap.textContent = cur.ans; gap.className = 'gap filled ' + cur.k; gap.removeAttribute('role'); gap.removeAttribute('aria-label');
  solve(cur, gap, cur.k);
  say(`${pick(PRAISE, 'praise')} <b>${cur.k}</b> ${cur.noun} → <span class="pill ${cur.k}">${cur.ans} ${cur.noun}</span>`, 'happy');
  later(next, 1600);
}

$('#tiles').addEventListener('click', e => {
  const t = e.target.closest('.tile');
  if (t) choose(t.dataset.k);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[12]$/.test(e.key)) return;
  choose(e.key === '1' ? 'ein' : 'eine');
});
