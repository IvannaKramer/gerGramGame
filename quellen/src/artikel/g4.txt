@@tipp
Artikel sind kleine Wörter, die ein Nomen begleiten. <b>der</b>, <b>die</b>, <b>das</b> sind <b>bestimmte Artikel</b>. <b>ein</b> und <b>eine</b> sind <b>unbestimmte Artikel</b>. Der Artikel steht vor seinem Nomen.

const KINDS = {
  best: ['bestimmt',   'bonus', '<b>der</b>, <b>die</b> und <b>das</b> sind bestimmte Artikel.'],
  unb:  ['unbestimmt', 'teal',  '<b>ein</b> und <b>eine</b> sind unbestimmte Artikel.']

const GAME = {
  id: 'jagd',
  unit: 'Sätze',
  next: { href: '5-neu-oder-bekannt.html', title: 'Neu oder bekannt?' },
  review: () => `<div class="review c2">${Object.keys(KINDS).map(k =>
    `<div class="col ${k}"><h3>${KINDS[k][0]}er Artikel</h3><ul>${all.filter(s => s.k === k).map(s => `<li>${full(s)}</li>`).join('')}</ul></div>`).join('')}</div>`

/* ---------- Spiel: Artikel-Jagd ---------- */
let all = [], queue = [], cur = null, found = false;
const full = s => s.words.map((w, i) => i === s.at ? `<b class="w ${s.k}">${w}</b>` : w).join(' ');
const tileEl = k => $(`.tile[data-k="${k}"]`);

function startGame() {
  all = SENTENCES.map(text => {
    const raw = text.split(' '), at = raw.findIndex(w => w[0] === '[');
    const words = raw.map(w => w.replace(/[\[\]]/g, ''));
    return { words, at, art: words[at], k: /^(der|die|das)$/i.test(words[at]) ? 'best' : 'unb', misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('In jedem Satz steckt genau ein Artikel. Tippe ihn an!');
  show('game');
  next();
}

function setTiles(on) { $$('.tile').forEach(t => { t.disabled = !on; t.classList.remove('hint'); }); }

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; found = false;
  setTiles(false);
  $('#step').textContent = '1. Tippe den Artikel an';
  $('#sentence').innerHTML = cur.words.map((w, i) => `<button class="wbtn" data-i="${i}">${w}</button>`).join('');
}

function tapWord(i, el) {
  if (state.busy || !cur || found) return;
  if (i !== cur.at) {
    const hint = miss(cur, el);
    if (hint) $(`.wbtn[data-i="${cur.at}"]`).classList.add('hint');
    say(hint ? 'Tipp: Artikel sind die kleinen Wörter <b>der</b>, <b>die</b>, <b>das</b>, <b>ein</b> und <b>eine</b>. Der Artikel leuchtet.'
      : `<b>${el.textContent.replace(/[.!?]/, '')}</b> ist kein Artikel. Such das kleine Wort, das vor einem Nomen steht.`, 'think');
    return;
  }
  found = true;
  sfx.pick();
  $$('.wbtn').forEach(b => { b.classList.remove('hint'); b.disabled = true; });
  el.classList.add('found');
  setTiles(true);
  $('#step').textContent = '2. Bestimmt oder unbestimmt?';
  say(`Gefunden! <b>${cur.art}</b> ist der Artikel. Ist er bestimmt oder unbestimmt?`, 'happy');
}

function choose(k) {
  if (state.busy || !cur || !found) return;
  if (k !== cur.k) {
    const hint = miss(cur, tileEl(k));
    if (hint) tileEl(cur.k).classList.add('hint');
    say(hint ? `Tipp: ${KINDS[cur.k][2]}` : `Fast! Schau dir den Artikel noch einmal an: <b>${cur.art}</b>.`, 'think');
    return;
  }
  state.busy = true;
  setTiles(false);
  const el = $(`.wbtn[data-i="${cur.at}"]`);
  el.className = 'wbtn solved ' + cur.k;
  solve(cur, el, KINDS[cur.k][1]);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${cur.k}">${cur.art}</span> ist ein ${KINDS[cur.k][0]}er Artikel: ${full(cur)}`, 'happy');
  later(next, 1900);
}

$('#sentence').addEventListener('click', e => {
  const b = e.target.closest('.wbtn');
  if (b) tapWord(+b.dataset.i, b);
});
$('#tiles').addEventListener('click', e => {
  const t = e.target.closest('.tile');
  if (t && !t.disabled) choose(t.dataset.k);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[12]$/.test(e.key)) return;
  choose(e.key === '1' ? 'best' : 'unb');
});
