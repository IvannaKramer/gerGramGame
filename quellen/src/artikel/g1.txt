const CARDS_PER_ROUND = 5;
const ARTICLES = ['der', 'die', 'das'];

const GAME = {
  id: 'koerbe',
  next: { href: '2-artikel-zwillinge.html', title: 'Artikel-Zwillinge' },
  review: () => `<div class="review">${ARTICLES.map(k =>
    `<div class="col ${k}"><h3>${k}</h3><ul>${all.filter(w => w.k === k).map(w => `<li><b>${w.k}</b> ${w.noun}</li>`).join('')}</ul></div>`).join('')}</div>`

/* ---------- Spiel: Drei Körbe ---------- */
let all = [], rounds = [], round = 0, left = 0, selCard = null, selKind = null;
const byId = id => all.find(w => w.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = k => $(`.basket[data-kind="${k}"]`);

function startGame() {
  all = WORDS.map((w, i) => ({ id: 'w' + i, noun: w[0], k: w[1], emoji: w[2], misses: 0, done: false }));
  /* In jeder Runde liegen Karten für alle drei Körbe */
  const n = Math.ceil(all.length / CARDS_PER_ROUND);
  rounds = Array.from({ length: n }, () => []);
  let r = 0;
  ARTICLES.forEach(k => shuffle(all.filter(w => w.k === k)).forEach(w => { rounds[r % n].push(w); r++; }));
  rounds.forEach(shuffle);
  resetState(all.length);
  round = 0;
  startRound();
  say('Tippe eine Karte an und dann den passenden Korb. Sprich das Wort leise mit <b>der</b>, <b>die</b> und <b>das</b>!');
  show('game');
}

function startRound() {
  selCard = selKind = null; left = rounds[round].length;
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#words').innerHTML = rounds[round].map(w =>
    `<button class="word-card" data-id="${w.id}" aria-pressed="false" style="--tilt:${(Math.random() * 3 - 1.5).toFixed(2)}deg">
      <span class="pic" aria-hidden="true">${w.emoji}</span><span class="word">${w.noun}</span></button>`).join('');
  renderSelection();
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selCard;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.kind === selKind;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const w = selCard && byId(selCard);
  if (w && w.misses >= 2) basketEl(w.k).classList.add('hint');
}

function tapCard(id) {
  const w = byId(id);
  if (!w || w.done || state.busy) return;
  if (selKind) { attempt(id, selKind); return; }
  selCard = selCard === id ? null : id;
  sfx.pick(); renderSelection();
  if (selCard) say(`Sprich leise: der ${w.noun}, die ${w.noun}, das ${w.noun}. Was klingt richtig?`);
}
function tapBasket(k) {
  if (state.busy) return;
  if (selCard) { attempt(selCard, k); return; }
  selKind = selKind === k ? null : k;
  sfx.pick(); renderSelection();
  if (selKind) say(`Welches Wort gehört in den Korb <span class="pill ${k}">${k}</span>?`);
}

function attempt(id, k) {
  const w = byId(id);
  if (!w || w.done) return;
  if (w.k !== k) {
    miss(w, cardEl(id));
    wobble(basketEl(k));
    selCard = id; selKind = null;
    renderSelection();
    say(w.misses < 2 ? pick(RETRY, 'retry')
      : `Tipp: Für den Artikel gibt es keine feste Regel. Du lernst ihn mit dem Nomen zusammen. Der richtige Korb leuchtet: <b>${w.k} ${w.noun}</b>.`, 'think');
    return;
  }
  const card = cardEl(id);
  selCard = selKind = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(w, basketEl(k), k);
  say(`${pick(PRAISE, 'praise')} Es heißt <span class="pill ${k}">${k} ${w.noun}</span>`, 'happy');
  if (--left === 0) {
    state.busy = true;
    if (round < rounds.length - 1) later(() => { round++; startRound(); state.busy = false; say('Runde geschafft! Hier kommen neue Karten.', 'happy'); }, 1700);
    else later(finish, 1500);
  }
}

$('#words').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapCard(c.dataset.id);
});
$$('.basket').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapBasket(b.dataset.kind); }));
enableDrag('.word-card:not(.done)', '.basket', (card, basket) => attempt(card.dataset.id, basket.dataset.kind));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-3]$/.test(e.key)) return;
  tapBasket(ARTICLES[+e.key - 1]);
});
