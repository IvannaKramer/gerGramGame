/* ---------- Spiel: Siegertreppchen ---------- */
const NAME = { 1: 'Grundstufe', 2: 'Vergleichsstufe', 3: 'Höchststufe' };
const COLOR = { 1: 'der', 2: 'das', 3: 'die' };
const HINT = {
  1: 'Tipp: Die <b>Grundstufe</b> ist das Wort ganz allein, ohne etwas dran. Sie gehört auf Stufe 1.',
  2: 'Tipp: Bei der <b>Vergleichsstufe</b> hängt ein <b>-er</b> am Wort. Sie gehört auf Stufe 2.',
  3: 'Tipp: Die <b>Höchststufe</b> beginnt mit <b>am</b> und endet auf <b>-sten</b>. Sie gehört auf Stufe 3.'
};
const AGAIN = ['Fast! Probier eine andere Stufe.', 'Das passt noch nicht. Schau dir das Wortende genau an.', 'Guter Versuch! Auf welche Stufe gehört das Wort?'];
let all = [], queue = [], cur = null, selCard = null, selStep = null, placed = 0;
const tri = w => `<span class="w s1">${w.f[1]}</span> – <span class="w s2">${w.f[2]}</span> – <span class="w s3">${w.f[3]}</span>`;
const triLi = w => `<li>${tri(w)}</li>`;
const cardEl = lv => $(`.word-card[data-lv="${lv}"]`);
const stepEl = lv => $(`.step[data-lv="${lv}"]`);

function startGame() {
  all = WORDS.map(w => ({ f: { 1: w[0], 2: w[1], 3: w[2] }, emoji: w[3], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Tippe ein Wort an und dann die passende Stufe. Schau genau auf das Wortende!');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  selCard = selStep = null; placed = 0;
  $('#ask').innerHTML = `<span class="pic" aria-hidden="true">${cur.emoji}</span><span>Welches Wort gehört auf welche Stufe?</span>`;
  $('#cards').innerHTML = shuffle([1, 2, 3]).map(lv =>
    `<button class="word-card" data-lv="${lv}" aria-pressed="false" style="--tilt:${(Math.random() * 3 - 1.5).toFixed(2)}deg"><span class="word">${cur.f[lv]}</span></button>`).join('');
  $$('.step').forEach(s => { s.classList.remove('filled'); s.disabled = false; $('.slot', s).textContent = ''; s.setAttribute('aria-label', `Stufe ${s.dataset.lv}: ${NAME[s.dataset.lv]}, noch leer`); });
  renderSelection();
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = +el.dataset.lv === selCard;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.step').forEach(el => {
    const on = +el.dataset.lv === selStep;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  if (selCard && cur.misses >= 2) stepEl(selCard).classList.add('hint');
}

function tapCard(lv) {
  if (state.busy || cardEl(lv).classList.contains('done')) return;
  if (selStep) { attempt(lv, selStep); return; }
  selCard = selCard === lv ? null : lv;
  sfx.pick(); renderSelection();
  if (selCard) say(cur.misses >= 2 ? HINT[lv] : `<b>${cur.f[lv]}</b> – auf welche Stufe gehört das Wort?`);
}
function tapStep(lv) {
  if (state.busy || stepEl(lv).classList.contains('filled')) return;
  if (selCard) { attempt(selCard, lv); return; }
  selStep = selStep === lv ? null : lv;
  sfx.pick(); renderSelection();
  if (selStep) say(`Welches Wort gehört auf die Stufe <span class="pill s${lv}">${NAME[lv]}</span>?`);
}

function attempt(cardLv, stepLv) {
  const card = cardEl(cardLv), step = stepEl(stepLv);
  if (state.busy || !card || card.classList.contains('done') || step.classList.contains('filled')) return;
  if (cardLv !== stepLv) {
    const hint = miss(cur, card);
    wobble(step);
    selCard = cardLv; selStep = null;
    renderSelection();
    say(hint ? HINT[cardLv] : pick(AGAIN, 'again'), 'think');
    return;
  }
  selCard = selStep = null;
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  step.classList.add('filled'); step.disabled = true;
  $('.slot', step).textContent = cur.f[stepLv];
  step.setAttribute('aria-label', `Stufe ${stepLv}: ${NAME[stepLv]}: ${cur.f[stepLv]}`);
  renderSelection();
  if (++placed < 3) {
    sfx.pick();
    say(`Genau! <span class="pill s${stepLv}">${cur.f[stepLv]}</span> ist die ${NAME[stepLv]}. Weiter so!`, 'happy');
    return;
  }
  state.busy = true;
  solve(cur, step, COLOR[stepLv]);
  say(`${pick(PRAISE, 'praise')} Die Treppe ist fertig: ${tri(cur)}`, 'happy');
  later(next, 2100);
}

$('#cards').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapCard(+c.dataset.lv);
});
$$('.step').forEach(s => s.addEventListener('click', () => { if (!suppressClick) tapStep(+s.dataset.lv); }));
enableDrag('.word-card:not(.done)', '.step:not(.filled)', (card, step) => attempt(+card.dataset.lv, +step.dataset.lv));
