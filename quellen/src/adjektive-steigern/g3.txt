@@tipp
Die <b>Vergleichsstufe</b> endet immer auf <b>-er</b>: wärmer. Die <b>Höchststufe</b> beginnt mit <b>am</b> und endet auf <b>-sten</b>: am wärmsten. Aus a, o, u wird manchmal ä, ö, ü. Sprich die Wörter leise vor, dein Ohr hilft dir!

/* ---------- Spiel: Stufe gesucht ---------- */
const NAME = { 1: 'Grundstufe', 2: 'Vergleichsstufe', 3: 'Höchststufe' };
const COLOR = { 1: 'der', 2: 'das', 3: 'die' };
let all = [], queue = [], cur = null;
const tri = w => `<span class="w s1">${w.f[1]}</span> – <span class="w s2">${w.f[2]}</span> – <span class="w s3">${w.f[3]}</span>`;
const triLi = w => `<li>${tri(w)}</li>`;

function startGame() {
  all = WORDS.map(w => ({ f: { 1: w[0], 2: w[1], 3: w[2] }, gap: w[3], wrong: w[4], emoji: w[5], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welches Wort fehlt auf der Treppe? Sprich die drei Stufen leise vor.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  cur.ans = cur.f[cur.gap];
  $('#ask').innerHTML = `<span class="pic" aria-hidden="true">${cur.emoji}</span><span>Welches Wort ist die <span class="pill s${cur.gap}">${NAME[cur.gap]}</span>?</span>`;
  $('#stairs').innerHTML = [3, 2, 1].map(lv =>
    `<div class="step s${lv}${lv === cur.gap ? ' open' : ''}" data-lv="${lv}"><span class="num" aria-hidden="true">${lv}</span><span class="nm">${NAME[lv]}</span>
      <span class="slot">${lv === cur.gap ? '<span role="img" aria-label="Hier fehlt ein Wort">?</span>' : cur.f[lv]}</span></div>`).join('');
  $('#opts').innerHTML = shuffle([cur.ans, ...cur.wrong]).map((o, i) =>
    `<button class="opt" data-o="${o}">${o}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function tipp(w) {
  return w.gap === 2
    ? `Tipp: Die Vergleichsstufe ist <b>ein</b> Wort und endet auf <b>-er</b>. Sprich laut: ${w.f[1]} – … – ${w.f[3]}. Die leuchtende Karte ist richtig.`
    : `Tipp: Die Höchststufe heißt <b>am</b> … <b>-sten</b>. Sprich die Karten laut: Welche kannst du gut aussprechen? Die leuchtende Karte ist richtig.`;
}

function choose(el) {
  if (!el || state.busy || el.disabled) return;
  if (el.dataset.o !== cur.ans) {
    const hint = miss(cur, el);
    el.disabled = true;
    if (hint) $(`.opt[data-o="${cur.ans}"]`).classList.add('hint');
    say(hint ? tipp(cur) : 'Fast! Sprich alle drei Stufen laut. Was klingt richtig?', 'think');
    return;
  }
  state.busy = true;
  $$('.opt').forEach(b => { b.classList.remove('hint'); if (b !== el) b.disabled = true; });
  el.classList.add('right', 's' + cur.gap);
  const step = $(`.step[data-lv="${cur.gap}"]`);
  step.classList.remove('open'); step.classList.add('pop');
  $('.slot', step).textContent = cur.ans;
  solve(cur, $('.slot', step), COLOR[cur.gap]);
  say(`${pick(PRAISE, 'praise')} ${tri(cur)}`, 'happy');
  later(next, 1900);
}

$('#opts').addEventListener('click', e => choose(e.target.closest('.opt')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-3]$/.test(e.key)) return;
  choose($$('.opt')[+e.key - 1]);
});
