@@tipp
Viele kurze Adjektive bekommen beim Steigern Pünktchen: kalt – k<b>ä</b>lter. Aber nicht alle: klar – klarer. Die Höchststufe endet auf <b>-sten</b>. Nach <b>t</b> und <b>z</b> heißt es meistens <b>-esten</b>, damit man das Wort gut sprechen kann: am kält<b>esten</b>.

/* ---------- Spiel: Stufen-Werkstatt ---------- */
const NAME = { 2: 'Vergleichsstufe', 3: 'Höchststufe' };
const COLOR = { 2: 'das', 3: 'die' };
let all = [], queue = [], cur = null, selStem = null, selEnd = null;
const tri = w => `<span class="w s1">${w.g}</span> – <span class="w s2">${w.v}</span> – <span class="w s3">${w.h}</span>`;
const triLi = w => `<li>${tri(w)}</li>`;

function startGame() {
  all = WORDS.map(w => ({ g: w[0], stem: w[1], bad: w[2], endH: w[3], lv: w[4], uml: w[0] !== w[1],
    v: w[1] + 'er', h: 'am ' + w[1] + w[3], end: w[4] === 2 ? 'er' : w[3], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Tippe einen Wortstamm und eine Endung an. Dann tippe auf „Prüfen“.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  selStem = selEnd = null;
  $('#from').textContent = cur.g;
  $('#goal').innerHTML = `Baue die <span class="pill s${cur.lv}">${NAME[cur.lv]}</span>`;
  $('#build').className = 'build';
  $('#stems').innerHTML = shuffle([cur.stem, cur.bad]).map((s, i) =>
    `<button class="part stem-tile" data-v="${s}" aria-pressed="false">${s}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
  $('#ends').innerHTML = ENDINGS.map((e, i) =>
    `<button class="part end-tile" data-v="${e}" aria-pressed="false" aria-label="Endung ${e}">-${e}<span class="kbd" aria-hidden="true">${i + 3}</span></button>`).join('');
  render();
}

function render() {
  $('#build').innerHTML = `${cur.lv === 3 ? '<span class="am">am</span>' : ''}<span class="slot${selStem ? ' on' : ''}">${selStem || '?'}</span><span class="slot${selEnd ? ' on' : ''}">${selEnd ? '-' + selEnd : '?'}</span>`;
  $$('.stem-tile').forEach(b => b.setAttribute('aria-pressed', String(b.dataset.v === selStem)));
  $$('.end-tile').forEach(b => b.setAttribute('aria-pressed', String(b.dataset.v === selEnd)));
}

function tapPart(el) {
  if (!el || state.busy) return;
  sfx.pick();
  if (el.classList.contains('stem-tile')) selStem = el.dataset.v; else selEnd = el.dataset.v;
  render();
}

function tipp(w) {
  const a = w.uml
    ? `Sprich laut: „${w.stem}er“ klingt richtig. Dieses Wort bekommt Pünktchen.`
    : `Sprich laut: „${w.stem}er“ klingt richtig. Dieses Wort bekommt keine Pünktchen.`;
  const b = w.lv === 2 ? 'Die Vergleichsstufe endet auf <b>-er</b>.'
    : w.endH === 'esten' ? `Die Höchststufe endet hier auf <b>-esten</b>, weil der Stamm auf <b>${w.stem.slice(-1)}</b> endet.`
    : 'Die Höchststufe endet hier auf <b>-sten</b>.';
  return `Tipp: ${a} ${b} Die leuchtenden Teile sind richtig.`;
}

function check() {
  if (state.busy || !cur) return;
  if (!selStem || !selEnd) { say('Du brauchst zwei Teile: einen Wortstamm und eine Endung.'); return; }
  if (selStem !== cur.stem || selEnd !== cur.end) {
    const hint = miss(cur, $('#build'));
    if (hint) { $(`.stem-tile[data-v="${cur.stem}"]`).classList.add('hint'); $(`.end-tile[data-v="${cur.end}"]`).classList.add('hint'); }
    say(hint ? tipp(cur)
      : selEnd !== cur.end && (selEnd === 'er' || cur.end === 'er') ? `Fast! Du sollst die <b>${NAME[cur.lv]}</b> bauen. Welche Endung gehört dazu?`
      : selEnd !== cur.end ? 'Fast! Sprich die Höchststufe laut. Mit welcher Endung kannst du sie gut aussprechen?'
      : 'Fast! Sprich das Wort laut. Braucht es Pünktchen oder nicht?', 'think');
    return;
  }
  state.busy = true;
  $$('.part').forEach(b => b.classList.remove('hint'));
  $('#build').classList.add('locked', 's' + cur.lv);
  $('#build').innerHTML = `${cur.lv === 3 ? '<span class="am">am</span>' : ''}<span class="slot">${cur.stem}${cur.end}</span>`;
  solve(cur, $('#build .slot'), COLOR[cur.lv]);
  say(`${pick(PRAISE, 'praise')} ${tri(cur)}`, 'happy');
  later(next, 2000);
}

$('.parts').addEventListener('click', e => tapPart(e.target.closest('.part')));
$('#check').addEventListener('click', check);
document.addEventListener('keydown', e => {
  if (!gameOn()) return;
  if (/^[1-5]$/.test(e.key)) tapPart($$('.part')[+e.key - 1]);
  else if (e.key === 'Enter' && !e.target.closest('button')) check();
});
