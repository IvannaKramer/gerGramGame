@@tipp
Schau auf die kleinen Wörter neben der Lücke! Bei <b>so … wie</b> steht die Grundstufe: so schnell wie. Vor <b>als</b> steht die Vergleichsstufe: schneller als. Nach <b>am</b> steht die Höchststufe: am schnellsten.

/* ---------- Spiel: Treppauf, treppab ---------- */
const NAME = { 1: 'Grundstufe', 2: 'Vergleichsstufe', 3: 'Höchststufe' };
const COLOR = { 1: 'der', 2: 'das', 3: 'die' };
const HINT = {
  1: 'Tipp: Im Satz steht <b>so … wie</b>. Dann sind beide gleich und das Adjektiv bleibt in der <b>Grundstufe</b>.',
  2: 'Tipp: Hinter der Lücke steht <b>als</b>. Davor gehört die <b>Vergleichsstufe</b> mit -er.',
  3: 'Tipp: Vor der Lücke steht <b>am</b>. Danach gehört die <b>Höchststufe</b> mit -sten.'
};
let all = [], queue = [], cur = null, sel = null;
const full = s => s.text.replace('___', `<b class="w s${s.lv}">${s.f[s.lv - 1]}</b>`);
const lvlEl = lv => $(`.lvl[data-lv="${lv}"]`);

function startGame() {
  all = SENTENCES.map(s => ({ text: s[0], f: s[1], lv: s[2], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Tippe auf eine Stufe. Lies dann den Satz: Klingt er richtig?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  sel = null;
  $('#lbl').innerHTML = `Setze ein: <b class="adj">${cur.f[0]}</b> – in welcher Stufe?`;
  $$('.lvl').forEach(b => b.classList.remove('hint'));
  render();
}

function render() {
  $('#sentence').innerHTML = cur.text.replace('___', sel
    ? `<span class="gap on" id="gap">${cur.f[sel - 1]}</span>`
    : `<span class="gap" id="gap"><span role="img" aria-label="Lücke">?</span></span>`);
  $$('.lvl').forEach(b => b.setAttribute('aria-pressed', String(+b.dataset.lv === sel)));
}

function tapLvl(lv) {
  if (state.busy || !cur) return;
  sfx.pick();
  sel = lv;
  render();
}

function check() {
  if (state.busy || !cur) return;
  if (!sel) { say('Tippe zuerst auf eine Stufe der Treppe.'); return; }
  if (sel !== cur.lv) {
    const hint = miss(cur, $('#gap'));
    if (hint) lvlEl(cur.lv).classList.add('hint');
    say(hint ? HINT[cur.lv] : 'Fast! Lies den Satz laut. Welches kleine Wort steht bei der Lücke: wie, als oder am?', 'think');
    return;
  }
  state.busy = true;
  $$('.lvl').forEach(b => b.classList.remove('hint'));
  const gap = $('#gap');
  gap.className = 'gap filled s' + cur.lv;
  solve(cur, gap, COLOR[cur.lv]);
  say(`${pick(PRAISE, 'praise')} Hier passt die <span class="pill s${cur.lv}">${NAME[cur.lv]}</span>: ${full(cur)}`, 'happy');
  later(next, 2100);
}

$$('.lvl').forEach(b => b.addEventListener('click', () => tapLvl(+b.dataset.lv)));
$('#check').addEventListener('click', check);
document.addEventListener('keydown', e => {
  if (!gameOn()) return;
  if (/^[1-3]$/.test(e.key)) tapLvl(+e.key);
  else if (e.key === 'Enter' && !e.target.closest('button')) check();
});
