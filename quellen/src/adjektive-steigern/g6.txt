@@tipp
Manche Adjektive ändern sich beim Steigern stark: <b>gut – besser – am besten</b>, <b>viel – mehr – am meisten</b>, <b>gern – lieber – am liebsten</b>, <b>hoch – höher – am höchsten</b>, <b>nah – näher – am nächsten</b>. Bei <b>teuer</b>, <b>dunkel</b> und <b>sauer</b> fällt in der Vergleichsstufe ein e weg: teurer, dunkler, saurer.

/* ---------- Spiel: Steigerungs-Meister ---------- */
const NAME = { 2: 'Vergleichsstufe', 3: 'Höchststufe' };
const COLOR = { 2: 'das', 3: 'die' };
let all = [], queue = [], cur = null;
const input = $('#answer');
const full = s => s.text.replace('___', `<b class="w s${s.lv}">${s.ans}</b>`);

function startGame() {
  all = SENTENCES.map(s => ({ text: s[0], g: s[1], ans: s[2], lv: s[3], k: s[4], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Schreibe die richtige Form in das Feld. Überlege zuerst: Vergleichsstufe oder Höchststufe?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#sentence').innerHTML = cur.text.replace('___', '<span class="gap" id="gap"><span role="img" aria-label="Lücke">?</span></span>');
  $('#lbl').innerHTML = `Setze ein: <b class="adj">${cur.g}</b>`;
  input.value = ''; input.readOnly = false; input.className = '';
  input.focus({ preventScroll: true });
}

function check() {
  if (state.busy || !cur) return;
  let v = input.value.trim().replace(/\s+/g, ' ');
  if (cur.lv === 3 && /^am /i.test(v)) v = v.slice(3);
  if (!v) { say('Schreibe zuerst das fehlende Wort in das Feld.'); input.focus(); return; }
  if (v !== cur.ans) {
    if (v.toLowerCase() === cur.ans) {
      wobble(input);
      say('Fast richtig! Denk daran: Adjektive schreibt man klein.', 'think');
    } else {
      miss(cur, input);
      say(cur.misses === 1
          ? (cur.lv === 2 ? 'Fast! Im Satz steht <b>als</b>. Du brauchst die Vergleichsstufe. Wie schreibt man sie?'
                          : 'Fast! Im Satz steht <b>am</b>. Du brauchst die Höchststufe. Wie schreibt man sie?')
        : cur.misses === 2 ? RULES[cur.k][1]
        : `So ist es richtig: <span class="pill s${cur.lv}">${cur.ans}</span> Schreibe das Wort genau so ab.`, 'think');
    }
    input.focus();
    return;
  }
  state.busy = true;
  input.value = cur.ans; input.readOnly = true; input.classList.add('good', 's' + cur.lv);
  const gap = $('#gap');
  gap.textContent = cur.ans; gap.className = 'gap filled s' + cur.lv;
  solve(cur, input, COLOR[cur.lv]);
  say(`${pick(PRAISE, 'praise')} ${NAME[cur.lv]} von <b>${cur.g}</b>: ${full(cur)}`, 'happy');
  later(next, 2100);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.keycap');
  if (!k || input.readOnly) return;
  const a = input.selectionStart ?? input.value.length, b = input.selectionEnd ?? a;
  input.value = input.value.slice(0, a) + k.textContent + input.value.slice(b);
  input.focus();
  input.setSelectionRange(a + 1, a + 1);
});
