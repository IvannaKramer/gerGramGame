@@tipp
Sind zwei Dinge <b>gleich</b>? Dann steht das Adjektiv in der Grundstufe und du sagst <b>so … wie</b>: so groß wie. Sind sie <b>verschieden</b>? Dann steht die Vergleichsstufe mit -er und du sagst <b>als</b>: größer als.

/* ---------- Spiel: Wie oder als? ---------- */
const CLS = { wie: 's1', als: 's2' };
const COLOR = { wie: 'der', als: 'das' };
let all = [], queue = [], cur = null;
const mark = t => t.replace(/\*(.+?)\*/g, '<u>$1</u>');
const plain = t => t.replace(/\*/g, '');
const full = s => plain(s.text).replace('___', `<b class="w ${CLS[s.k]}">${s.k}</b>`);
const tileEl = k => $(`.tile[data-k="${k}"]`);

function startGame() {
  all = SENTENCES.map(s => ({ text: s[0], k: s[1], a: s[2], b: s[3], sig: (s[0].match(/\*(.+?)\*/) || [])[1], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Gleich oder verschieden? Schau auf die Bilder und auf die unterstrichenen Wörter.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  const same = cur.k === 'wie';
  $('#pics').innerHTML = `<span aria-hidden="true">${cur.a}</span><span class="rel" role="img" aria-label="${same ? 'gleich' : 'verschieden'}">${same ? '=' : '≠'}</span><span aria-hidden="true">${cur.b}</span>`;
  $('#sentence').innerHTML = mark(cur.text).replace('___', '<span class="gap" id="gap" role="img" aria-label="Lücke">?</span>');
  $$('.tile').forEach(t => t.classList.remove('hint'));
}

function tipp(s) {
  return s.k === 'wie'
    ? `Tipp: Hier steht <b>${s.sig}</b>. Beide sind gleich. Bei „so“ und der Grundstufe heißt es <b>wie</b>.`
    : `Tipp: <b>${s.sig}</b> ist die Vergleichsstufe mit -er. Die beiden sind verschieden. Dann heißt es <b>als</b>.`;
}

function choose(k) {
  if (state.busy || !cur) return;
  if (k !== cur.k) {
    const hint = miss(cur, tileEl(k));
    if (hint) tileEl(cur.k).classList.add('hint');
    say(hint ? tipp(cur) : `Fast! Schau auf die unterstrichenen Wörter: <b>${cur.sig}</b>. Sind die beiden gleich oder verschieden?`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const gap = $('#gap');
  gap.textContent = k; gap.classList.add('filled', CLS[k]); gap.removeAttribute('role'); gap.removeAttribute('aria-label');
  solve(cur, gap, COLOR[k]);
  say(`${pick(PRAISE, 'praise')} ${k === 'wie' ? 'Die beiden sind gleich:' : 'Die beiden sind verschieden:'} ${full(cur)}`, 'happy');
  later(next, 1900);
}

$$('.tile').forEach(t => t.addEventListener('click', () => choose(t.dataset.k)));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[12]$/.test(e.key)) return;
  choose(e.key === '1' ? 'wie' : 'als');
});
