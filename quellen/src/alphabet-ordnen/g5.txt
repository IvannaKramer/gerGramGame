/* Aufbau: [erstes Leitwort, letztes Leitwort, [vier Wörter]]
   Das Spiel rechnet selbst aus, ob ein Wort vor der Seite, auf der Seite oder nach der Seite steht
   (ä, ö, ü zählen wie a, o, u). Achte darauf, dass alle drei Antworten ungefähr gleich oft vorkommen. */
const PAGES = [
  ['Garten', 'Gold',   ['Gabel', 'Geld', 'Glück', 'Gruppe']],
  ['Laden',  'Licht',  ['Labor', 'Leiter', 'Liste', 'Löffel']],
  ['Rakete', 'Rose',   ['Rabe', 'Räuber', 'Riese', 'Rücken']],
  ['Wagen',  'Wetter', ['Waffel', 'Wärme', 'Wind', 'Wurst']],
  ['Tanz',   'Tinte',  ['Tabelle', 'Tal', 'Tasche', 'Tür']]
];
/* Aufbau: Kürzel: Überschrift */
const KINDS = { vor: 'weiter vorne', auf: 'auf dieser Seite', nach: 'weiter hinten' };

const GAME = {
  id: 'leitwort',
  next: { href: '6-grundform-profi.html', title: 'Grundform-Profi' },
  review: () => `<div class="review">${Object.keys(KINDS).map(k =>
    `<div class="col ${k}"><h3>${KINDS[k]}</h3><ul>${all.filter(c => c.k === k).map(c => `<li><b>${c.w}</b> <span>(${c.a} – ${c.b})</span></li>`).join('')}</ul></div>`).join('')}</div>`
};

/* ---------- Spiel: Leitwort-Detektiv ---------- */
/* Sortier-Regel wie im Wörterbuch: ä, ö, ü zählen wie a, o, u; ß wie ss; groß und klein ist egal */
const key = s => s.toLowerCase().replace(/ä/g, 'a').replace(/ö/g, 'o').replace(/ü/g, 'u').replace(/ß/g, 'ss');
const cmp = (a, b) => key(a) < key(b) ? -1 : key(a) > key(b) ? 1 : 0;
const ORD = ['erste', 'zweite', 'dritte', 'vierte', 'fünfte', 'sechste'];
let all = [], queue = [], cur = null;

/* Begründung: a steht vor b – welcher Buchstabe entscheidet? */
function why(a, b) {
  const ka = key(a), kb = key(b);
  let i = 0;
  while (i < ka.length && ka[i] === kb[i]) i++;
  const um = (a.slice(0, i + 1) + b.slice(0, i + 1)).match(/[äöü]/i);
  return `Der ${ORD[i] || 'nächste'} Buchstabe entscheidet: <b>${ka[i]}</b> kommt vor <b>${kb[i]}</b>.` +
    (um ? ` Denk daran: <b>${um[0].toLowerCase()}</b> wird wie <b>${key(um[0])}</b> einsortiert.` : '');
}

function startGame() {
  all = [];
  PAGES.forEach((p, pi) => shuffle(p[2].slice()).forEach(w =>
    all.push({ w, a: p[0], b: p[1], pi, k: cmp(w, p[0]) < 0 ? 'vor' : cmp(w, p[1]) > 0 ? 'nach' : 'auf', misses: 0 })));
  queue = all.slice();
  resetState(all.length);
  say('Schau auf die Leitwörter oben. Steht das gesuchte Wort weiter vorne, auf dieser Seite oder weiter hinten?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#round').textContent = `Wörterbuch-Seite ${cur.pi + 1} von ${PAGES.length}`;
  $('#lw1').textContent = cur.a; $('#lw2').textContent = cur.b;
  const word = $('#word');
  word.textContent = cur.w; word.className = 'seek';
  $$('.wbtn').forEach(b => { b.disabled = false; b.classList.remove('hint'); });
}

function choose(el) {
  if (!el || state.busy || el.disabled || !cur) return;
  const { w, a, b, k } = cur;
  if (el.dataset.k !== k) {
    const hint = miss(cur, el);
    el.disabled = true;
    if (hint) $(`.wbtn[data-k="${k}"]`).classList.add('hint');
    const tip = k === 'vor' ? `<b>${w}</b> steht vor <b>${a}</b>. ${why(w, a)}`
      : k === 'nach' ? `<b>${w}</b> steht nach <b>${b}</b>. ${why(b, w)}`
      : `<b>${w}</b> steht nach <b>${a}</b> und vor <b>${b}</b>. Vergleiche mit <b>${a}</b>: ${why(a, w)}`;
    say(hint ? `Tipp: ${tip}`
      : `Noch nicht. Vergleiche <b>${w}</b> zuerst mit <b>${a}</b>, dann mit <b>${b}</b>. Schau Buchstabe für Buchstabe.`, 'think');
    return;
  }
  state.busy = true;
  $$('.wbtn').forEach(x => { x.classList.remove('hint'); if (x !== el) x.disabled = true; });
  $('#word').classList.add(k);
  solve(cur, $('#word'), k === 'vor' ? 'der' : k === 'auf' ? 'das' : 'bonus');
  const pill = `<span class="pill ${k}">${w}</span>`;
  say(`${pick(PRAISE, 'praise')} ` + (k === 'vor' ? `${pill} steht weiter vorne, vor <b>${a}</b>. ${why(w, a)}`
    : k === 'nach' ? `${pill} steht weiter hinten, nach <b>${b}</b>. ${why(b, w)}`
    : `${pill} steht auf dieser Seite: nach <b>${a}</b> und vor <b>${b}</b>.`), 'happy');
  later(next, 2800);
}

$('#where').addEventListener('click', e => choose(e.target.closest('.wbtn')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-3]$/.test(e.key)) return;
  choose($$('.wbtn')[+e.key - 1]);
});
