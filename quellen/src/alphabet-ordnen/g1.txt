/* Aufbau: drei Nachbarn aus dem ABC. Der Unterstrich _ steht für den Buchstaben, der gesucht wird.
   Beispiel: 'K_M' – gesucht ist L. Das Spiel rechnet den fehlenden Buchstaben selbst aus. */
const ROWS = [
  'BC_', 'FG_', 'KL_', 'QR_', 'VW_',
  '_FG', '_JK', '_OP', '_UV', '_XY',
  'B_D', 'F_H', 'K_M', 'P_R', 'T_V'
];

const GAME = {
  id: 'nachbarn',
  unit: 'Buchstaben',
  next: { href: '2-abc-raupe.html', title: 'ABC-Raupe' },
  review: () => {
    const li = c => `<li>${[...c.letters].map((ch, i) => i === c.pos ? `<b>${ch}</b>` : ch).join(' ')}</li>`;
    const col = (h, p) => `<div class="col ab"><h3>${h}</h3><ul>${all.filter(c => c.pos === p).map(li).join('')}</ul></div>`;
    return `<div class="review">${col('davor', 0)}${col('dazwischen', 1)}${col('danach', 2)}</div>`;
  }
};

/* ---------- Spiel: ABC-Nachbarn ---------- */
const ABC = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
const ASK = ['Welcher Buchstabe kommt davor?', 'Welcher Buchstabe steht dazwischen?', 'Welcher Buchstabe kommt danach?'];
const TRY = ['Fast! Sag das ABC leise auf.', 'Noch nicht. Schau in die ABC-Leiste oben.', 'Guter Versuch! Such die Nachbarn in der ABC-Leiste.'];
let all = [], queue = [], cur = null;

function startGame() {
  all = ROWS.map(r => {
    const pos = r.indexOf('_'), anchor = pos === 0 ? 1 : 0;
    const start = ABC.indexOf(r[anchor]) - anchor;
    const letters = ABC.slice(start, start + 3);
    return { letters, pos, target: letters[pos], misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  $('#abc').innerHTML = [...ABC].map(ch => `<span data-ch="${ch}">${ch}</span>`).join('');
  say('Welcher Buchstabe fehlt? Die ABC-Leiste oben hilft dir.');
  show('game');
  next();
}

function markABC(on) {
  $$('#abc span').forEach(s => {
    s.classList.toggle('on', on && cur.letters.includes(s.dataset.ch) && s.dataset.ch !== cur.target);
    s.classList.toggle('hit', on === 'all' && s.dataset.ch === cur.target);
  });
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  markABC(false);
  $('#ask').textContent = ASK[cur.pos];
  $('#trio').innerHTML = [...cur.letters].map((ch, i) => i === cur.pos
    ? '<span class="lt gap" id="gap" aria-label="gesuchter Buchstabe">?</span>' : `<span class="lt">${ch}</span>`).join('');
  const t = ABC.indexOf(cur.target);
  const near = [...ABC].filter((ch, i) => Math.abs(i - t) <= 4 && !cur.letters.includes(ch));
  $('#opts').innerHTML = shuffle([cur.target, ...shuffle(near).slice(0, 3)]).map((ch, i) =>
    `<button class="opt" data-ch="${ch}" aria-label="Buchstabe ${ch}">${ch}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function choose(el) {
  if (!el || state.busy || el.disabled) return;
  const L = cur.letters;
  if (el.dataset.ch !== cur.target) {
    const hint = miss(cur, el);
    el.disabled = true;
    if (hint) { $(`.opt[data-ch="${cur.target}"]`).classList.add('hint'); markABC(true); }
    const known = cur.pos === 1 ? L[0] : L[1];
    say(hint ? `Tipp: Such <b>${known}</b> in der ABC-Leiste. Der Buchstabe ${(cur.pos === 0 ? 'direkt davor' : 'direkt danach')} fehlt. Die richtige Karte leuchtet.`
      : pick(TRY, 'try'), 'think');
    return;
  }
  state.busy = true;
  el.classList.add('good');
  $$('.opt').forEach(o => { o.classList.remove('hint'); if (o !== el) o.disabled = true; });
  const gap = $('#gap');
  gap.textContent = cur.target; gap.classList.add('filled');
  markABC('all');
  solve(cur, gap, 'die');
  const why = cur.pos === 0 ? `<b>${L[0]}</b> kommt im ABC vor <b>${L[1]}</b>.`
    : cur.pos === 1 ? `<b>${L[1]}</b> steht im ABC zwischen <b>${L[0]}</b> und <b>${L[2]}</b>.`
    : `<b>${L[2]}</b> kommt im ABC nach <b>${L[1]}</b>.`;
  say(`${pick(PRAISE, 'praise')} <span class="pill ab">${[...L].join(' ')}</span> ${why}`, 'happy');
  later(next, 1900);
}

$('#opts').addEventListener('click', e => choose(e.target.closest('.opt')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-4]$/.test(e.key)) return;
  choose($$('.opt')[+e.key - 1]);
});
