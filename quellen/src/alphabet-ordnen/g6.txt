@@tipp
Im Wörterbuch stehen die Wörter in der <b>Grundform</b>. Nomen stehen in der <b>Einzahl</b>: Bäume → Baum. Verben stehen in der Grundform mit <b>-en</b>: sie hat gespielt → spielen. Adjektive stehen <b>ungesteigert</b>: schneller → schnell.

/* ---------- Spiel: Grundform-Profi ---------- */
let all = [], queue = [], cur = null;
const input = $('#answer');

function startGame() {
  all = WORDS.map(w => ({ ez: w[0], mz: w[1], k: w[2], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Unter welchem Wort schlägst du nach? Schreibe die Grundform in das Feld.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#from').textContent = cur.ez;
  input.value = ''; input.readOnly = false; input.className = '';
  input.focus({ preventScroll: true });
}

function check() {
  if (state.busy || !cur) return;
  const v = input.value.trim().replace(/\s+/g, ' ').replace(/^(der|die|das) /i, '');
  if (!v) { say('Schreibe zuerst die Grundform in das Feld.'); input.focus(); return; }
  if (v !== cur.mz) {
    if (v.toLowerCase() === cur.mz.toLowerCase()) {
      wobble(input);
      say(cur.k === 'nom' ? 'Fast richtig! Denk daran: Nomen schreibt man groß.' : 'Fast richtig! Denk daran: Verben und Adjektive schreibt man klein.', 'think');
    } else {
      miss(cur, input);
      say(cur.misses === 1 ? RULES[cur.k][1]
        : cur.misses === 2 ? RULES[cur.k][2]
        : `So ist es richtig: <span class="pill ${cur.k}">${cur.mz}</span> Schreibe das Wort genau so ab.`, 'think');
    }
    input.focus();
    return;
  }
  state.busy = true;
  input.value = cur.mz; input.readOnly = true; input.classList.add('good', cur.k);
  solve(cur, input, cur.k === 'nom' ? 'der' : cur.k === 'vrb' ? 'die' : 'das');
  say(`${pick(PRAISE, 'praise')} <b>${cur.ez}</b> → <span class="pill ${cur.k}">${cur.mz}</span> ${RULES[cur.k][3]}`, 'happy');
  later(next, 2100);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.keycap');
  if (!k || input.readOnly) return;
  const a = input.selectionStart ?? input.value.length, b = input.selectionEnd ?? a;
  input.value = input.value.slice(0, a) + k.textContent + input.value.slice(b);
  input.focus();
  input.setSelectionRange(a + 1, a + 1);
});
