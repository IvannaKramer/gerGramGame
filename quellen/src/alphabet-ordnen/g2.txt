/* ---------- Spiel: ABC-Raupe ---------- */
const ABC = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
/* Sortier-Regel wie im Wörterbuch: ä, ö, ü zählen wie a, o, u; ß wie ss; groß und klein ist egal */
const key = s => s.toLowerCase().replace(/ä/g, 'a').replace(/ö/g, 'o').replace(/ü/g, 'u').replace(/ß/g, 'ss');
const cmp = (a, b) => key(a) < key(b) ? -1 : key(a) > key(b) ? 1 : 0;
const wordHTML = (w, tag) => `<${tag}${tag === 'b' ? '' : ' class="f"'}>${w.charAt(0)}</${tag}>${w.slice(1)}`;
let all = [], queue = [], cur = null, order = [], sel = null;

function startGame() {
  all = ROWS.map(r => ({
    words: r.map(s => { const i = s.indexOf(' '); return { emoji: s.slice(0, i), w: s.slice(i + 1).trim() }; }).sort((a, b) => cmp(a.w, b.w)),
    misses: 0
  }));
  queue = shuffle(all.slice());
  resetState(all.length);
  $('#abc').innerHTML = [...ABC].map(ch => `<span data-ch="${ch}">${ch}</span>`).join('');
  say('Ordne die Wörter nach dem ABC. Tippe zwei Wörter an, dann tauschen sie die Plätze.');
  show('game');
  next();
}

function markABC(on) {
  const firsts = cur ? cur.words.map(x => x.w.charAt(0).toUpperCase()) : [];
  $$('#abc span').forEach(s => s.classList.toggle('on', on && firsts.includes(s.dataset.ch)));
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; sel = null;
  do { order = shuffle(cur.words.map((_, i) => i)); } while (order.every((wi, pos) => wi === pos));
  markABC(false);
  $('#row').classList.remove('solved');
  render();
}

function render(focusPos) {
  $('#row').innerHTML = order.map((wi, pos) => {
    const x = cur.words[wi];
    return `<button class="wcard${sel === pos ? ' selected' : ''}" data-pos="${pos}" aria-pressed="${sel === pos}" aria-label="Platz ${pos + 1}: ${x.w}">` +
      `<span class="nr" aria-hidden="true">${pos + 1}</span><span class="pic" aria-hidden="true">${x.emoji}</span><span class="w">${wordHTML(x.w, 'span')}</span></button>`;
  }).join('');
  if (focusPos != null) { const c = $(`.wcard[data-pos="${focusPos}"]`); if (c) c.focus({ preventScroll: true }); }
}

function swap(a, b, focus) {
  if (state.busy || a === b || a < 0 || b < 0 || a >= order.length || b >= order.length) return;
  [order[a], order[b]] = [order[b], order[a]];
  sel = null; sfx.pick();
  render(focus ? b : null);
}

function tap(pos) {
  if (state.busy) return;
  if (sel === null) {
    sel = pos; sfx.pick(); render(pos);
    say(`Tippe jetzt das Wort an, mit dem <b>${cur.words[order[pos]].w}</b> den Platz tauschen soll.`);
  } else if (sel === pos) { sel = null; render(pos); }
  else swap(sel, pos, true);
}

function check() {
  if (state.busy || !cur) return;
  const wrong = order.findIndex((wi, pos) => wi !== pos);
  if (wrong !== -1) {
    sel = null; render();
    const right = order.filter((wi, pos) => wi === pos).length;
    const hint = miss(cur, $(`.wcard[data-pos="${wrong}"]`));
    if (hint) {
      $(`.wcard[data-pos="${order.indexOf(wrong)}"]`).classList.add('hint');
      markABC(true);
      say(`Tipp: Such die Anfangsbuchstaben in der ABC-Leiste. So ist die Reihenfolge: ${cur.words.map(x => `<b>${x.w.charAt(0)}</b>`).join(', ')}. Das leuchtende Wort gehört auf Platz ${wrong + 1}.`, 'think');
    } else {
      say(right === 0 ? 'Noch nicht ganz. Welcher Anfangsbuchstabe kommt im ABC zuerst?'
        : right === 1 ? 'Fast! Ein Wort steht schon richtig. Schau auf die blauen Anfangsbuchstaben.'
        : `Fast! ${right} Wörter stehen schon richtig. Schau auf die blauen Anfangsbuchstaben.`, 'think');
    }
    return;
  }
  state.busy = true; sel = null; render();
  $('#row').classList.add('solved');
  markABC(true);
  solve(cur, $('#row'), 'der');
  say(`${pick(PRAISE, 'praise')} ${cur.words.map(x => `<span class="pill ab">${x.w}</span>`).join(' ')} So stehen die Wörter im Wörterbuch.`, 'happy');
  later(next, 2600);
}

$('#row').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.wcard');
  if (c) tap(+c.dataset.pos);
});
$('#row').addEventListener('keydown', e => {
  const c = e.target.closest('.wcard');
  const step = { ArrowLeft: -1, ArrowUp: -1, ArrowRight: 1, ArrowDown: 1 }[e.key];
  if (!c || !step) return;
  e.preventDefault();
  const pos = +c.dataset.pos;
  swap(pos, pos + step, true);
});
$('#check').addEventListener('click', check);
enableDrag('.wcard', '.wcard', (src, tgt) => swap(+src.dataset.pos, +tgt.dataset.pos, false));
