   Die Reihenfolge der Liste ordnet das Spiel selbst. Wichtig: kein Wort doppelt. */

/* ---------- Spiel: Wo ist mein Platz? ---------- */
/* Sortier-Regel wie im Wörterbuch: ä, ö, ü zählen wie a, o, u; ß wie ss; groß und klein ist egal */
const key = s => s.toLowerCase().replace(/ä/g, 'a').replace(/ö/g, 'o').replace(/ü/g, 'u').replace(/ß/g, 'ss');
const cmp = (a, b) => key(a) < key(b) ? -1 : key(a) > key(b) ? 1 : 0;
const ORD = ['erste', 'zweite', 'dritte', 'vierte', 'fünfte', 'sechste'];
let all = [], queue = [], cur = null;

/* Begründung: a steht vor b – welcher Buchstabe entscheidet? */
function why(a, b) {
  const ka = key(a), kb = key(b);
  let i = 0;
  while (i < ka.length && ka[i] === kb[i]) i++;
  const um = (a.slice(0, i + 1) + b.slice(0, i + 1)).match(/[äöü]/i);
  return `Der ${ORD[i] || 'nächste'} Buchstabe entscheidet: <b>${ka[i]}</b> kommt vor <b>${kb[i]}</b>.` +
    (um ? ` Denk daran: <b>${um[0].toLowerCase()}</b> wird wie <b>${key(um[0])}</b> einsortiert.` : '');
}

function startGame() {
  all = [];
  ROUNDS.forEach((r, ri) => {
    const list = r[0].slice().sort(cmp);
    shuffle(r[1].slice()).forEach(w => all.push({ w, list, ri, gap: list.filter(x => cmp(x, w) < 0).length, misses: 0 }));
  });
  queue = all.slice();
  resetState(all.length);
  say('In welche Lücke gehört das neue Wort? Tippe auf die Lücke oder zieh das Wort dorthin.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#round').textContent = `Runde ${cur.ri + 1} von ${ROUNDS.length}`;
  $('#new').innerHTML = `<button class="word-card" aria-label="Neues Wort: ${cur.w}">${cur.w}</button>`;
  render(false);
}

function render(solved) {
  const L = cur.list, parts = [];
  for (let g = 0; g <= L.length; g++) {
    if (solved && g === cur.gap) parts.push(`<div class="entry new">${cur.w}</div>`);
    else {
      const where = g === 0 ? `vor ${L[0]}` : g === L.length ? `nach ${L[g - 1]}` : `zwischen ${L[g - 1]} und ${L[g]}`;
      parts.push(`<button class="gap" data-g="${g}" aria-label="Lücke ${where}"${solved ? ' disabled' : ''}><span class="kbd" aria-hidden="true">${g + 1}</span>hier?</button>`);
    }
    if (g < L.length) parts.push(`<div class="entry">${L[g]}</div>`);
  }
  $('#list').classList.toggle('solved', solved);
  $('#list').innerHTML = parts.join('');
}

function place(g) {
  const el = $(`.gap[data-g="${g}"]`);
  if (!el || state.busy || el.disabled) return;
  const prev = cur.list[cur.gap - 1], nxt = cur.list[cur.gap], w = cur.w;
  const reason = prev ? `<b>${w}</b> kommt nach <b>${prev}</b>. ${why(prev, w)}` : `<b>${w}</b> kommt vor <b>${nxt}</b>. ${why(w, nxt)}`;
  if (g !== cur.gap) {
    const hint = miss(cur, el);
    el.disabled = true;
    if (hint) $(`.gap[data-g="${cur.gap}"]`).classList.add('hint');
    say(hint ? `Tipp: ${reason} Die richtige Lücke leuchtet.`
      : `Noch nicht. <b>${w}</b> steht weiter ${g < cur.gap ? 'hinten' : 'vorne'}. Vergleiche Buchstabe für Buchstabe.`, 'think');
    return;
  }
  state.busy = true;
  render(true);
  $('#new').innerHTML = '<span class="placed">einsortiert</span>';
  solve(cur, $('.entry.new'), 'die');
  const where = prev && nxt ? `zwischen <b>${prev}</b> und <b>${nxt}</b>` : prev ? `ganz hinten, nach <b>${prev}</b>` : `ganz vorne, vor <b>${nxt}</b>`;
  say(`${pick(PRAISE, 'praise')} <span class="pill neu">${w}</span> steht ${where}. ${prev ? why(prev, w) : why(w, nxt)}`, 'happy');
  later(next, 2800);
}

$('#list').addEventListener('click', e => { const b = e.target.closest('.gap'); if (b) place(+b.dataset.g); });
$('#new').addEventListener('click', e => {
  if (suppressClick || state.busy || !e.target.closest('.word-card')) return;
  say('Tippe auf die Lücke, in die das Wort gehört. Du kannst das Wort auch dorthin ziehen.');
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-5]$/.test(e.key)) return;
  place(+e.key - 1);
});
enableDrag('#new .word-card', '.gap', (src, tgt) => place(+tgt.dataset.g));
