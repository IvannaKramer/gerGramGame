/* ---------- Spiel: Was steht zuerst? ---------- */
const ABC = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ';
/* Sortier-Regel wie im Wörterbuch: ä, ö, ü zählen wie a, o, u; ß wie ss; groß und klein ist egal */
const key = s => s.toLowerCase().replace(/ä/g, 'a').replace(/ö/g, 'o').replace(/ü/g, 'u').replace(/ß/g, 'ss');
const cmp = (a, b) => key(a) < key(b) ? -1 : key(a) > key(b) ? 1 : 0;
const TRY = ['Fast! Schau auf den zweiten Buchstaben. Welcher kommt im ABC früher?', 'Noch nicht. Vergleiche die pinken Buchstaben mit der ABC-Leiste.',
  'Guter Versuch! Sag das ABC leise auf: Welchen pinken Buchstaben hörst du zuerst?'];
let all = [], queue = [], cur = null;

function startGame() {
  all = PAIRS.map(p => { const [a, b] = p.slice().sort(cmp); return { a, b, misses: 0 }; });
  queue = shuffle(all.slice());
  resetState(all.length);
  $('#abc').innerHTML = [...ABC].map(ch => `<span data-ch="${ch}">${ch}</span>`).join('');
  say('Welches Wort steht im Wörterbuch zuerst? Schau auf den zweiten Buchstaben.');
  show('game');
  next();
}

function markABC(on) {
  const two = cur ? [cur.a[1].toUpperCase(), cur.b[1].toUpperCase()] : [];
  $$('#abc span').forEach(s => s.classList.toggle('on', on && two.includes(s.dataset.ch)));
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  markABC(false);
  $('#duo').classList.remove('solved');
  $('#duo').innerHTML = shuffle([cur.a, cur.b]).map((w, i) =>
    `<button class="pickw" data-w="${w}" aria-label="${w}"><span class="kbd" aria-hidden="true">${i + 1}</span>` +
    `<span class="w">${w[0]}<mark>${w[1]}</mark>${w.slice(2)}</span><span class="place">${w === cur.a ? 'steht zuerst' : 'steht danach'}</span></button>`).join('');
}

function choose(el) {
  if (!el || state.busy) return;
  const l1 = cur.a[1], l2 = cur.b[1];
  if (el.dataset.w !== cur.a) {
    const hint = miss(cur, el);
    if (hint) { $(`.pickw[data-w="${cur.a}"]`).classList.add('hint'); markABC(true); }
    say(hint ? `Tipp: Beide Wörter beginnen mit <b>${cur.a[0]}</b>. Such die zweiten Buchstaben in der ABC-Leiste: <b>${l1}</b> kommt vor <b>${l2}</b>. Das leuchtende Wort steht zuerst.`
      : pick(TRY, 'try'), 'think');
    return;
  }
  state.busy = true;
  el.classList.remove('hint'); el.classList.add('good');
  $('#duo').classList.add('solved');
  markABC(true);
  solve(cur, el, 'der');
  say(`${pick(PRAISE, 'praise')} <span class="pill ab">${cur.a}</span> steht vor <b>${cur.b}</b>. Der zweite Buchstabe entscheidet: <b>${l1}</b> kommt im ABC vor <b>${l2}</b>.`, 'happy');
  later(next, 2300);
}

$('#duo').addEventListener('click', e => choose(e.target.closest('.pickw')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[12]$/.test(e.key)) return;
  choose($$('.pickw')[+e.key - 1]);
});
