/* ---------- Spiel: Mehrzahl-Memory ---------- */
const KIND = { ez: 'Einzahl', mz: 'Mehrzahl' };
const BACK = { ez: '●', mz: '●●●' };
let rounds = [], round = 0, cards = [], open = [], tries = 0, left = 0;
const closedLabel = c => `${KIND[c.kind]}-Karte, verdeckt`;

function startGame() {
  const all = shuffle(WORDS.map(w => ({ ez: w[0], mz: w[1], emoji: w[2], misses: 0 })));
  rounds = [];
  for (let i = 0; i < all.length; i += PAIRS_PER_ROUND) rounds.push(all.slice(i, i + PAIRS_PER_ROUND));
  resetState(all.length);
  round = 0; tries = 0;
  startRound();
  say('Dreh zwei Karten um: eine blaue und eine pinke. Passen Einzahl und Mehrzahl zusammen?');
  show('game');
}

function startRound() {
  open = []; left = rounds[round].length;
  cards = shuffle(rounds[round].flatMap(w => [{ w, kind: 'ez' }, { w, kind: 'mz' }]));
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#memo').innerHTML = cards.map((c, i) => {
    const text = c.kind === 'ez' ? c.w.ez : c.w.mz;
    const pic = c.kind === 'ez' ? c.w.emoji : c.w.emoji + c.w.emoji;
    return `<button class="mcard ${c.kind}" data-i="${i}" aria-label="${closedLabel(c)}">
      <span class="inner">
        <span class="back" aria-hidden="true"><b>${BACK[c.kind]}</b><small>${KIND[c.kind]}</small></span>
        <span class="face" aria-hidden="true"><span class="pic">${pic}</span><span class="word">${text}</span></span>
      </span></button>`;
  }).join('');
}

$('#memo').addEventListener('click', e => {
  const el = e.target.closest('.mcard');
  if (!el || state.busy) return;
  const c = cards[+el.dataset.i];
  if (c.open || c.found) return;
  sfx.pick();
  c.open = true;
  el.classList.add('open');
  el.setAttribute('aria-label', `${KIND[c.kind]}: ${c.kind === 'ez' ? c.w.ez : c.w.mz}`);
  open.push({ c, el });
  if (open.length < 2) return;

  tries++;
  const [a, b] = open; open = [];
  state.busy = true;
  if (a.c.w === b.c.w) {
    a.c.found = b.c.found = true;
    later(() => {
      a.el.classList.add('found'); b.el.classList.add('found');
      solve(a.c.w, b.el, 'die');
      say(`${pick(PRAISE, 'praise')} <span class="pill ez">${a.c.w.ez}</span> und <span class="pill mz">${a.c.w.mz}</span> gehören zusammen.`, 'happy');
      state.busy = false;
      if (--left === 0) roundDone();
    }, 450);
  } else {
    const same = a.c.kind === b.c.kind;
    say(same
      ? `Das sind zwei ${KIND[a.c.kind]}-Karten. Dreh eine blaue und eine pinke Karte um!`
      : 'Das passt noch nicht. Merk dir, wo die Karten liegen!', 'think');
    later(() => {
      [a, b].forEach(o => { o.c.open = false; o.el.classList.remove('open'); o.el.setAttribute('aria-label', closedLabel(o.c)); });
      state.busy = false;
    }, 1700);
  }
});

function roundDone() {
  state.busy = true;
  if (round < rounds.length - 1) {
    later(() => { say('Runde geschafft! Hier kommen neue Karten.', 'happy'); round++; startRound(); state.busy = false; }, 1600);
  } else later(finish, 1300);
}
