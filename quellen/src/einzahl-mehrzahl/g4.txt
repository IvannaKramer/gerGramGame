@@tipp
Für die Mehrzahl gibt es verschiedene Endungen: <b>-e</b>, <b>-er</b>, <b>-n</b> oder <b>-en</b> und <b>-s</b>. Manche Wörter bleiben auch gleich. Dann ändert sich nur der Artikel: Er heißt in der Mehrzahl immer <b>die</b>.

/* ---------- Spiel: Endungs-Werkstatt ---------- */
let all = [], queue = [], cur = null;
const ending = k => ENDINGS.find(e => e[0] === k);
const tileEl = k => $(`.tile[data-k="${k}"]`);

$('#tiles').innerHTML = ENDINGS.map(([k, label, sub, cls], i) =>
  `<button class="tile ${cls}" data-k="${k}" aria-label="Endung ${label}${sub ? ', ' + sub : ''}"><b>${label}</b>${sub ? `<small>${sub}</small>` : ''}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');

function startGame() {
  all = WORDS.map(w => ({ ez: w[0], mz: w[1], k: w[2], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welche Endung braucht das Wort in der Mehrzahl? Tippe sie an oder zieh sie an das Wort.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#from').textContent = cur.ez;
  $('#build').innerHTML = `<span class="art">die</span><span class="stem" id="stem">${cur.ez.split(' ')[1]}</span><span class="slot" id="slot">?</span>`;
}

function choose(k) {
  if (state.busy || !cur) return;
  const [, label, , cls, tip] = ending(cur.k);
  if (k !== cur.k) {
    const hint = miss(cur, tileEl(k));
    if (hint) tileEl(cur.k).classList.add('hint');
    say(hint ? tip : pick(RETRY, 'retry'), 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const stem = cur.ez.split(' ')[1], suffix = cur.mz.split(' ')[1].slice(stem.length);
  const slot = $('#slot');
  if (suffix) { slot.textContent = suffix; slot.className = 'slot filled ' + cls; }
  else { slot.remove(); $('#stem').className = 'stem same ' + cls; }
  solve(cur, $('#build'), 'sun');
  say(`${pick(PRAISE, 'praise')} <b>${cur.ez}</b> → <span class="pill ${cls}">${cur.mz}</span>` +
    (suffix ? '' : ' Das Wort bleibt gleich, nur der Artikel ändert sich.'), 'happy');
  later(next, 1700);
}

$('#tiles').addEventListener('click', e => {
  if (suppressClick) return;
  const t = e.target.closest('.tile');
  if (t) choose(t.dataset.k);
});
enableDrag('.tile', '.bench', tile => choose(tile.dataset.k));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-5]$/.test(e.key)) return;
  choose(ENDINGS[+e.key - 1][0]);
});
