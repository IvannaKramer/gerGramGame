@@tipp
Viele Nomen bekommen in der Mehrzahl einen <b>Umlaut</b>: aus <b>a</b> wird <b>ä</b>, aus <b>o</b> wird <b>ö</b>, aus <b>u</b> wird <b>ü</b> und aus <b>au</b> wird <b>äu</b>. Sprich das Wort laut, dann hörst du es.

/* ---------- Spiel: Umlaut-Zauber ---------- */
const UML = { a: 'ä', o: 'ö', u: 'ü', A: 'Ä', O: 'Ö', U: 'Ü' };
const PLAIN = { 'ä': 'a', 'ö': 'o', 'ü': 'u', 'Ä': 'A', 'Ö': 'O', 'Ü': 'U' };
const plain = s => s.replace(/[äöüÄÖÜ]/g, ch => PLAIN[ch]);
let all = [], queue = [], cur = null, letters = [];

function startGame() {
  all = WORDS.map(w => { const noun = w[1].split(' ')[1]; return { ez: w[0], mz: w[1], noun, need: noun !== plain(noun), misses: 0 }; });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Sprich die Mehrzahl laut. Braucht ein Buchstabe Pünktchen? Dann tippe ihn an.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  letters = [...plain(cur.noun)].map((ch, i) => ({ base: ch, on: false, can: ch in UML, target: cur.noun[i] !== ch }));
  $('#from').textContent = cur.ez;
  $('#spell').classList.remove('locked');
  renderLetters();
}

function renderLetters(hint) {
  $('#spell').innerHTML = '<span>die</span><span class="letters">' + letters.map((l, i) => {
    const ch = l.on ? UML[l.base] : l.base;
    return l.can
      ? `<button class="lt${l.on ? ' on' : ''}${hint && l.target ? ' hint' : ''}" data-i="${i}" aria-pressed="${l.on}" aria-label="Buchstabe ${ch}">${ch}</button>`
      : `<span class="lt">${ch}</span>`;
  }).join('') + '</span>';
}

/* Beschreibt den Wechsel, z. B. „au → äu“ */
function change() {
  const i = letters.findIndex(l => l.target), l = letters[i];
  const au = l.base.toLowerCase() === 'a' && letters[i + 1] && letters[i + 1].base === 'u';
  return au ? `Aus <b>${l.base}u</b> wird <b>${UML[l.base]}u</b>.` : `Aus <b>${l.base}</b> wird <b>${UML[l.base]}</b>.`;
}

function check() {
  if (state.busy || !cur) return;
  const built = letters.map(l => l.on ? UML[l.base] : l.base).join('');
  const has = letters.some(l => l.on);
  if (built !== cur.noun) {
    const hint = miss(cur, $('#spell .letters'));
    let msg;
    if (hint) msg = cur.need ? `Tipp: ${change()} Tippe auf den leuchtenden Buchstaben.` : 'Tipp: Dieses Wort braucht keinen Umlaut. Nimm alle Pünktchen wieder weg.';
    else if (cur.need && !has) msg = 'Hmm, hier fehlen noch Pünktchen. Sprich das Wort laut: Welcher Laut verändert sich?';
    else if (!cur.need) msg = 'Fast! Sprich das Wort noch einmal laut. Braucht es wirklich Pünktchen?';
    else msg = 'Fast! Ein anderer Buchstabe bekommt die Pünktchen.';
    if (hint) renderLetters(true);
    say(msg, 'think');
    return;
  }
  state.busy = true;
  renderLetters();
  $('#spell').classList.add('locked');
  solve(cur, $('#spell'), 'die');
  say(`${pick(PRAISE, 'praise')} <b>${cur.ez}</b> → <span class="pill mz">${cur.mz}</span> ` +
    (cur.need ? change() : 'Hier bleibt alles ohne Umlaut.'), 'happy');
  later(next, 1900);
}

$('#spell').addEventListener('click', e => {
  const b = e.target.closest('button.lt');
  if (!b || state.busy) return;
  const l = letters[+b.dataset.i];
  l.on = !l.on;
  sfx.pick();
  renderLetters();
  const again = $(`button.lt[data-i="${b.dataset.i}"]`);
  if (again) again.focus({ preventScroll: true });
});
$('#check').addEventListener('click', check);
