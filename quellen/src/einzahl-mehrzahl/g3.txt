/* ---------- Spiel: Ballon-Mehrzahl ---------- */
let all = [], queue = [], cur = null;

function startGame() {
  all = WORDS.map(w => ({ ez: w[0], mz: w[1], emoji: w[2], wrong: w[3], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welcher Ballon zeigt die richtige Mehrzahl? Tippe ihn an!');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#ask').innerHTML = `<span class="pic" aria-hidden="true">${cur.emoji}</span><span class="lbl">Einzahl</span>
    <span class="big">${cur.ez}</span><span class="q" id="ask-q">Wie heißt die Mehrzahl?</span>`;
  $('#balloons').innerHTML = shuffle([cur.mz, ...cur.wrong]).map((o, i) => {
    const [art, noun] = o.split(' ');
    return `<button class="balloon b${i + 1}" data-o="${o}" aria-label="${o}">
      <span class="bl"><span class="art">${art}</span><span>${noun}</span><span class="kbd" aria-hidden="true">${i + 1}</span></span>
      <span class="string"></span></button>`;
  }).join('');
}

function choose(el) {
  if (!el || state.busy || el.disabled) return;
  if (el.dataset.o !== cur.mz) {
    const hint = miss(cur, el);
    el.disabled = true; el.classList.add('limp');
    if (hint) $(`.balloon[data-o="${cur.mz}"]`).classList.add('hint');
    say(hint ? `Tipp: Sprich laut „${cur.ez.startsWith('die ') ? 'eine' : 'ein'} ${cur.ez.split(' ')[1]}, viele …“. Der leuchtende Ballon ist richtig.` : pick(RETRY, 'retry'), 'think');
    return;
  }
  state.busy = true;
  solve(cur, $('.bl', el), 'die');
  el.classList.add('pop');
  $$('.balloon').forEach(b => { if (b !== el) { b.classList.remove('limp'); b.classList.add('away'); } });
  $('#ask-q').innerHTML = `<span class="pill mz">${cur.mz}</span>`;
  say(`${pick(PRAISE, 'praise')} Ein Wort: <b>${cur.ez}</b>. Viele: <span class="pill mz">${cur.mz}</span>`, 'happy');
  later(next, 1700);
}

$('#balloons').addEventListener('click', e => choose(e.target.closest('.balloon')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-3]$/.test(e.key)) return;
  choose($$('.balloon')[+e.key - 1]);
});
