@@tipp
Manche Wörter haben eine besondere Mehrzahl. Wörter mit <b>-y</b> bekommen nur ein <b>-s</b>. Aus <b>-in</b> wird <b>-innen</b>. Aus <b>-nis</b>, <b>-is</b> und <b>-us</b> wird <b>-sse</b>. Und aus <b>-um</b> wird <b>-en</b>.

/* ---------- Spiel: Mehrzahl-Meister ---------- */
let all = [], queue = [], cur = null;
const input = $('#answer');

function startGame() {
  all = WORDS.map(w => ({ ez: w[0], mz: w[1], k: w[2], noun: w[1].slice(4), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Schreibe die Mehrzahl in das Feld. Überlege zuerst: Welche Regel passt zu dem Wort?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#from').textContent = cur.ez;
  input.value = ''; input.readOnly = false; input.classList.remove('good');
  input.focus({ preventScroll: true });
}

function check() {
  if (state.busy || !cur) return;
  let v = input.value.trim().replace(/\s+/g, ' ');
  if (/^die /i.test(v)) v = v.slice(4);
  if (!v) { say('Schreibe zuerst die Mehrzahl in das Feld.'); input.focus(); return; }
  if (v !== cur.noun) {
    if (v.toLowerCase() === cur.noun.toLowerCase()) {
      wobble(input);
      say('Fast richtig! Denk daran: Nomen schreibt man groß.', 'think');
    } else {
      miss(cur, input);
      say(cur.misses === 1 ? 'Fast! Schau dir das Wortende genau an und probier es noch einmal.'
        : cur.misses === 2 ? RULES[cur.k][2]
        : `So ist es richtig: <span class="pill mz">${cur.mz}</span> Schreibe das Wort genau so ab.`, 'think');
    }
    input.focus();
    return;
  }
  state.busy = true;
  input.value = cur.noun; input.readOnly = true; input.classList.add('good');
  solve(cur, input, 'die');
  say(`${pick(PRAISE, 'praise')} <b>${cur.ez}</b> → <span class="pill mz">${cur.mz}</span>`, 'happy');
  later(next, 1700);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.keycap');
  if (!k || input.readOnly) return;
  const a = input.selectionStart ?? input.value.length, b = input.selectionEnd ?? a;
  input.value = input.value.slice(0, a) + k.textContent + input.value.slice(b);
  input.focus();
  input.setSelectionRange(a + 1, a + 1);
});
