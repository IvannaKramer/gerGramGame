/* ---------- Spiel: Schreib-Profi ---------- */
const GAPS = /\[(\w+)(?:\|(\w+))?\]/g;
let all = [], queue = [], cur = null;
const kind = ans => ans.toLowerCase();
const full = (s, cls) => { let i = 0; return s.text.replace(GAPS, () => { const g = s.gaps[i++]; return `<${cls === 'w' ? 'b' : 'span'} class="${cls} ${g.k}">${g.ans}</${cls === 'w' ? 'b' : 'span'}>`; }); };
const inputs = () => $$('#sentence input');

function startGame() {
  all = SENTENCES.map(text => ({ text, misses: 0,
    gaps: [...text.matchAll(GAPS)].map(m => ({ ans: m[1], probe: m[2] || '', k: kind(m[1]), start: m.index === 0 })) }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Schreibe das oder dass in jede Lücke. Mach bei jeder Lücke die Probe!');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  let i = 0;
  $('#sentence').innerHTML = cur.text.replace(GAPS, () => { i++;
    return `<input type="text" autocapitalize="off" autocorrect="off" autocomplete="off" spellcheck="false" maxlength="6" aria-label="Lücke ${i}${cur.gaps.length > 1 ? ' von ' + cur.gaps.length : ''}">`; });
  inputs()[0].focus({ preventScroll: true });
}

function tip(i) {
  const g = cur.gaps[i], where = cur.gaps.length > 1 ? ` für Lücke ${i + 1}` : '';
  return g.k === 'das'
    ? `Tipp${where}: Hier passt die Probe mit <b>${g.probe.toLowerCase()}</b>. Also schreibst du <b>${g.ans}</b> mit einem s.`
    : `Tipp${where}: Hier passt weder „dieses“ noch „welches“. Also schreibst du <b>${g.ans}</b> mit zwei s.`;
}

function check() {
  if (state.busy || !cur) return;
  const ins = inputs();
  const empty = ins.find(inp => !inp.value.trim());
  if (empty) { say(ins.length > 1 ? 'Fülle zuerst alle Lücken aus.' : 'Schreibe zuerst das Wort in die Lücke.'); empty.focus(); return; }
  const wrong = [], nearly = [];
  ins.forEach((inp, i) => {
    const v = inp.value.trim(), g = cur.gaps[i];
    inp.classList.remove('hint');
    if (v === g.ans) { inp.value = g.ans; inp.readOnly = true; inp.classList.add('good', g.k); }
    else if (v.toLowerCase() === g.ans.toLowerCase() || v.toLowerCase() === g.ans.toLowerCase().replace('ss', 'ß')) nearly.push(i);
    else wrong.push(i);
  });
  if (wrong.length) {
    const first = ins[wrong[0]];
    miss(cur, first);
    wrong.slice(1).forEach(i => wobble(ins[i]));
    if (cur.misses === 2) first.classList.add('hint');
    say(cur.misses === 1 ? 'Fast! Mach bei jeder Lücke die Probe: Passt „dieses“ oder „welches“?'
      : cur.misses === 2 ? tip(wrong[0])
      : `So ist es richtig: ${full(cur, 'pill')} Schreibe die Wörter genau so ab.`, 'think');
    first.focus();
    return;
  }
  if (nearly.length) {
    const inp = ins[nearly[0]], g = cur.gaps[nearly[0]];
    wobble(inp);
    say(/ß/.test(inp.value) ? 'Fast richtig! Heute schreibt man <b>dass</b> mit ss.'
      : g.start ? 'Fast richtig! Am Satzanfang schreibt man groß.'
      : 'Fast richtig! Mitten im Satz schreibt man das Wort klein.', 'think');
    inp.focus();
    return;
  }
  state.busy = true;
  solve(cur, ins[ins.length - 1], 'sun');
  say(`${pick(PRAISE, 'praise')} ${full(cur, 'pill')}`, 'happy');
  later(next, 2600);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
