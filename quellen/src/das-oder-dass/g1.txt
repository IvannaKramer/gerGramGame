/* ---------- Spiel: Proben-Lupe ---------- */
const COLOR = { das: 'das', dass: 'bonus' };
let all = [], queue = [], cur = null;
const tileEl = k => $(`.tile[data-k="${k}"]`);
const full = s => s.text.replace('___', `<b class="w ${s.k}">${s.ans}</b>`);
const probeText = s => s.text.replace('___', `<b>${s.probe}</b>`);

function startGame() {
  all = SENTENCES.map(s => ({ text: s[0], ans: s[1], probe: s[2], emoji: s[3], k: s[1].toLowerCase(), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies den Satz mit der Lücke. Lies dann den Probe-Satz leise vor. Klingt er richtig?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#pic').textContent = cur.emoji;
  $('#sentence').innerHTML = cur.text.replace('___', '<span class="gap" id="gap">?</span>');
  $('#probe').innerHTML = cur.text.replace('___', `<mark>${cur.probe}</mark>`);
}

function choose(k) {
  if (state.busy || !cur) return;
  if (k !== cur.k) {
    const hint = miss(cur, tileEl(k));
    if (hint) tileEl(cur.k).classList.add('hint');
    say(!hint ? pick(['Fast! Lies den Probe-Satz noch einmal leise vor.', 'Hmm, das passt noch nicht. Hör genau hin: Klingt der Probe-Satz richtig?'], 'retry')
      : cur.k === 'das'
        ? `Tipp: Der Probe-Satz klingt richtig: „${probeText(cur)}“ Dann schreibst du <b>das</b> mit einem s.`
        : `Tipp: Der Probe-Satz klingt falsch: „${probeText(cur)}“ So spricht niemand! Dann schreibst du <b>dass</b> mit zwei s.`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const gap = $('#gap');
  gap.textContent = cur.ans; gap.className = 'gap filled ' + k;
  solve(cur, gap, COLOR[k]);
  say(`${pick(PRAISE, 'praise')} ${k === 'das' ? `„${cur.probe}“ passt.` : `„${cur.probe}“ passt nicht.`} Also <span class="pill ${k}">${k}</span>: ${full(cur)}`, 'happy');
  later(next, 2400);
}

$('#tiles').addEventListener('click', e => {
  const t = e.target.closest('.tile');
  if (t) choose(t.dataset.k);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[12]$/.test(e.key)) return;
  choose(e.key === '1' ? 'das' : 'dass');
});
