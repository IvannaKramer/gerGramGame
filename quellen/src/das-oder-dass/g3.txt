/* Aufbau: der richtige Satz – mit Komma, wenn eins nötig ist. Das Spiel nimmt das Komma heraus.
   In jedem Satz steht genau einmal das oder dass und höchstens ein Komma. */
const SENTENCES = [
  'Das Kind lacht laut.',
  'Wir besuchen das Museum.',
  'Der Hund frisst das Futter.',
  'Das Wetter ist heute schön.',
  'Emma holt das Brot vom Bäcker.',
  'Das macht mir Spaß.',
  'Im Garten steht das Zelt.',
  'Ich glaube, dass du gewinnst.',
  'Lea hofft, dass die Sonne scheint.',
  'Der Lehrer sagt, dass wir leise sein sollen.',
  'Wir wissen, dass Hunde gern spielen.',
  'Es ist toll, dass du mitkommst.',
  'Mir gefällt, dass ihr teilt.',
  'Jonas merkt, dass sein Schuh offen ist.',
  'Ich bin froh, dass heute Freitag ist.'
];

const GAME = {
  id: 'komma',
  unit: 'Sätze',
  next: { href: '4-wort-detektiv.html', title: 'Wort-Detektiv' },
  review: () => `<div class="review c2">
    <div class="col dass"><h3>dass – mit Komma</h3><ul>${all.filter(s => s.k === 'dass').map(s => `<li>${full(s)}</li>`).join('')}</ul></div>
    <div class="col das"><h3>das – ohne Komma</h3><ul>${all.filter(s => s.k === 'das').map(s => `<li>${full(s)}</li>`).join('')}</ul></div></div>`
};

/* ---------- Spiel: Komma-Wächter ---------- */
const COLOR = { das: 'das', dass: 'bonus' };
let all = [], queue = [], cur = null;
const isSignal = w => /^dass?$/i.test(w.replace(/[.,!?]/g, ''));
const mark = (w, k) => isSignal(w) ? `<b class="w ${k}">${w}</b>` : w;
const full = s => s.words.map((w, i) => mark(w, s.k) + (i === s.at ? '<b>,</b>' : '')).join(' ');

function startGame() {
  all = SENTENCES.map(t => {
    const raw = t.split(' ');
    const at = raw.findIndex(w => w.endsWith(','));
    const words = raw.map(w => w.replace(/,$/, ''));
    return { words, at, k: words.some(w => /^dass$/i.test(w)) ? 'dass' : 'das', misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies den Satz. Fehlt ein Komma? Tippe auf die richtige Stelle. Oder tippe auf „Kein Komma nötig“.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#none').classList.remove('hint');
  const box = $('#sentence');
  box.classList.remove('solved');
  box.innerHTML = cur.words.map((w, i) => `<span class="wd">${mark(w, cur.k)}` + (i < cur.words.length - 1
    ? `<button class="cgap" data-i="${i}" aria-label="Komma nach „${w}“ setzen"><i></i></button>` : '') + '</span>').join('');
}

function hint() {
  if (cur.at >= 0) {
    $(`.cgap[data-i="${cur.at}"]`).classList.add('hint');
    return 'Tipp: Vor <b class="w dass">dass</b> steht immer ein Komma. Such das lila Wort. Das Komma gehört direkt davor.';
  }
  $('#none').classList.add('hint');
  return 'Tipp: In diesem Satz steht <b class="w das">das</b> mit einem s. Du kannst <b>dieses</b> dafür einsetzen. Hier fehlt kein Komma.';
}

function choose(i, el) {
  if (state.busy || !cur) return;
  if (i !== cur.at) {
    const tip = miss(cur, el);
    say(tip ? hint()
      : cur.at < 0 ? 'Hmm. In diesem Satz steht <b class="w das">das</b> mit einem s. Braucht der Satz wirklich ein Komma?'
      : i < 0 ? 'Schau genau hin: Im Satz steht <b class="w dass">dass</b> mit zwei s. Was gehört davor?'
      : 'Fast! Such das Wort <b class="w dass">dass</b>. Wo muss das Komma stehen?', 'think');
    return;
  }
  state.busy = true;
  $('#none').classList.remove('hint');
  const box = $('#sentence');
  const target = i >= 0 ? $(`.cgap[data-i="${i}"]`) : $('#none');
  solve(cur, target, COLOR[cur.k]);
  if (i >= 0) target.classList.add('set');
  box.classList.add('solved');
  say(cur.k === 'dass'
    ? `${pick(PRAISE, 'praise')} Vor <span class="pill dass">dass</span> steht ein Komma: ${full(cur)}`
    : `${pick(PRAISE, 'praise')} Hier steht <span class="pill das">das</span> mit einem s. Der Satz braucht kein Komma: ${full(cur)}`, 'happy');
  later(next, 2400);
}

$('#sentence').addEventListener('click', e => {
  const g = e.target.closest('.cgap');
  if (g) choose(+g.dataset.i, g);
});
$('#none').addEventListener('click', () => choose(-1, $('#none')));
document.addEventListener('keydown', e => {
  if (gameOn() && e.key === '0') choose(-1, $('#none'));
});
