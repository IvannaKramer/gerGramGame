@@tipp
<b>das</b> kann ein <b>Artikel</b> sein (das Haus), ein <b>Pronomen</b> (Das gefällt mir.) oder ein <b>Relativpronomen</b> (das Buch, das ich lese). <b>dass</b> ist ein <b>Bindewort</b>. Es verbindet zwei Sätze: Ich hoffe, dass du kommst.

/* ---------- Spiel: Wort-Detektiv ---------- */
const ORDER = Object.keys(JOBS);
let all = [], queue = [], cur = null;
const tileEl = j => $(`.tile[data-j="${j}"]`);
const full = s => s.text.replace('___', `<b class="w ${JOBS[s.j][2]}">${s.ans}</b>`);

$('#tiles').innerHTML = ORDER.map((j, i) => {
  const [word, job, cls] = JOBS[j];
  return `<button class="tile ${cls}" data-j="${j}" aria-label="${word}, ${job}"><b>${word}</b><small>${job.replace('Relativ', 'Relativ&shy;')}</small><span class="kbd" aria-hidden="true">${i + 1}</span></button>`;
}).join('');

function startGame() {
  all = SENTENCES.map(s => ({ text: s[0], ans: s[1], j: s[2], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welches Wort fehlt? Und welche Aufgabe hat es im Satz? Tippe auf die passende Karte.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#sentence').innerHTML = cur.text.replace('___', '<span class="gap" id="gap">?</span>');
}

function choose(j) {
  if (state.busy || !cur) return;
  const [word, job, cls, why, tip] = JOBS[cur.j];
  if (j !== cur.j) {
    const hint = miss(cur, tileEl(j));
    if (hint) tileEl(cur.j).classList.add('hint');
    say(hint ? tip
      : JOBS[j][0] !== word ? 'Mach die Probe: Passt „dieses“ oder „welches“ in die Lücke? Dann fehlt das. Sonst fehlt dass.'
      : 'Fast! <b class="w das">das</b> mit einem s stimmt. Aber welche Aufgabe hat es hier?', 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const gap = $('#gap');
  gap.textContent = cur.ans; gap.className = 'gap filled ' + cls;
  solve(cur, gap, cls === 'das' ? 'das' : 'bonus');
  say(`${pick(PRAISE, 'praise')} <span class="pill ${cls}">${word}</span> ist hier ein <b>${job}</b>. ${why}<br>${full(cur)}`, 'happy');
  later(next, 2600);
}

$('#tiles').addEventListener('click', e => {
  const t = e.target.closest('.tile');
  if (t) choose(t.dataset.j);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-4]$/.test(e.key)) return;
  choose(ORDER[+e.key - 1]);
});
