const CARDS_PER_ROUND = 5;

const GAME = {
  id: 'koerbe',
  unit: 'Sätze',
  next: { href: '3-komma-waechter.html', title: 'Komma-Wächter' },
  review: () => `<div class="review c2">${['das', 'dass'].map(k =>
    `<div class="col ${k}"><h3>${k}</h3><ul>${all.filter(s => s.k === k).map(s => `<li>${full(s)}</li>`).join('')}</ul></div>`).join('')}</div>`

/* ---------- Spiel: Zwei Körbe ---------- */
const COLOR = { das: 'das', dass: 'bonus' };
let all = [], rounds = [], round = 0, left = 0, selCard = null, selKind = null;
const byId = id => all.find(s => s.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = k => $(`.basket[data-kind="${k}"]`);
const full = s => s.text.replace('___', `<b class="w ${s.k}">${s.ans}</b>`);
const probeText = s => s.text.replace('___', `<b>${s.probe}</b>`);

function startGame() {
  all = SENTENCES.map((s, i) => ({ id: 's' + i, text: s[0], ans: s[1], probe: s[2], emoji: s[3], k: s[1].toLowerCase(), misses: 0, done: false }));
  /* In jeder Runde liegen Karten für beide Körbe */
  const pools = { das: shuffle(all.filter(s => s.k === 'das')), dass: shuffle(all.filter(s => s.k === 'dass')) };
  const n = Math.ceil(all.length / CARDS_PER_ROUND);
  rounds = Array.from({ length: n }, () => []);
  let r = 0;
  ['das', 'dass'].forEach(k => pools[k].forEach(s => { rounds[r % n].push(s); r++; }));
  rounds.forEach(shuffle);
  resetState(all.length);
  round = 0;
  startRound();
  say('Tippe eine Karte an. Ich sage dir dann den Probe-Satz.');
  show('game');
}

function startRound() {
  selCard = selKind = null; left = rounds[round].length;
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#words').innerHTML = rounds[round].map(s =>
    `<button class="word-card" data-id="${s.id}" aria-pressed="false" style="--tilt:${(Math.random() * 2 - 1).toFixed(2)}deg">
      <span class="pic" aria-hidden="true">${s.emoji}</span><span class="word">${s.text.replace('___', '<span class="blank" role="img" aria-label="Lücke"></span>')}</span></button>`).join('');
  renderSelection();
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selCard;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.kind === selKind;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const s = selCard && byId(selCard);
  if (s && s.misses >= 2) basketEl(s.k).classList.add('hint');
}

function tapCard(id) {
  const s = byId(id);
  if (!s || s.done || state.busy) return;
  if (selKind) { attempt(id, selKind); return; }
  selCard = selCard === id ? null : id;
  sfx.pick(); renderSelection();
  if (selCard) say(`Mach die Probe: „${probeText(s)}“ Klingt das richtig?`);
}
function tapBasket(k) {
  if (state.busy) return;
  if (selCard) { attempt(selCard, k); return; }
  selKind = selKind === k ? null : k;
  sfx.pick(); renderSelection();
  if (selKind) say(`Welcher Satz gehört in den Korb <span class="pill ${k}">${k}</span>?`);
}

function attempt(id, k) {
  const s = byId(id);
  if (!s || s.done) return;
  if (s.k !== k) {
    miss(s, cardEl(id));
    wobble(basketEl(k));
    selCard = id; selKind = null;
    renderSelection();
    say(s.misses < 2 ? `Fast! Mach die Probe noch einmal: „${probeText(s)}“ Klingt das richtig?`
      : s.k === 'das'
        ? `Tipp: Die Probe klingt richtig: „${probeText(s)}“ Dann schreibst du <b>das</b> mit einem s.`
        : `Tipp: Die Probe klingt falsch: „${probeText(s)}“ Dann schreibst du <b>dass</b> mit zwei s.`, 'think');
    return;
  }
  const card = cardEl(id);
  selCard = selKind = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(s, basketEl(k), COLOR[k]);
  say(`${pick(PRAISE, 'praise')} Hier fehlt <span class="pill ${k}">${k}</span>: ${full(s)}`, 'happy');
  if (--left === 0) {
    state.busy = true;
    if (round < rounds.length - 1) later(() => { round++; startRound(); state.busy = false; say('Runde geschafft! Hier kommen neue Karten.', 'happy'); }, 2000);
    else later(finish, 1800);
  }
}

$('#words').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapCard(c.dataset.id);
});
$$('.basket').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapBasket(b.dataset.kind); }));
enableDrag('.word-card:not(.done)', '.basket', (card, basket) => attempt(card.dataset.id, basket.dataset.kind));
