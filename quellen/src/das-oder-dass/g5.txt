/* ---------- Spiel: Lücken-Geschichten ---------- */
const COLOR = { das: 'das', dass: 'bonus' };
const GAP = /\[(\w+)\|(\w+)\]/;
let texts = [], t = 0, sel = null;
const fill = (s, html) => s.text.replace(GAP, html);
const gapEl = g => $(`.lgap[data-g="${g.id}"]`);
const tileEl = k => $(`.tile[data-k="${k}"]`);
const openGaps = () => texts[t].sents.filter(s => s.g && !s.g.done).map(s => s.g);

function startGame() {
  let id = 0;
  texts = TEXTS.map(([title, list]) => ({ title, sents: list.map(text => {
    const m = text.match(GAP);
    const s = { text, g: m ? { id: id++, ans: m[1], probe: m[2], k: m[1].toLowerCase(), misses: 0, done: false } : null };
    if (s.g) s.g.s = s;
    return s;
  }) }));
  resetState(id);
  t = 0;
  startText();
  say('Lies die Geschichte. Welches Wort fehlt in der gelben Lücke: das oder dass?');
  show('game');
}

function startText() {
  const cur = texts[t];
  $('#round').textContent = `Geschichte ${t + 1} von ${texts.length}: ${cur.title}`;
  $('#story').innerHTML = cur.sents.map(s => s.g
    ? fill(s, `<button class="lgap" data-g="${s.g.id}" aria-label="Lücke"><span aria-hidden="true">?</span></button>`) : s.text).join(' ');
  select(openGaps()[0]);
}

function select(g) {
  sel = g || null;
  $$('.tile').forEach(el => el.classList.remove('hint'));
  $$('.lgap').forEach(el => {
    const on = !!sel && +el.dataset.g === sel.id;
    el.classList.toggle('cur', on);
    el.setAttribute('aria-label', el.classList.contains('filled') ? el.textContent : on ? 'Lücke, ausgewählt' : 'Lücke');
  });
  if (sel && sel.misses >= 2) tileEl(sel.k).classList.add('hint');
  /* Auf kleinen Bildschirmen die Lücke in die Mitte holen, falls sie verdeckt ist */
  const el = sel && gapEl(sel), r = el && el.getBoundingClientRect();
  if (r && (r.top < 80 || r.bottom > innerHeight - 260)) el.scrollIntoView({ block: 'center', behavior: reduced ? 'auto' : 'smooth' });
}

function probe() {
  if (state.busy || !sel) return;
  sfx.pick();
  say(`<span aria-hidden="true">🔍</span> Probe: „${fill(sel.s, `<mark>${sel.probe}</mark>`)}“<br>Klingt das richtig? Dann fehlt <b>das</b>. Klingt es falsch? Dann fehlt <b>dass</b>.`, 'think');
}

function choose(k) {
  if (state.busy || !sel) return;
  const g = sel;
  if (k !== g.k) {
    const hint = miss(g, tileEl(k));
    wobble(gapEl(g));
    if (hint) tileEl(g.k).classList.add('hint');
    say(!hint ? 'Fast! Mach die Probe: Passt „dieses“ oder „welches“ in die Lücke?'
      : g.k === 'das'
        ? `Tipp: Hier passt die Probe: „${fill(g.s, `<b>${g.probe}</b>`)}“ Dann schreibst du <b>das</b> mit einem s.`
        : 'Tipp: Hier passt weder „dieses“ noch „welches“. Dann schreibst du <b>dass</b> mit zwei s. Davor steht ein Komma.', 'think');
    return;
  }
  g.done = true;
  const el = gapEl(g);
  el.textContent = g.ans; el.className = 'lgap filled ' + k; el.disabled = true;
  solve(g, el, COLOR[k]);
  say(`${pick(PRAISE, 'praise')} ${fill(g.s, `<span class="pill ${k}">${g.ans}</span>`)}`, 'happy');
  const open = openGaps();
  if (open.length) { select(open.find(o => o.id > g.id) || open[0]); return; }
  select(null);
  state.busy = true;
  if (t < texts.length - 1) later(() => { t++; startText(); state.busy = false; say('Geschichte geschafft! Hier kommt die nächste.', 'happy'); }, 2400);
  else later(finish, 2200);
}

$('#story').addEventListener('click', e => {
  const el = e.target.closest('.lgap');
  if (!el || state.busy || el.classList.contains('filled')) return;
  const g = openGaps().find(o => o.id === +el.dataset.g);
  if (g) { sfx.pick(); select(g); }
});
$('#tiles').addEventListener('click', e => {
  const tile = e.target.closest('.tile');
  if (!tile) return;
  if (tile.id === 'lupe') probe(); else choose(tile.dataset.k);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[123]$/.test(e.key)) return;
  if (e.key === '3') probe(); else choose(e.key === '1' ? 'das' : 'dass');
});
