const WORDS = [
  ['der [Fahr]er',     'fahr'],
  ['die Ab[fahr]t',    'fahr'],
  ['das [Fahr]rad',    'fahr'],
  ['die [Fähr]e',      'fahr'],

/* ---------- Spiel: Wortfamilien-Häuser ---------- */
let items = [], selWord = null, selFam = null;
const byId = id => items.find(w => w.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const houseEl = f => $(`.house[data-fam="${f}"]`);

$('#houses').innerHTML = FAMILIES.map(([f, base, emo]) =>
  `<button class="house ws" data-fam="${f}" aria-pressed="false" aria-label="Wortfamilie ${f}, wie ${base}"><span class="emo" aria-hidden="true">${emo}</span><b>${f}</b><small>${base}</small></button>`).join('');

function startGame() {
  items = shuffle(WORDS.map((w, i) => {
    const stem = w[0].match(/\[(.+?)\]/)[1];
    return { id: 'w' + i, text: w[0].replace(/[\[\]]/g, ''), html: w[0].replace(/\[(.+?)\]/, '<span class="t ws">$1</span>'),
      stem, fam: w[1], changed: stem.toLowerCase() !== w[1], misses: 0, done: false };
  }));
  resetState(items.length);
  selWord = selFam = null;
  $('#words').innerHTML = items.map(w =>
    `<button class="word-card" data-id="${w.id}" aria-pressed="false" style="--tilt:${(Math.random() * 3 - 1.5).toFixed(2)}deg"><span class="word">${w.text}</span></button>`).join('');
  renderSelection();
  say('Tippe eine Karte an und dann das passende Haus. Welcher Wortstamm steckt im Wort?');
  show('game');
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selWord;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.house').forEach(el => {
    const on = el.dataset.fam === selFam;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const w = selWord && byId(selWord);
  if (w && w.misses >= 2) houseEl(w.fam).classList.add('hint');
}

function tapWord(id) {
  const w = byId(id);
  if (!w || w.done || state.busy) return;
  if (selFam) { attempt(id, selFam); return; }
  selWord = selWord === id ? null : id;
  sfx.pick(); renderSelection();
  if (selWord) say(`<b>${w.text}</b> – zu welcher Wortfamilie gehört das Wort?`);
}
function tapHouse(f) {
  if (state.busy) return;
  if (selWord) { attempt(selWord, f); return; }
  selFam = selFam === f ? null : f;
  sfx.pick(); renderSelection();
  if (selFam) say(`Welches Wort gehört zur Wortfamilie <span class="pill ws">${f}</span>?`);
}

/* Umlaut im Wortstamm erklären: a → ä, au → äu */
const change = w => w.fam.includes('au') ? 'Aus <b>au</b> wird hier <b>äu</b>.' : 'Aus <b>a</b> wird hier <b>ä</b>.';

function attempt(id, f) {
  const w = byId(id);
  if (!w || w.done) return;
  if (w.fam !== f) {
    miss(w, cardEl(id));
    wobble(houseEl(f));
    selWord = id; selFam = null;
    renderSelection();
    say(w.misses >= 2
      ? `Tipp: Such den Wortstamm! In <b>${w.html}</b> steckt <span class="pill ws">${w.stem.toLowerCase()}</span>. ${w.changed ? change(w) + ' ' : ''}Das leuchtende Haus ist richtig.`
      : 'Das passt noch nicht. Sprich das Wort langsam: Welcher Wortstamm steckt darin?', 'think');
    return;
  }
  const card = cardEl(id);
  selWord = selFam = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(w, houseEl(f), 'das');
  say(`${pick(PRAISE, 'praise')} <b>${w.html}</b> gehört zur Wortfamilie <span class="pill ws">${f}</span>.${w.changed ? ' ' + change(w) : ''}`, 'happy');
  if (state.solved === state.total) { state.busy = true; later(finish, 1500); }
}

$('#words').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapWord(c.dataset.id);
});
$$('.house').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapHouse(b.dataset.fam); }));
enableDrag('.word-card:not(.done)', '.house', (card, house) => attempt(card.dataset.id, house.dataset.fam));
