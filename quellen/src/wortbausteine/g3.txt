/* ---------- Spiel: Groß oder klein? ---------- */
let all = [], queue = [], cur = null;
const colored = w => `${w.base}<b>${w.suffix}</b>`;
const btn = up => $(`.case[data-up="${up ? 1 : 0}"]`);

function startGame() {
  all = WORDS.map(w => {
    const [base, suffix] = w[0].split('·');
    return { base, suffix, noun: w[1] === 'N', emoji: w[2], art: w[3], word: base + suffix, misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Groß oder klein? Schau auf die pinke Nachsilbe und tippe den passenden Buchstaben an.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.case').forEach(b => b.classList.remove('hint'));
  const first = cur.base[0];
  $('#pic').textContent = cur.emoji;
  $('#puzzle').innerHTML = `<span class="slot" id="slot">?</span><span>${cur.base.slice(1)}</span><span class="t ns">${cur.suffix}</span>`;
  $('#puzzle').setAttribute('aria-label', `Wort ohne ersten Buchstaben: ${cur.word.slice(1)}`);
  $('#help').innerHTML = `Das Wort endet auf <span class="pill ns">-${cur.suffix}</span>`;
  $('#b-up').textContent = first.toUpperCase();
  $('#b-lo').textContent = first.toLowerCase();
  btn(true).setAttribute('aria-label', `großes ${first.toUpperCase()}`);
  btn(false).setAttribute('aria-label', `kleines ${first.toLowerCase()}`);
}

function choose(up) {
  if (state.busy || !cur) return;
  const sfxTxt = `<span class="pill ns">-${cur.suffix}</span>`;
  if (up !== cur.noun) {
    const hint = miss(cur, btn(up));
    if (hint) btn(cur.noun).classList.add('hint');
    say(hint
      ? (cur.noun ? `Tipp: Wörter mit ${sfxTxt} sind Nomen. Nomen schreibst du groß.` : `Tipp: Wörter mit ${sfxTxt} sind Adjektive. Adjektive schreibst du klein.`)
      : `Schau noch einmal auf die Nachsilbe ${sfxTxt}. Ist das Wort ein Nomen?`, 'think');
    return;
  }
  state.busy = true;
  $$('.case').forEach(b => b.classList.remove('hint'));
  const slot = $('#slot');
  slot.textContent = cur.base[0]; slot.classList.add('filled');
  if (cur.noun) slot.insertAdjacentHTML('beforebegin', `<span class="art">${cur.art}</span>`);
  solve(cur, btn(up), cur.noun ? 'bonus' : 'teal');
  say(cur.noun
    ? `${pick(PRAISE, 'praise')} <b>${cur.art} ${cur.base}<span class="t ns">${cur.suffix}</span></b> ist ein Nomen. Darum schreibst du es groß.`
    : `${pick(PRAISE, 'praise')} <b>${cur.base}<span class="t ns">${cur.suffix}</span></b> ist ein Adjektiv. Darum schreibst du es klein.`, 'happy');
  later(next, 1900);
}

$$('.case').forEach(b => b.addEventListener('click', () => choose(b.dataset.up === '1')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[12]$/.test(e.key)) return;
  choose(e.key === '1');
});
