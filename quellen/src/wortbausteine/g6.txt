@@tipp
Wörter mit <b>-ung</b>, <b>-heit</b>, <b>-keit</b> und <b>-nis</b> sind Nomen. Du schreibst sie <b>groß</b>. Aufgepasst: Manchmal fällt beim Bauen ein Buchstabe weg oder es kommt einer dazu: samm<b>e</b>ln → die Sammlung, müd<b>e</b> → die Müd<b>ig</b>keit.

/* ---------- Spiel: Nomen-Meister ---------- */
let all = [], queue = [], cur = null;
const input = $('#answer');

function startGame() {
  all = WORDS.map(w => {
    const [art, noun] = w[3].split(' ');
    return { from: w[0], verb: w[1] === 'V', sfx: w[2], full: w[3], art, noun, k: w[4], tip: w[5], misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Baue aus dem Wort und der Nachsilbe ein Nomen. Schreibe es in das Feld.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#from-lbl').textContent = cur.verb ? 'Verb + Nachsilbe' : 'Adjektiv + Nachsilbe';
  $('#from').innerHTML = `<span>${cur.from}</span><span class="plus" aria-hidden="true">+</span><span class="pill ns">${cur.sfx}</span>`;
  $('#art').textContent = cur.art;
  input.value = ''; input.readOnly = false; input.classList.remove('good');
  input.focus({ preventScroll: true });
}

function check() {
  if (state.busy || !cur) return;
  let v = input.value.trim().replace(/\s+/g, ' ');
  if (/^(die|das|der) /i.test(v)) v = v.slice(4);
  if (!v) { say('Schreibe zuerst das Nomen in das Feld.'); input.focus(); return; }
  if (v !== cur.noun) {
    if (v.toLowerCase() === cur.noun.toLowerCase()) {
      wobble(input);
      say(`Fast richtig! Wörter mit <span class="pill ns">${cur.sfx}</span> sind Nomen. Nomen schreibt man groß.`, 'think');
    } else {
      miss(cur, input);
      say(cur.misses === 1 ? 'Fast! Sprich das neue Wort langsam vor und probier es noch einmal.'
        : cur.misses === 2 ? cur.tip
        : `So ist es richtig: <span class="pill ns sol">${cur.full}</span> Schreibe das Wort genau so ab.`, 'think');
    }
    input.focus();
    return;
  }
  state.busy = true;
  input.value = cur.noun; input.readOnly = true; input.classList.add('good');
  solve(cur, input, 'die');
  say(`${pick(PRAISE, 'praise')} <b>${cur.from}</b> → <span class="pill ns">${cur.full}</span>`, 'happy');
  later(next, 1800);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.keycap');
  if (!k || input.readOnly) return;
  const a = input.selectionStart ?? input.value.length, b = input.selectionEnd ?? a;
  input.value = input.value.slice(0, a) + k.textContent + input.value.slice(b);
  input.focus();
  input.setSelectionRange(a + 1, a + 1);
});
