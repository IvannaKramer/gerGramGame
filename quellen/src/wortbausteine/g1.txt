/* ---------- Spiel: Baustein-Detektiv ---------- */
let all = [], queue = [], cur = null;
const colored = w => w.parts.map((p, i) => `<span class="t ${KINDS[w.kinds[i]][1]}">${p}</span>`).join('');
const pill = (w, i) => `<span class="pill ${KINDS[w.kinds[i]][1]}">${w.parts[i]}</span>`;

function startGame() {
  all = WORDS.map(w => {
    const parts = w[0].split('·');
    return { parts, kinds: w[1].split(''), ask: w[2], emoji: w[3], rel: w[4], word: parts.join(''), misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welchen Baustein suche ich? Tippe ihn im Wort an!');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  const k = KINDS[cur.ask];
  $('#pic').textContent = cur.emoji;
  $('#ask').innerHTML = `Tippe ${k[2].split(' ')[0]} <span class="pill ${k[1]}">${k[0]}</span> an:`;
  $('#blocks').innerHTML = cur.parts.map((p, i) =>
    `<button class="blk ${KINDS[cur.kinds[i]][1]}" data-i="${i}" aria-label="Baustein ${p}"><b>${p}</b><small>${KINDS[cur.kinds[i]][0]}</small><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
  $('#rel').innerHTML = `Verwandtes Wort: <b>${cur.rel}</b>`;
}

function choose(i) {
  const el = $$('.blk')[i];
  if (state.busy || !cur || !el) return;
  const want = cur.kinds.indexOf(cur.ask), k = KINDS[cur.ask];
  if (i !== want) {
    const hint = miss(cur, el);
    if (hint) $$('.blk')[want].classList.add('hint');
    say(hint ? `${k[4]} Der leuchtende Baustein ist richtig.`
      : `<b>${cur.parts[i]}</b> ist ${KINDS[cur.kinds[i]][3]}. Ich suche ${k[2]}. Probier es noch einmal!`, 'think');
    return;
  }
  state.busy = true;
  $$('.blk').forEach(b => { b.classList.remove('hint'); b.classList.add('on'); b.disabled = true; });
  solve(cur, el, cur.ask === 'v' ? 'der' : cur.ask === 's' ? 'das' : 'die');
  say(`${pick(PRAISE, 'praise')} ${pill(cur, want)} ist ${k[3]} von <b>${colored(cur)}</b>.`, 'happy');
  later(next, 2000);
}

$('#blocks').addEventListener('click', e => { const b = e.target.closest('.blk'); if (b) choose(+b.dataset.i); });
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-3]$/.test(e.key)) return;
  choose(+e.key - 1);
});
