@@tipp
Mit den Nachsilben <b>-ung</b>, <b>-heit</b>, <b>-keit</b> und <b>-nis</b> baust du aus Verben und Adjektiven <b>Nomen</b>: heilen → die Heil<b>ung</b>, frei → die Frei<b>heit</b>. Das neue Wort schreibst du <b>groß</b>.

/* ---------- Spiel: Baustein-Werkstatt ---------- */
let all = [], queue = [], cur = null;
const tileEl = k => $(`.tile[data-k="${k}"]`);
const cap = s => s[0].toUpperCase() + s.slice(1);

$('#tiles').innerHTML = ENDINGS.map(([k], i) =>
  `<button class="tile ns" data-k="${k}" aria-label="Nachsilbe ${k}"><b>-${k}</b><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');

function startGame() {
  all = WORDS.map(w => {
    const parts = w[2].split('·');
    return { from: w[0], verb: w[1] === 'V', parts, kinds: w[3].split(''), k: w[4], art: w[5],
      stemCap: cap(parts.join('')), noun: `${w[5]} ${cap(parts.join(''))}${w[4]}`, misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welche Nachsilbe macht aus dem Wort ein Nomen? Tippe sie an oder zieh sie an das Wort.');
  show('game');
  next();
}

const partsHtml = (w, upper) => w.parts.map((p, i) =>
  `<span class="part ${KIND_CLS[w.kinds[i]]}">${upper && i === 0 ? cap(p) : p}</span>`).join('');

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#from-lbl').textContent = cur.verb ? 'Verb' : 'Adjektiv';
  $('#from').textContent = cur.from;
  $('#build').innerHTML = `<span class="art" id="art"></span><span id="stem">${partsHtml(cur, false)}</span><span class="slot" id="slot">?</span>`;
}

function choose(k) {
  if (state.busy || !cur || !tileEl(k)) return;
  if (k !== cur.k) {
    const hint = miss(cur, tileEl(k));
    if (hint) tileEl(cur.k).classList.add('hint');
    say(hint ? ENDINGS.find(e => e[0] === cur.k)[1]
      : `Hmm, „${cur.stemCap}${k}“ gibt es nicht. Sprich das Wort mit einer anderen Nachsilbe vor!`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#art').textContent = cur.art;
  $('#stem').innerHTML = partsHtml(cur, true);
  const slot = $('#slot');
  slot.textContent = k; slot.className = 'slot filled ns';
  solve(cur, $('#build'), 'die');
  say(`${pick(PRAISE, 'praise')} <b>${cur.from}</b> → <span class="pill ns">${cur.noun}</span>. Das ist ein Nomen. Darum schreibst du es groß.`, 'happy');
  later(next, 2000);
}

$('#tiles').addEventListener('click', e => {
  if (suppressClick) return;
  const t = e.target.closest('.tile');
  if (t) choose(t.dataset.k);
});
enableDrag('.tile', '.bench', tile => choose(tile.dataset.k));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-4]$/.test(e.key)) return;
  choose(ENDINGS[+e.key - 1][0]);
});
