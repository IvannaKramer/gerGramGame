@@tipp
Eine <b>Vorsilbe</b> steht vor dem Wortstamm und verändert die Bedeutung: <b>ver</b>-, <b>be</b>-, <b>ent</b>-, <b>er</b>-, <b>zer</b>-, <b>vor</b>- und <b>un</b>-. Aus <b>brechen</b> wird <b>zer</b>brechen, aus <b>möglich</b> wird <b>un</b>möglich.

/* ---------- Spiel: Vorsilben-Lücke ---------- */
let all = [], queue = [], cur = null, nr = 0;
const optEl = p => $(`.opt[data-p="${p}"]`);

function startGame() {
  all = SENTENCES.map(s => ({ a: s[0], rest: s[1], z: s[2], pre: s[3], wrong: s[4], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  nr = 0;
  say('Welche Vorsilbe fehlt? Probiere alle drei leise aus und tippe die passende an.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  nr++;
  $('#count').textContent = `Satz ${nr} von ${all.length}`;
  $('#sent').innerHTML = `${cur.a}<span class="w"><span class="gap vs" id="gap" aria-label="Lücke">&nbsp;</span>${cur.rest}</span>${cur.z}`;
  $('#opts').innerHTML = shuffle([cur.pre, ...cur.wrong]).map((p, i) =>
    `<button class="opt vs" data-p="${p}" aria-label="Vorsilbe ${p}"><b>${p}-</b><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function choose(el) {
  if (!el || state.busy || el.disabled || !cur) return;
  const p = el.dataset.p;
  if (p !== cur.pre) {
    const hint = miss(cur, el);
    el.disabled = true;
    if (hint) optEl(cur.pre).classList.add('hint');
    say(hint ? `${TIPS[cur.pre]} Die leuchtende Vorsilbe ist richtig.`
      : `Hmm, „${p}${cur.rest}“ gibt es nicht. Probiere eine andere Vorsilbe!`, 'think');
    return;
  }
  state.busy = true;
  $$('.opt').forEach(b => b.classList.remove('hint'));
  const gap = $('#gap');
  gap.textContent = p; gap.classList.add('filled'); gap.removeAttribute('aria-label');
  solve(cur, el, 'der');
  say(`${pick(PRAISE, 'praise')} <span class="pill vs">${p}</span> + ${cur.rest} = <b><span class="t vs">${p}</span>${cur.rest}</b>`, 'happy');
  later(next, 2000);
}

$('#opts').addEventListener('click', e => choose(e.target.closest('.opt')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-3]$/.test(e.key)) return;
  choose($$('.opt')[+e.key - 1]);
});
