@@tipp
Der Artikel verändert sich mit dem Fall: Wer oder was? <b>der</b> Tiger · Wessen? <b>des</b> Tigers · Wem? <b>dem</b> Tiger · Wen oder was? <b>den</b> Tiger.

/* ---------- Spiel: Artikel-Dreher ---------- */
const FALL = {
  nom: { name: 'Nominativ', q: 'Wer oder was?', art: 'der', color: 'der' },
  gen: { name: 'Genitiv', q: 'Wessen?', art: 'des', color: 'bonus' },
  dat: { name: 'Dativ', q: 'Wem?', art: 'dem', color: 'das' },
  akk: { name: 'Akkusativ', q: 'Wen oder was?', art: 'den', color: 'die' }
};
const ORDER = ['nom', 'gen', 'dat', 'akk'];
const ARTS = ORDER.map(k => FALL[k].art);
let all = [], queue = [], cur = null, idx = -1;
const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
const shown = (s, art) => s.start ? cap(art) : art;
const full = s => s.text.replace('___', `<b class="w ${s.k}">${shown(s, s.art)}</b>`);

$('#ladder').innerHTML = ARTS.map((a, i) => `<span data-i="${i}">${a}</span>`).join('<i>→</i>');

function startGame() {
  all = SENTENCES.map(s => ({ text: s[0], art: s[1], k: ORDER[ARTS.indexOf(s[1])], q: s[2], emoji: s[3], start: s[0].startsWith('___'), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies die Frage. Dreh dann den Artikel im gelben Feld, bis er passt.');
  show('game');
  next();
}

function render() {
  const sp = $('#spin');
  sp.textContent = idx < 0 ? '?' : shown(cur, ARTS[idx]);
  sp.setAttribute('aria-label', idx < 0 ? 'Artikel wählen' : `Artikel: ${ARTS[idx]}. Tippen dreht weiter.`);
  $$('#ladder span').forEach((el, i) => el.classList.toggle('on', i === idx));
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; idx = -1;
  $$('#ladder span').forEach(el => el.classList.remove('hint'));
  $('#pic').textContent = cur.emoji;
  const qp = $('#qpill');
  qp.className = 'pill ' + cur.k; qp.textContent = FALL[cur.k].q;
  $('#qtext').textContent = cur.q;
  $('#sentence').innerHTML = cur.text.replace('___', '<button class="spin" id="spin" type="button"></button>');
  render();
}

function turn(to) {
  if (state.busy || !cur) return;
  idx = to === undefined ? (idx + 1) % ARTS.length : to;
  sfx.pick(); render();
}

function check() {
  if (state.busy || !cur) return;
  const sp = $('#spin'), f = FALL[cur.k];
  if (idx < 0) { say('Tippe zuerst auf das gelbe Feld mit dem Fragezeichen.'); wobble(sp); return; }
  if (ARTS[idx] !== cur.art) {
    const hint = miss(cur, sp);
    if (hint) $(`#ladder span[data-i="${ARTS.indexOf(cur.art)}"]`).classList.add('hint');
    say(!hint ? `Fast! Lies die Frage noch einmal: „${cur.q}“ Dreh den Artikel weiter.`
      : `Tipp: Auf „${f.q}“ antwortet der ${f.name}. Im ${f.name} heißt der Artikel <b>${cur.art}</b>.`, 'think');
    return;
  }
  state.busy = true;
  $$('#ladder span').forEach(el => el.classList.remove('hint'));
  sp.classList.add('solved', cur.k); sp.disabled = true;
  solve(cur, sp, f.color);
  say(`${pick(PRAISE, 'praise')} <b>${cur.q}</b> – ${f.name}, also <span class="pill ${cur.k}">${cur.art}</span>: ${full(cur)}`, 'happy');
  later(next, 2600);
}

$('#sentence').addEventListener('click', e => { if (e.target.closest('#spin')) turn(); });
$('#check').addEventListener('click', check);
document.addEventListener('keydown', e => {
  if (!gameOn()) return;
  if (/^[1-4]$/.test(e.key)) turn(+e.key - 1);
});
