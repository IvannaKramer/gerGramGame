@@tipp
Manche Nomen bekommen eine Endung: <b>des</b> Fluss<b>es</b>, <b>dem</b> Rabe<b>n</b>, <b>den</b> Zuschauer<b>n</b>. Kleine Wörter wie <b>mit</b> und <b>nach</b> verlangen immer den Dativ, <b>für</b> und <b>ohne</b> immer den Akkusativ.

/* ---------- Spiel: Fall-Profi ---------- */
const FALL = {
  nom: { name: 'Nominativ', q: 'Wer oder was?', color: 'der' },
  gen: { name: 'Genitiv', q: 'Wessen?', color: 'bonus' },
  dat: { name: 'Dativ', q: 'Wem?', color: 'das' },
  akk: { name: 'Akkusativ', q: 'Wen oder was?', color: 'die' }
};
const ORDER = ['nom', 'gen', 'dat', 'akk'];
let all = [], queue = [], cur = null;
const full = s => s.text.replace('___', `<b class="w ${s.k}">${s.ans}</b>`);
const input = () => $('#sentence input');

function startGame() {
  all = SENTENCES.map(s => ({ text: s[0], base: s[1], ans: s[2], k: s[3], help: s[4], rule: s[5], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Frag nach der Lücke. Schreibe dann Artikel und Nomen im richtigen Fall hinein.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#base').textContent = cur.base;
  $('#sentence').innerHTML = cur.text.replace('___',
    '<input type="text" autocapitalize="off" autocorrect="off" autocomplete="off" spellcheck="false" maxlength="24" aria-label="Artikel und Nomen im richtigen Fall">');
  input().focus({ preventScroll: true });
}

function check() {
  if (state.busy || !cur) return;
  const inp = input(), f = FALL[cur.k];
  const v = inp.value.trim().replace(/\s+/g, ' ');
  if (!v) { say('Schreibe zuerst Artikel und Nomen in die Lücke.'); inp.focus(); return; }
  inp.classList.remove('hint');
  if (v !== cur.ans) {
    if (v.toLowerCase() === cur.ans.toLowerCase()) {
      wobble(inp);
      say('Fast richtig! Achte auf die Großschreibung: Den Artikel schreibst du klein, das Nomen groß.', 'think');
      inp.focus();
      return;
    }
    if (!v.includes(' ')) {
      wobble(inp);
      say('Da fehlt noch etwas. Schreibe den Artikel <b>und</b> das Nomen in die Lücke.', 'think');
      inp.focus();
      return;
    }
    miss(cur, inp);
    if (cur.misses === 2) inp.classList.add('hint');
    say(cur.misses === 1 ? `Fast! ${cur.help}`
      : cur.misses === 2 ? `Tipp: Hier steht der <b>${f.name}</b>. ${cur.rule}`
      : `So ist es richtig: <span class="pill ${cur.k}">${cur.ans}</span> Schreibe es genau so ab.`, 'think');
    inp.focus();
    return;
  }
  state.busy = true;
  inp.readOnly = true; inp.classList.add('good', cur.k);
  solve(cur, inp, f.color);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${cur.k}">${cur.ans}</span> steht im <b>${f.name}</b>: ${full(cur)}`, 'happy');
  later(next, 2800);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
