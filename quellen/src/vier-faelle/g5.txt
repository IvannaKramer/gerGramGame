/* ---------- Spiel: Wort-Verwandler ---------- */
const FALL = {
  nom: { name: 'Nominativ', q: 'Wer oder was?', color: 'der' },
  gen: { name: 'Genitiv', q: 'Wessen?', color: 'bonus' },
  dat: { name: 'Dativ', q: 'Wem?', color: 'das' },
  akk: { name: 'Akkusativ', q: 'Wen oder was?', color: 'die' }
};
const ORDER = ['nom', 'gen', 'dat', 'akk'];
let all = [], queue = [], cur = null;
const full = s => s.text.replace('___', `<b class="w ${s.k}">${s.ans}</b>`);
const formEl = i => $(`.form[data-i="${i}"]`);

function startGame() {
  all = SENTENCES.map(s => ({ text: s[0], base: s[1], forms: s[2], ans: s[3], k: s[4], q: s[5], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies den Satz und frag nach der Lücke. Welche Form passt hinein?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#base').textContent = cur.base;
  $('#sentence').innerHTML = cur.text.replace('___', '<span class="gap" id="gap">?</span>');
  $('#forms').innerHTML = cur.forms.map((f, i) =>
    `<button class="form" type="button" data-i="${i}">${f}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function choose(i) {
  if (state.busy || !cur) return;
  const el = formEl(i), f = FALL[cur.k];
  if (!el) return;
  if (cur.forms[i] !== cur.ans) {
    const hint = miss(cur, el);
    if (hint) formEl(cur.forms.indexOf(cur.ans)).classList.add('hint');
    say(!hint ? 'Fast! Frag nach der Lücke: Wer oder was? Wessen? Wem? Wen oder was?'
      : `Tipp: Frag so: „${cur.q}“ Das ist der <b>${f.name}</b>. Im ${f.name} heißt es <b>${cur.ans.charAt(0).toLowerCase() + cur.ans.slice(1)}</b>.`, 'think');
    return;
  }
  state.busy = true;
  $$('.form').forEach(b => { b.classList.remove('hint'); b.disabled = true; });
  const gap = $('#gap');
  gap.textContent = cur.ans; gap.className = 'gap filled ' + cur.k;
  solve(cur, gap, f.color);
  say(`${pick(PRAISE, 'praise')} <b>${cur.q}</b> – <span class="pill ${cur.k}">${cur.ans.charAt(0).toLowerCase() + cur.ans.slice(1)}</span>. Das ist der <b>${f.name}</b>.`, 'happy');
  later(next, 2500);
}

$('#forms').addEventListener('click', e => {
  if (suppressClick) return;
  const b = e.target.closest('.form');
  if (b) choose(+b.dataset.i);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-4]$/.test(e.key)) return;
  choose(+e.key - 1);
});
enableDrag('.form:not(:disabled)', '#gap', src => choose(+src.dataset.i));
