/* ---------- Spiel: Fall-Detektiv ---------- */
const FALL = {
  nom: { name: 'Nominativ', q: 'Wer oder was?', art: 'der', color: 'der' },
  gen: { name: 'Genitiv', q: 'Wessen?', art: 'des', color: 'bonus' },
  dat: { name: 'Dativ', q: 'Wem?', art: 'dem', color: 'das' },
  akk: { name: 'Akkusativ', q: 'Wen oder was?', art: 'den', color: 'die' }
};
const ORDER = ['nom', 'gen', 'dat', 'akk'];
const MARK = /\[([^\]]+)\]/;
let all = [], queue = [], cur = null;
const tileEl = k => $(`.tile[data-k="${k}"]`);
const full = s => s.text.replace(MARK, `<b class="w ${s.k}">$1</b>`);
/* der markierte Satzteil, so wie er mitten im Satz steht */
const answer = text => { const m = text.match(MARK); return m.index === 0 ? m[1].charAt(0).toLowerCase() + m[1].slice(1) : m[1]; };

function startGame() {
  all = SENTENCES.map(s => ({ text: s[0], k: s[1], q: s[2], emoji: s[3], ans: answer(s[0]), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies den Satz. Frag nach dem gelben Satzteil. Welche Frage passt?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#pic').textContent = cur.emoji;
  $('#sentence').innerHTML = cur.text.replace(MARK, '<span class="mk" id="mk">$1</span>');
}

function choose(k) {
  if (state.busy || !cur) return;
  const f = FALL[cur.k];
  if (k !== cur.k) {
    const hint = miss(cur, tileEl(k));
    if (hint) tileEl(cur.k).classList.add('hint');
    say(!hint ? pick(['Fast! Frag nach dem gelben Satzteil. Welche Frage klingt richtig?', 'Hmm, das passt noch nicht. Schau auf den Artikel: der, des, dem oder den?'], 'retry')
      : `Tipp: Frag so: „${cur.q}“ Die Antwort ist <b>${cur.ans}</b>. Der Artikel <b>${f.art}</b> zeigt dir den ${f.name}.`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const mk = $('#mk');
  mk.className = 'mk solved ' + k;
  solve(cur, mk, f.color);
  say(`${pick(PRAISE, 'praise')} <b>${cur.q}</b> – <span class="pill ${k}">${cur.ans}</span>. Das ist der <b>${f.name}</b>.`, 'happy');
  later(next, 2400);
}

$('#tiles').addEventListener('click', e => {
  const t = e.target.closest('.tile');
  if (t) choose(t.dataset.k);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-4]$/.test(e.key)) return;
  choose(ORDER[+e.key - 1]);
});
