@@tipp
Auf jede Frage antwortet ein anderer Fall: <b>Wer oder was?</b> → Nominativ (der) · <b>Wessen?</b> → Genitiv (des) · <b>Wem?</b> → Dativ (dem) · <b>Wen oder was?</b> → Akkusativ (den).

/* ---------- Spiel: Frage-Lupe ---------- */
const FALL = {
  nom: { name: 'Nominativ', q: 'Wer oder was?', color: 'der' },
  gen: { name: 'Genitiv', q: 'Wessen?', color: 'bonus' },
  dat: { name: 'Dativ', q: 'Wem?', color: 'das' },
  akk: { name: 'Akkusativ', q: 'Wen oder was?', color: 'die' }
};
const ORDER = ['nom', 'gen', 'dat', 'akk'];
let all = [], queue = [], cur = null;
const low = s => s.charAt(0).toLowerCase() + s.slice(1);
const full = s => s.parts.map((p, i) => i === s.target ? `<b class="w ${s.k}">${p}</b>` : p).join(' ') + '.';

function startGame() {
  all = SENTENCES.map(s => {
    const raw = s[0].replace(/\.$/, '').split(' | ');
    const target = raw.findIndex(p => p.startsWith('['));
    const parts = raw.map(p => p.replace(/[\[\]]/g, ''));
    /* Antwort so, wie sie mitten im Satz steht (Artikel klein) */
    return { parts, target, k: s[1], q: s[2], emoji: s[3], ans: target === 0 ? low(parts[0]) : parts[target], misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies die Frage. Welcher Satzteil antwortet darauf? Tippe ihn an!');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#pic').textContent = cur.emoji;
  const qp = $('#qpill');
  qp.className = 'pill ' + cur.k; qp.textContent = FALL[cur.k].name;
  $('#qtext').textContent = cur.q;
  $('#chunks').innerHTML = cur.parts.map((p, i) => `<button class="chunk" type="button" data-i="${i}">${p}</button>`).join('') + '<span aria-hidden="true">.</span>';
}

function choose(i) {
  if (state.busy || !cur) return;
  const el = $(`.chunk[data-i="${i}"]`), f = FALL[cur.k];
  if (!el) return;
  if (i !== cur.target) {
    const hint = miss(cur, el);
    if (hint) $(`.chunk[data-i="${cur.target}"]`).classList.add('hint');
    say(!hint ? `Fast! <b>${cur.parts[i]}</b> antwortet nicht auf „${f.q}“ Lies die Frage noch einmal.`
      : `Tipp: Auf „${f.q}“ antwortet der ${f.name}. Such das Wort <b>${cur.ans.split(' ')[0]}</b>!`, 'think');
    return;
  }
  state.busy = true;
  $$('.chunk').forEach(c => { c.classList.remove('hint'); c.disabled = true; });
  el.classList.add('good', cur.k);
  solve(cur, el, f.color);
  say(`${pick(PRAISE, 'praise')} <b>${cur.q}</b> – <span class="pill ${cur.k}">${cur.ans}</span>. Das ist der <b>${f.name}</b>.`, 'happy');
  later(next, 2400);
}

$('#chunks').addEventListener('click', e => {
  const c = e.target.closest('.chunk');
  if (c) choose(+c.dataset.i);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-6]$/.test(e.key)) return;
  choose(+e.key - 1);
});
