/* ---------- Spiel: Fall-Kästen ---------- */
const FALL = {
  nom: { name: 'Nominativ', q: 'Wer oder was?', color: 'der' },
  gen: { name: 'Genitiv', q: 'Wessen?', color: 'bonus' },
  dat: { name: 'Dativ', q: 'Wem?', color: 'das' },
  akk: { name: 'Akkusativ', q: 'Wen oder was?', color: 'die' }
};
const ORDER = ['nom', 'gen', 'dat', 'akk'];
const MARK = /\[([^\]]+)\]/;
let all = [], rounds = [], round = 0, left = 0, selCard = null, selKind = null;
const byId = id => all.find(s => s.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = k => $(`.basket[data-kind="${k}"]`);
const full = s => s.text.replace(MARK, `<b class="w ${s.k}">$1</b>`);
const answer = text => { const m = text.match(MARK); return m.index === 0 ? m[1].charAt(0).toLowerCase() + m[1].slice(1) : m[1]; };

function startGame() {
  all = SENTENCES.map((s, i) => ({ id: 's' + i, text: s[0], k: s[1], q: s[2], ans: answer(s[0]), misses: 0, done: false }));
  /* In jeder Runde liegen Karten aus allen vier Fällen */
  const n = Math.ceil(all.length / CARDS_PER_ROUND);
  rounds = Array.from({ length: n }, () => []);
  let r = 0;
  ORDER.forEach(k => shuffle(all.filter(s => s.k === k)).forEach(s => { rounds[r % n].push(s); r++; }));
  rounds.forEach(shuffle);
  resetState(all.length);
  round = 0;
  startRound();
  say('Tippe eine Karte an. Frag nach dem gelben Satzteil und wähle den Kasten.');
  show('game');
}

function startRound() {
  selCard = selKind = null; left = rounds[round].length;
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#words').innerHTML = rounds[round].map(s =>
    `<button class="word-card" data-id="${s.id}" aria-pressed="false" style="--tilt:${(Math.random() * 2 - 1).toFixed(2)}deg">
      <span class="word">${s.text.replace(MARK, '<span class="mk">$1</span>')}</span></button>`).join('');
  renderSelection();
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selCard;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.kind === selKind;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const s = selCard && byId(selCard);
  if (s && s.misses >= 2) basketEl(s.k).classList.add('hint');
}

function tapCard(id) {
  const s = byId(id);
  if (!s || s.done || state.busy) return;
  if (selKind) { attempt(id, selKind); return; }
  selCard = selCard === id ? null : id;
  sfx.pick(); renderSelection();
  if (selCard) say(`<b>${s.ans}</b> – mit welcher Frage fragst du danach?`);
}
function tapBasket(k) {
  if (state.busy) return;
  if (selCard) { attempt(selCard, k); return; }
  selKind = selKind === k ? null : k;
  sfx.pick(); renderSelection();
  if (selKind) say(`Welche Karte gehört in den Kasten <span class="pill ${k}">${FALL[k].name}</span>?`);
}

function attempt(id, k) {
  const s = byId(id);
  if (!s || s.done) return;
  const f = FALL[s.k];
  if (s.k !== k) {
    miss(s, cardEl(id));
    wobble(basketEl(k));
    selCard = id; selKind = null;
    renderSelection();
    say(s.misses < 2 ? `Fast! Frag nach <b>${s.ans}</b>: Wer oder was? Wessen? Wem? Wen oder was?`
      : `Tipp: Frag so: „${s.q}“ – <b>${s.ans}</b>. Auf „${f.q}“ antwortet der <b>${f.name}</b>.`, 'think');
    return;
  }
  const card = cardEl(id);
  selCard = selKind = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(s, basketEl(k), f.color);
  say(`${pick(PRAISE, 'praise')} <b>${s.q}</b> – <span class="pill ${k}">${s.ans}</span>. Das ist der <b>${f.name}</b>.`, 'happy');
  if (--left === 0) {
    state.busy = true;
    if (round < rounds.length - 1) later(() => { round++; startRound(); state.busy = false; say('Runde geschafft! Hier kommen neue Karten.', 'happy'); }, 2200);
    else later(finish, 1800);
  }
}

$('#words').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapCard(c.dataset.id);
});
$$('.basket').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapBasket(b.dataset.kind); }));
enableDrag('.word-card:not(.done)', '.basket', (card, basket) => attempt(card.dataset.id, basket.dataset.kind));
