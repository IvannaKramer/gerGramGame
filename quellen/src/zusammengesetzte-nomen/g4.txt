/* ---------- Spiel: Wörter-Baukasten ---------- */
const ARTS = ['der', 'die', 'das'];
let all = [], queue = [], cur = null, step = 1, stepMiss = 0, opts = [];
const tileEl = v => $$('.tile').find(t => t.dataset.v === v);

function startGame() {
  all = WORDS.map(w => ({ bw: w[0], fuge: w[1], gw: w[2], art: w[3], gap: w[4], wrong: w[5], misses: 0,
    word: w[0] + w[1] + w[2].toLowerCase(),
    html: w[0] + (w[1] ? `<span class="fuge">${w[1]}</span>` : '') + w[2].toLowerCase() }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welches Wort passt in die Lücke? Tippe es an oder zieh es nach oben.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; step = 1; stepMiss = 0;
  const open = '<span class="blk open" id="slot">?</span>';
  const given = (w, cls, name) => `<span class="blk ${cls}">${w}<small>${name}</small></span>`;
  $('#lbl').textContent = 'Welcher Baustein fehlt?';
  $('#build').innerHTML = cur.gap === 'gw'
    ? given(cur.bw, 'bw', 'Bestimmungswort') + '<span class="plus" aria-hidden="true">+</span>' + open
    : open + '<span class="plus" aria-hidden="true">+</span>' + given(cur.gw, 'gw', 'Grundwort');
  opts = shuffle([cur.gap === 'gw' ? cur.gw : cur.bw, ...cur.wrong]);
  const tiles = $('#tiles');
  tiles.classList.remove('three');
  tiles.innerHTML = opts.map((o, i) => `<button class="tile" data-v="${o}"><b>${o}</b><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function choose(v) {
  if (state.busy || !cur) return;
  if (step === 1) {
    const right = cur.gap === 'gw' ? cur.gw : cur.bw, given = cur.gap === 'gw' ? cur.bw : cur.gw;
    if (v !== right) {
      miss(cur, tileEl(v));
      const hint = ++stepMiss >= 2;
      if (hint) tileEl(right).classList.add('hint');
      const tried = cur.gap === 'gw' ? cur.bw + v.toLowerCase() : v + cur.gw.toLowerCase();
      say(hint
        ? `Tipp: Sprich jedes Wort mit <b>${given}</b> zusammen. Nur ein Wort gibt es wirklich. Der leuchtende Baustein passt.`
        : `Hmm, „${tried}“ gibt es nicht. Probier einen anderen Baustein!`, 'think');
      return;
    }
    step = 2; stepMiss = 0;
    sfx.pick();
    $('#lbl').textContent = 'Welcher Artikel passt?';
    $('#build').innerHTML = `<span class="built"><span class="aslot" id="aslot">?</span><span class="w">${cur.html}</span></span>`;
    opts = ARTS;
    const tiles = $('#tiles');
    tiles.classList.add('three');
    tiles.innerHTML = ARTS.map((a, i) => `<button class="tile art ${a}" data-v="${a}"><b>${a}</b><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
    say(`Gut gebaut: <b>${cur.word}</b>. ` + (cur.fuge ? `Zwischen den Wörtern steht hier noch <b>${cur.fuge}</b>. ` : '') + 'Welcher Artikel passt? Denk an das Grundwort!');
    return;
  }
  if (v !== cur.art) {
    miss(cur, tileEl(v));
    const hint = ++stepMiss >= 2;
    if (hint) tileEl(cur.art).classList.add('hint');
    say(hint
      ? `Tipp: Das Grundwort steht hinten. Es heißt <b>${cur.art} ${cur.gw}</b>. Also heißt es auch <b>${cur.art}</b> ${cur.word}.`
      : `Fast! Das Grundwort ist <b>${cur.gw}</b>. Welchen Artikel hat es?`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const slot = $('#aslot'); slot.textContent = cur.art; slot.className = 'aslot filled ' + cur.art;
  solve(cur, slot, cur.art);
  say(`${pick(PRAISE, 'praise')} ${cur.bw} + ${cur.gw} = <span class="pill ${cur.art}">${cur.art} ${cur.word}</span> Das Grundwort bestimmt den Artikel: <b>${cur.art} ${cur.gw}</b>.`, 'happy');
  later(next, 2100);
}

$('#tiles').addEventListener('click', e => {
  if (suppressClick) return;
  const t = e.target.closest('.tile');
  if (t) choose(t.dataset.v);
});
enableDrag('.tile', '.bench', tile => choose(tile.dataset.v));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-4]$/.test(e.key)) return;
  const v = opts[+e.key - 1];
  if (v) choose(v);
});
