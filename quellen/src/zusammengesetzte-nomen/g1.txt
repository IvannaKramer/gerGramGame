@@tipp
Aus zwei Nomen kannst du ein neues Nomen bauen. Vorne steht das <b>Bestimmungswort</b>, hinten das <b>Grundwort</b>. Das neue Wort schreibst du zusammen und vorne groß: Apfel + Saft = <b>Apfelsaft</b>.

/* ---------- Spiel: Wörter-Puzzle ---------- */
let all = [], rounds = [], round = 0, selL = null, selR = null, left = 0;
const byId = id => all.find(w => w.id === id);
const cardEl = (side, id) => $(`#${side} .pc[data-id="${id}"]`);

function startGame() {
  all = WORDS.map((w, i) => ({ id: 'w' + i, bw: w[0], e1: w[1], gw: w[2], e2: w[3], art: w[4], word: w[0] + w[2].toLowerCase(), misses: 0 }));
  rounds = [];
  for (let i = 0; i < all.length; i += PER_ROUND) rounds.push(all.slice(i, i + PER_ROUND));
  shuffle(rounds);
  resetState(all.length);
  round = 0;
  startRound();
  say('Tippe links ein Wort an und dann rechts das Wort, das dazu passt. Welches neue Wort gibt es wirklich?');
  show('game');
}

function startRound() {
  selL = selR = null; left = rounds[round].length;
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  const card = (w, word, pic) => `<button class="pc" data-id="${w.id}" aria-pressed="false"><span class="pic" aria-hidden="true">${pic}</span><span class="word">${word}</span></button>`;
  $('#left').innerHTML = shuffle(rounds[round].slice()).map(w => card(w, w.bw, w.e1)).join('');
  $('#right').innerHTML = shuffle(rounds[round].slice()).map(w => card(w, w.gw, w.e2)).join('');
  $('#shelf').innerHTML = '';
}

function renderSelection() {
  $$('.pc').forEach(el => {
    const on = el.dataset.id === (el.closest('#left') ? selL : selR);
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const w = selL && byId(selL);
  if (w && w.misses >= 2) cardEl('right', w.id).classList.add('hint');
}

function tap(el) {
  if (!el || state.busy || el.classList.contains('done')) return;
  const isLeft = !!el.closest('#left'), id = el.dataset.id;
  if (isLeft) {
    if (selR) { attempt(id, selR); return; }
    selL = selL === id ? null : id;
    sfx.pick(); renderSelection();
    if (selL) say(`<b>${byId(id).bw}</b> + … Welches Wort passt dahinter?`);
  } else {
    if (selL) { attempt(selL, id); return; }
    selR = selR === id ? null : id;
    sfx.pick(); renderSelection();
    if (selR) say(`… + <b>${byId(id).gw}</b>. Welches Wort passt davor?`);
  }
}

function attempt(l, r) {
  const w = byId(l), other = byId(r);
  if (!w || !other || w.done || other.done) return;
  if (l !== r) {
    const hint = miss(w, cardEl('left', l));
    wobble(cardEl('right', r));
    selL = l; selR = null;
    renderSelection();
    say(hint
      ? `Tipp: Sprich <b>${w.bw}</b> mit jedem Wort rechts zusammen. Nur ein Wort gibt es wirklich. Die leuchtende Karte passt.`
      : `Hmm, „${w.bw + other.gw.toLowerCase()}“ gibt es nicht. Probier eine andere Karte!`, 'think');
    return;
  }
  selL = selR = null;
  renderSelection();
  const a = cardEl('left', l), b = cardEl('right', r);
  [a, b].forEach(c => { c.classList.add('done'); c.disabled = true; });
  $('#shelf').insertAdjacentHTML('beforeend', `<span class="chip ${w.art}"><span aria-hidden="true">${w.e1}${w.e2}</span> <b>${w.art} ${w.word}</b></span>`);
  solve(w, b, w.art);
  say(`${pick(PRAISE, 'praise')} ${w.bw} + ${w.gw} = <span class="pill ${w.art}">${w.art} ${w.word}</span> Der Artikel kommt vom Grundwort: <b>${w.art} ${w.gw}</b>.`, 'happy');
  if (--left === 0) {
    state.busy = true;
    if (round < rounds.length - 1) later(() => { round++; startRound(); state.busy = false; say('Runde geschafft! Hier kommen neue Karten.', 'happy'); }, 2000);
    else later(finish, 1700);
  }
}

$('.puzzle').addEventListener('click', e => { if (!suppressClick) tap(e.target.closest('.pc')); });
enableDrag('.pc:not(.done)', '.pc:not(.done)', (src, tgt) => {
  const sl = !!src.closest('#left'), tl = !!tgt.closest('#left');
  if (sl === tl) return;
  attempt(sl ? src.dataset.id : tgt.dataset.id, sl ? tgt.dataset.id : src.dataset.id);
});
