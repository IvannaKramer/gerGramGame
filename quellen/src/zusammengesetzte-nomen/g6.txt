@@tipp
Vorne kann auch ein <b>Verb</b> oder ein <b>Adjektiv</b> stehen: schreiben + Tisch = der <b>Schreib</b>tisch, hoch + Haus = das <b>Hoch</b>haus. Manchmal kommt ein <b>Verbindungs-Buchstabe</b> dazwischen: Sonne + <b>n</b> + Schirm. Das neue Wort ist immer ein Nomen. Du schreibst es groß und das Grundwort bestimmt den Artikel.

   Art: 'f' = Nomen + Nomen mit Verbindungs-Buchstabe, 'v' = Verb + Nomen, 'a' = Adjektiv + Nomen
   Bei Verben fällt die Endung -en weg (schreiben → Schreib-). Das berechnet das Spiel selbst. */

const KINDS = {
  f: ['Nomen + Nomen mit Verbindungs-Buchstabe', 'Nomen + Nomen'],
  v: ['Verb + Nomen', 'Verb + Nomen'],
  a: ['Adjektiv + Nomen', 'Adjektiv + Nomen']

/* ---------- Spiel: Profi-Werkstatt ---------- */
let all = [], queue = [], cur = null;
const input = $('#answer');
const cap = s => s[0].toUpperCase() + s.slice(1);

function startGame() {
  all = WORDS.map(w => {
    const stem = cap(w[4] === 'v' ? w[0].replace(/en$/, '') : w[0]);
    return { first: w[0], fuge: w[1], gw: w[2], art: w[3], k: w[4], stem, misses: 0,
      word: stem + w[1] + w[2].toLowerCase(),
      html: stem + (w[1] ? `<span class="fuge">${w[1]}</span>` : '') + w[2].toLowerCase() };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Baue aus den zwei Wörtern ein neues Nomen. Schreibe es mit Artikel in das Feld.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#from').innerHTML = `<small>${KINDS[cur.k][1]}</small>${cur.first}<span class="plus" aria-hidden="true">+</span>${cur.gw}`;
  input.value = ''; input.readOnly = false; input.className = '';
  input.focus({ preventScroll: true });
}

function ruleTip() {
  if (cur.k === 'f') return `Tipp: Zwischen die beiden Wörter kommt hier noch <b>${cur.fuge}</b>: ${cur.first} + <b>${cur.fuge}</b> + ${cur.gw.toLowerCase()}. Schreibe alles zusammen.`;
  if (cur.k === 'v') return `Tipp: Vom Verb bleibt nur der Wortstamm: aus <b>${cur.first}</b> wird <b>${cur.stem}</b>-. Das neue Wort ist ein Nomen. Du schreibst es vorne groß.`;
  return `Tipp: Das Adjektiv <b>${cur.first}</b> steht vorne. Das neue Wort ist ein Nomen. Du schreibst es zusammen und vorne groß.`;
}

function check() {
  if (state.busy || !cur) return;
  const v = input.value.trim().replace(/\s+/g, ' ');
  if (!v) { say('Schreibe zuerst das neue Wort mit Artikel in das Feld.'); input.focus(); return; }
  const m = v.match(/^(der|die|das) (.+)$/i);
  if (!m) { wobble(input); say('Schreibe auch den Artikel dazu: <b>der</b>, <b>die</b> oder <b>das</b>?', 'think'); input.focus(); return; }
  const art = m[1].toLowerCase(), noun = m[2];
  if (noun !== cur.word || art !== cur.art) {
    if (noun !== cur.word && noun.toLowerCase() === cur.word.toLowerCase()) {
      wobble(input);
      say('Fast richtig! Das neue Wort ist ein Nomen. Du schreibst es vorne groß und danach klein.', 'think');
    } else if (noun === cur.word) {
      miss(cur, input);
      say(`Das Wort stimmt! Aber der Artikel passt noch nicht. Das Grundwort ist <b>${cur.gw}</b>. Welchen Artikel hat es?`, 'think');
    } else {
      miss(cur, input);
      say(cur.misses === 1 ? 'Fast! Sprich das neue Wort langsam vor und probier es noch einmal.'
        : cur.misses === 2 ? ruleTip()
        : `So ist es richtig: <span class="pill ${cur.art}">${cur.art} ${cur.word}</span> Schreibe das Wort genau so ab.`, 'think');
    }
    input.focus();
    return;
  }
  state.busy = true;
  input.value = `${cur.art} ${cur.word}`; input.readOnly = true; input.className = 'good ' + cur.art;
  solve(cur, input, cur.art);
  say(`${pick(PRAISE, 'praise')} ${cur.first} + ${cur.gw} = <span class="pill ${cur.art}">${cur.art} ${cur.word}</span> Der Artikel kommt vom Grundwort: <b>${cur.art} ${cur.gw}</b>.`, 'happy');
  later(next, 2100);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.keycap');
  if (!k || input.readOnly) return;
  const a = input.selectionStart ?? input.value.length, b = input.selectionEnd ?? a;
  input.value = input.value.slice(0, a) + k.textContent + input.value.slice(b);
  input.focus();
  input.setSelectionRange(a + 1, a + 1);
});
