/* ---------- Spiel: Artikel-Chef ---------- */
let all = [], queue = [], cur = null;
const btn = a => $(`.art-btn[data-a="${a}"]`);

function startGame() {
  all = WORDS.map(w => {
    const [a1, bw] = w[0].split(' '), [art, gw] = w[1].split(' ');
    return { a1, bw, art, gw, emoji: w[2], word: bw + gw.toLowerCase(), misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welcher Artikel gehört zum neuen Wort? Schau auf das Grundwort. Es ist der Chef!');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.art-btn').forEach(b => b.classList.remove('hint'));
  $('#eq').innerHTML = `<div class="eqparts">
      <span class="pcard bw"><small>Bestimmungswort</small><span class="w">${cur.a1} ${cur.bw}</span></span>
      <span class="plus" aria-hidden="true">+</span>
      <span class="pcard gw chef" id="chef" style="--ac:var(--${cur.art})"><small>Grundwort</small><span class="w"><i>${cur.art}</i> ${cur.gw}</span></span>
    </div>
    <svg class="down" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="3" stroke-linecap="round" stroke-linejoin="round" aria-hidden="true"><path d="M12 4v15M5 13l7 7 7-7"/></svg>
    <p class="result"><span class="aslot" id="aslot">?</span><span>${cur.word}</span>${cur.emoji ? `<span class="pic" aria-hidden="true">${cur.emoji}</span>` : ''}</p>`;
}

function choose(a) {
  if (state.busy || !cur) return;
  if (a !== cur.art) {
    const hint = miss(cur, btn(a));
    if (hint) { btn(cur.art).classList.add('hint'); $('#chef').classList.add('hint'); }
    say(hint
      ? `Tipp: Das Grundwort steht hinten. Es heißt <b>${cur.art} ${cur.gw}</b>. Also heißt es auch <b>${cur.art}</b> ${cur.word}.`
      : a === cur.a1
        ? `Fast! <b>${a}</b> gehört zu ${cur.bw}. Das ist aber nur das Bestimmungswort. Schau auf das Grundwort!`
        : 'Hmm, das passt noch nicht. Schau auf das Grundwort. Es steht hinten.', 'think');
    return;
  }
  state.busy = true;
  $$('.art-btn').forEach(b => b.classList.remove('hint'));
  const chef = $('#chef'); chef.classList.remove('hint'); chef.classList.add('show');
  const slot = $('#aslot'); slot.textContent = cur.art; slot.className = 'aslot filled ' + cur.art;
  solve(cur, slot, cur.art);
  say(`${pick(PRAISE, 'praise')} Das Grundwort ist der Chef: <b>${cur.art} ${cur.gw}</b> → <span class="pill ${cur.art}">${cur.art} ${cur.word}</span>`, 'happy');
  later(next, 2000);
}

$('#arts').addEventListener('click', e => { const b = e.target.closest('.art-btn'); if (b) choose(b.dataset.a); });
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-3]$/.test(e.key)) return;
  choose(['der', 'die', 'das'][+e.key - 1]);
});
