@@tipp
Das <b>Grundwort</b> steht hinten. Es sagt, <b>was es ist</b>. Eine Milch<b>kuh</b> ist eine Kuh. Kuh<b>milch</b> ist Milch. Wenn du die Wörter umdrehst, bekommst du ein ganz anderes Wort!

/* ---------- Spiel: Wort-Dreher ---------- */
let pairs = [], queue = [], cur = null, slots = [null, null], tiles = [];
const SLOT = [['bw', 'vorne'], ['gw', 'hinten']];

function startGame() {
  const mk = (first, second, [art, fuge, riddle]) => ({ first, second, art, fuge, riddle, misses: 0,
    word: first + fuge + second.toLowerCase(),
    html: first + (fuge ? `<span class="fuge">${fuge}</span>` : '') + second.toLowerCase() });
  pairs = PAIRS.map(p => [mk(p[0], p[1], p[2]), mk(p[1], p[0], p[3])]);
  /* Erst von jedem Paar eine Aufgabe, dann die andere: So kommen die Partner nicht direkt hintereinander. */
  const order = shuffle(pairs.map(p => shuffle(p.slice())));
  queue = [...shuffle(order.map(p => p[0])), ...shuffle(order.map(p => p[1]))];
  resetState(queue.length);
  say('Was ist gesucht? Dieses Wort ist das Grundwort und kommt nach hinten. Tippe zuerst das Wort für vorne an.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; slots = [null, null];
  tiles = shuffle([cur.first, cur.second]);
  $('#riddle').textContent = cur.riddle;
  $('#tiles').innerHTML = tiles.map((t, i) => `<button class="tile" data-v="${t}"><b>${t}</b><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
  render();
}

function render() {
  $('#build').innerHTML = slots.map((v, i) =>
    `<span class="dslot ${SLOT[i][0]}${v ? ' full' : ''}" data-s="${i}">${v || '?'}<small>${SLOT[i][1]}</small></span>`
  ).join('<span class="plus" aria-hidden="true" style="align-self:center;color:var(--ink-soft)">+</span>');
  $$('.tile').forEach(t => t.classList.toggle('used', slots.includes(t.dataset.v)));
}

function place(v, s) {
  if (state.busy || !cur || slots.includes(v)) return;
  if (s === undefined) s = slots[0] === null ? 0 : 1;
  if (slots[s] !== null) return;
  slots[s] = v;
  sfx.pick();
  /* Ist ein Feld gefüllt, wandert das zweite Wort von selbst in das andere Feld */
  const other = tiles.find(t => t !== v);
  slots[1 - s] = other;
  render();
  check();
}

function check() {
  if (slots[0] !== cur.first) {
    state.busy = true;
    miss(cur, $('#build'));
    say(`Tipp: Das Grundwort sagt, <b>was es ist</b>. Gesucht ist hier <b>${cur.art} ${cur.second}</b>. Dieses Wort gehört nach hinten.`, 'think');
    later(() => {
      slots = [null, null]; state.busy = false; render();
      const t = $$('.tile').find(x => x.dataset.v === cur.first);
      if (t) t.classList.add('hint');
    }, 1500);
    return;
  }
  state.busy = true;
  $('#build').innerHTML = `<span class="built"><span class="pill ${cur.art}">${cur.art} ${cur.html}</span></span>`;
  solve(cur, $('#build .pill'), cur.art);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${cur.art}">${cur.art} ${cur.word}</span> Das Grundwort ist <b>${cur.second}</b>.` +
    (cur.fuge ? ` Zwischen den Wörtern steht hier noch <b>${cur.fuge}</b>.` : ''), 'happy');
  later(next, 2200);
}

$('#tiles').addEventListener('click', e => {
  if (suppressClick) return;
  const t = e.target.closest('.tile');
  if (t) place(t.dataset.v);
});
enableDrag('.tile:not(.used)', '.dslot', (tile, slot) => place(tile.dataset.v, +slot.dataset.s));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[12]$/.test(e.key)) return;
  place(tiles[+e.key - 1]);
});
