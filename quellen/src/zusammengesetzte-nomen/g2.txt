@@tipp
Ein zusammengesetztes Nomen besteht aus zwei Teilen. Hinten steht das <b>Grundwort</b>. Es sagt, was es ist: Eine Fischsuppe ist eine <b>Suppe</b>. Vorne steht das <b>Bestimmungswort</b>. Es sagt genauer, was für eine Suppe es ist: eine Suppe mit <b>Fisch</b>.

/* ---------- Spiel: Wort-Schere ---------- */
let all = [], queue = [], cur = null, pos = null, hintOn = false;

function startGame() {
  all = WORDS.map(w => ({ bw: w[0], gw: w[1], a1: w[2], art: w[3], e1: w[4], e2: w[5], word: w[0] + w[1].toLowerCase(), cut: w[0].length, misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Wo hört das erste Wort auf? Tippe auf den Buchstaben, mit dem das zweite Wort anfängt.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; pos = null; hintOn = false;
  $('.bench').classList.remove('locked');
  $('#cutword').classList.remove('split');
  $('#pics').innerHTML = `${cur.e1}<small>+</small>${cur.e2}`;
  render();
}

function render() {
  $('#cutword').innerHTML = [...cur.word].map((ch, i) =>
    (i === pos ? '<span class="cut"></span>' : '') +
    `<button class="ltr${i === 0 ? ' first' : ''}${hintOn && i === cur.cut ? ' hint' : ''}" data-i="${i}" aria-label="Buchstabe ${ch}${i ? ', hier schneiden' : ''}">${ch}</button>`).join('');
}

function move(to) {
  if (state.busy || !cur) return;
  pos = Math.max(1, Math.min(cur.word.length - 1, to));
  sfx.pick();
  render();
}

function check() {
  if (state.busy || !cur) return;
  if (pos === null) { say('Setz zuerst die Schere: Tippe auf den Buchstaben, mit dem das zweite Wort anfängt.'); return; }
  if (pos !== cur.cut) {
    const hint = miss(cur, $('#cutword'));
    if (hint) { hintOn = true; render(); }
    const a = cur.word.slice(0, pos), b = cur.word.slice(pos);
    say(hint
      ? 'Tipp: Schau auf die zwei Bilder. Das zweite Wort beginnt mit dem leuchtenden Buchstaben.'
      : `Hmm: „${a}“ und „${b}“ – sind das zwei Nomen? Schieb die Schere noch ein Stück.`, 'think');
    return;
  }
  state.busy = true;
  $('.bench').classList.add('locked');
  const cw = $('#cutword');
  cw.classList.add('split');
  cw.innerHTML = `<div class="parts"><span class="part bw"><b>${cur.bw}</b><small>Bestimmungswort</small></span><span class="plus">+</span>
    <span class="part gw"><b>${cur.gw}</b><small>Grundwort</small></span></div>`;
  solve(cur, cw, cur.art);
  say(`${pick(PRAISE, 'praise')} Im Wort stecken ${cur.a1} ${cur.bw} und <b>${cur.art} ${cur.gw}</b>: <span class="pill ${cur.art}">${cur.art} ${cur.word}</span> Das Grundwort ist <b>${cur.gw}</b>.`, 'happy');
  later(next, 2400);
}

$('#cutword').addEventListener('click', e => {
  const b = e.target.closest('.ltr');
  if (!b || state.busy) return;
  const i = +b.dataset.i;
  if (i === 0) { say('Vor dem ersten Buchstaben kannst du nicht schneiden. Wo fängt das zweite Wort an?'); return; }
  move(i);
});
$('#mv-left').addEventListener('click', () => move(pos === null ? cur.word.length - 1 : pos - 1));
$('#mv-right').addEventListener('click', () => move(pos === null ? 1 : pos + 1));
$('#check').addEventListener('click', check);
document.addEventListener('keydown', e => {
  if (!gameOn() || state.busy || !cur) return;
  if (e.key === 'ArrowLeft') { e.preventDefault(); move(pos === null ? cur.word.length - 1 : pos - 1); }
  else if (e.key === 'ArrowRight') { e.preventDefault(); move(pos === null ? 1 : pos + 1); }
  else if (e.key === 'Enter' && !e.target.closest('button')) { e.preventDefault(); check(); }
});
