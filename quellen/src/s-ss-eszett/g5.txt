@@tipp
Verwandte Wörter schreibt man nicht immer gleich. Hör auf den Vokal: <b>e</b>ssen und er <b>i</b>sst klingen kurz, also <b>ss</b>. Aber er <b>a</b>ß klingt lang, also <b>ß</b>. Bei einfachem <b>s</b> hilft die Grundform: er liest → le·sen. Dort summt das s.

/* ---------- Spiel: Wortfamilien-Werkstatt ---------- */
const KL = { s: 'ks', ss: 'kss', 'ß': 'ksz' }, COLOR = { s: 'der', ss: 'die', 'ß': 'das' };
let all = [], fams = [], fi = 0, cur = [];

function startGame() {
  all = [];
  fams = shuffle(FAMILIES.map(([head, note, forms]) => ({ head, note,
    forms: forms.map(f => { const w = { head, gap: f[0], l: f[1], tip: f[2], full: f[0].replace('_', f[1]), html: f[0].replace('_', `<b>${f[1]}</b>`), misses: 0, pick: null }; all.push(w); return w; }) })));
  resetState(all.length);
  fi = 0;
  show('game');
  startFamily();
  say('Sprich jede Form laut. Tippe neben jeder Form auf s, ss oder ß. Tippe dann auf „Prüfen“.');
}

function startFamily() {
  const f = fams[fi];
  cur = f.forms;
  state.busy = false;
  $('#round').textContent = `Wortfamilie ${fi + 1} von ${fams.length}`;
  $('#head').innerHTML = f.head + (f.note ? `<small>(${f.note})</small>` : '');
  $('#rows').innerHTML = cur.map((w, i) =>
    `<div class="frow" data-i="${i}"><span class="form">${w.gap.replace('_', '<span class="tmp">&nbsp;</span>')}</span>
      <span class="opts" role="group" aria-label="Was fehlt in ${w.gap.replace('_', ' Lücke ')}?">${LETTERS.map(([l, k]) =>
        `<button type="button" class="opt ${k}" data-l="${l}" aria-pressed="false">${l}</button>`).join('')}</span></div>`).join('');
}

function setPick(row, l) {
  const w = cur[+row.dataset.i];
  if (state.busy || w.done) return;
  sfx.pick();
  w.pick = l;
  $$('.opt', row).forEach(o => { o.setAttribute('aria-pressed', String(o.dataset.l === l)); o.classList.remove('hint'); });
  const tmp = $('.tmp', row);
  tmp.textContent = l; tmp.classList.add('set');
}

function check() {
  if (state.busy) return;
  const open = cur.filter(w => !w.done);
  if (open.some(w => !w.pick)) { wobble($('#check')); say('Fülle zuerst alle Lücken. Tippe neben jeder Form auf s, ss oder ß.', 'think'); return; }
  let wrong = [];
  cur.forEach((w, i) => {
    if (w.done) return;
    const row = $(`.frow[data-i="${i}"]`);
    if (w.pick === w.l) {
      row.classList.add('ok', KL[w.l]);
      $('.form', row).innerHTML = w.gap.replace('_', `<b class="hit ${KL[w.l]}">${w.l}</b>`);
      solve(w, row, COLOR[w.l]);
    } else {
      miss(w, row);
      w.pick = null;
      $$('.opt', row).forEach(o => o.setAttribute('aria-pressed', 'false'));
      const tmp = $('.tmp', row); tmp.innerHTML = '&nbsp;'; tmp.classList.remove('set');
      if (w.misses >= 2) $(`.opt[data-l="${w.l}"]`, row).classList.add('hint');
      wrong.push(w);
    }
  });
  if (wrong.length) {
    const tip = wrong.find(w => w.misses >= 2);
    say(tip ? `Tipp für <b>${tip.gap.replace('_', '<i class="gap"></i>')}</b>: ${tip.tip} Der leuchtende Baustein ist richtig.`
      : wrong.length === 1 ? 'Fast! Eine Lücke stimmt noch nicht. Sprich die Form laut: Ist der Vokal kurz oder lang? Summt es oder zischt es?'
      : `Fast! ${wrong.length} Lücken stimmen noch nicht. Sprich die Formen laut: Ist der Vokal kurz oder lang?`, 'think');
    return;
  }
  state.busy = true;
  const f = fams[fi];
  say(`${pick(PRAISE, 'praise')} <b>${f.head}</b>: ${f.forms.map(w => `<span class="pill ${KL[w.l]}">${w.full}</span>`).join(' ')}`, 'happy');
  if (fi < fams.length - 1) later(() => { fi++; startFamily(); }, 2600);
  else later(finish, 2200);
}

$('#rows').addEventListener('click', e => { const o = e.target.closest('.opt'); if (o) setPick(o.closest('.frow'), o.dataset.l); });
$('#check').addEventListener('click', check);
