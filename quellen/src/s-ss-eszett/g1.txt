/* ---------- Spiel: Kurz oder lang? ---------- */
const KL = { s: 'ks', ss: 'kss', 'ß': 'ksz' }, COLOR = { s: 'der', ss: 'die', 'ß': 'das' };
/* Vorlesen ist nur ein Zusatz: Ohne Sprachausgabe bleibt der Knopf versteckt. */
const canSpeak = (() => { try { return 'speechSynthesis' in window && typeof SpeechSynthesisUtterance === 'function'; } catch (e) { return false; } })();
function speak(text) {
  try { if (!canSpeak) return; speechSynthesis.cancel(); const u = new SpeechSynthesisUtterance(text); u.lang = 'de-DE'; u.rate = .8; speechSynthesis.speak(u); } catch (e) { /* ohne Vorlesen weiterspielen */ }
}
let all = [], queue = [], cur = null;
$('#listen').hidden = !canSpeak;

function startGame() {
  all = WORDS.map(w => {
    const v = w[0].match(/\[(.+?)\]/)[1], plain = w[0].replace(/[\[\]]/g, '');
    return { gap: w[0], l: w[1], emoji: w[2], v, zw: v.length === 2, full: plain.replace('_', w[1]),
      html: plain.replace('_', `<b>${w[1]}</b>`), misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Sprich das Wort laut. Klingt der unterstrichene Vokal kurz oder lang?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.pickbtn').forEach(b => b.classList.remove('hint'));
  $('#pic').textContent = cur.emoji;
  $('#word').innerHTML = cur.gap.replace(/\[(.+?)\]/, '<u>$1</u>').replace('_', '<span class="slot" id="slot">?</span>');
}

function choose(l) {
  if (state.busy || !cur) return;
  const btn = $(`.pickbtn[data-l="${l}"]`);
  if (l !== cur.l) {
    const hint = miss(cur, btn);
    if (hint) $(`.pickbtn[data-l="${cur.l}"]`).classList.add('hint');
    const why = cur.l === 'ss' ? `Das <b>${cur.v}</b> klingt kurz und knapp. Du kannst es nicht lang ziehen.`
      : cur.zw ? `<b>${cur.v}</b> ist ein Zwielaut. Er zählt wie ein langer Vokal.`
      : `Das <b>${cur.v}</b> kannst du lang ziehen.`;
    say(hint ? `Tipp: ${why} Der leuchtende Knopf ist richtig.`
      : cur.zw ? `Hmm, schau genau hin: Vor der Lücke steht <b>${cur.v}</b>. Nach au, ei und eu steht nie ss.`
      : `Hmm, sprich das Wort noch einmal laut. Kannst du das <b>${cur.v}</b> lang ziehen?`, 'think');
    return;
  }
  state.busy = true;
  $$('.pickbtn').forEach(b => b.classList.remove('hint'));
  const slot = $('#slot');
  slot.textContent = cur.l; slot.className = 'slot filled ' + KL[cur.l];
  solve(cur, slot, COLOR[cur.l]);
  const why = cur.l === 'ss' ? `Das <b>${cur.v}</b> klingt kurz. Also <b>ss</b>:`
    : cur.zw ? `Nach <b>${cur.v}</b> steht <b>ß</b>:` : `Das <b>${cur.v}</b> klingt lang. Also <b>ß</b>:`;
  say(`${pick(PRAISE, 'praise')} ${why} <span class="pill ${KL[cur.l]}">${cur.full}</span>`, 'happy');
  later(next, 2000);
}

$$('.pickbtn').forEach(b => b.addEventListener('click', () => choose(b.dataset.l)));
$('#listen').addEventListener('click', () => { if (cur) speak(cur.full); });
document.addEventListener('keydown', e => {
  if (!gameOn() || !cur) return;
  if (e.key === '1') choose('ss'); else if (e.key === '2') choose('ß');
});
