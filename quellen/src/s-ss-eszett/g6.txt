@@tipp
Prüfe das Wort in zwei Schritten. Erstens: Klingt der Vokal vor der Lücke <b>kurz</b>? Dann <b>ss</b>: hässlich. Zweitens: Ist er <b>lang</b> oder steht dort au, ei, eu? Dann sprich das Wort oder seine Verlängerung laut. <b>Zischt</b> es, schreibst du <b>ß</b>: außen. <b>Summt</b> es, schreibst du <b>s</b>: Beweis → Beweise.

/* ---------- Spiel: Schreib-Meister ---------- */
const GAP = '<i class="gap"></i>';
const KL = { s: 'ks', ss: 'kss', 'ß': 'ksz' }, COLOR = { s: 'der', ss: 'die', 'ß': 'das' };
let all = [], queue = [], cur = null;
const input = $('#answer');

function startGame() {
  all = SENTENCES.map(s => {
    const gapw = s[0].match(/\{(.+?)\}/)[1];
    return { sent: s[0], gapw, l: s[1], tip: s[2], right: gapw.replace('_', s[1]), html: gapw.replace('_', `<b>${s[1]}</b>`),
      variants: ['s', 'ss', 'ß'].map(x => gapw.replace('_', x).toLowerCase()), misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Sprich das gelbe Wort laut. Fehlt s, ss oder ß? Schreibe das ganze Wort in das Feld.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#sentence').innerHTML = cur.sent.replace(/\{(.+?)\}/, (m, w) => `<mark id="mark">${w.replace('_', GAP)}</mark>`);
  input.value = ''; input.readOnly = false; input.classList.remove('good');
  input.focus({ preventScroll: true });
}

function check() {
  if (state.busy || !cur) return;
  const v = input.value.trim().replace(/\s+/g, ' ');
  if (!v) { say('Schreibe zuerst das Wort in das Feld.'); input.focus(); return; }
  if (v !== cur.right) {
    if (v.toLowerCase() === cur.right.toLowerCase()) {
      wobble(input);
      say(/^[A-ZÄÖÜ]/.test(cur.right) ? 'Fast richtig! Denk daran: Nomen schreibt man groß.' : 'Fast richtig! Dieses Wort schreibt man klein.', 'think');
    } else {
      miss(cur, input);
      say(cur.misses === 1 ? (cur.variants.includes(v.toLowerCase())
          ? 'Fast! Der s-Laut stimmt noch nicht. Sprich das Wort laut: Ist der Vokal kurz oder lang? Summt es oder zischt es?'
          : 'Fast! Schau dir jeden Buchstaben genau an und probier es noch einmal.')
        : cur.misses === 2 ? `Tipp: ${cur.tip}`
        : `So ist es richtig: <span class="pill ${KL[cur.l]}">${cur.right}</span> Schreibe das Wort genau so ab.`, 'think');
    }
    input.focus();
    return;
  }
  state.busy = true;
  input.readOnly = true; input.classList.add('good');
  const mark = $('#mark');
  mark.className = 'fixed ' + KL[cur.l];
  mark.innerHTML = cur.gapw.replace('_', `<b class="hit ${KL[cur.l]}">${cur.l}</b>`);
  solve(cur, mark, COLOR[cur.l]);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${KL[cur.l]}">${cur.right}</span> mit <b>${cur.l}</b>.`, 'happy');
  later(next, 2000);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.keycap');
  if (!k || input.readOnly) return;
  const a = input.selectionStart ?? input.value.length, b = input.selectionEnd ?? a;
  input.value = input.value.slice(0, a) + k.textContent + input.value.slice(b);
  input.focus();
  input.setSelectionRange(a + 1, a + 1);
});
