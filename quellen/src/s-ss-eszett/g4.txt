/* ---------- Spiel: Lücken-Geschichte ---------- */
const GAP = '<i class="gap"></i>';
const KL = { s: 'ks', ss: 'kss', 'ß': 'ksz' }, COLOR = { s: 'der', ss: 'die', 'ß': 'das' };
const RULE = Object.fromEntries(LETTERS.map(b => [b[0], b[3]]));
const RX = /\[(ss|s|ß)\]/;
let all = [], stories = [], si = 0, gi = 0, cur = null;

$('#letters').innerHTML = LETTERS.map(([l, k], i) =>
  `<button class="tile ${k}" data-l="${l}" aria-label="${l}"><b>${l}</b><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');

function startGame() {
  all = [];
  stories = STORIES.map(([title, text]) => {
    const gaps = [];
    const html = text.split(' ').map(tok => {
      const m = tok.match(RX);
      if (!m) return tok;
      const [, word, punct] = tok.match(/^(.*?)([.,!?]*)$/);
      const w = { l: m[1], gapw: word.replace(RX, '_'), full: word.replace(RX, m[1]), html: word.replace(RX, `<b>${m[1]}</b>`), misses: 0 };
      gaps.push(w); all.push(w);
      w.id = 'g' + all.length;
      return `<span class="gw">${word.replace(RX, `<span class="sl" id="${w.id}">&nbsp;</span>`)}</span>${punct}`;
    }).join(' ');
    return { title, html, gaps };
  });
  resetState(all.length);
  si = 0;
  show('game');
  startStory();
  say('Lies die Geschichte. Welcher Buchstabe fehlt in der gelben Lücke: s, ss oder ß?');
}

function startStory() {
  const st = stories[si];
  gi = 0;
  $('#round').textContent = `Geschichte ${si + 1} von ${stories.length}`;
  $('#story-title').textContent = st.title;
  $('#story').innerHTML = st.html;
  nextGap();
}

function nextGap() {
  cur = stories[si].gaps[gi];
  state.busy = false;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#' + cur.id).classList.add('now');
}

function choose(l) {
  if (state.busy || !cur) return;
  const tile = $(`.tile[data-l="${l}"]`);
  if (l !== cur.l) {
    const hint = miss(cur, tile);
    if (hint) $(`.tile[data-l="${cur.l}"]`).classList.add('hint');
    say(hint ? `Tipp: ${RULE[cur.l]} Der leuchtende Baustein ist richtig.`
      : `Hmm, sprich <b>${cur.gapw.replace('_', GAP)}</b> noch einmal leise vor. Ist der Vokal kurz oder lang? Summt es oder zischt es?`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const sl = $('#' + cur.id);
  sl.textContent = cur.l; sl.className = 'sl filled ' + KL[cur.l];
  solve(cur, sl, COLOR[cur.l]);
  const why = l === 's' ? 'lang und summt' : l === 'ss' ? 'kurzer Vokal' : 'lang und zischt';
  say(`${pick(PRAISE, 'praise')} <span class="pill ${KL[l]}">${cur.full}</span> mit <b>${l}</b>: ${why}.`, 'happy');
  gi++;
  if (gi < stories[si].gaps.length) { later(nextGap, 900); return; }
  cur = null;
  if (si < stories.length - 1) later(() => { si++; startStory(); say('Geschichte geschafft! Hier kommt die nächste.', 'happy'); }, 2200);
  else later(finish, 1900);
}

$('#letters').addEventListener('click', e => { const t = e.target.closest('.tile'); if (t) choose(t.dataset.l); });
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-3]$/.test(e.key)) return;
  choose(LETTERS[+e.key - 1][0]);
});
