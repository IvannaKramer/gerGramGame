@@tipp
Sprich das Wort laut. Klingt der Vokal <b>kurz</b>? Dann fehlt <b>ss</b>: Schüssel. Klingt der Vokal <b>lang</b> und der s-Laut <b>zischt</b> scharf wie eine Schlange? Dann fehlt <b>ß</b>: grüßen. Klingt der Vokal lang und das s <b>summt</b> weich wie eine Biene? Dann fehlt <b>s</b>: Nase.

/* ---------- Spiel: Drei Körbe ---------- */
const GAP = '<i class="gap"></i>';
const KL = { s: 'ks', ss: 'kss', 'ß': 'ksz' }, COLOR = { s: 'der', ss: 'die', 'ß': 'das' };
const RULE = Object.fromEntries(BASKETS.map(b => [b[0], b[3]]));
/* Vorlesen ist nur ein Zusatz: Ohne Sprachausgabe gibt es den Knopf nicht. */
const canSpeak = (() => { try { return 'speechSynthesis' in window && typeof SpeechSynthesisUtterance === 'function'; } catch (e) { return false; } })();
function speak(text) {
  try { if (!canSpeak) return; speechSynthesis.cancel(); const u = new SpeechSynthesisUtterance(text); u.lang = 'de-DE'; u.rate = .8; speechSynthesis.speak(u); } catch (e) { /* ohne Vorlesen weiterspielen */ }
}
const listenBtn = w => canSpeak ? ` <button type="button" class="listen" data-id="${w.id}"><span aria-hidden="true">🔊</span> Anhören</button>` : '';
let items = [], selWord = null, selL = null;
const byId = id => items.find(w => w.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = l => $(`.basket[data-l="${l}"]`);

$('#baskets').innerHTML = BASKETS.map(([l, k, h], i) =>
  `<button class="basket ${k}" data-l="${l}" aria-pressed="false" aria-label="Korb ${l}"><b>${l}</b><small>${h}</small><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');

function startGame() {
  items = shuffle(WORDS.map((w, i) => ({ id: 'w' + i, gap: w[0], l: w[1], emoji: w[2], help: w[3], full: w[0].replace('_', w[1]),
    html: w[0].replace('_', `<b>${w[1]}</b>`), misses: 0, done: false })));
  resetState(items.length);
  selWord = selL = null;
  $('#words').innerHTML = items.map(w =>
    `<button class="word-card" data-id="${w.id}" aria-pressed="false" aria-label="${w.gap.replace('_', ' Lücke ')}" style="--tilt:${(Math.random() * 3 - 1.5).toFixed(2)}deg">
      <span class="pic" aria-hidden="true">${w.emoji}</span><span class="word">${w.gap.replace('_', GAP)}</span>
      <span class="long">${w.help}</span></button>`).join('');
  renderSelection();
  say('Tippe eine Karte an. Sprich das Wort laut und tippe dann auf den passenden Korb.');
  show('game');
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selWord;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.l === selL;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const w = selWord && byId(selWord);
  if (w && w.misses >= 2) basketEl(w.l).classList.add('hint');
}

function tapWord(id) {
  const w = byId(id);
  if (!w || w.done || state.busy) return;
  if (selL) { attempt(id, selL); return; }
  selWord = selWord === id ? null : id;
  sfx.pick(); renderSelection();
  if (selWord) say(`Sprich laut: <b>${w.gap.replace('_', GAP)}</b>. Ist der Vokal kurz oder lang? Summt das s oder zischt es?${listenBtn(w)}`);
}
function tapBasket(l) {
  if (state.busy) return;
  if (selWord) { attempt(selWord, l); return; }
  selL = selL === l ? null : l;
  sfx.pick(); renderSelection();
  if (selL) say(`Welche Karte gehört in den Korb <span class="pill ${KL[l]}">${l}</span>?`);
}

function attempt(id, l) {
  const w = byId(id);
  if (!w || w.done) return;
  if (w.l !== l) {
    miss(w, cardEl(id));
    wobble(basketEl(l));
    selWord = id; selL = null;
    renderSelection();
    say(w.misses >= 2
      ? `Tipp: ${RULE[w.l]} Der leuchtende Korb ist richtig.`
      : `Fast! Lies die kleine Hilfe unter dem Wort: <b>${w.help}</b>. Welcher Korb passt dazu?${listenBtn(w)}`, 'think');
    return;
  }
  const card = cardEl(id);
  selWord = selL = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(w, basketEl(l), COLOR[l]);
  say(`${pick(PRAISE, 'praise')} ${w.help}: <span class="pill ${KL[l]}">${w.full}</span> mit <b>${l}</b>.`, 'happy');
  if (state.solved === state.total) { state.busy = true; later(finish, 1700); }
}

$('#words').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapWord(c.dataset.id);
});
$$('.basket').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapBasket(b.dataset.l); }));
$('#bubble').addEventListener('click', e => { const b = e.target.closest('.listen'); const w = b && byId(b.dataset.id); if (w) speak(w.full); });
enableDrag('.word-card:not(.done)', '.basket', (card, basket) => attempt(card.dataset.id, basket.dataset.l));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-3]$/.test(e.key)) return;
  tapBasket(BASKETS[+e.key - 1][0]);
});
