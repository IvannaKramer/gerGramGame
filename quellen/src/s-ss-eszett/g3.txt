@@tipp
Am Wortende klingen s, ss und ß gleich. Der Trick: <b>Verlängere</b> das Wort und sprich es laut! <b>Summt</b> das s weich, schreibst du <b>s</b>: Gras → Gräser. <b>Zischt</b> es scharf nach kurzem Vokal, schreibst du <b>ss</b>: Fass → Fässer. Zischt es nach langem Vokal, schreibst du <b>ß</b>: Floß → Flöße.

/* ---------- Spiel: Verlängerungs-Zauber ---------- */
const GAP = '<i class="gap"></i>';
const KL = { s: 'ks', ss: 'kss', 'ß': 'ksz' }, COLOR = { s: 'der', ss: 'die', 'ß': 'das' };
const RULE = Object.fromEntries(LETTERS.map(b => [b[0], b[3]]));
const colorExt = w => w.ext.replace('_', `<b class="hit ${KL[w.l]}">${w.l}</b>`);
/* Vorlesen ist nur ein Zusatz: Ohne Sprachausgabe bleibt der Knopf versteckt. */
const canSpeak = (() => { try { return 'speechSynthesis' in window && typeof SpeechSynthesisUtterance === 'function'; } catch (e) { return false; } })();
function speak(text) {
  try { if (!canSpeak) return; speechSynthesis.cancel(); const u = new SpeechSynthesisUtterance(text); u.lang = 'de-DE'; u.rate = .8; speechSynthesis.speak(u); } catch (e) { /* ohne Vorlesen weiterspielen */ }
}
let all = [], queue = [], cur = null, longer = false;

$('#letters').innerHTML = LETTERS.map(([l, k], i) =>
  `<button class="tile wait ${k}" data-l="${l}" aria-label="${l}"><b>${l}</b><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');

function startGame() {
  all = WORDS.map(w => ({ gap: w[0], l: w[1], emoji: w[2], ext: w[3], full: w[0].replace('_', w[1]), extFull: w[3].replace('_', w[1]), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies das Wort. Tippe dann auf „Verlängern“ und sprich das lange Wort laut.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; longer = false;
  $('#pic').textContent = cur.emoji;
  $('#word').innerHTML = cur.gap.replace('_', '<span class="slot" id="slot">?</span>');
  $('#ext').innerHTML = '<span class="ph">Tippe unten auf „Verlängern“.</span>';
  $('#listen').hidden = true;
  $('#stretch').disabled = false;
  $$('.tile').forEach(t => { t.classList.add('wait'); t.classList.remove('hint'); });
}

function stretch() {
  if (state.busy || !cur || longer) return;
  longer = true;
  sfx.pick();
  $('#ext').innerHTML = cur.ext.replace('_', GAP);
  $('#listen').hidden = !canSpeak;
  $('#stretch').disabled = true;
  $$('.tile').forEach(t => t.classList.remove('wait'));
  say(`Sprich laut: <b>${cur.ext.replace('_', GAP)}</b>. Summt das s? Oder zischt es? Ist der Vokal davor kurz oder lang?`);
}

function choose(l) {
  if (state.busy || !cur) return;
  if (!longer) { wobble($('#stretch')); say('Erst verlängern! Tippe auf „Verlängern“. Dann hörst du den s-Laut besser.', 'think'); return; }
  const tile = $(`.tile[data-l="${l}"]`);
  if (l !== cur.l) {
    const hint = miss(cur, tile);
    if (hint) $(`.tile[data-l="${cur.l}"]`).classList.add('hint');
    say(hint ? `Tipp: ${RULE[cur.l]} Der leuchtende Baustein ist richtig.`
      : `Hmm, sprich das lange Wort noch einmal laut: <b>${cur.ext.replace('_', GAP)}</b>. Summt es oder zischt es?`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const slot = $('#slot');
  slot.textContent = cur.l; slot.className = 'slot filled ' + KL[cur.l];
  $('#ext').innerHTML = colorExt(cur);
  solve(cur, slot, COLOR[cur.l]);
  say(`${pick(PRAISE, 'praise')} ${colorExt(cur)} → <span class="pill ${KL[cur.l]}">${cur.full}</span> mit <b>${cur.l}</b>.`, 'happy');
  later(next, 2000);
}

$('#stretch').addEventListener('click', stretch);
$('#listen').addEventListener('click', () => { if (cur && longer) speak(cur.extFull); });
$('#letters').addEventListener('click', e => { const t = e.target.closest('.tile'); if (t) choose(t.dataset.l); });
document.addEventListener('keydown', e => {
  if (!gameOn() || !cur) return;
  if (e.key === 'v' || e.key === 'V') stretch();
  else if (/^[1-3]$/.test(e.key)) choose(LETTERS[+e.key - 1][0]);
});
