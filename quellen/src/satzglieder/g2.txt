/* ---------- Spiel: Welcher Satz stimmt? ---------- */
const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
const sentence = parts => cap(parts.join(' ')) + '.';
const TRY_V = ['Fast! Zähle die Satzglieder vor dem Verb. Es darf nur eins sein.', 'Hmm. Hier steht das Verb nicht an zweiter Stelle.'];
const TRY_S = ['Fast! Hier wurde ein Satzglied zerrissen. Lies den Satz laut.', 'Hmm. Hier stehen Wörter nicht mehr zusammen, die zusammengehören.'];
let all = [], queue = [], cur = null;

/* Alle Reihenfolgen der Satzglieder (als Listen von Platznummern) */
function perms(n) {
  if (n === 1) return [[0]];
  const out = [];
  perms(n - 1).forEach(p => { for (let i = 0; i <= p.length; i++) out.push([...p.slice(0, i), n - 1, ...p.slice(i)]); });
  return out;
}
/* Findet die Reihenfolge der Satzglieder in einem Satz; null, wenn ein Satzglied zerrissen ist */
const orderOf = (s, text) => perms(s.parts.length).find(o => sentence(o.map(i => s.parts[i])) === text) || null;
const pills = (s, ord) => ord.map((i, pos) => `<span class="pill ${i === 1 ? 'vb' : 'sg'}">${pos === 0 ? cap(s.parts[i]) : s.parts[i]}</span>`).join(' ') + '.';

function startGame() {
  all = SENTENCES.map(s => ({ parts: s[0], right: s[1], wrong: [s[2], s[3]], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies die drei Sätze genau. Nur einer ist richtig umgestellt. Tippe ihn an!');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#base').innerHTML = cur.parts.map((p, i) => `<span class="chip ${i === 1 ? 'vb' : 'sg'}">${i === 0 ? cap(p) : p}</span>`).join('') + '.';
  cur.opts = shuffle([cur.right, ...cur.wrong]);
  $('#opts').innerHTML = cur.opts.map((t, i) =>
    `<button class="opt" data-i="${i}">${t}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function choose(i) {
  const btn = $(`.opt[data-i="${i}"]`);
  if (state.busy || !cur || !btn || btn.disabled) return;
  const text = cur.opts[i];
  if (text !== cur.right) {
    const hint = miss(cur, btn);
    btn.classList.add('away'); btn.disabled = true;
    if (hint) {
      $$('.opt').find(b => cur.opts[+b.dataset.i] === cur.right).classList.add('hint');
      say(`Tipp: Das Verb <span class="pill vb">${cur.parts[1]}</span> steht an zweiter Stelle. Davor steht genau <b>ein</b> ganzes Satzglied.`, 'think');
    } else say(pick(orderOf(cur, text) ? TRY_V : TRY_S, 'try'), 'think');
    return;
  }
  state.busy = true;
  $$('.opt').forEach(b => { b.classList.remove('hint'); if (b !== btn) { b.classList.add('away'); b.disabled = true; } });
  btn.classList.add('good');
  solve(cur, btn, 'das');
  say(`${pick(PRAISE, 'praise')} ${pills(cur, orderOf(cur, text))} Das Verb steht an zweiter Stelle.`, 'happy');
  later(next, 2400);
}

$('#opts').addEventListener('click', e => {
  const b = e.target.closest('.opt');
  if (b) choose(+b.dataset.i);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-3]$/.test(e.key)) return;
  choose(+e.key - 1);
});
