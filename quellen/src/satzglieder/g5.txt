/* ---------- Spiel: Neuer Satzanfang ---------- */
const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
const sentence = parts => cap(parts.join(' ')) + '.';
let all = [], queue = [], cur = null, pool = [], placed = [];

function perms(a) {
  if (a.length <= 1) return [a];
  return a.flatMap((x, i) => perms([...a.slice(0, i), ...a.slice(i + 1)]).map(p => [x, ...p]));
}

function startGame() {
  all = SENTENCES.map(([parts, start]) => {
    const first = parts.indexOf(start), rest = parts.map((_, i) => i).filter(i => i !== 1 && i !== first);
    /* alle richtigen Fortsetzungen: zuerst das Verb, dann die übrigen Satzglieder in beliebiger Reihenfolge */
    const valid = perms(rest).map(p => [1, ...p].map(i => parts[i]).join(' '));
    return { parts, first, rest, valid, misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Tippe die Wörter in der richtigen Reihenfolge an. Denk daran: Nach dem ersten Satzglied kommt das Verb.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; placed = [];
  const words = [1, ...cur.rest].flatMap(i => cur.parts[i].split(' '));
  let order;
  do { order = shuffle(words.map((_, i) => i)); } while (words.length > 2 && order.every((v, i) => v === i));
  pool = order.map((wi, id) => ({ id, text: words[wi] }));
  $('#base').textContent = sentence(cur.parts);
  render();
}

function render(focusId) {
  $('#build').innerHTML = `<span class="start sg">${cap(cur.parts[cur.first])}</span>` +
    placed.map(id => `<button class="wtile" data-id="${id}" aria-label="${pool[id].text} zurücklegen">${pool[id].text}</button>`).join('') +
    (placed.length < pool.length ? '<span class="rest" aria-hidden="true">…</span>' : '<span>.</span>');
  $('#tiles').innerHTML = pool.map(w =>
    `<button class="wtile${placed.includes(w.id) ? ' used' : ''}" data-id="${w.id}"${placed.includes(w.id) ? ' disabled' : ''}>${w.text}</button>`).join('');
  if (focusId != null) { const t = $(`#tiles .wtile[data-id="${focusId}"]:not(.used)`) || $('#tiles .wtile:not(.used)') || $('#check'); t.focus(); }
}

function tapTile(id, back) {
  if (state.busy || !cur) return;
  if (back) placed = placed.filter(p => p !== id);
  else if (!placed.includes(id)) placed.push(id);
  sfx.pick(); render(id);
}

function check() {
  if (state.busy || !cur) return;
  if (placed.length < pool.length) { say('Benutze zuerst alle Wörter. Tippe sie der Reihe nach an.'); return; }
  const mine = placed.map(id => pool[id].text), verb = `<span class="pill vb">${cur.parts[1]}</span>`;
  if (!cur.valid.includes(mine.join(' '))) {
    const hint = miss(cur, $('#build'));
    if (hint) {
      say(`Tipp: Nach <b>${cap(cur.parts[cur.first])}</b> kommt das Verb ${verb}. Diese Wörter bleiben zusammen: ` +
        cur.rest.map(i => `<span class="pill sg">${cur.parts[i]}</span>`).join(' '), 'think');
      const v = $$('#build .wtile').find(b => b.textContent === cur.parts[1]);
      if (v) v.classList.add('hint');
    } else say(mine[0] !== cur.parts[1] ? `Fast! Nach dem ersten Satzglied kommt sofort das Verb. Wo steht ${verb} bei dir?`
      : 'Fast! Hier wurde ein Satzglied zerrissen. Welche Wörter gehören zusammen? Tippe ein Wort im Satz an, dann springt es zurück.', 'think');
    return;
  }
  state.busy = true;
  /* die gebaute Reihenfolge der Satzglieder wiederfinden */
  const ord = perms(cur.rest).find(p => [1, ...p].map(i => cur.parts[i]).join(' ') === mine.join(' '));
  $('#build').innerHTML = [cur.first, 1, ...ord].map((i, pos) => `<span class="chip ${i === 1 ? 'vb' : 'sg'}">${pos === 0 ? cap(cur.parts[i]) : cur.parts[i]}</span>`).join('') + '<span>.</span>';
  solve(cur, $('#build'), 'der');
  say(`${pick(PRAISE, 'praise')} ${sentence([cur.first, 1, ...ord].map(i => cur.parts[i]))} Das Verb ${verb} steht an zweiter Stelle.`, 'happy');
  later(next, 2400);
}

$('#tiles').addEventListener('click', e => {
  const t = e.target.closest('.wtile');
  if (t && !t.disabled) tapTile(+t.dataset.id, false);
});
$('#build').addEventListener('click', e => {
  const t = e.target.closest('.wtile');
  if (t) tapTile(+t.dataset.id, true);
});
$('#check').addEventListener('click', check);
document.addEventListener('keydown', e => {
  if (!gameOn() || e.key !== 'Backspace' || state.busy || !placed.length) return;
  e.preventDefault();
  tapTile(placed[placed.length - 1], true);
});
