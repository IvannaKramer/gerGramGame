@@tipp
Wörter, die beim Umstellen <b>zusammenbleiben</b>, sind zusammen <b>ein</b> Satzglied. Auch das <b>Verb</b> ist ein Satzglied. <b>Mia | tanzt | gern.</b> hat also 3 Satzglieder.

/* ---------- Spiel: Satzglieder zählen ---------- */
const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
const sentence = parts => cap(parts.join(' ')) + '.';
let all = [], queue = [], cur = null, variant = 0;
const numEl = n => $(`.num[data-n="${n}"]`);
const pills = s => s.parts.map((p, i) => `<span class="pill ${i === 1 ? 'vb' : 'sg'}">${i === 0 ? cap(p) : p}</span>`).join(' ') + '.';
const chips = s => s.parts.map((p, i) => `<span class="chip ${i === 1 ? 'vb' : 'sg'}">${i === 0 ? cap(p) : p}</span>`).join('') + '.';

$('#nums').innerHTML = [2, 3, 4, 5].map(n => `<button class="num" data-n="${n}" aria-label="${n} Satzglieder">${n}</button>`).join('');

function startGame() {
  all = SENTENCES.map(p => ({ parts: p, misses: 0 }));
  /* jede Umstellung: ein anderes Satzglied wandert nach vorn, das Verb bleibt an zweiter Stelle */
  all.forEach(s => {
    s.variants = s.parts.map((_, i) => i).filter(i => i > 1).map(first =>
      sentence([first, 1, ...s.parts.map((_, i) => i).filter(i => i !== 1 && i !== first)].map(i => s.parts[i])));
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Vergleiche die beiden Sätze. Welche Wörter bleiben zusammen? Zähle die Satzglieder!');
  show('game');
  next();
}

function showVariant() {
  const m = $('#moved'), has = cur.variants.length > 0;
  m.classList.toggle('none', !has);
  m.textContent = has ? cur.variants[variant % cur.variants.length] : 'Hier kannst du nichts umstellen. Das Verb muss an zweiter Stelle bleiben.';
  $('#again').hidden = cur.variants.length < 2;
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; variant = 0;
  $$('.num').forEach(b => b.classList.remove('hint'));
  $('#base').textContent = sentence(cur.parts);
  showVariant();
}

function choose(n) {
  if (state.busy || !cur) return;
  const right = cur.parts.length;
  if (n !== right) {
    const hint = miss(cur, numEl(n));
    if (hint) { numEl(right).classList.add('hint'); $('#base').innerHTML = chips(cur); }
    say(hint ? 'Tipp: Jedes Kästchen ist ein Satzglied. Auch das Verb zählt mit. Zähle die Kästchen!'
      : n < right ? 'Fast! Es sind mehr Satzglieder. Hast du das Verb mitgezählt?'
      : 'Fast! Es sind weniger. Wörter, die zusammenbleiben, zählen zusammen nur als ein Satzglied.', 'think');
    return;
  }
  state.busy = true;
  $$('.num').forEach(b => b.classList.remove('hint'));
  $('#base').innerHTML = chips(cur);
  solve(cur, numEl(n), 'sun');
  say(`${pick(PRAISE, 'praise')} Das sind <b>${n} Satzglieder</b>: ${pills(cur)}`, 'happy');
  later(next, 2400);
}

$('#nums').addEventListener('click', e => {
  const b = e.target.closest('.num');
  if (b) choose(+b.dataset.n);
});
$('#again').addEventListener('click', () => { if (state.busy || !cur) return; variant++; sfx.pick(); showVariant(); });
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[2-5]$/.test(e.key)) return;
  choose(+e.key);
});
