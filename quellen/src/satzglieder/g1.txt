@@tipp
Im Aussagesatz steht das <b>Verb</b> immer an <b>zweiter Stelle</b>. Die anderen Satzglieder dürfen ihre Plätze tauschen: <b>Der Hund bellt laut.</b> oder <b>Laut bellt der Hund.</b>

/* ---------- Spiel: Satz-Zug ---------- */
const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
const sentence = parts => cap(parts.join(' ')) + '.';
const TRY = ['Fast! Zähle nach: Auf welchem Platz steht das Verb?', 'Noch nicht ganz. Das pinke Verb braucht einen anderen Platz.', 'Guter Versuch! Schau auf den pinken Wagen.'];
let all = [], queue = [], cur = null, order = [], sel = null;

/* Eine andere richtige Reihenfolge als die übergebene: ein anderes Satzglied wandert nach vorn */
function altOrder(s, now) {
  const n = s.parts.length, first = now[0] === 2 ? (n > 3 ? 3 : 0) : 2;
  const rest = s.parts.map((_, i) => i).filter(i => i !== 1 && i !== first);
  return [first, 1, ...rest];
}
const pills = (s, ord) => ord.map((i, pos) => `<span class="pill ${i === 1 ? 'vb' : 'sg'}">${pos === 0 ? cap(s.parts[i]) : s.parts[i]}</span>`).join(' ') + '.';

function startGame() {
  all = SENTENCES.map(s => ({ emoji: s[0], parts: s.slice(1), misses: 0 }));
  /* zuerst die kurzen Sätze, dann die langen */
  queue = shuffle(all.filter(s => s.parts.length === 3)).concat(shuffle(all.filter(s => s.parts.length > 3)));
  resetState(all.length);
  say('Stelle die Wagen um. Tippe zwei Wagen an, dann tauschen sie die Plätze. Das <span class="pill vb">Verb</span> gehört auf Platz 2.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; sel = null;
  do { order = shuffle(cur.parts.map((_, i) => i)); } while (order[1] === 1);
  $('#pic').textContent = cur.emoji;
  $('#train').classList.remove('solved');
  render();
}

function render(focusPos) {
  $('#train').innerHTML = '<span class="loco" aria-hidden="true">🚂</span>' + order.map((pi, pos) =>
    `<button class="wagon ${pi === 1 ? 'vb' : 'sg'}${sel === pos ? ' selected' : ''}" data-pos="${pos}" aria-pressed="${sel === pos}" aria-label="Platz ${pos + 1}: ${cur.parts[pi]}${pi === 1 ? ' (Verb)' : ''}">` +
    `<span class="nr" aria-hidden="true">${pos + 1}</span>${pos === 0 ? cap(cur.parts[pi]) : cur.parts[pi]}</button>`).join('') + '<span class="stop" aria-hidden="true">.</span>';
  if (focusPos != null) { const w = $(`.wagon[data-pos="${focusPos}"]`); if (w) w.focus(); }
}

function swap(a, b, focus) {
  if (state.busy || a === b || a < 0 || b < 0 || a >= order.length || b >= order.length) return;
  [order[a], order[b]] = [order[b], order[a]];
  sel = null; sfx.pick();
  render(focus ? b : null);
}

function tap(pos) {
  if (state.busy) return;
  if (sel === null) {
    sel = pos; sfx.pick(); render(pos);
    say(`Tippe jetzt den Wagen an, mit dem <b>${cur.parts[order[pos]]}</b> den Platz tauschen soll.`);
  } else if (sel === pos) { sel = null; render(pos); }
  else swap(sel, pos, true);
}

function check() {
  if (state.busy || !cur) return;
  const vpos = order.indexOf(1), verb = `<span class="pill vb">${cur.parts[1]}</span>`;
  if (vpos !== 1) {
    sel = null; render();
    const hint = miss(cur, $('.wagon.vb'));
    if (hint) $('.wagon.vb').classList.add('hint');
    say(hint ? `Tipp: Das Verb ${verb} gehört auf <b>Platz 2</b>. Davor steht genau ein Satzglied.`
      : vpos === 0 ? `So beginnt eine Frage. Im Aussagesatz steht das Verb ${verb} an zweiter Stelle.`
      : pick(TRY, 'try'), 'think');
    return;
  }
  state.busy = true; sel = null; render();
  $('#train').classList.add('solved');
  solve(cur, $('#train'), 'die');
  say(`${pick(PRAISE, 'praise')} ${pills(cur, order)}<span class="also">So geht es auch: ${sentence(altOrder(cur, order).map(i => cur.parts[i]))}</span>`, 'happy');
  later(next, 2600);
}

$('#train').addEventListener('click', e => {
  if (suppressClick) return;
  const w = e.target.closest('.wagon');
  if (w) tap(+w.dataset.pos);
});
$('#train').addEventListener('keydown', e => {
  const w = e.target.closest('.wagon');
  if (!w || (e.key !== 'ArrowLeft' && e.key !== 'ArrowRight')) return;
  e.preventDefault();
  const pos = +w.dataset.pos;
  swap(pos, pos + (e.key === 'ArrowLeft' ? -1 : 1), true);
});
$('#check').addEventListener('click', check);
enableDrag('.wagon', '.wagon', (src, tgt) => swap(+src.dataset.pos, +tgt.dataset.pos, false));
