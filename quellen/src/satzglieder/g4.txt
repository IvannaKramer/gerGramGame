Mach im Kopf die <b>Umstellprobe</b>: Welche Wörter wandern nur gemeinsam nach vorn? Sie sind zusammen ein Satzglied. Das <b>Verb</b> ist immer ein eigenes Satzglied. Davor und dahinter kommt also ein Schnitt.

/* ---------- Spiel: Satz-Schere ---------- */
const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
const sentence = parts => cap(parts.join(' ')) + '.';
let all = [], queue = [], cur = null;
const gapEl = i => $(`.gap[data-i="${i}"]`);
const cuts = () => $$('.gap[aria-pressed="true"]').map(g => +g.dataset.i);

function startGame() {
  all = SENTENCES.map(p => {
    const words = p.join(' ').split(' ');
    /* Schnittstellen: nach dem wievielten Wort hört ein Satzglied auf? */
    const ends = []; let n = 0;
    p.slice(0, -1).forEach(part => { n += part.split(' ').length; ends.push(n - 1); });
    return { parts: p, words, ends, misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Wo hört ein Satzglied auf? Tippe auf die Lücken und schneide den Satz in Satzglieder.');
  show('game');
  next();
}

function showCount() {
  const n = cuts().length + 1;
  $('#count').textContent = n === 1 ? 'Noch kein Schnitt.' : `Du hast den Satz in ${n} Teile geschnitten.`;
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#cut').innerHTML = cur.words.map((w, i) => {
    const word = i === 0 ? cap(w) : w, last = i === cur.words.length - 1;
    return `<span class="unit"><span class="w">${word}${last ? '.' : ''}</span>` +
      (last ? '' : `<button class="gap" data-i="${i}" aria-pressed="false" aria-label="Schnitt nach ${word}"></button>`) + '</span>';
  }).join('');
  showCount();
}

function check() {
  if (state.busy || !cur) return;
  const mine = cuts();
  if (!mine.length) { say('Setze zuerst deine Schnitte. Tippe dazu auf die Lücken zwischen den Wörtern.'); return; }
  const wrong = mine.filter(i => !cur.ends.includes(i)), missing = cur.ends.filter(i => !mine.includes(i));
  if (wrong.length || missing.length) {
    const hint = miss(cur, $('#cut'));
    if (hint) {
      wrong.forEach(i => gapEl(i).setAttribute('aria-pressed', 'false'));
      missing.forEach(i => gapEl(i).classList.add('hint'));
      showCount();
      say(`Tipp: Das Verb <span class="pill vb">${cur.parts[1]}</span> ist ein eigenes Satzglied. ` +
        (missing.length ? 'An den leuchtenden Stellen fehlt noch ein Schnitt.' : 'Die falschen Schnitte habe ich weggenommen.'), 'think');
    } else say(wrong.length ? 'Fast! Ein Schnitt trennt Wörter, die zusammengehören. Mach die Umstellprobe!'
      : 'Fast! Da fehlt noch ein Schnitt. Welche Wörter können allein nach vorn wandern?', 'think');
    return;
  }
  state.busy = true;
  $('#cut').innerHTML = cur.parts.map((p, i) => `<span class="chip ${i === 1 ? 'vb' : 'sg'}">${i === 0 ? cap(p) : p}</span>`).join('') + '.';
  $('#count').textContent = `${cur.parts.length} Satzglieder`;
  solve(cur, $('#cut'), 'der');
  const n = cur.parts.length, probe = [n - 1, 1, 0, ...cur.parts.map((_, i) => i).filter(i => i > 1 && i < n - 1)];
  say(`${pick(PRAISE, 'praise')} Der Satz hat <b>${n} Satzglieder</b>.<span class="also">Umstellprobe: ${sentence(probe.map(i => cur.parts[i]))}</span>`, 'happy');
  later(next, 2600);
}

$('#cut').addEventListener('click', e => {
  const g = e.target.closest('.gap');
  if (!g || state.busy) return;
  g.setAttribute('aria-pressed', String(g.getAttribute('aria-pressed') !== 'true'));
  g.classList.remove('hint');
  sfx.pick(); showCount();
});
$('#check').addEventListener('click', check);
