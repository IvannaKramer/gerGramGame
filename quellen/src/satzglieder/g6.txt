/* ---------- Spiel: Satzanfang-Profi ---------- */
const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
const sentence = parts => cap(parts.join(' ')) + '.';
const EMPTY = 'Tippe eine Karte an. Dann steht hier der Satz mit dieser Wortgruppe am Anfang.';
let all = [], queue = [], cur = null;
const candEl = i => $(`.cand[data-i="${i}"]`);
const chosen = () => $$('.cand[aria-pressed="true"]').map(c => +c.dataset.i);

/* Umstellprobe: die Wortgruppe wandert nach vorn, dann das Verb, dann alle übrigen Wörter */
function probe(s, group) {
  const words = s.parts.join(' ').split(' '), g = group.split(' '), vi = s.parts[0].split(' ').length;
  let at = -1;
  for (let i = 0; i + g.length <= words.length && at < 0; i++) if (g.every((w, k) => words[i + k] === w)) at = i;
  const rest = words.filter((_, i) => i !== vi && (i < at || i >= at + g.length));
  return `<b>${cap(group)}</b> ${words[vi]} ${rest.join(' ')}.`;
}

function startGame() {
  all = SENTENCES.map(([parts, cands]) => ({ parts, cands, misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Tippe die Karten an und lies die Umstellprobe. Wähle alle Wortgruppen aus, bei denen der Satz richtig klingt.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  cur.order = shuffle(cur.cands.slice());
  cur.right = cur.order.map((c, i) => i).filter(i => i !== undefined && cur.parts.includes(cur.order[i]) && cur.order[i] !== cur.parts[1]);
  $('#base').textContent = sentence(cur.parts);
  const p = $('#probe'); p.textContent = EMPTY; p.classList.add('empty');
  $('#cands').innerHTML = cur.order.map((c, i) =>
    `<button class="cand" data-i="${i}" aria-pressed="false">${c}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function showProbe(i) {
  const p = $('#probe');
  p.innerHTML = probe(cur, cur.order[i]); p.classList.remove('empty');
}

function toggle(i) {
  const c = candEl(i);
  if (state.busy || !cur || !c) return;
  c.setAttribute('aria-pressed', String(c.getAttribute('aria-pressed') !== 'true'));
  c.classList.remove('hint');
  sfx.pick(); showProbe(i);
}

function check() {
  if (state.busy || !cur) return;
  const mine = chosen();
  if (!mine.length) { say('Wähle zuerst die Karten aus, die allein am Satzanfang stehen können.'); return; }
  const wrong = mine.filter(i => !cur.right.includes(i)), missing = cur.right.filter(i => !mine.includes(i));
  if (wrong.length || missing.length) {
    const hint = miss(cur, wrong.length ? candEl(wrong[0]) : $('#cands'));
    if (wrong.length) showProbe(wrong[0]);
    if (hint) {
      wrong.forEach(i => candEl(i).setAttribute('aria-pressed', 'false'));
      missing.forEach(i => candEl(i).classList.add('hint'));
      say(`Tipp: Ein ganzes Satzglied kann allein vor dem Verb <span class="pill vb">${cur.parts[1]}</span> stehen. ` +
        (missing.length ? 'Die leuchtenden Karten fehlen dir noch.' : 'Die falschen Karten habe ich abgewählt.'), 'think');
    } else say(wrong.length ? `Fast! Lies die Umstellprobe mit <b>${cur.order[wrong[0]]}</b> noch einmal laut. Klingt der Satz richtig?`
      : 'Fast! Alle ausgewählten Karten stimmen. Aber es gibt noch mehr ganze Satzglieder.', 'think');
    return;
  }
  state.busy = true;
  $$('.cand').forEach(c => { c.classList.remove('hint'); c.classList.add(c.getAttribute('aria-pressed') === 'true' ? 'good' : 'off'); c.setAttribute('aria-pressed', 'false'); });
  solve(cur, $('#cands'), 'der');
  say(`${pick(PRAISE, 'praise')} ` + cur.parts.map((p, i) => `<span class="pill ${i === 1 ? 'vb' : 'sg'}">${i === 0 ? cap(p) : p}</span>`).join(' ') + '.', 'happy');
  later(next, 2600);
}

$('#cands').addEventListener('click', e => {
  const c = e.target.closest('.cand');
  if (c) toggle(+c.dataset.i);
});
$('#check').addEventListener('click', check);
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-6]$/.test(e.key)) return;
  toggle(+e.key - 1);
});
