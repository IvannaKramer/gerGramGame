@@tipp
Das <b>Perfekt</b> hat zwei Teile: <b>hat</b> oder <b>ist</b> und das <b>Partizip</b> (meistens mit ge-). Die meisten Verben brauchen <b>hat</b>. Bewegt sich jemand von einem Ort zum anderen oder verändert sich etwas, heißt es <b>ist</b>: Lea ist gelaufen. Das Eis ist getaut.

/* ---------- Spiel: Perfekt-Baukasten ---------- */
let all = [], queue = [], cur = null;

function startGame() {
  all = SENTENCES.map(r => ({ who: r[0], verb: r[1], rest: r[2], aux: r[3], part: r[4], wrong: r[5], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Baue den Satz im Perfekt. Wähle zuerst <b>hat</b> oder <b>ist</b>, dann das Partizip.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  cur.auxOk = cur.partOk = false;
  $('#from').textContent = `${cur.who} ${cur.verb} ${cur.rest}.`;
  $('#build').innerHTML = `${cur.who} <span class="slot now" id="s1">?</span> ${cur.rest} <span class="slot" id="s2">?</span>.`;
  $('#aux').innerHTML = AUX.map((a, i) => `<button data-t="aux" data-v="${a}">${a}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
  $('#parts').innerHTML = shuffle([cur.part, ...cur.wrong]).map((p, i) => `<button data-t="part" data-v="${p}">${p}<span class="kbd" aria-hidden="true">${i + 3}</span></button>`).join('');
}

function choose(el) {
  if (!el || state.busy || el.disabled || !cur) return;
  const t = el.dataset.t, v = el.dataset.v, want = t === 'aux' ? cur.aux : cur.part;
  if (v !== want) {
    const hint = miss(cur, el);
    el.disabled = true; el.classList.add('no');
    if (hint) $(`#kit button[data-t="${t}"][data-v="${want}"]`).classList.add('hint');
    say(hint ? (t === 'aux' ? TIPS[cur.aux] : TIPS.part)
      : (t === 'aux' ? 'Fast! Sprich den Satz leise mit dem anderen Hilfsverb.' : 'Fast! Dieses Partizip klingt noch nicht richtig. Probier ein anderes.'), 'think');
    return;
  }
  /* Richtiger Teil: in die Lücke einbauen */
  $$(`#kit button[data-t="${t}"]`).forEach(b => { b.disabled = true; b.classList.remove('hint'); b.classList.add(b === el ? 'yes' : 'off'); });
  const slot = $(t === 'aux' ? '#s1' : '#s2');
  slot.textContent = v; slot.className = 'slot filled';
  if (t === 'aux') cur.auxOk = true; else cur.partOk = true;
  if (!(cur.auxOk && cur.partOk)) {
    sfx.pick();
    $(cur.auxOk ? '#s2' : '#s1').classList.add('now');
    say(t === 'aux' ? `Genau, es heißt <b>${v}</b>. Jetzt fehlt noch das Partizip.` : `Genau, <b>${v}</b> ist das Partizip. Heißt es hat oder ist?`);
    return;
  }
  state.busy = true;
  solve(cur, $('#build'), 'die');
  say(`${pick(PRAISE, 'praise')} ${cur.who} <span class="pill pf">${cur.aux}</span> ${cur.rest} <span class="pill pf">${cur.part}</span>.`, 'happy');
  later(next, 2200);
}

$('#kit').addEventListener('click', e => choose(e.target.closest('button')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-5]$/.test(e.key)) return;
  choose($$('#kit button')[+e.key - 1]);
});
