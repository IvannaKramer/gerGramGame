@@tipp
Im <b>Perfekt</b> und im <b>Futur</b> besteht das Verb aus <b>zwei Teilen</b>. Der erste Teil steht vorn: hat, ist oder wird. Der zweite Teil steht ganz <b>am Ende</b> vom Satz: Der Hund <b>hat</b> den Ball <b>geholt</b>.

      <div class="ex fu" style="--tilt:-.5deg"><div class="ex-head"><b>Futur</b><span>wird + Grundform</span></div><div class="ex-body">Ida <span class="a">wird</span> Tee <span class="a">kochen</span>.</div></div>
@@howto
        <li>Lies den Satz. Jedes Wort steht auf einem eigenen Kärtchen.</li>

        <li>Hast du beide Teile gefunden, zeigt dir die Farbe die Zeitform: pink für Perfekt, lila für Futur.</li>

/* ---------- Spiel: Zwei-Teile-Detektiv ---------- */
const MISS = ['Fast! Dieses Wort gehört nicht zum Verb. Probier ein anderes.', 'Hmm, das ist kein Teil vom Verb. Was tut hier jemand?', 'Guter Versuch! Schau mal ganz ans Ende vom Satz.'];
let all = [], queue = [], cur = null;

function startGame() {
  all = SENTENCES.map(([text, k, emoji]) => ({ k, emoji, misses: 0,
    words: text.replace(/\.$/, '').split(' ').map(w => w[0] === '*' ? { t: w.slice(1), v: true } : { t: w, v: false }) }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Das Verb hat hier zwei Teile. Tippe beide Teile an!');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  cur.words.forEach(w => { w.hit = false; });
  $('#pic').textContent = cur.emoji;
  $('#found').textContent = '';
  const toks = $('#toks');
  toks.className = 'toks';
  toks.innerHTML = cur.words.map((w, i) => `<button class="tok" data-i="${i}">${w.t}</button>`).join('') + '<span class="dot" aria-hidden="true">.</span>';
}

$('#toks').addEventListener('click', e => {
  const el = e.target.closest('.tok');
  if (!el || state.busy || el.disabled || el.classList.contains('hit')) return;
  const w = cur.words[+el.dataset.i];
  if (!w.v) {
    const hint = miss(cur, el);
    el.disabled = true; el.classList.add('no');
    if (hint) cur.words.forEach((x, i) => { if (x.v && !x.hit) $(`.tok[data-i="${i}"]`).classList.add('hint'); });
    say(hint ? TENSES[cur.k][2] : pick(MISS, 'miss'), 'think');
    return;
  }
  w.hit = true;
  el.classList.remove('hint'); el.classList.add('hit');
  const parts = cur.words.filter(x => x.v);
  if (parts.some(x => !x.hit)) {
    sfx.pick();
    say(`Genau, <b>${w.t}</b> gehört zum Verb. Wo ist der zweite Teil?`);
    return;
  }
  state.busy = true;
  const verb = parts.map(x => x.t).join(' … ');
  $$('.tok').forEach(t => t.classList.remove('hint'));
  $('#toks').classList.add('solved', cur.k);
  $('#found').innerHTML = `<span class="pill ${cur.k}">${verb}</span> = ${TENSES[cur.k][0]}`;
  solve(cur, el, TENSES[cur.k][1]);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${cur.k}">${verb}</span> ist ${TENSES[cur.k][0]}.`, 'happy');
  later(next, 2000);
});
