@@tipp
Ein Satz kann in jede Zeit reisen. Dabei ändert sich nur das <b>Verb</b>: Präsens: sie <b>lacht</b>. Präteritum: sie <b>lachte</b>. Perfekt: sie <b>hat gelacht</b>. Futur: sie <b>wird lachen</b>. Im Perfekt und im Futur rutscht der zweite Teil ans Satzende.

/* ---------- Spiel: Zeitmaschine ---------- */
const KEYS = Object.keys(TENSES);
const MISS = ['Fast! Diese Form gehört zu einer anderen Zeit.', 'Das passt noch nicht zum Reiseziel. Probier eine andere Form.', 'Guter Versuch! Schau noch einmal auf das Reiseziel.'];
let all = [], queue = [], cur = null;

/* Baut den Satz in einer Zeitform; mark = Verb hervorheben */
function sentence(s, k, mark) {
  const [a, b] = s.forms[k].split(' ');
  const m = t => mark ? `<b>${t}</b>` : t;
  return `${s.who} ${m(a)} ${s.rest}${b ? ' ' + m(b) : ''}.`;
}
const label = f => f.replace(' ', ' … ');

function startGame() {
  all = SENTENCES.map(r => ({ who: r[0], rest: r[1], forms: { pr: r[2], pt: r[3], pf: r[4], fu: r[5] }, from: r[6], to: r[7], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Wohin reist der Satz? Tippe auf die Verbform, die zum Reiseziel passt.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  const name = TENSES[cur.to][0];
  $('#src').textContent = sentence(cur, cur.from, false);
  $('#dest').className = 'dest ' + cur.to;
  $('#dest').innerHTML = `Reiseziel: <b>${name}</b>`;
  $('#outcard').className = 'tm-card out';
  $('#out').innerHTML = '<span class="q">?</span>';
  $('#opts').innerHTML = shuffle(KEYS.slice()).map((k, i) =>
    `<button class="opt" data-k="${k}">${label(cur.forms[k])}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function choose(el) {
  if (!el || state.busy || el.disabled || !cur) return;
  const k = el.dataset.k;
  if (k !== cur.to) {
    const hint = miss(cur, el);
    el.disabled = true; el.classList.add('no');
    if (hint) $(`.opt[data-k="${cur.to}"]`).classList.add('hint');
    say(hint ? TENSES[cur.to][2] : (k === cur.from ? 'So heißt der Satz jetzt schon. Er soll aber in eine andere Zeit reisen!' : pick(MISS, 'miss')), 'think');
    return;
  }
  state.busy = true;
  $$('.opt').forEach(o => { o.classList.remove('hint'); if (o !== el) { o.disabled = true; o.classList.add('no'); } });
  el.classList.add('yes', k);
  $('#outcard').classList.add('arrived', k);
  $('#out').innerHTML = sentence(cur, k, true);
  solve(cur, $('#outcard'), TENSES[k][1]);
  say(`${pick(PRAISE, 'praise')} Im ${TENSES[k][0]} heißt es: <span class="pill ${k}">${label(cur.forms[k])}</span>`, 'happy');
  later(next, 2100);
}

$('#opts').addEventListener('click', e => choose(e.target.closest('.opt')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-4]$/.test(e.key)) return;
  choose($$('.opt')[+e.key - 1]);
});
