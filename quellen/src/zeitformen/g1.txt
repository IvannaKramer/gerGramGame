@@tipp
Schau auf das <b>Verb</b>! Im Spiel ist es gelb markiert. Hat es nur <b>einen Teil</b>, ist es Präsens (er spielt) oder Präteritum (er spielte). Hat es <b>zwei Teile</b>, ist es Perfekt (er hat gespielt) oder Futur (er wird spielen).

@@howto
        <li>Lies den Satz. Das Verb ist gelb markiert. Wörter wie „gestern“, „jetzt“ oder „morgen“ helfen dir auch.</li>
        <li>Tippe auf den Korb mit der passenden Zeitform. Du kannst den Satz auch mit dem Finger in den Korb ziehen.</li>
        <li>Passt es, kommt der nächste Satz. Passt es nicht, probierst du einfach einen anderen Korb.</li>

/* ---------- Spiel: Zeit-Sortierer ---------- */
const MISS = ['Fast! Schau dir das gelbe Verb noch einmal an.', 'Das passt noch nicht. Hat das Verb einen Teil oder zwei Teile?', 'Guter Versuch! Probier einen anderen Korb.'];
let all = [], queue = [], cur = null;
const basketEl = k => $(`.basket[data-k="${k}"]`);

$('#baskets').innerHTML = Object.keys(TENSES).map((k, i) =>
  `<button class="basket ${k}" data-k="${k}" aria-label="${TENSES[k][0]}, zum Beispiel: ${TENSES[k][1]}"><b>${TENSES[k][0]}</b><span class="eg">${TENSES[k][1]}</span><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');

function startGame() {
  all = SENTENCES.map(([text, k, emoji]) => {
    const words = text.replace(/\.$/, '').split(' ');
    return { k, emoji, misses: 0,
      html: words.map(w => w[0] === '*' ? `<mark>${w.slice(1)}</mark>` : w).join(' ') + '.',
      verb: words.filter(w => w[0] === '*').map(w => w.slice(1)).join(' … ') };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('In welcher Zeitform steht der Satz? Schau auf das gelbe Verb und tippe auf den passenden Korb.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.basket').forEach(b => b.classList.remove('hint'));
  const card = $('#card');
  card.className = 'sent-card';
  card.innerHTML = `<span class="pic" aria-hidden="true">${cur.emoji}</span><span class="sent">${cur.html}</span>`;
}

function choose(k) {
  if (state.busy || !cur) return;
  if (k !== cur.k) {
    const hint = miss(cur, basketEl(k));
    if (hint) basketEl(cur.k).classList.add('hint');
    say(hint ? TENSES[cur.k][3] : pick(MISS, 'miss'), 'think');
    return;
  }
  state.busy = true;
  $$('.basket').forEach(b => b.classList.remove('hint'));
  $('#card').classList.add('solved', k);
  solve(cur, basketEl(k), TENSES[k][2]);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${k}">${cur.verb}</span> ist ${TENSES[k][0]}.`, 'happy');
  later(next, 1900);
}

$('#baskets').addEventListener('click', e => {
  if (suppressClick) return;
  const b = e.target.closest('.basket');
  if (b) choose(b.dataset.k);
});
enableDrag('.sent-card', '.basket', (card, basket) => choose(basket.dataset.k));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-4]$/.test(e.key)) return;
  choose(Object.keys(TENSES)[+e.key - 1]);
});
