@@tipp
Manche Verben verändern in der Vergangenheit ihren <b>Vokal</b>: singen – sang – hat gesungen. Das Partizip endet bei ihnen auf <b>-en</b>. Viele Verben folgen dem gleichen Muster. Kennst du eins, kennst du viele!

/* ---------- Spiel: Verb-Meister ---------- */
let all = [], queue = [], cur = null;
const in1 = $('#a1'), in2 = $('#a2');
let lastInput = in1;
const clean = el => el.value.trim().replace(/\s+/g, ' ').replace(/^(er|sie|es) /i, '');

function startGame() {
  all = VERBS.map(w => ({ gf: w[0], pt: w[1], pf: w[2], k: w[3], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Schreibe das Präteritum und das Perfekt. Überlege zuerst: Wie wechselt der Vokal?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  cur.ok1 = cur.ok2 = false;
  $('#from').textContent = cur.gf;
  [in1, in2].forEach(i => { i.value = ''; i.readOnly = false; i.classList.remove('good'); });
  lastInput = in1;
  in1.focus({ preventScroll: true });
}

function lock(input, text) { input.value = text; input.readOnly = true; input.classList.add('good'); }

function check() {
  if (state.busy || !cur) return;
  const v1 = clean(in1), v2 = clean(in2);
  if ((!cur.ok1 && !v1) || (!cur.ok2 && !v2)) {
    say('Fülle zuerst beide Felder aus: Präteritum und Perfekt.');
    (!cur.ok1 && !v1 ? in1 : in2).focus();
    return;
  }
  const part = cur.pf.split(' ')[1];
  let soft = '', wrong = [], auxOnly = false;
  if (!cur.ok1) {
    if (v1 === cur.pt) { cur.ok1 = true; lock(in1, cur.pt); }
    else if (v1.toLowerCase() === cur.pt) soft = 'Fast richtig! Denk daran: Verben schreibt man klein.';
    else wrong.push(in1);
  }
  if (!cur.ok2) {
    if (v2 === cur.pf) { cur.ok2 = true; lock(in2, cur.pf); }
    else if (v2.toLowerCase() === cur.pf) soft = 'Fast richtig! Denk daran: Verben schreibt man klein.';
    else if (v2.toLowerCase() === part) soft = 'Das Partizip stimmt! Im Perfekt fehlt aber noch <b>hat</b> oder <b>ist</b> davor.';
    else { wrong.push(in2); auxOnly = /^(hat|ist) /.test(v2) && v2.split(' ')[1] === part; }
  }
  if (cur.ok1 && cur.ok2) {
    state.busy = true;
    solve(cur, in2, 'die');
    say(`${pick(PRAISE, 'praise')} <b>${cur.gf}</b> – <span class="pill pt">${cur.pt}</span> – <span class="pill pf">${cur.pf}</span>`, 'happy');
    later(next, 2100);
    return;
  }
  if (!wrong.length) {
    wobble(cur.ok1 ? in2 : in1);
    say(soft, 'think');
    (cur.ok1 ? in2 : in1).focus();
    return;
  }
  miss(cur, wrong[0]);
  if (wrong.length > 1) wobble(wrong[1]);
  const which = wrong.length > 1 ? 'Beide Formen stimmen noch nicht.'
    : wrong[0] === in1 ? 'Das Perfekt passt. Schau dir das Präteritum noch einmal an.'
    : auxOnly ? 'Das Partizip stimmt schon. Aber heißt es <b>hat</b> oder <b>ist</b>?'
    : (cur.ok1 ? 'Das Präteritum stimmt schon. ' : '') + 'Schau dir das Perfekt noch einmal an.';
  say(cur.misses === 1 ? `Fast! ${which}`
    : cur.misses === 2 ? RULES[cur.k][1]
    : `So ist es richtig: <span class="pill pt">${cur.pt}</span> <span class="pill pf">${cur.pf}</span> Schreibe die Formen genau so ab.`, 'think');
  wrong[0].focus();
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
[in1, in2].forEach(i => i.addEventListener('focus', () => { lastInput = i; }));
$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.keycap');
  const input = lastInput;
  if (!k || input.readOnly) return;
  const a = input.selectionStart ?? input.value.length, b = input.selectionEnd ?? a;
  input.value = input.value.slice(0, a) + k.textContent + input.value.slice(b);
  input.focus();
  input.setSelectionRange(a + 1, a + 1);
});
