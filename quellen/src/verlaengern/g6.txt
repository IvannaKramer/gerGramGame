/* ---------- Spiel: Fehler-Detektiv ---------- */
const COLOR = { pb: 'der', pd: 'die', pg: 'das' };
const plainExt = w => w.ext.replace(/[\[\]]/g, '');
const colorExt = w => w.ext.replace(/\[(.)\]/, `<b class="hit ${w.pair}">$1</b>`);
let all = [], queue = [], cur = null, step = 1;
const input = $('#answer');

function startGame() {
  all = SENTENCES.map(s => {
    const l = (s[3].match(/\[(.)\]/) || [])[1];
    /* Stelle des geänderten Buchstabens im richtigen Wort */
    let pos = 0; while (pos < s[2].length && s[1][pos] === s[2][pos]) pos++;
    return { sent: s[0], wrong: s[1], right: s[2], ext: s[3], l, pos, pair: PAIR[l], misses: 0, wmiss: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  show('game');
  next();
}

function setStep(n) {
  step = n;
  $('#form').hidden = $('#keys').hidden = $('#check').hidden = n !== 2;
  $('#sentence').classList.toggle('locked', n !== 1);
  $('#lbl').textContent = n === 1 ? 'Welches Wort ist falsch geschrieben? Tippe es an.' : 'Schreibe das markierte Wort richtig auf.';
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#sentence').innerHTML = cur.sent.split(' ').map(t => {
    const [, w, punct] = t.match(/^(.*?)([.,!?]*)$/);
    return `<span class="tokw"><button type="button" class="tok" data-w="${w}">${w}</button>${punct}</span>`;
  }).join('');
  input.value = ''; input.readOnly = false; input.classList.remove('good');
  setStep(1);
  say('Ein Wort im Satz ist falsch geschrieben. Verlängere die Wörter im Kopf und tippe das Fehlerwort an.');
}

function tapWord(el) {
  if (state.busy || step !== 1 || el.disabled) return;
  if (el.dataset.w !== cur.wrong) {
    const hint = miss(cur, el);
    el.disabled = true; el.classList.add('limp');
    if (hint) $$('.tok').find(t => t.dataset.w === cur.wrong).classList.add('hint');
    say(hint ? `Tipp: Schau dir das leuchtende Wort an. Denk an <b>${plainExt(cur)}</b>.`
      : `<b>${el.dataset.w}</b> ist richtig geschrieben. Prüfe die Wörter mit b, p, d, t, g oder k: Welches kannst du verlängern?`, 'think');
    return;
  }
  sfx.pick();
  $$('.tok').forEach(t => { t.classList.remove('hint', 'limp'); t.disabled = true; });
  el.classList.add('found');
  setStep(2);
  say(`Gefunden! <b>${cur.wrong}</b> stimmt so nicht. Verlängere das Wort: Welchen Buchstaben hörst du? Schreibe es richtig auf.`, 'happy');
  input.focus({ preventScroll: true });
}

function check() {
  if (state.busy || !cur || step !== 2) return;
  const v = input.value.trim().replace(/\s+/g, ' ');
  if (!v) { say('Schreibe zuerst das Wort in das Feld.'); input.focus(); return; }
  if (v !== cur.right) {
    if (v.toLowerCase() === cur.right.toLowerCase()) {
      wobble(input);
      say(/^[A-ZÄÖÜ]/.test(cur.right) ? 'Fast richtig! Denk daran: Nomen schreibt man groß.' : 'Fast richtig! Dieses Wort schreibt man klein.', 'think');
    } else {
      miss(cur, input); cur.wmiss++;
      say(cur.wmiss === 1 ? (v.toLowerCase() === cur.wrong.toLowerCase()
          ? 'So stand es im Satz. Aber ein Buchstabe ist falsch. Verlängere das Wort!'
          : 'Fast! Schau dir jeden Buchstaben genau an und probier es noch einmal.')
        : cur.wmiss === 2 ? `Tipp: Verlängere! ${colorExt(cur)} – nach dem Punkt hörst du ein <b>${cur.l}</b>.`
        : `So ist es richtig: <span class="pill ${cur.pair}">${cur.right}</span> Schreibe das Wort genau so ab.`, 'think');
    }
    input.focus();
    return;
  }
  state.busy = true;
  input.readOnly = true; input.classList.add('good');
  const tok = $('.tok.found');
  tok.className = 'tok fixed ' + cur.pair;
  tok.innerHTML = `${cur.right.slice(0, cur.pos)}<b class="hit ${cur.pair}">${cur.right[cur.pos]}</b>${cur.right.slice(cur.pos + 1)}`;
  solve(cur, tok, COLOR[cur.pair]);
  say(`${pick(PRAISE, 'praise')} ${colorExt(cur)} → <span class="pill ${cur.pair}">${cur.right}</span> mit <b>${cur.l}</b>.`, 'happy');
  later(next, 2300);
}

$('#sentence').addEventListener('click', e => { const t = e.target.closest('.tok'); if (t) tapWord(t); });
$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.keycap');
  if (!k || input.readOnly) return;
  const a = input.selectionStart ?? input.value.length, b = input.selectionEnd ?? a;
  input.value = input.value.slice(0, a) + k.textContent + input.value.slice(b);
  input.focus();
  input.setSelectionRange(a + 1, a + 1);
});
