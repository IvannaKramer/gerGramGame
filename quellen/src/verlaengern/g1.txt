@@tipp
Am Wortende klingt <b>b</b> wie p, <b>d</b> wie t und <b>g</b> wie k. Der Trick: <b>Verlängere</b> das Wort und sprich es in Silben! Bei „Hun·de“ hörst du das <b>d</b> ganz deutlich. Also schreibst du auch „Hund“ mit <b>d</b>.

/* ---------- Spiel: Verlängerungs-Trick ---------- */
const pairOf = l => Object.keys(PAIRS).find(p => PAIRS[p].includes(l));
const COLOR = { pb: 'der', pd: 'die', pg: 'das' };
const plainExt = w => w.ext.replace(/[\[\]]/g, '');
const colorExt = w => w.ext.replace(/\[(.)\]/, `<b class="hit ${w.pair}">$1</b>`);
let all = [], queue = [], cur = null, longer = false;

function startGame() {
  all = WORDS.map(w => ({ gap: w[0], l: w[1], emoji: w[2], ext: w[3], pair: pairOf(w[1]), full: w[0].replace('_', w[1]), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies das Wort. Tippe dann auf „Verlängern“ und sprich das lange Wort laut.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; longer = false;
  $('#pic').textContent = cur.emoji;
  $('#word').innerHTML = cur.gap.replace('_', '<span class="slot" id="slot">?</span>');
  $('#ext').innerHTML = '<span class="ph">Tippe unten auf „Verlängern“.</span>';
  $('#stretch').disabled = false;
  $('#letters').innerHTML = PAIRS[cur.pair].slice(0, 2).map((l, i) =>
    `<button class="tile wait ${cur.pair}" data-l="${l}" aria-label="Buchstabe ${l}"><b>${l}</b><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function stretch() {
  if (state.busy || !cur || longer) return;
  longer = true;
  sfx.pick();
  $('#ext').textContent = plainExt(cur);
  $('#stretch').disabled = true;
  $$('.tile').forEach(t => t.classList.remove('wait'));
  say(`Sprich laut in Silben: <b>${plainExt(cur)}</b>. Welchen Buchstaben hörst du nach dem Punkt?`);
}

function choose(l) {
  if (state.busy || !cur) return;
  if (!longer) { wobble($('#stretch')); say('Erst verlängern! Tippe auf „Verlängern“. Dann hörst du den Buchstaben.', 'think'); return; }
  const tile = $(`.tile[data-l="${l}"]`);
  if (l !== cur.l) {
    const hint = miss(cur, tile);
    if (hint) $(`.tile[data-l="${cur.l}"]`).classList.add('hint');
    say(hint ? `Tipp: Sprich ganz deutlich <b>${plainExt(cur)}</b>. Nach dem Punkt hörst du ein <b>${cur.l}</b>. Der leuchtende Buchstabe ist richtig.`
      : `Hmm, hör noch einmal genau hin: <b>${plainExt(cur)}</b>.`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const slot = $('#slot');
  slot.textContent = cur.l; slot.className = 'slot filled ' + cur.pair;
  $('#ext').innerHTML = colorExt(cur);
  solve(cur, slot, COLOR[cur.pair]);
  say(`${pick(PRAISE, 'praise')} ${colorExt(cur)} → <span class="pill ${cur.pair}">${cur.full}</span> mit <b>${cur.l}</b>.`, 'happy');
  later(next, 2000);
}

$('#stretch').addEventListener('click', stretch);
$('#letters').addEventListener('click', e => { const t = e.target.closest('.tile'); if (t) choose(t.dataset.l); });
document.addEventListener('keydown', e => {
  if (!gameOn() || !cur) return;
  if (e.key === 'v' || e.key === 'V') stretch();
  else if (/^[12]$/.test(e.key)) choose(PAIRS[cur.pair][+e.key - 1]);
});
