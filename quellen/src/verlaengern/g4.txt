@@tipp
Eine echte <b>Verlängerung</b> hilft dir nur, wenn nach der Lücke ein Selbstlaut kommt (a, e, i, o, u, ä, ö, ü). Dann beginnt mit dem gesuchten Buchstaben eine neue Silbe und du hörst ihn deutlich: <b>ge·ben</b>. Bei „du gibst“ oder „er gab“ hörst du ihn nicht.

/* ---------- Spiel: Welches Wort hilft? ---------- */
const GAP = '<i class="gap"></i>';
const COLOR = { pb: 'der', pd: 'die', pg: 'das' };
const pairOf = l => Object.keys(PAIRS).find(p => PAIRS[p].includes(l));
const plainExt = w => w.ext.replace(/[\[\]]/g, '');
const gapExt = w => w.ext.replace(/\[.\]/, GAP);
const colorExt = w => w.ext.replace(/\[(.)\]/, `<b class="hit ${w.pair}">$1</b>`);
let all = [], queue = [], cur = null, step = 1, options = [];

function startGame() {
  all = WORDS.map(w => ({ gap: w[0], l: w[1], ext: w[2], wrong: w[3], pair: pairOf(w[1]), full: w[0].replace('_', w[1]), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welches der drei Wörter hilft dir? Such das Wort, bei dem nach der Lücke ein Selbstlaut kommt.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; step = 1;
  $('#word').innerHTML = cur.gap.replace('_', '<span class="slot" id="slot">?</span>');
  $('#ext').innerHTML = '<span class="ph">Wähle unten das Wort, das dir hilft.</span>';
  $('#ask').textContent = 'Welches Wort hilft dir?';
  /* Im ersten Schritt zeigen alle drei Wörter eine Lücke, damit die Schreibung nichts verrät */
  options = shuffle([{ ok: true, html: gapExt(cur).replace('·', ''), label: plainExt(cur).replace('·', '') },
    ...cur.wrong.map(o => ({ ok: false, html: o.replace('_', GAP), label: o }))]);
  $('#opts').className = 'opts words';
  $('#opts').innerHTML = options.map((o, i) =>
    `<button class="opt" data-i="${i}" aria-label="Wort ${i + 1}: ${o.label.replace('_', ' Lücke ')}">${o.html}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function chooseWord(i) {
  const o = options[i], el = $(`.opt[data-i="${i}"]`);
  if (!o || !el || el.disabled) return;
  if (!o.ok) {
    const hint = miss(cur, el);
    el.disabled = true; el.classList.add('limp');
    if (hint) $(`.opt[data-i="${options.findIndex(x => x.ok)}"]`).classList.add('hint');
    say(hint ? 'Tipp: Such das Wort, bei dem nach der Lücke ein Selbstlaut kommt. Das leuchtende Wort ist die echte Verlängerung.'
      : `Bei <b>${o.html}</b> steht die Lücke am Ende einer Silbe. Dort hörst du den Buchstaben nicht deutlich. Welches Wort hilft besser?`, 'think');
    return;
  }
  sfx.pick();
  step = 2;
  $('#ext').innerHTML = gapExt(cur);
  $('#ask').textContent = 'Welchen Buchstaben hörst du?';
  $('#opts').className = 'opts';
  $('#opts').innerHTML = PAIRS[cur.pair].slice(0, 2).map((l, k) =>
    `<button class="tile ${cur.pair}" data-l="${l}" aria-label="Buchstabe ${l}"><b>${l}</b><span class="kbd" aria-hidden="true">${k + 1}</span></button>`).join('');
  say(`Genau, dieses Wort hilft! Sprich es laut in Silben: <b>${gapExt(cur)}</b>. Welchen Buchstaben hörst du nach dem Punkt?`, 'happy');
}

function chooseLetter(l) {
  const tile = $(`.tile[data-l="${l}"]`);
  if (!tile) return;
  if (l !== cur.l) {
    const hint = miss(cur, tile);
    if (hint) $(`.tile[data-l="${cur.l}"]`).classList.add('hint');
    say(hint ? `Tipp: Sprich ganz deutlich <b>${plainExt(cur)}</b>. Nach dem Punkt hörst du ein <b>${cur.l}</b>. Der leuchtende Buchstabe ist richtig.`
      : 'Hmm, hör noch einmal genau hin. Sprich das lange Wort langsam in Silben.', 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const slot = $('#slot');
  slot.textContent = cur.l; slot.className = 'slot filled ' + cur.pair;
  $('#ext').innerHTML = colorExt(cur);
  solve(cur, slot, COLOR[cur.pair]);
  say(`${pick(PRAISE, 'praise')} ${colorExt(cur)} → <span class="pill ${cur.pair}">${cur.full}</span> mit <b>${cur.l}</b>.`, 'happy');
  later(next, 2000);
}

$('#opts').addEventListener('click', e => {
  if (state.busy || !cur) return;
  const o = e.target.closest('.opt'), t = e.target.closest('.tile');
  if (o) chooseWord(+o.dataset.i); else if (t) chooseLetter(t.dataset.l);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || state.busy || !cur || !/^[1-3]$/.test(e.key)) return;
  if (step === 1) chooseWord(+e.key - 1);
  else if (e.key !== '3') chooseLetter(PAIRS[cur.pair][+e.key - 1]);
});
