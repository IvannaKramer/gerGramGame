@@tipp
Auch mitten im Wort klingt <b>b</b> vor einem t wie p und <b>g</b> wie k: „er gibt“, „sie fragt“. Bilde die <b>Grundform</b> und sprich sie in Silben: ge·<b>b</b>en, fra·<b>g</b>en, hu·<b>p</b>en, den·<b>k</b>en. Jetzt hörst du den Buchstaben.

/* ---------- Spiel: Grundform-Trick ---------- */
const COLOR = { pb: 'der', pd: 'die', pg: 'das' };
const pairOf = l => LETTERS.find(x => x[0] === l)[1];
const plainExt = w => w.ext.replace(/[\[\]]/g, '');
const colorExt = w => w.ext.replace(/\[(.)\]/, `<b class="hit ${w.pair}">$1</b>`);
let all = [], queue = [], cur = null;
const tileEl = l => $(`.tile[data-l="${l}"]`);

$('#tiles').innerHTML = LETTERS.map(([l, p], i) =>
  `<button class="tile ${p}" data-l="${l}" aria-label="Buchstabe ${l}"><b>${l}</b><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');

function startGame() {
  all = SENTENCES.map(s => {
    const gapVerb = s[0].split(' ').find(x => x.includes('_')).replace(/[.,!?]/g, '');
    return { sent: s[0], l: s[1], ext: s[2], pair: pairOf(s[1]), gapVerb, verb: gapVerb.replace('_', s[1]), misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welcher Buchstabe fehlt im Verb? Denk an die Grundform und sprich sie in Silben.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#sentence').innerHTML = cur.sent.split(' ').map(x => x.includes('_')
    ? `<span class="verb">${x.replace('_', '<span class="slot" id="slot">?</span>')}</span>` : x).join(' ');
  $('#base').innerHTML = '&nbsp;';
}

function choose(l) {
  if (state.busy || !cur) return;
  if (l !== cur.l) {
    const hint = miss(cur, tileEl(l));
    if (hint) { tileEl(cur.l).classList.add('hint'); $('#base').innerHTML = `Grundform: <b>${plainExt(cur)}</b>`; }
    say(hint ? `Tipp: Die Grundform heißt <b>${plainExt(cur)}</b>. Nach dem Punkt hörst du ein <b>${cur.l}</b>. Der leuchtende Buchstabe ist richtig.`
      : pairOf(l) !== cur.pair ? 'Hmm, dieser Buchstabe klingt ganz anders. Lies das Verb noch einmal laut.'
      : `Fast! Wie heißt die Grundform von <b>${cur.gapVerb.replace('_', '…')}</b>? Sprich sie langsam in Silben.`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const slot = $('#slot');
  slot.textContent = cur.l; slot.className = 'slot filled ' + cur.pair;
  $('#base').innerHTML = `Grundform: <b>${colorExt(cur)}</b>`;
  solve(cur, slot, COLOR[cur.pair]);
  say(`${pick(PRAISE, 'praise')} ${colorExt(cur)} → <span class="pill ${cur.pair}">${cur.verb}</span> mit <b>${cur.l}</b>.`, 'happy');
  later(next, 2000);
}

$('#tiles').addEventListener('click', e => {
  if (suppressClick) return;
  const t = e.target.closest('.tile');
  if (t) choose(t.dataset.l);
});
enableDrag('.tile', '.bench', tile => choose(tile.dataset.l));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-4]$/.test(e.key)) return;
  choose(LETTERS[+e.key - 1][0]);
});
