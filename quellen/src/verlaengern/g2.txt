/* ---------- Spiel: Buchstaben-Körbe ---------- */
const GAP = '<i class="gap"></i>';
const COLOR = { pb: 'der', pd: 'die', pg: 'das' };
const pairOf = l => BASKETS.find(b => b[0] === l)[1];
const plainExt = w => w.ext.replace(/[\[\]]/g, '');
const colorExt = w => w.ext.replace(/\[(.)\]/, `<b class="hit ${w.pair}">$1</b>`);
let items = [], selWord = null, selL = null;
const byId = id => items.find(w => w.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = l => $(`.basket[data-l="${l}"]`);

$('#baskets').innerHTML = BASKETS.map(([l, p]) =>
  `<button class="basket ${p}" data-l="${l}" aria-pressed="false" aria-label="Korb ${l}"><b>${l}</b></button>`).join('');

function startGame() {
  items = shuffle(WORDS.map((w, i) => ({ id: 'w' + i, gap: w[0], l: w[1], emoji: w[2], ext: w[3], pair: pairOf(w[1]), full: w[0].replace('_', w[1]), misses: 0, done: false })));
  resetState(items.length);
  selWord = selL = null;
  $('#words').innerHTML = items.map(w =>
    `<button class="word-card" data-id="${w.id}" aria-pressed="false" aria-label="${w.gap.replace('_', ' Lücke ')}" style="--tilt:${(Math.random() * 3 - 1.5).toFixed(2)}deg">
      <span class="pic" aria-hidden="true">${w.emoji}</span><span class="word">${w.gap.replace('_', GAP)}</span>
      <span class="long">${w.ext.replace(/\[.\]/, GAP)}</span></button>`).join('');
  renderSelection();
  say('Tippe eine Karte an. Sprich die Verlängerung laut und tippe dann auf den passenden Korb.');
  show('game');
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selWord;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.l === selL;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const w = selWord && byId(selWord);
  if (w && w.misses >= 2) basketEl(w.l).classList.add('hint');
}

function tapWord(id) {
  const w = byId(id);
  if (!w || w.done || state.busy) return;
  if (selL) { attempt(id, selL); return; }
  selWord = selWord === id ? null : id;
  sfx.pick(); renderSelection();
  if (selWord) say(`Sprich die Verlängerung laut in Silben. Welcher Buchstabe fehlt bei <b>${w.gap.replace('_', GAP)}</b>?`);
}
function tapBasket(l) {
  if (state.busy) return;
  if (selWord) { attempt(selWord, l); return; }
  selL = selL === l ? null : l;
  sfx.pick(); renderSelection();
  if (selL) say(`Welche Karte gehört in den Korb <span class="pill ${pairOf(l)}">${l}</span>?`);
}

function attempt(id, l) {
  const w = byId(id);
  if (!w || w.done) return;
  if (w.l !== l) {
    miss(w, cardEl(id));
    wobble(basketEl(l));
    selWord = id; selL = null;
    renderSelection();
    say(w.misses >= 2
      ? `Tipp: Sprich ganz deutlich <b>${plainExt(w)}</b>. Nach dem Punkt hörst du ein <b>${w.l}</b>. Der leuchtende Korb ist richtig.`
      : pairOf(l) !== w.pair ? 'Hmm, dieser Buchstabe klingt ganz anders. Sprich die Verlängerung noch einmal laut.'
      : 'Fast! Sprich die Verlängerung langsam in Silben. Was hörst du nach dem Punkt?', 'think');
    return;
  }
  const card = cardEl(id);
  selWord = selL = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(w, basketEl(l), COLOR[w.pair]);
  say(`${pick(PRAISE, 'praise')} ${colorExt(w)} → <span class="pill ${w.pair}">${w.full}</span> mit <b>${w.l}</b>.`, 'happy');
  if (state.solved === state.total) { state.busy = true; later(finish, 1700); }
}

$('#words').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapWord(c.dataset.id);
});
$$('.basket').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapBasket(b.dataset.l); }));
enableDrag('.word-card:not(.done)', '.basket', (card, basket) => attempt(card.dataset.id, basket.dataset.l));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-6]$/.test(e.key)) return;
  tapBasket(BASKETS[+e.key - 1][0]);
});
