@@tipp
Jedes Wort mit Lücke hat einen Partner: seine <b>Verlängerung</b>. Bei Nomen ist das die Mehrzahl (Hand → Hän·de), bei Verben die Grundform (sie schreibt → schrei·ben). Adjektive kannst du steigern (klug → klü·ger).

@@howto
        <li>Tippe eine lila Karte an. Sie dreht sich um und zeigt ein Wort mit Lücke.</li>
        <li>Tippe dann eine türkise Karte an. Sie zeigt eine Verlängerung.</li>
        <li>Gehören beide zusammen? Dann bleibt das Paar offen und die Lücke wird gefüllt. Wenn nicht, drehen sich die Karten wieder um.</li>

/* ---------- Spiel: Verlängerungs-Memory ---------- */
const GAP = '<i class="gap"></i>';
const KIND = { kw: 'Wort mit Lücke', lw: 'Verlängerung' };
const BACK = { kw: 'Wort', lw: '↔' };
const plainExt = w => w.ext.replace(/[\[\]]/g, '');
const colorExt = w => w.ext.replace(/\[(.)\]/, `<b class="hit ${w.pair}">$1</b>`);
const colorFull = w => w.gap.replace('_', `<b class="hit ${w.pair}">${w.l}</b>`);
let rounds = [], round = 0, cards = [], open = [], tries = 0, left = 0;
const closedLabel = c => `Karte ${KIND[c.kind]}, verdeckt`;
const faceText = c => c.kind === 'kw' ? c.w.gap.replace('_', ' Lücke ') : plainExt(c.w);

function startGame() {
  const all = shuffle(WORDS.map((w, i) => ({ nr: i, gap: w[0], l: w[1], emoji: w[2], ext: w[3], pair: PAIR[w[1]], full: w[0].replace('_', w[1]), misses: 0 })));
  rounds = [];
  for (let i = 0; i < all.length; i += PAIRS_PER_ROUND) rounds.push(all.slice(i, i + PAIRS_PER_ROUND));
  resetState(all.length);
  round = 0; tries = 0;
  startRound();
  say('Dreh zwei Karten um: eine lila Karte und eine türkise. Passt die Verlängerung zum Wort?');
  show('game');
}

function startRound() {
  open = []; left = rounds[round].length;
  cards = shuffle(rounds[round].flatMap(w => [{ w, kind: 'kw' }, { w, kind: 'lw' }]));
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#memo').innerHTML = cards.map((c, i) => {
    const text = c.kind === 'kw' ? c.w.gap.replace('_', GAP) : plainExt(c.w);
    const pic = c.kind === 'kw' ? c.w.emoji : '↔';
    return `<button class="mcard ${c.kind}" data-i="${i}" data-p="${c.w.nr}" aria-label="${closedLabel(c)}">
      <span class="inner">
        <span class="back" aria-hidden="true"><b>${BACK[c.kind]}</b><small>${c.kind === 'kw' ? 'mit Lücke' : 'Verlängerung'}</small></span>
        <span class="face" aria-hidden="true"><span class="pic">${pic}</span><span class="word">${text}</span></span>
      </span></button>`;
  }).join('');
}

$('#memo').addEventListener('click', e => {
  const el = e.target.closest('.mcard');
  if (!el || state.busy) return;
  const c = cards[+el.dataset.i];
  if (c.open || c.found) return;
  sfx.pick();
  c.open = true;
  el.classList.add('open');
  el.setAttribute('aria-label', `${KIND[c.kind]}: ${faceText(c)}`);
  open.push({ c, el });
  if (open.length < 2) return;

  tries++;
  const [a, b] = open; open = [];
  state.busy = true;
  if (a.c.w === b.c.w) {
    const w = a.c.w;
    a.c.found = b.c.found = true;
    later(() => {
      [a, b].forEach(o => {
        o.el.classList.add('found');
        $('.word', o.el).innerHTML = o.c.kind === 'kw' ? colorFull(w) : colorExt(w);
        o.el.setAttribute('aria-label', `Gefunden: ${o.c.kind === 'kw' ? w.full : plainExt(w)}`);
      });
      solve(w, b.el, 'bonus');
      say(`${pick(PRAISE, 'praise')} ${colorExt(w)} → <span class="pill ${w.pair}">${w.full}</span> mit <b>${w.l}</b>.`, 'happy');
      state.busy = false;
      if (--left === 0) roundDone();
    }, 450);
  } else {
    const same = a.c.kind === b.c.kind;
    say(same
      ? 'Das sind zwei Karten mit derselben Farbe. Dreh eine lila und eine türkise Karte um!'
      : 'Das passt noch nicht. Merk dir, wo die Karten liegen!', 'think');
    later(() => {
      [a, b].forEach(o => { o.c.open = false; o.el.classList.remove('open'); o.el.setAttribute('aria-label', closedLabel(o.c)); });
      state.busy = false;
    }, 1700);
  }
});

function roundDone() {
  state.busy = true;
  if (round < rounds.length - 1) {
    later(() => { say('Runde geschafft! Hier kommen neue Karten.', 'happy'); round++; startRound(); state.busy = false; }, 1600);
  } else later(finish, 1300);
}
