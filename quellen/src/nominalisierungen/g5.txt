@@tipp
Nach <b>etwas, nichts, viel, wenig</b> wird ein Adjektiv zum Nomen: Es wird groß und bekommt meistens die Endung <b>-es</b> (etwas Neu<b>es</b>). Nach <b>alles</b> hat es die Endung <b>-e</b> (alles Gut<b>e</b>). Achtung, Falle: Kommt nach dem Adjektiv noch ein Nomen, bleibt es klein (etwas <b>warme</b> Milch).

/* ---------- Gemeinsame Helfer ---------- */
const cap = w => w[0].toUpperCase() + w.slice(1);
const low = w => w[0].toLowerCase() + w.slice(1);
const VERSTECKT = { beim: 'bei dem', zum: 'zu dem', im: 'in dem', vom: 'von dem', ans: 'an das', am: 'an dem' };
const MENGE = ['etwas', 'nichts', 'viel', 'wenig', 'alles'];
/* Erklärt ein Signalwort in einem kurzen Satz */
function sigText(sig) {
  const l = sig.toLowerCase();
  return VERSTECKT[l] ? `<span class="pill sig">${l}</span> ist ein versteckter Artikel (${l} = ${VERSTECKT[l]})`
    : MENGE.includes(l) ? `<span class="pill sig">${l}</span> ist ein Mengenwort`
    : `<span class="pill sig">${l}</span> ist ein Artikel`;
}
/* Zerlegt einen Satz in Wörter. Wörter in [Klammern] sind die fraglichen Wörter.
   Großes Wort: hint ist das Signalwort. Kleines Wort: hint ist die Erklärung. */
function make(text, hints) {
  const toks = text.split(' ').map(tok => {
    const m = tok.replace(/[\[\]]/g, '').match(/^(.*?)([.,!?:;]*)$/);
    return { w: m[1], post: m[2], target: tok.includes('[') };
  });
  const gaps = [];
  toks.forEach((t, ti) => {
    if (!t.target) return;
    const hint = hints[gaps.length] || '', big = t.w[0] !== t.w[0].toLowerCase();
    let si = -1;
    if (big) for (let i = ti - 1; i >= 0; i--) if (toks[i].w === hint) { si = i; break; }
    gaps.push({ ti, si, word: t.w, big, k: big ? 'gr' : 'kl', sig: big ? hint : '', why: big ? '' : hint });
    t.gap = gaps.length - 1;
  });
  return Object.assign({ text, toks, gaps, misses: 0, done: false }, gaps[0]);
}
/* Der ganze Satz in Farbe: Signalwort lila, großes Wort blau, kleines Wort türkis */
function full(s, tag = 'b', cls = 'w') {
  return s.toks.map((t, i) => {
    const g = s.gaps.find(x => x.ti === i), sg = s.gaps.some(x => x.si === i);
    return (g ? `<${tag} class="${cls} ${g.k}">${t.w}</${tag}>` : sg ? `<b class="w sig">${t.w}</b>` : t.w) + t.post;
  }).join(' ');
}

/* ---------- Spiel: Adjektiv-Werkstatt ---------- */
let all = [], queue = [], cur = null, opts = [];
const tileFor = v => $$('.tile').find(t => t.dataset.v === v);

function startGame() {
  all = SENTENCES.map(s => Object.assign(make(s[0], [s[1]]), { base: s[2], wrong: s[3] }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welche Form passt in die Lücke? Achte auf die Endung und auf groß oder klein.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#base').textContent = cur.base;
  $('#sentence').innerHTML = cur.toks.map((t, i) => (i === cur.ti ? '<span class="gap" id="gap">?</span>' : t.w) + t.post).join(' ');
  /* Vier Antworten: zwei Endungen, jede klein (türkis) und groß (blau) */
  const pair = shuffle([low(cur.word), low(cur.wrong)]);
  opts = [pair[0], pair[1], cap(pair[0]), cap(pair[1])];
  $('#tiles').innerHTML = opts.map((v, i) =>
    `<button class="tile ${i < 2 ? 'kl' : 'gr'}" data-v="${v}"><b>${v}</b><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function choose(v) {
  if (state.busy || !cur) return;
  if (v !== cur.word) {
    const hint = miss(cur, tileFor(v));
    if (hint) tileFor(cur.word).classList.add('hint');
    const sameEnding = low(v) === low(cur.word);
    say(hint
      ? (cur.big ? `Tipp: ${sigText(cur.sig)}. Danach kommt kein Nomen mehr. Also wird das Adjektiv selbst zum Nomen: groß mit der Endung <b>-${cur.word.slice(cur.base.replace(/e$/, '').length)}</b>.`
                 : `Tipp: ${cur.why} Also bleibt es klein.`)
      : sameEnding
        ? (cur.big ? 'Die Endung stimmt schon! Aber schau auf das Wort vor der Lücke: Ist es ein Signalwort?'
                   : 'Die Endung stimmt schon! Aber schau, was nach der Lücke kommt: Steht dort ein Nomen?')
        : 'Hmm, lies den Satz leise mit diesem Wort. Klingt die Endung richtig?', 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const gap = $('#gap');
  gap.textContent = cur.word; gap.className = 'gap filled ' + cur.k;
  solve(cur, gap, cur.big ? 'der' : 'teal');
  say(`${pick(PRAISE, 'praise')} ${full(cur)}<br>${cur.big ? 'Nach dem Mengenwort kommt kein Nomen mehr. Das Adjektiv wird selbst zum Nomen.' : cur.why}`, 'happy');
  later(next, 3200);
}

$('#tiles').addEventListener('click', e => {
  const t = e.target.closest('.tile');
  if (t) choose(t.dataset.v);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-4]$/.test(e.key)) return;
  choose(opts[+e.key - 1]);
});
