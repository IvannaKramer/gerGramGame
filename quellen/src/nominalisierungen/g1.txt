@@tipp
Ein <b>Signalwort</b> macht aus einem Verb oder Adjektiv ein <b>Nomen</b>. Dann schreibst du groß. Signalwörter sind: <b>das, ein</b> (Artikel), <b>beim, zum, im, vom, ans</b> (versteckte Artikel) und <b>etwas, nichts, viel, wenig, alles</b> (Mengenwörter).

/* ---------- Gemeinsame Helfer ---------- */
const cap = w => w[0].toUpperCase() + w.slice(1);
const low = w => w[0].toLowerCase() + w.slice(1);
const VERSTECKT = { beim: 'bei dem', zum: 'zu dem', im: 'in dem', vom: 'von dem', ans: 'an das', am: 'an dem' };
const MENGE = ['etwas', 'nichts', 'viel', 'wenig', 'alles'];
/* Erklärt ein Signalwort in einem kurzen Satz */
function sigText(sig) {
  const l = sig.toLowerCase();
  return VERSTECKT[l] ? `<span class="pill sig">${l}</span> ist ein versteckter Artikel (${l} = ${VERSTECKT[l]})`
    : MENGE.includes(l) ? `<span class="pill sig">${l}</span> ist ein Mengenwort`
    : `<span class="pill sig">${l}</span> ist ein Artikel`;
}
/* Zerlegt einen Satz in Wörter. Wörter in [Klammern] sind die fraglichen Wörter.
   Großes Wort: hint ist das Signalwort. Kleines Wort: hint ist die Erklärung. */
function make(text, hints) {
  const toks = text.split(' ').map(tok => {
    const m = tok.replace(/[\[\]]/g, '').match(/^(.*?)([.,!?:;]*)$/);
    return { w: m[1], post: m[2], target: tok.includes('[') };
  });
  const gaps = [];
  toks.forEach((t, ti) => {
    if (!t.target) return;
    const hint = hints[gaps.length] || '', big = t.w[0] !== t.w[0].toLowerCase();
    let si = -1;
    if (big) for (let i = ti - 1; i >= 0; i--) if (toks[i].w === hint) { si = i; break; }
    gaps.push({ ti, si, word: t.w, big, k: big ? 'gr' : 'kl', sig: big ? hint : '', why: big ? '' : hint });
    t.gap = gaps.length - 1;
  });
  return Object.assign({ text, toks, gaps, misses: 0, done: false }, gaps[0]);
}
/* Der ganze Satz in Farbe: Signalwort lila, großes Wort blau, kleines Wort türkis */
function full(s, tag = 'b', cls = 'w') {
  return s.toks.map((t, i) => {
    const g = s.gaps.find(x => x.ti === i), sg = s.gaps.some(x => x.si === i);
    return (g ? `<${tag} class="${cls} ${g.k}">${t.w}</${tag}>` : sg ? `<b class="w sig">${t.w}</b>` : t.w) + t.post;
  }).join(' ');
}

/* ---------- Spiel: Signalwörter-Jagd ---------- */
let all = [], queue = [], cur = null;
const tokEl = i => $(`.tok[data-i="${i}"]`);

function startGame() {
  all = SENTENCES.map(s => Object.assign(make(s[0], [s[1]]), { emoji: s[2] }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Schau auf das Wort direkt vor dem gelben Wort. Ist es ein Signalwort? Dann tippe es an.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#none').classList.remove('hint');
  $('#pic').textContent = cur.emoji;
  $('#sentence').innerHTML = cur.toks.map((t, i) => i === cur.ti
    ? `<span class="caps" id="target">${t.w.toUpperCase()}</span>${t.post}`
    : `<button class="tok" data-i="${i}">${t.w}</button>${t.post}`).join(' ');
}

function choose(i) {
  if (state.busy || !cur) return;
  const el = i < 0 ? $('#none') : tokEl(i);
  const right = cur.big ? i === cur.si : i < 0;
  const tw = `<b>${cur.word.toUpperCase()}</b>`, before = cur.toks[cur.ti - 1].w;
  if (!right) {
    const hint = miss(cur, el);
    if (hint) (cur.big ? tokEl(cur.si) : $('#none')).classList.add('hint');
    say(hint
      ? (cur.big ? `Tipp: Vor ${tw} steht „${before}“. Das ist ein Signalwort. Tippe es an!`
                 : `Tipp: Vor ${tw} steht „${before}“. Das ist kein Signalwort. Tippe unten auf <b>Kein Signalwort</b>.`)
      : i < 0 ? `Schau noch einmal hin: Welches Wort steht direkt vor ${tw}? Findest du es in der Liste?`
              : `Fast! Schau auf das Wort direkt vor ${tw}. Ist es ein Signalwort?`, 'think');
    return;
  }
  state.busy = true;
  $$('.hint').forEach(h => h.classList.remove('hint'));
  const t = $('#target');
  t.textContent = cur.word; t.className = 'gap filled ' + cur.k;
  if (cur.big) el.classList.add('found', 'sig');
  solve(cur, t, cur.big ? 'der' : 'teal');
  say(cur.big
    ? `${pick(PRAISE, 'praise')} ${sigText(cur.sig)}. Danach schreibst du groß: ${full(cur)}`
    : `${pick(PRAISE, 'praise')} „${before}“ ist kein Signalwort. Das Wort bleibt klein: ${full(cur)}`, 'happy');
  later(next, 2800);
}

$('#sentence').addEventListener('click', e => {
  const t = e.target.closest('.tok');
  if (t) { sfx.pick(); choose(+t.dataset.i); }
});
$('#none').addEventListener('click', () => choose(-1));
document.addEventListener('keydown', e => { if (gameOn() && e.key === '0') choose(-1); });
