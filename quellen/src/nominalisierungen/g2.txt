   Bei großen Wörtern steht das Signalwort direkt davor. Bei kleinen Wörtern bleibt das Signalwort leer.
   Immer 5 Karten bilden eine Runde. */

  ['Wir lachen beim [Turnen].',        'beim',  '🤸'],
  ['Ich höre das [Bellen].',           'das',   '🐕'],

const CARDS_PER_ROUND = 5;

const GAME = {

/* ---------- Gemeinsame Helfer ---------- */
const cap = w => w[0].toUpperCase() + w.slice(1);
const low = w => w[0].toLowerCase() + w.slice(1);
const VERSTECKT = { beim: 'bei dem', zum: 'zu dem', im: 'in dem', vom: 'von dem', ans: 'an das', am: 'an dem' };
const MENGE = ['etwas', 'nichts', 'viel', 'wenig', 'alles'];
/* Erklärt ein Signalwort in einem kurzen Satz */
function sigText(sig) {
  const l = sig.toLowerCase();
  return VERSTECKT[l] ? `<span class="pill sig">${l}</span> ist ein versteckter Artikel (${l} = ${VERSTECKT[l]})`
    : MENGE.includes(l) ? `<span class="pill sig">${l}</span> ist ein Mengenwort`
    : `<span class="pill sig">${l}</span> ist ein Artikel`;
}
/* Zerlegt einen Satz in Wörter. Wörter in [Klammern] sind die fraglichen Wörter.
   Großes Wort: hint ist das Signalwort. Kleines Wort: hint ist die Erklärung. */
function make(text, hints) {
  const toks = text.split(' ').map(tok => {
    const m = tok.replace(/[\[\]]/g, '').match(/^(.*?)([.,!?:;]*)$/);
    return { w: m[1], post: m[2], target: tok.includes('[') };
  });
  const gaps = [];
  toks.forEach((t, ti) => {
    if (!t.target) return;
    const hint = hints[gaps.length] || '', big = t.w[0] !== t.w[0].toLowerCase();
    let si = -1;
    if (big) for (let i = ti - 1; i >= 0; i--) if (toks[i].w === hint) { si = i; break; }
    gaps.push({ ti, si, word: t.w, big, k: big ? 'gr' : 'kl', sig: big ? hint : '', why: big ? '' : hint });
    t.gap = gaps.length - 1;
  });
  return Object.assign({ text, toks, gaps, misses: 0, done: false }, gaps[0]);
}
/* Der ganze Satz in Farbe: Signalwort lila, großes Wort blau, kleines Wort türkis */
function full(s, tag = 'b', cls = 'w') {
  return s.toks.map((t, i) => {
    const g = s.gaps.find(x => x.ti === i), sg = s.gaps.some(x => x.si === i);
    return (g ? `<${tag} class="${cls} ${g.k}">${t.w}</${tag}>` : sg ? `<b class="w sig">${t.w}</b>` : t.w) + t.post;
  }).join(' ');
}

/* ---------- Spiel: Zwei Körbe ---------- */
const COLOR = { gr: 'der', kl: 'teal' }, NAME = { gr: 'groß', kl: 'klein' };
let all = [], rounds = [], round = 0, left = 0, selCard = null, selKind = null;
const byId = id => all.find(s => s.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = k => $(`.basket[data-kind="${k}"]`);
const before = s => s.toks[s.ti - 1].w;
const tw = s => `<b>${s.word.toUpperCase()}</b>`;

function startGame() {
  all = SENTENCES.map((s, i) => Object.assign(make(s[0], [s[1]]), { id: 's' + i, emoji: s[2] }));
  /* In jeder Runde liegen Karten für beide Körbe */
  const pools = { gr: shuffle(all.filter(s => s.k === 'gr')), kl: shuffle(all.filter(s => s.k === 'kl')) };
  const n = Math.ceil(all.length / CARDS_PER_ROUND);
  rounds = Array.from({ length: n }, () => []);
  let r = 0;
  ['gr', 'kl'].forEach(k => pools[k].forEach(s => { rounds[r % n].push(s); r++; }));
  rounds.forEach(shuffle);
  resetState(all.length);
  round = 0;
  startRound();
  say('Tippe eine Karte an. Ich zeige dir dann, welches Wort vor dem gelben Wort steht.');
  show('game');
}

function startRound() {
  selCard = selKind = null; left = rounds[round].length;
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#words').innerHTML = rounds[round].map(s =>
    `<button class="word-card" data-id="${s.id}" aria-pressed="false" style="--tilt:${(Math.random() * 2 - 1).toFixed(2)}deg">
      <span class="pic" aria-hidden="true">${s.emoji}</span><span class="word">${s.toks.map((t, i) =>
        (i === s.ti ? `<span class="caps">${t.w.toUpperCase()}</span>` : t.w) + t.post).join(' ')}</span></button>`).join('');
  renderSelection();
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selCard;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.kind === selKind;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const s = selCard && byId(selCard);
  if (s && s.misses >= 2) basketEl(s.k).classList.add('hint');
}

function tapCard(id) {
  const s = byId(id);
  if (!s || s.done || state.busy) return;
  if (selKind) { attempt(id, selKind); return; }
  selCard = selCard === id ? null : id;
  sfx.pick(); renderSelection();
  if (selCard) say(`Vor ${tw(s)} steht „${before(s)}“. Ist das ein Signalwort?`);
}
function tapBasket(k) {
  if (state.busy) return;
  if (selCard) { attempt(selCard, k); return; }
  selKind = selKind === k ? null : k;
  sfx.pick(); renderSelection();
  if (selKind) say(`Welcher Satz gehört in den Korb <span class="pill ${k}">${NAME[k]}</span>?`);
}

function attempt(id, k) {
  const s = byId(id);
  if (!s || s.done) return;
  if (s.k !== k) {
    miss(s, cardEl(id));
    wobble(basketEl(k));
    selCard = id; selKind = null;
    renderSelection();
    say(s.misses < 2 ? `Fast! Schau noch einmal auf „${before(s)}“. Ist das ein Signalwort?`
      : s.big ? `Tipp: ${sigText(s.sig)}. Danach schreibst du <b>groß</b>.`
              : `Tipp: „${before(s)}“ ist kein Signalwort. Das Wort bleibt <b>klein</b>.`, 'think');
    return;
  }
  const card = cardEl(id);
  selCard = selKind = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(s, basketEl(k), COLOR[k]);
  say(`${pick(PRAISE, 'praise')} ${s.big ? 'Groß nach dem Signalwort:' : 'Kein Signalwort, also klein:'} ${full(s)}`, 'happy');
  if (--left === 0) {
    state.busy = true;
    if (round < rounds.length - 1) later(() => { round++; startRound(); state.busy = false; say('Runde geschafft! Hier kommen neue Karten.', 'happy'); }, 2200);
    else later(finish, 2000);
  }
}

$('#words').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapCard(c.dataset.id);
});
$$('.basket').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapBasket(b.dataset.kind); }));
enableDrag('.word-card:not(.done)', '.basket', (card, basket) => attempt(card.dataset.id, basket.dataset.kind));
