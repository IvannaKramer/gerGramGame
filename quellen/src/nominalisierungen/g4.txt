/* ---------- Gemeinsame Helfer ---------- */
const cap = w => w[0].toUpperCase() + w.slice(1);
const low = w => w[0].toLowerCase() + w.slice(1);
const VERSTECKT = { beim: 'bei dem', zum: 'zu dem', im: 'in dem', vom: 'von dem', ans: 'an das', am: 'an dem' };
const MENGE = ['etwas', 'nichts', 'viel', 'wenig', 'alles'];
/* Erklärt ein Signalwort in einem kurzen Satz */
function sigText(sig) {
  const l = sig.toLowerCase();
  return VERSTECKT[l] ? `<span class="pill sig">${l}</span> ist ein versteckter Artikel (${l} = ${VERSTECKT[l]})`
    : MENGE.includes(l) ? `<span class="pill sig">${l}</span> ist ein Mengenwort`
    : `<span class="pill sig">${l}</span> ist ein Artikel`;
}
/* Zerlegt einen Satz in Wörter. Wörter in [Klammern] sind die fraglichen Wörter.
   Großes Wort: hint ist das Signalwort. Kleines Wort: hint ist die Erklärung. */
function make(text, hints) {
  const toks = text.split(' ').map(tok => {
    const m = tok.replace(/[\[\]]/g, '').match(/^(.*?)([.,!?:;]*)$/);
    return { w: m[1], post: m[2], target: tok.includes('[') };
  });
  const gaps = [];
  toks.forEach((t, ti) => {
    if (!t.target) return;
    const hint = hints[gaps.length] || '', big = t.w[0] !== t.w[0].toLowerCase();
    let si = -1;
    if (big) for (let i = ti - 1; i >= 0; i--) if (toks[i].w === hint) { si = i; break; }
    gaps.push({ ti, si, word: t.w, big, k: big ? 'gr' : 'kl', sig: big ? hint : '', why: big ? '' : hint });
    t.gap = gaps.length - 1;
  });
  return Object.assign({ text, toks, gaps, misses: 0, done: false }, gaps[0]);
}
/* Der ganze Satz in Farbe: Signalwort lila, großes Wort blau, kleines Wort türkis */
function full(s, tag = 'b', cls = 'w') {
  return s.toks.map((t, i) => {
    const g = s.gaps.find(x => x.ti === i), sg = s.gaps.some(x => x.si === i);
    return (g ? `<${tag} class="${cls} ${g.k}">${t.w}</${tag}>` : sg ? `<b class="w sig">${t.w}</b>` : t.w) + t.post;
  }).join(' ');
}

/* ---------- Spiel: Fehler-Detektiv ---------- */
let all = [], queue = [], cur = null;
const tokEl = i => $(`.tok[data-i="${i}"]`);
const pillW = s => `<span class="pill ${s.k}">${s.word}</span>`;
/* Warum ist das Wort groß oder klein? */
const reason = s => s.big ? `${sigText(s.sig)} und gehört zu ${pillW(s)}. Also groß!` : s.why;

function startGame() {
  all = SENTENCES.map(s => Object.assign(make(s[0], [s[1]]), { err: s[2] === 'falsch' }));
  all.forEach(s => { s.shown = s.err ? (s.big ? low(s.word) : cap(s.word)) : s.word; });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies den Satz genau. Ist ein Wort falsch geschrieben? Tippe es an. Sonst tippe auf „Alles richtig!“.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#ok').classList.remove('hint');
  const sen = $('#sentence');
  sen.classList.remove('solved');
  sen.innerHTML = cur.toks.map((t, i) => `<button class="tok" data-i="${i}">${i === cur.ti ? cur.shown : t.w}</button>${t.post}`).join(' ');
}

function choose(i) {
  if (state.busy || !cur) return;
  const el = i < 0 ? $('#ok') : tokEl(i);
  const right = cur.err ? i === cur.ti : i < 0;
  if (!right) {
    const hint = miss(cur, el);
    if (hint) (cur.err ? tokEl(cur.ti) : $('#ok')).classList.add('hint');
    say(hint
      ? (cur.err ? `Tipp: Schau dir „${cur.shown}“ an. ${cur.big ? `Dazu gehört das Signalwort „${cur.sig}“.` : cur.why}`
                 : `Tipp: Hier ist alles richtig. ${cur.big ? `Zu „${cur.word}“ gehört das Signalwort „${cur.sig}“.` : cur.why}`)
      : i < 0 ? 'Hier hat sich ein Fehler versteckt. Suche die Verben und Adjektive: Gehört ein Signalwort dazu?'
              : 'Dieses Wort ist richtig geschrieben. Suche die Verben und Adjektive: Gehört ein Signalwort dazu?', 'think');
    return;
  }
  state.busy = true;
  $$('.hint').forEach(h => h.classList.remove('hint'));
  const sen = $('#sentence');
  sen.classList.add('solved');
  sen.innerHTML = full(cur);
  solve(cur, $('.w.' + cur.k, sen), cur.big ? 'der' : 'teal');
  say(cur.err
    ? `${pick(PRAISE, 'praise')} Richtig heißt es ${pillW(cur)}. ${reason(cur)}`
    : `${pick(PRAISE, 'praise')} Hier ist alles richtig. ${reason(cur)}`, 'happy');
  later(next, 3400);
}

  const t = e.target.closest('.tok');
  if (t) { sfx.pick(); choose(+t.dataset.i); }
});
$('#ok').addEventListener('click', () => choose(-1));
document.addEventListener('keydown', e => { if (gameOn() && e.key === '0') choose(-1); });
