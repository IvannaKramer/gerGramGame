@@tipp
Ein Signalwort allein reicht nicht – es muss zu dem Wort <b>gehören</b>. Zwischen Signalwort und Nomen darf ein Adjektiv stehen: <b>das</b> laute <b>Bellen</b>. Kommt danach aber noch ein anderes Nomen, bleibt das Adjektiv klein: das <b>schöne</b> Haus. Und Verben bleiben klein, wenn sie sagen, was jemand tut: Ich will etwas <b>trinken</b>.

/* ---------- Gemeinsame Helfer ---------- */
const cap = w => w[0].toUpperCase() + w.slice(1);
const low = w => w[0].toLowerCase() + w.slice(1);
const VERSTECKT = { beim: 'bei dem', zum: 'zu dem', im: 'in dem', vom: 'von dem', ans: 'an das', am: 'an dem' };
const MENGE = ['etwas', 'nichts', 'viel', 'wenig', 'alles'];
/* Erklärt ein Signalwort in einem kurzen Satz */
function sigText(sig) {
  const l = sig.toLowerCase();
  return VERSTECKT[l] ? `<span class="pill sig">${l}</span> ist ein versteckter Artikel (${l} = ${VERSTECKT[l]})`
    : MENGE.includes(l) ? `<span class="pill sig">${l}</span> ist ein Mengenwort`
    : `<span class="pill sig">${l}</span> ist ein Artikel`;
}
/* Zerlegt einen Satz in Wörter. Wörter in [Klammern] sind die fraglichen Wörter.
   Großes Wort: hint ist das Signalwort. Kleines Wort: hint ist die Erklärung. */
function make(text, hints) {
  const toks = text.split(' ').map(tok => {
    const m = tok.replace(/[\[\]]/g, '').match(/^(.*?)([.,!?:;]*)$/);
    return { w: m[1], post: m[2], target: tok.includes('[') };
  });
  const gaps = [];
  toks.forEach((t, ti) => {
    if (!t.target) return;
    const hint = hints[gaps.length] || '', big = t.w[0] !== t.w[0].toLowerCase();
    let si = -1;
    if (big) for (let i = ti - 1; i >= 0; i--) if (toks[i].w === hint) { si = i; break; }
    gaps.push({ ti, si, word: t.w, big, k: big ? 'gr' : 'kl', sig: big ? hint : '', why: big ? '' : hint });
    t.gap = gaps.length - 1;
  });
  return Object.assign({ text, toks, gaps, misses: 0, done: false }, gaps[0]);
}
/* Der ganze Satz in Farbe: Signalwort lila, großes Wort blau, kleines Wort türkis */
function full(s, tag = 'b', cls = 'w') {
  return s.toks.map((t, i) => {
    const g = s.gaps.find(x => x.ti === i), sg = s.gaps.some(x => x.si === i);
    return (g ? `<${tag} class="${cls} ${g.k}">${t.w}</${tag}>` : sg ? `<b class="w sig">${t.w}</b>` : t.w) + t.post;
  }).join(' ');
}

/* ---------- Spiel: Großschreib-Profi ---------- */
let all = [], queue = [], cur = null;
const inputs = () => $$('#sentence input');
const upper = g => g.word.toUpperCase();

function startGame() {
  all = SENTENCES.map(s => make(s[0], s.slice(1)));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Schreibe jedes Wort richtig in die Lücke. Gehört ein Signalwort dazu?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#sentence').innerHTML = cur.toks.map(t => {
    if (!t.target) return `<span class="txt">${t.w}${t.post}</span>`;
    const g = cur.gaps[t.gap];
    return `<span class="wgap"><input type="text" autocapitalize="off" autocorrect="off" autocomplete="off" spellcheck="false" maxlength="${g.word.length + 3}"
      style="width:${(g.word.length * .68 + 1.6).toFixed(1)}em" aria-label="Lücke${cur.gaps.length > 1 ? ' ' + (t.gap + 1) : ''}: ${upper(g)}"><span class="capl" aria-hidden="true">${upper(g)}</span></span>${t.post ? `<span class="txt">${t.post}</span>` : ''}`;
  }).join(' ');
  inputs()[0].focus({ preventScroll: true });
}

function tip(i) {
  const g = cur.gaps[i];
  return g.big ? `Tipp für <b>${upper(g)}</b>: ${sigText(g.sig)} und gehört zu diesem Wort. Also groß!`
               : `Tipp für <b>${upper(g)}</b>: ${g.why} Also klein!`;
}

function check() {
  if (state.busy || !cur) return;
  const ins = inputs();
  const empty = ins.find(inp => !inp.value.trim());
  if (empty) { say(ins.length > 1 ? 'Fülle zuerst alle Lücken aus.' : 'Schreibe zuerst das Wort in die Lücke.'); empty.focus(); return; }
  const wrong = [], typo = [];
  ins.forEach((inp, i) => {
    const v = inp.value.trim(), g = cur.gaps[i];
    inp.classList.remove('hint');
    if (v === g.word) { inp.value = g.word; inp.readOnly = true; inp.classList.add('good', g.k); }
    else if (v.toLowerCase() === g.word.toLowerCase()) wrong.push(i);
    else typo.push(i);
  });
  if (typo.length) {
    const inp = ins[typo[0]];
    wobble(inp);
    say(`Schau genau hin: Hier fehlt das Wort <b>${upper(cur.gaps[typo[0]])}</b>. Schreibe es mit denselben Buchstaben.`, 'think');
    inp.focus();
    return;
  }
  if (wrong.length) {
    const first = ins[wrong[0]];
    miss(cur, first);
    wrong.slice(1).forEach(i => wobble(ins[i]));
    if (cur.misses === 2) first.classList.add('hint');
    say(cur.misses === 1 ? 'Fast! Überlege noch einmal: Gehört ein Signalwort zu dem Wort? Oder kommt danach noch ein Nomen?'
      : cur.misses === 2 ? tip(wrong[0])
      : `So ist es richtig: ${full(cur, 'span', 'pill')}<br>Schreibe die Wörter genau so ab.`, 'think');
    first.focus();
    return;
  }
  state.busy = true;
  solve(cur, ins[ins.length - 1], 'sun');
  say(`${pick(PRAISE, 'praise')} ${full(cur, 'span', 'pill')}`, 'happy');
  later(next, 3000);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
