@@tipp
Suche im Kopf ein <b>verwandtes Wort</b>: die Mehrzahl zurück in die Einzahl (Wälder → Wald), das Verb in die Grundform (er fährt → fahren) oder ein kürzeres Wort (häufig → Haufen). Findest du ein Wort mit <b>a</b> oder <b>au</b>, schreibst du <b>ä</b> oder <b>äu</b>. Sonst schreibst du <b>e</b> oder <b>eu</b>.

/* ---------- Spiel: Lücken-Blitz ---------- */
let all = [], queue = [], cur = null;
const ORDER = Object.keys(SOUNDS);
const cls = l => SOUNDS[l][0];
const baseOf = l => l.length === 2 ? 'au' : 'a';
const qmark = w => w.gap.replace('_', '?');

function startGame() {
  all = WORDS.map(w => ({ gap: w[0], l: w[1], rel: w[2], full: w[0].replace('_', w[1]), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Suche im Kopf ein verwandtes Wort mit a oder au. Welcher Laut gehört in die Lücke?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#word').innerHTML = cur.gap.replace('_', '<span class="slot" id="slot">?</span>');
  $('#proof').textContent = '';
}

function proof(w) {
  return w.rel ? `verwandt: ${w.rel}` : `kein verwandtes Wort mit ${baseOf(w.l)}`;
}

function choose(l) {
  if (state.busy || !cur) return;
  const tile = $(`.tile[data-l="${l}"]`);
  if (!tile) return;
  if (l !== cur.l) {
    const hint = miss(cur, tile);
    if (hint) $(`.tile[data-l="${cur.l}"]`).classList.add('hint');
    say(hint ? (cur.rel ? `Tipp: <b>${qmark(cur)}</b> ist verwandt mit <b>${cur.rel}</b>. Aus ${baseOf(cur.l)} wird ${cur.l}. Der leuchtende Laut ist richtig.`
        : `Tipp: Zu <b>${qmark(cur)}</b> gibt es kein verwandtes Wort mit ${baseOf(cur.l)}. Darum schreibst du ${cur.l}. Der leuchtende Laut ist richtig.`)
      : (l.length !== cur.l.length ? `Hmm, sprich das Wort laut: <b>${qmark(cur)}</b>. Passt dieser Laut in die Lücke?`
        : `Fast! Suche ein verwandtes Wort: Gibt es eins mit ${baseOf(cur.l)}?`), 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const slot = $('#slot');
  slot.textContent = cur.l; slot.className = 'slot filled ' + cls(cur.l);
  $('#proof').textContent = proof(cur);
  solve(cur, slot, SOUNDS[cur.l][1]);
  say(`${pick(PRAISE, 'praise')} ` + (cur.rel ? `${cur.rel} → <span class="pill ${cls(cur.l)}">${cur.full}</span> mit <b>${cur.l}</b>.`
    : `<span class="pill ${cls(cur.l)}">${cur.full}</span> mit <b>${cur.l}</b>, denn es gibt kein verwandtes Wort mit ${baseOf(cur.l)}.`), 'happy');
  later(next, 2100);
}

$('#letters').addEventListener('click', e => { const t = e.target.closest('.tile'); if (t) choose(t.dataset.l); });
document.addEventListener('keydown', e => {
  if (!gameOn() || !cur || !/^[1-4]$/.test(e.key)) return;
  choose(ORDER[+e.key - 1]);
});
