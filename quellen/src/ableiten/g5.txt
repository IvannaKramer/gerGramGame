@@tipp
Wörter einer <b>Wortfamilie</b> haben denselben Wortstamm und etwas miteinander zu tun: <b>Land</b> – Länder, Gelände, ländlich. Wer zu einer Familie mit <b>a</b> gehört, wird mit <b>ä</b> geschrieben. Wer zu einer Familie mit <b>au</b> gehört, bekommt <b>äu</b>. Ein Wort wie Kalender klingt ähnlich, hat aber mit Land nichts zu tun.

/* ---------- Spiel: Familien-Treffen ---------- */
const GAP = '<i class="gap"></i>';
let rounds = [], ri = 0, cur = null, penalty = 0, wrongTaps = 0;
const cls = l => SOUNDS[l][0];
const colored = c => c.gap.replace('_', `<b class="hit ${cls(c.l)}">${c.l}</b>`);
const markHead = r => r.head.replace(r.l === 'äu' ? /au/i : /a/i, '<u>$&</u>');

function startGame() {
  rounds = FAMILIES.map(f => ({ head: f[0], l: f[1],
    cards: [...f[2].map(g => ({ gap: g, l: f[1], member: true })), ...f[3].map(o => ({ gap: o[0], l: o[1], member: false }))]
      .map(c => ({ ...c, full: c.gap.replace('_', c.l), misses: 0, done: false })) }));
  resetState(rounds.reduce((n, r) => n + r.cards.filter(c => c.member).length, 0));
  ri = -1;
  show('game');
  nextRound();
}

function nextRound() {
  ri++;
  cur = rounds[ri];
  if (!cur) { finish(); return; }
  state.busy = false; penalty = 0; wrongTaps = 0;
  cur.order = shuffle(cur.cards.slice());
  $('#round').textContent = `Familie ${ri + 1} von ${rounds.length}`;
  $('#head').innerHTML = markHead(cur);
  $('#fam').innerHTML = cur.order.map((c, i) =>
    `<button class="fcard" data-i="${i}" aria-label="${c.gap.replace('_', ' Lücke ')}">${c.gap.replace('_', GAP)}</button>`).join('');
  say(`Vier Wörter sind mit <b>${cur.head}</b> verwandt. Tippe sie an! Zwei Wörter gehören nicht dazu.`);
}

function tap(i) {
  if (state.busy || !cur) return;
  const c = cur.order[i], el = $(`.fcard[data-i="${i}"]`);
  if (!c || !el || el.disabled) return;
  const base = cur.l === 'äu' ? 'au' : 'a';
  if (!c.member) {
    miss(c, el); penalty++; wrongTaps++;
    el.disabled = true; el.classList.add('limp');
    const hint = wrongTaps >= 2;
    if (hint) $$('.fcard').forEach((b, k) => { if (cur.order[k].member && !cur.order[k].done) b.classList.add('hint'); });
    say(hint ? `Tipp: Zur Familie gehören nur Wörter, die mit <b>${cur.head}</b> zu tun haben. Die leuchtenden Karten sind verwandt. Dort kommt ${cur.l} in die Lücke.`
      : `<b>${c.gap.replace('_', GAP)}</b> klingt ähnlich, hat aber mit <b>${cur.head}</b> nichts zu tun. Die Bedeutung muss passen.`, 'think');
    return;
  }
  c.done = true;
  c.misses = penalty > 0 ? 1 : 0; penalty = Math.max(0, penalty - 1);
  el.disabled = true; el.classList.remove('hint'); el.classList.add('in', cls(c.l));
  el.innerHTML = colored(c);
  solve(c, el, SOUNDS[c.l][1]);
  const left = cur.cards.filter(x => x.member && !x.done).length;
  if (left) {
    say(`${pick(PRAISE, 'praise')} ${cur.head} → <span class="pill ${cls(c.l)}">${c.full}</span> mit <b>${c.l}</b>. Noch ${left === 1 ? 'ein Wort gehört' : left + ' Wörter gehören'} zur Familie.`, 'happy');
    return;
  }
  /* Familie komplett: Die beiden übrigen Wörter zeigen jetzt ihren Laut */
  state.busy = true;
  const rest = [];
  $$('.fcard').forEach((b, k) => {
    const o = cur.order[k];
    if (o.member) return;
    b.disabled = true; b.classList.remove('limp'); b.classList.add('out');
    b.innerHTML = colored(o); rest.push(o);
  });
  const l = rest[0].l;
  say(`Familie komplett! ${rest.map(o => `<span class="pill ${cls(o.l)}">${o.full}</span>`).join(' und ')} haben kein verwandtes Wort mit ${base}. Darum schreibst du sie mit <b>${l}</b>.`, 'happy');
  later(nextRound, 3600);
}

$('#fam').addEventListener('click', e => { const b = e.target.closest('.fcard'); if (b) tap(+b.dataset.i); });
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-6]$/.test(e.key)) return;
  tap(+e.key - 1);
});
