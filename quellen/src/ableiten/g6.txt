/* ---------- Spiel: Schreib-Profi ---------- */
let all = [], queue = [], cur = null;
const input = $('#answer');
const cls = l => SOUNDS[l][0];
const baseOf = l => l.length === 2 ? 'au' : 'a';
const coloredWord = w => `${w.right.slice(0, w.pos)}<b class="hit ${cls(w.l)}">${w.l}</b>${w.right.slice(w.pos + w.l.length)}`;

function startGame() {
  all = SENTENCES.map(s => {
    const tok = s[0].split(' ').find(t => t.includes('_'));
    const pos = tok.indexOf('_');
    const two = s[1].slice(pos, pos + 2);
    const l = (two === 'äu' || two === 'eu') ? two : s[1][pos];
    return { sent: s[0], right: s[1], merk: s[2] === 'M', rel: s[2] === 'M' ? '' : s[2], l, pos, misses: 0, wmiss: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#sentence').innerHTML = cur.sent.split(' ').map(t => {
    if (!t.includes('_')) return t;
    const [, w, punct] = t.match(/^(.*?)([.,!?]*)$/);
    return `<span class="gapw" id="gapw">${w.replace('_', '<i class="gap"></i>')}</span>${punct}`;
  }).join(' ');
  input.value = ''; input.readOnly = false; input.classList.remove('good', 'hint');
  say('Suche ein verwandtes Wort mit a oder au. Schreibe dann das ganze Wort auf.');
  if (matchMedia('(hover:hover)').matches) input.focus({ preventScroll: true });
}

function tip(w) {
  if (w.merk) return `Tipp: Das ist ein Merkwort. Du schreibst es mit <b>${w.l}</b>, obwohl es kein verwandtes Wort mit ${baseOf(w.l)} gibt.`;
  if (w.rel) return `Tipp: Das Wort ist verwandt mit <b>${w.rel}</b>. Aus ${baseOf(w.l)} wird <b>${w.l}</b>.`;
  return `Tipp: Hier gibt es kein verwandtes Wort mit ${baseOf(w.l)}. Darum schreibst du <b>${w.l}</b>.`;
}

function check() {
  if (state.busy || !cur) return;
  const v = input.value.trim().replace(/\s+/g, ' ');
  if (!v) { say('Schreibe zuerst das Wort in das Feld.'); input.focus(); return; }
  if (v !== cur.right) {
    if (v.toLowerCase() === cur.right.toLowerCase()) {
      wobble(input);
      say(/^[A-ZÄÖÜ]/.test(cur.right) ? 'Fast richtig! Denk daran: Nomen schreibt man groß.' : 'Fast richtig! Dieses Wort schreibt man hier klein.', 'think');
    } else {
      miss(cur, input); cur.wmiss++;
      if (cur.wmiss >= 3) input.classList.add('hint');
      say(cur.wmiss === 1 ? 'Fast! Schau dir jeden Buchstaben genau an. Gibt es ein verwandtes Wort mit a oder au?'
        : cur.wmiss === 2 ? tip(cur)
        : `So ist es richtig: <span class="pill ${cls(cur.l)}">${cur.right}</span> Schreibe das Wort genau so ab.`, 'think');
    }
    input.focus();
    return;
  }
  state.busy = true;
  input.readOnly = true; input.classList.remove('hint'); input.classList.add('good');
  const g = $('#gapw');
  g.className = 'gapw fixed ' + cls(cur.l);
  g.innerHTML = coloredWord(cur);
  solve(cur, g, SOUNDS[cur.l][1]);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${cls(cur.l)}">${cur.right}</span> ` + (cur.merk ? `ist ein Merkwort mit <b>${cur.l}</b>. Gut gemerkt!`
    : cur.rel ? `mit <b>${cur.l}</b>, denn es ist verwandt mit <b>${cur.rel}</b>.`
    : `mit <b>${cur.l}</b>, denn es gibt kein verwandtes Wort mit ${baseOf(cur.l)}.`), 'happy');
  later(next, 2600);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.keycap');
  if (!k || input.readOnly) return;
  const a = input.selectionStart ?? input.value.length, b = input.selectionEnd ?? a;
  input.value = input.value.slice(0, a) + k.textContent + input.value.slice(b);
  input.focus();
  input.setSelectionRange(a + 1, a + 1);
});
