@@tipp
Bei den meisten Wörtern mit <b>ä</b> oder <b>äu</b> findest du ein verwandtes Wort mit <b>a</b> oder <b>au</b>: die Äpfel → der Apfel. Bei einigen Wörtern gibt es keins, zum Beispiel bei <b>Käse</b>, <b>Bär</b> und <b>Mädchen</b>. Das sind <b>Merkwörter</b>. Du musst sie dir einprägen.

/* ---------- Spiel: Merkwort-Körbe ---------- */
const KIND = { ab: 'Ableiten', mw: 'Merkwort' };
const KCLS = { ab: 'kab', mw: 'kmw' };
let items = [], selWord = null, selKind = null;
const byId = id => items.find(w => w.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = k => $(`.basket[data-kind="${k}"]`);
/* ä ist im ganzen Thema blau, äu grün */
const colored = w => w.text.replace(/äu|ä/i, m => `<b class="hit ${m.length === 2 ? 'kau' : 'ka'}">${m}</b>`);
const bare = w => w.text.replace(/^(der|die|das) /, '');

function startGame() {
  items = shuffle(WORDS.map((w, i) => ({ id: 'w' + i, text: w[0], emoji: w[1], rel: w[2], kind: w[2] ? 'ab' : 'mw',
    sound: /äu/i.test(w[0]) ? 'äu' : 'ä', misses: 0, done: false })));
  resetState(items.length);
  selWord = selKind = null;
  $('#words').innerHTML = items.map(w =>
    `<button class="word-card" data-id="${w.id}" aria-pressed="false" style="--tilt:${(Math.random() * 3 - 1.5).toFixed(2)}deg">
      <span class="pic" aria-hidden="true">${w.emoji}</span><span class="word">${w.text}</span></button>`).join('');
  renderSelection();
  say('Tippe eine Karte an. Findest du ein verwandtes Wort mit a oder au? Dann tippe auf „Ableiten“. Sonst ist es ein Merkwort.');
  show('game');
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selWord;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.kind === selKind;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const w = selWord && byId(selWord);
  if (w && w.misses >= 2) basketEl(w.kind).classList.add('hint');
}

function tapWord(id) {
  const w = byId(id);
  if (!w || w.done || state.busy) return;
  if (selKind) { attempt(id, selKind); return; }
  selWord = selWord === id ? null : id;
  sfx.pick(); renderSelection();
  if (selWord) say(`<b>${w.text}</b> – findest du ein verwandtes Wort mit ${w.sound === 'äu' ? 'au' : 'a'}?`);
}
function tapBasket(k) {
  if (state.busy) return;
  if (selWord) { attempt(selWord, k); return; }
  selKind = selKind === k ? null : k;
  sfx.pick(); renderSelection();
  if (selKind) say(`Welche Karte gehört in den Korb <span class="pill ${KCLS[k]}">${KIND[k]}</span>?`);
}

function attempt(id, k) {
  const w = byId(id);
  if (!w || w.done) return;
  const a = w.sound === 'äu' ? 'au' : 'a';
  if (w.kind !== k) {
    miss(w, cardEl(id));
    wobble(basketEl(k));
    selWord = id; selKind = null;
    renderSelection();
    say(w.misses >= 2
      ? (w.kind === 'ab' ? `Tipp: Denk an <b>${w.rel}</b>. Das ist verwandt und hat ein ${a}. Der leuchtende Korb ist richtig.`
        : `Tipp: Zu <b>${bare(w)}</b> gibt es kein verwandtes Wort mit ${a}. Der leuchtende Korb ist richtig.`)
      : (w.kind === 'ab' ? `Hmm, such noch einmal: Gibt es zu <b>${bare(w)}</b> ein verwandtes Wort mit ${a}? Denk an die Einzahl oder an ein kürzeres Wort.`
        : `Welches verwandte Wort mit ${a} meinst du? Überlege noch einmal: Hat es wirklich mit <b>${bare(w)}</b> zu tun?`), 'think');
    return;
  }
  const card = cardEl(id);
  selWord = selKind = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(w, basketEl(k), k === 'ab' ? 'teal' : 'sun');
  say(`${pick(PRAISE, 'praise')} ` + (k === 'ab'
    ? `${w.rel} → <b>${colored(w)}</b>. Aus ${a} wird ${w.sound}.`
    : `<b>${colored(w)}</b> ist ein <span class="pill kmw">Merkwort</span>. Es gibt kein verwandtes Wort mit ${a}. Merk dir das ${w.sound}!`), 'happy');
  if (state.solved === state.total) { state.busy = true; later(finish, 1700); }
}

$('#words').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapWord(c.dataset.id);
});
$$('.basket').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapBasket(b.dataset.kind); }));
enableDrag('.word-card:not(.done)', '.basket', (card, basket) => attempt(card.dataset.id, basket.dataset.kind));
