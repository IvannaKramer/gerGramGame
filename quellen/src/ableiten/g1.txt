@@tipp
In der Mehrzahl wird aus <b>a</b> oft <b>ä</b> und aus <b>au</b> oft <b>äu</b>: der Ball → viele Bälle, der Baum → viele Bäume. Steht im Wort schon <b>e</b> oder <b>eu</b>, dann bleibt es so: das Bett → viele Betten, der Freund → viele Freunde.

/* ---------- Spiel: Umlaut-Zauber ---------- */
let all = [], queue = [], cur = null;
const cls = l => SOUNDS[l][0];
const colored = w => w.gap.replace('_', `<b class="hit ${cls(w.l)}">${w.l}</b>`);

function startGame() {
  all = WORDS.map(w => ({ base: w[0], plain: w[0].replace(/[\[\]]/g, ''), from: (w[0].match(/\[(.+)\]/) || [])[1], emoji: w[1], gap: w[2], l: w[3],
    full: w[2].replace('_', w[3]), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies das Wort oben und sprich die Mehrzahl laut. Welcher Laut gehört in die Lücke?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#pic').textContent = cur.emoji;
  $('#base').innerHTML = cur.base.replace(/\[(.+)\]/, '<u>$1</u>');
  $('#word').innerHTML = cur.gap.replace('_', '<span class="slot" id="slot">?</span>');
  $('#letters').innerHTML = PAIRS[cur.l].map((l, i) =>
    `<button class="tile ${cls(l)}" data-l="${l}" aria-label="${l}"><b>${l}</b><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

/* Die Regel als Satz, z. B. „Aus a wird ä“ oder „e bleibt e“ */
const rule = w => w.from === w.l ? `<b>${w.l}</b> bleibt <b>${w.l}</b>` : `Aus <b>${w.from}</b> wird <b>${w.l}</b>`;

function choose(l) {
  if (state.busy || !cur) return;
  const tile = $(`.tile[data-l="${l}"]`);
  if (!tile) return;
  if (l !== cur.l) {
    const hint = miss(cur, tile);
    if (hint) $(`.tile[data-l="${cur.l}"]`).classList.add('hint');
    say(hint ? `Tipp: In <b>${cur.plain}</b> steht ein <b>${cur.from}</b>. ${rule(cur)}. Der leuchtende Laut ist richtig.`
      : `Hmm, schau noch einmal auf das Wort oben: <b>${cur.plain}</b>. Welcher Laut ist unterstrichen?`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const slot = $('#slot');
  slot.textContent = cur.l; slot.className = 'slot filled ' + cls(cur.l);
  solve(cur, slot, SOUNDS[cur.l][1]);
  say(`${pick(PRAISE, 'praise')} ${rule(cur)}: ${cur.plain} → <span class="pill ${cls(cur.l)}">${cur.full}</span>`, 'happy');
  later(next, 2000);
}

$('#letters').addEventListener('click', e => { const t = e.target.closest('.tile'); if (t) choose(t.dataset.l); });
document.addEventListener('keydown', e => {
  if (!gameOn() || !cur || !/^[12]$/.test(e.key)) return;
  choose(PAIRS[cur.l][+e.key - 1]);
});
