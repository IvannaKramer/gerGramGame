@@tipp
Verwandte Wörter gehören zur selben <b>Wortfamilie</b>. Sie haben etwas miteinander zu tun: Der <b>Bäcker</b> kann <b>backen</b>. Findest du ein verwandtes Wort mit <b>a</b>, schreibst du <b>ä</b>. Findest du eins mit <b>au</b>, schreibst du <b>äu</b>. Findest du keins, schreibst du <b>e</b> oder <b>eu</b>.

/* ---------- Spiel: Wortfamilien-Suche ---------- */
let all = [], queue = [], cur = null, step = 1, options = [];
const cls = l => SOUNDS[l][0];
const colored = w => w.gap.replace('_', `<b class="hit ${cls(w.l)}">${w.l}</b>`);
const qmark = w => w.gap.replace('_', '?');
/* a oder au im verwandten Wort unterstreichen */
const markRel = w => w.rel.replace(w.base === 'au' ? /au/i : /a/i, '<u>$&</u>');

function startGame() {
  all = WORDS.map(w => ({ gap: w[0], l: w[1], emoji: w[2], rel: w[3], wrong: w[4], base: PAIRS[w[1]][0] === 'ä' ? 'a' : 'au',
    full: w[0].replace('_', w[1]), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Gibt es ein verwandtes Wort? Tippe es an. Wenn keins der beiden Wörter passt, tippe auf „Keins passt“.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; step = 1;
  $('#pic').textContent = cur.emoji;
  $('#word').innerHTML = cur.gap.replace('_', '<span class="slot" id="slot">?</span>');
  $('#ext').innerHTML = '<span class="ph">Suche unten ein verwandtes Wort.</span>';
  $('#ask').textContent = 'Welches Wort ist verwandt?';
  options = shuffle([...(cur.rel ? [{ ok: true, text: cur.rel }] : []), ...cur.wrong.map(t => ({ ok: false, text: t }))]);
  options.push({ ok: !cur.rel, text: 'Keins passt', none: true });
  $('#opts').className = 'opts words';
  $('#opts').innerHTML = options.map((o, i) =>
    `<button class="opt${o.none ? ' none' : ''}" data-i="${i}">${o.text}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function chooseWord(i) {
  const o = options[i], el = $(`.opt[data-i="${i}"]`);
  if (!o || !el || el.disabled) return;
  if (!o.ok) {
    const hint = miss(cur, el);
    el.disabled = true; el.classList.add('limp');
    if (hint) $(`.opt[data-i="${options.findIndex(x => x.ok)}"]`).classList.add('hint');
    say(hint ? (cur.rel ? `Tipp: Verwandte Wörter haben etwas miteinander zu tun. Das leuchtende Wort gehört zu <b>${qmark(cur)}</b>.`
        : `Tipp: Keins der Wörter hat etwas mit <b>${qmark(cur)}</b> zu tun. Tippe auf das leuchtende Feld.`)
      : o.none ? `Doch, eins der beiden Wörter ist mit <b>${qmark(cur)}</b> verwandt. Welches hat damit zu tun?`
      : `<b>${o.text}</b> sieht ähnlich aus, hat aber mit <b>${qmark(cur)}</b> nichts zu tun. Die Bedeutung muss passen.`, 'think');
    return;
  }
  sfx.pick();
  step = 2;
  $('#ext').innerHTML = cur.rel ? markRel(cur) : `<span class="none">kein verwandtes Wort mit ${cur.base}</span>`;
  $('#ask').textContent = 'Was gehört in die Lücke?';
  $('#opts').className = 'opts';
  $('#opts').innerHTML = PAIRS[cur.l].map((l, k) =>
    `<button class="tile ${cls(l)}" data-l="${l}" aria-label="${l}"><b>${l}</b><span class="kbd" aria-hidden="true">${k + 1}</span></button>`).join('');
  say(cur.rel ? `Genau! <b>${markRel(cur)}</b> gehört zur selben Wortfamilie. Du siehst ein <b>${cur.base}</b>. Was schreibst du also in die Lücke?`
    : `Genau! Es gibt kein verwandtes Wort mit <b>${cur.base}</b>. Was schreibst du dann in die Lücke?`, 'happy');
}

function chooseLetter(l) {
  const tile = $(`.tile[data-l="${l}"]`);
  if (!tile) return;
  if (l !== cur.l) {
    const hint = miss(cur, tile);
    if (hint) $(`.tile[data-l="${cur.l}"]`).classList.add('hint');
    say(hint ? (cur.rel ? `Tipp: Das verwandte Wort <b>${cur.rel}</b> hat ein <b>${cur.base}</b>. Darum schreibst du <b>${cur.l}</b>. Der leuchtende Laut ist richtig.`
        : `Tipp: Ohne verwandtes Wort mit <b>${cur.base}</b> schreibst du <b>${cur.l}</b>. Der leuchtende Laut ist richtig.`)
      : (cur.rel ? `Hmm, schau auf das verwandte Wort: <b>${markRel(cur)}</b>. Aus ${cur.base} wird …?`
        : `Hmm, denk an die Regel: Nur wenn es ein verwandtes Wort mit ${cur.base} gibt, schreibst du ${PAIRS[cur.l][0]}.`), 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const slot = $('#slot');
  slot.textContent = cur.l; slot.className = 'slot filled ' + cls(cur.l);
  solve(cur, slot, SOUNDS[cur.l][1]);
  say(`${pick(PRAISE, 'praise')} ` + (cur.rel ? `${cur.rel} → <span class="pill ${cls(cur.l)}">${cur.full}</span> mit <b>${cur.l}</b>.`
    : `<span class="pill ${cls(cur.l)}">${cur.full}</span> mit <b>${cur.l}</b>, denn es gibt kein verwandtes Wort mit ${cur.base}.`), 'happy');
  later(next, 2100);
}

$('#opts').addEventListener('click', e => {
  if (state.busy || !cur) return;
  const o = e.target.closest('.opt'), t = e.target.closest('.tile');
  if (o) chooseWord(+o.dataset.i); else if (t) chooseLetter(t.dataset.l);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || state.busy || !cur || !/^[1-3]$/.test(e.key)) return;
  if (step === 1) chooseWord(+e.key - 1);
  else if (e.key !== '3') chooseLetter(PAIRS[cur.l][+e.key - 1]);
});
