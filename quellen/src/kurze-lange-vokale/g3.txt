/* ---------- Spiel: Langes i gesucht ---------- */
const GAP = '<i class="gap"></i>';
let longs = [], shorts = [], rounds = [], round = 0, cards = [], left = 0, roundMiss = { misses: 0 };
const item = (w, long) => ({ long, emoji: w[1], misses: 0, plain: w[0].replace(/[\[\]]/g, ''),
  gapped: w[0].replace(/\[.+?\]/, GAP),
  marked: w[0].replace(/\[(.+?)\]/, `<span class="${long ? 'vl' : 'vk'}">$1</span>`) });

function startGame() {
  longs = LONG.map(w => item(w, true)); shorts = SHORT.map(w => item(w, false));
  const L = shuffle(longs.slice()), S = shuffle(shorts.slice());
  rounds = shuffle(LONG_PER_ROUND.slice()).map(n => shuffle([...L.splice(0, n), ...S.splice(0, CARDS_PER_ROUND - n)]));
  resetState(longs.length);
  round = 0;
  startRound();
  say('Sprich jedes Wort laut. Tippe alle Karten an, in denen du ein <b>langes i</b> hörst.');
  show('game');
}

function startRound() {
  cards = rounds[round];
  left = cards.filter(c => c.long).length;
  roundMiss = { misses: 0 };
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#field').innerHTML = cards.map((c, i) =>
    `<button class="scard" data-i="${i}" aria-label="Bildkarte ${c.plain}">
      <span class="pic" aria-hidden="true">${c.emoji}</span><span class="word" aria-hidden="true">${c.gapped}</span></button>`).join('');
}

function reveal(c, el) {
  c.shown = true;
  el.classList.remove('hint');
  el.classList.add('shown', c.long ? 'la' : 'ku');
  el.disabled = true;
  el.querySelector('.word').innerHTML = c.marked;
  el.insertAdjacentHTML('beforeend', `<span class="tag" aria-hidden="true">${c.long ? 'langes i' : 'kurzes i'}</span>`);
  el.setAttribute('aria-label', `${c.plain}, ${c.long ? 'langes i, mit ie' : 'kurzes i, nur i'}`);
}

$('#field').addEventListener('click', e => {
  const el = e.target.closest('.scard');
  if (!el || state.busy) return;
  const c = cards[+el.dataset.i];
  if (c.shown) return;
  if (c.long) {
    reveal(c, el);
    solve(c, el, 'der');
    say(`${pick(PRAISE, 'praise')} <span class="pill la">${c.marked}</span> Du hörst ein langes i. Darum schreibst du <b>ie</b>.`, 'happy');
    if (--left === 0) roundDone();
    return;
  }
  const hint = miss(roundMiss, el);
  reveal(c, el);
  let txt = `<span class="pill ku">${c.marked}</span> Hier ist das i kurz. Ein kurzes i schreibst du nur <b>i</b>. Such weiter nach dem langen i!`;
  if (hint) {
    const i = cards.findIndex(x => x.long && !x.shown);
    if (i >= 0) { cards[i].misses = 1; $(`.scard[data-i="${i}"]`).classList.add('hint'); txt += ' Tipp: Sprich das Wort auf der leuchtenden Karte. Dort kannst du das i lang ziehen.'; }
  }
  say(txt, 'think');
});

function roundDone() {
  state.busy = true;
  later(() => {
    cards.forEach((c, i) => { if (!c.shown) reveal(c, $(`.scard[data-i="${i}"]`)); });
    if (round < rounds.length - 1) {
      say('Runde geschafft! Die anderen Wörter haben ein <b>kurzes i</b>. Schau sie dir an. Gleich kommen neue Karten.', 'happy');
      later(() => { round++; startRound(); state.busy = false; say('Neue Karten! Wo hörst du ein <b>langes i</b>?'); }, 3200);
    } else {
      say('Alle gefunden! Die anderen Wörter haben ein <b>kurzes i</b>.', 'happy');
      later(finish, 2400);
    }
  }, 900);
}
