@@tipp
<b>ck</b> und <b>tz</b> stehen nur <b>direkt nach einem kurzen Vokal</b>: Jacke, Katze. Nach <b>l, n, r</b>, das merke ja, steht nie tz und nie ck: Onkel, Holz. Auch nach einem <b>langen Vokal</b> und nach den Zwielauten <b>au, ei, eu</b> schreibst du nur k oder z: Haken, Pauke, Kreuz.

/* Aufbau: [Satz, Wort, Regel] – im Satz steht ___ für das gesuchte Wort.
   Im Wort steht in eckigen Klammern der Buchstabe oder Vokal, auf den es ankommt (l, n, r, kurzer Vokal, langer Vokal, Zwielaut).
   In runden Klammern stehen die Buchstaben, die im Satz als Lücke gezeigt werden. Die Regeln stehen darunter. */
const WORDS = [
  ['Im Park sitzt Opa auf einer ___.',                     'Ba[n](k)',       'lk'],
  ['Am Himmel zieht eine graue ___ vorbei.',               'Wo[l](k)e',      'lk'],
  ['Mama schneidet eine grüne ___ in Scheiben.',           'Gu[r](k)e',      'lk'],
  ['Zum Geburtstag bekommt Mia ein tolles ___.',           'Gesche[n](k)',   'lk'],
  ['Ohne Lampe ist es im Keller ganz ___.',                'du[n](k)el',     'lk'],
  ['Vor Aufregung klopft mein ___ ganz schnell.',          'He[r](z)',       'lz'],
  ['Unter der Tanne wächst ein kleiner ___.',              'Pi[l](z)',       'lz'],
  ['Auf dem Geburtstagskuchen brennt eine ___.',           'Ke[r](z)e',      'lz'],
  ['Auf dem Schulfest ___ alle Kinder zur Musik.',         'ta[n](z)en',     'lz'],
  ['An den Pommes fehlt noch etwas ___.',                  'Sa[l](z)',       'lz'],
  ['Die ___ kriecht langsam über den Weg.',                'Schn[e](ck)e',   'ku'],
  ['Jeden Morgen klingelt mein ___ um sieben Uhr.',        'W[e](ck)er',     'ku'],
  ['Tom hat das Rennen gewonnen und ist sehr ___.',        'gl[ü](ck)lich',  'ku'],
  ['Beim Gewitter leuchtet ein heller ___ am Himmel.',     'Bl[i](tz)',      'ku'],
  ['Das Licht geht ___ aus und alle staunen.',             'pl[ö](tz)lich',  'ku'],

/* Aufbau: Kürzel: [Überschrift, Farbe] */
const RULES = {
  lk: ['Nach l, n, r: nur k',               'lnr'],
  lz: ['Nach l, n, r: nur z',               'lnr'],
  ku: ['Kurzer Vokal: ck oder tz',          'ku'],
  la: ['Zwielaut au, ei, eu: nur k oder z', 'la']
};

const GAME = {
  id: 'meister',
  next: null,
  review: () => `<div class="review c2">${Object.keys(RULES).map(k =>
    `<div class="col ${RULES[k][1]}"><h3>${RULES[k][0]}</h3><ul>${all.filter(w => w.k === k).map(w => `<li>${w.marked}</li>`).join('')}</ul></div>`).join('')}</div>`
};

/* ---------- Spiel: Regel-Meister ---------- */
const GAP = '<i class="gap"></i>';
const COLOR = { lnr: 'das', ku: 'die', la: 'der' };
let all = [], queue = [], cur = null;
const input = $('#answer');

function ruleText(w) {
  const wrong = w.fill.endsWith('k') ? 'ck' : 'tz', right = wrong === 'ck' ? 'k' : 'z';
  if (w.cls === 'lnr') return `Vor der Lücke steht ein <b>${w.key}</b>. Nach l, n, r, das merke ja, steht nie tz und nie ck. Du schreibst nur <b>${right}</b>.`;
  if (w.cls === 'ku') return `Das <b>${w.key}</b> vor der Lücke ist kurz. Direkt nach einem kurzen Vokal schreibst du <b>${wrong}</b>.`;
  return `Vor der Lücke steht der Zwielaut <b>${w.key}</b>. Nach au, ei und eu schreibst du nur <b>${right}</b>, nie ${wrong}.`;
}

function startGame() {
  all = WORDS.map(w => {
    const key = w[1].match(/\[(.+?)\]/)[1], fill = w[1].match(/\((.+?)\)/)[1], cls = RULES[w[2]][1];
    const mark = cls === 'lnr' ? '<span class="hl">$1</span>' : `<span class="${cls === 'ku' ? 'vk' : 'vl'}">$1</span>`;
    return { k: w[2], cls, key, fill, misses: 0,
      word: w[1].replace(/[\[\]()]/g, ''),
      marked: w[1].replace(/\[(.+?)\]/, mark).replace(/[()]/g, ''),
      html: w[0].replace('___', `<span class="tw">${w[1].replace(/\(.+?\)/, GAP).replace(/[\[\]]/g, '')}</span>`),
      done: w[0].replace('___', `<span class="tw">${w[1].replace(/\[(.+?)\]/, mark).replace(/[()]/g, '')}</span>`) };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Bei einem Wort fehlen Buchstaben. Schau auf den Buchstaben vor der Lücke und schreibe das ganze Wort in das Feld.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#sent').innerHTML = cur.html;
  input.value = ''; input.readOnly = false; input.classList.remove('good');
  input.focus({ preventScroll: true });
}

function check() {
  if (state.busy || !cur) return;
  const v = input.value.trim();
  if (!v) { say('Schreibe zuerst das ganze Wort in das Feld.'); input.focus(); return; }
  if (v.length <= 2) { wobble(input); say('Schreibe bitte das <b>ganze Wort</b> in das Feld, nicht nur die fehlenden Buchstaben.', 'think'); input.focus(); return; }
  if (v !== cur.word) {
    if (v.toLowerCase() === cur.word.toLowerCase()) {
      wobble(input);
      say(/^[A-ZÄÖÜ]/.test(cur.word) ? 'Fast richtig! Denk daran: Nomen schreibt man groß.' : 'Fast richtig! Dieses Wort schreibt man klein.', 'think');
    } else {
      miss(cur, input);
      say(cur.misses === 1 ? 'Fast! Schau dir den Buchstaben vor der Lücke genau an und probier es noch einmal.'
        : cur.misses === 2 ? 'Tipp: ' + ruleText(cur)
        : `So ist es richtig: <span class="pill ${cur.cls}">${cur.word}</span> Schreibe das Wort genau so ab.`, 'think');
    }
    input.focus();
    return;
  }
  state.busy = true;
  input.readOnly = true; input.classList.add('good');
  $('#sent').innerHTML = cur.done;
  solve(cur, input, COLOR[cur.cls]);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${cur.cls}">${cur.marked}</span> ${ruleText(cur)}`, 'happy');
  later(next, 2600);
}

$('#form').addEventListener('submit', e => { e.preventDefault(); check(); });
$('#keys').addEventListener('click', e => {
  const k = e.target.closest('.keycap');
  if (!k || input.readOnly) return;
  const a = input.selectionStart ?? input.value.length, b = input.selectionEnd ?? a;
  input.value = input.value.slice(0, a) + k.textContent + input.value.slice(b);
  input.focus();
  input.setSelectionRange(a + 1, a + 1);
});
