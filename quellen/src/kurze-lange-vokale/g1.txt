/* ---------- Spiel: Punkt oder Strich? ---------- */
const KIND = { k: 'kurz', l: 'lang' };
const CLS = { k: 'ku', l: 'la' };
const GAP = '<i class="gap"></i>';
let items = [], selWord = null, selKind = null;
const byId = id => items.find(w => w.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = k => $(`.basket[data-kind="${k}"]`);

/* Vorlesen als Zusatz: Das Spiel geht auch ohne Sprachausgabe. */
const canSpeak = (() => { try { return 'speechSynthesis' in window && typeof SpeechSynthesisUtterance === 'function'; } catch (e) { return false; } })();
function speak(text) {
  if (!canSpeak) return;
  try {
    const voices = speechSynthesis.getVoices();
    const de = voices.find(v => /^de([-_]|$)/i.test(v.lang));
    if (voices.length && !de) { say('Vorlesen geht auf diesem Gerät leider nicht. Sprich das Wort selbst laut.'); return; }
    const u = new SpeechSynthesisUtterance(text);
    u.lang = 'de-DE'; u.rate = .8; if (de) u.voice = de;
    speechSynthesis.cancel(); speechSynthesis.speak(u);
  } catch (e) { /* ohne Sprachausgabe weiterspielen */ }
}
const speakBtn = w => canSpeak ? ` <button type="button" class="speakbtn" data-say="${w.plain}">🔊 Anhören</button>` : '';

function startGame() {
  items = shuffle(WORDS.map((w, i) => {
    const vowel = w[0].match(/\[(.+?)\]/)[1], hidden = w[0].match(/\((.+?)\)/)[1];
    return { id: 'w' + i, kind: w[1], emoji: w[2], vowel: vowel.toLowerCase(), hidden, misses: 0, done: false,
      plain: w[0].replace(/[\[\]()]/g, ''),
      gapped: w[0].replace(/\(.+?\)/, GAP).replace(/[\[\]]/g, ''),
      marked: w[0].replace(/\[(.+?)\]/, `<span class="${w[1] === 'k' ? 'vk' : 'vl'}">$1</span>`).replace(/[()]/g, '') };
  }));
  resetState(items.length);
  selWord = selKind = null;
  $('#words').innerHTML = items.map(w =>
    `<button class="word-card" data-id="${w.id}" aria-pressed="false" aria-label="Bildkarte ${w.plain}" style="--tilt:${(Math.random() * 3 - 1.5).toFixed(2)}deg">
      <span class="pic" aria-hidden="true">${w.emoji}</span><span class="word" aria-hidden="true">${w.gapped}</span></button>`).join('');
  renderSelection();
  say('Tippe eine Karte an und sprich das Wort laut. Ist der Vokal vor der Lücke kurz oder lang?');
  show('game');
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selWord;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.kind === selKind;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const w = selWord && byId(selWord);
  if (w && w.misses >= 2) basketEl(w.kind).classList.add('hint');
}

function tapWord(id) {
  const w = byId(id);
  if (!w || w.done || state.busy) return;
  if (selKind) { attempt(id, selKind); return; }
  selWord = selWord === id ? null : id;
  sfx.pick(); renderSelection();
  if (selWord) say(`Sprich das Wort zum Bild ${w.emoji} laut. Ist das <b>${w.vowel}</b> kurz oder lang?${speakBtn(w)}`);
}
function tapBasket(k) {
  if (state.busy) return;
  if (selWord) { attempt(selWord, k); return; }
  selKind = selKind === k ? null : k;
  sfx.pick(); renderSelection();
  if (selKind) say(`Welche Karte gehört in den Korb <span class="pill ${CLS[k]}">${KIND[k]}</span>?`);
}

/* Der Vokal dreimal hintereinander: So hört man, ob er sich lang ziehen lässt. */
const stretch = w => w.plain.replace(new RegExp(w.vowel, 'i'), m => m + w.vowel + w.vowel);
function hintText(w) {
  return w.kind === 'l'
    ? `Tipp: Zieh das ${w.vowel} lang: „${stretch(w)}“. Das Wort klingt immer noch richtig. Das <b>${w.vowel}</b> ist also <b>lang</b>. Der blaue Korb leuchtet.`
    : `Tipp: Zieh das ${w.vowel} lang: „${stretch(w)}“. Das klingt komisch! Das <b>${w.vowel}</b> ist also <b>kurz</b>. Der pinke Korb leuchtet.`;
}
function praiseText(w) {
  const double = w.hidden.length === 2 && w.hidden[0] === w.hidden[1];
  return `${pick(PRAISE, 'praise')} <span class="pill ${CLS[w.kind]}">${w.marked}</span> Das ${w.vowel} ist ${KIND[w.kind]}.` +
    (w.kind === 'k' && double ? ` Danach kommt <b>${w.hidden}</b>.` : w.kind === 'l' && w.hidden.length === 1 ? ` Danach kommt nur ein <b>${w.hidden}</b>.` : '');
}

function attempt(id, k) {
  const w = byId(id);
  if (!w || w.done) return;
  if (w.kind !== k) {
    miss(w, cardEl(id));
    wobble(basketEl(k));
    selWord = id; selKind = null;
    renderSelection();
    say(w.misses >= 2 ? hintText(w) : pick(['Hmm, hör noch einmal genau hin. Sprich das Wort laut.', 'Das passt noch nicht. Kannst du den Vokal lang ziehen?', 'Guter Versuch! Sprich das Wort langsam und hör auf den Vokal.'], 'retry'), 'think');
    return;
  }
  const card = cardEl(id);
  selWord = selKind = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(w, basketEl(k), k === 'k' ? 'die' : 'der');
  say(praiseText(w), 'happy');
  if (state.solved === state.total) { state.busy = true; later(finish, 1800); }
}

$('#words').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapWord(c.dataset.id);
});
$('#bubble').addEventListener('click', e => { const b = e.target.closest('.speakbtn'); if (b) speak(b.dataset.say); });
$$('.basket').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapBasket(b.dataset.kind); }));
enableDrag('.word-card:not(.done)', '.basket', (card, basket) => attempt(card.dataset.id, basket.dataset.kind));
