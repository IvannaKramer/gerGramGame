/* ---------- Spiel: Kurz-oder-lang-Entscheider ---------- */
const KIND = { k: 'kurz', l: 'lang' };
const CLS = { k: 'ku', l: 'la' };
let all = [], queue = [], cur = null, step = 1, options = [];

/* Vorlesen als Zusatz: Das Spiel geht auch ohne Sprachausgabe. */
const canSpeak = (() => { try { return 'speechSynthesis' in window && typeof SpeechSynthesisUtterance === 'function'; } catch (e) { return false; } })();
function speak(text) {
  if (!canSpeak) return;
  try {
    const voices = speechSynthesis.getVoices();
    const de = voices.find(v => /^de([-_]|$)/i.test(v.lang));
    if (voices.length && !de) { say('Vorlesen geht auf diesem Gerät leider nicht. Sprich das Wort selbst laut.'); return; }
    const u = new SpeechSynthesisUtterance(text);
    u.lang = 'de-DE'; u.rate = .8; if (de) u.voice = de;
    speechSynthesis.cancel(); speechSynthesis.speak(u);
  } catch (e) { /* ohne Sprachausgabe weiterspielen */ }
}

/* Welche zwei Möglichkeiten gibt es für die Lücke? [kurze Schreibung, lange Schreibung] */
function pairFor(fill) {
  if (fill === 'ck' || fill === 'k') return ['ck', 'k'];
  if (fill === 'tz' || fill === 'z') return ['tz', 'z'];
  if (fill === 'i' || fill === 'ie') return ['i', 'ie'];
  return [fill[0] + fill[0], fill[0]];
}
function ruleText(w) {
  const [short, long] = w.pair;
  if (short === 'i') return w.kind === 'l' ? 'Ein langes i schreibst du meist <b>ie</b>.' : 'Ein kurzes i schreibst du nur <b>i</b>.';
  if (short === 'ck' || short === 'tz') return w.kind === 'k' ? `Nach einem kurzen Vokal schreibst du <b>${short}</b>.` : `Nach einem langen Vokal schreibst du <b>${long}</b>, nie ${short}.`;
  return w.kind === 'k' ? `Nach einem kurzen Vokal wird der Mitlaut verdoppelt: <b>${short}</b>.` : `Nach einem langen Vokal steht nur ein <b>${long}</b>.`;
}

function startGame() {
  all = WORDS.map(w => {
    const vowel = w[0].match(/\[(.+?)\]/)[1], par = w[0].match(/\((.+?)\)/);
    const fill = par ? par[1] : vowel.toLowerCase();
    const mark = `<span class="${w[1] === 'k' ? 'vk' : 'vl'}">$1</span>`;
    const slot = '<span class="slot" id="slot">?</span>';
    return { kind: w[1], clue: w[2], fill, pair: pairFor(fill), sound: par ? vowel.toLowerCase() : 'i', vowelGap: !par, misses: 0,
      plain: w[0].replace(/[\[\]()]/g, ''),
      marked: w[0].replace(/\[(.+?)\]/, mark).replace(/[()]/g, ''),
      gapped: par ? w[0].replace(/\(.+?\)/, slot).replace(/[\[\]]/g, '') : w[0].replace(/\[.+?\]/, slot),
      solved: par ? w[0].replace(/\[(.+?)\]/, mark).replace(/\((.+?)\)/, `<span class="slot filled ${CLS[w[1]]}" id="slot">$1</span>`)
                  : w[0].replace(/\[(.+?)\]/, `<span class="slot filled ${CLS[w[1]]}" id="slot">$1</span>`) };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  $('#speak').hidden = !canSpeak;
  say('Lies den Hinweis und sprich das gesuchte Wort laut. Ist der Vokal kurz oder lang?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; step = 1;
  $('#clue').textContent = cur.clue;
  $('#word').innerHTML = cur.gapped;
  $('#len').innerHTML = '';
  $('#ask').innerHTML = cur.vowelGap ? 'Schritt 1: Ist das <b>i</b> in der Lücke kurz oder lang?' : `Schritt 1: Ist das <b>${cur.sound}</b> vor der Lücke kurz oder lang?`;
  $('#opts').innerHTML =
    `<button class="tile ku" data-v="k" aria-label="kurz"><b><i class="sym dot" aria-hidden="true"></i>kurz</b><span class="kbd" aria-hidden="true">1</span></button>
     <button class="tile la" data-v="l" aria-label="lang"><b><i class="sym bar" aria-hidden="true"></i>lang</b><span class="kbd" aria-hidden="true">2</span></button>`;
}

function chooseLength(k) {
  const tile = $(`.tile[data-v="${k}"]`);
  if (!tile) return;
  if (k !== cur.kind) {
    const hint = miss(cur, tile);
    if (hint) $(`.tile[data-v="${cur.kind}"]`).classList.add('hint');
    say(hint
      ? (cur.kind === 'l' ? `Tipp: Sprich das Wort ganz langsam. Das <b>${cur.sound}</b> kannst du lang ziehen. Es ist <b>lang</b>.`
                          : `Tipp: Sprich das Wort ganz langsam. Das <b>${cur.sound}</b> klingt kurz und knapp. Es ist <b>kurz</b>.`)
      : `Hmm, hör noch einmal genau hin. Sprich das Wort laut: Kannst du das <b>${cur.sound}</b> lang ziehen?`, 'think');
    return;
  }
  sfx.pick();
  step = 2;
  $('#len').innerHTML = `<span class="pill ${CLS[k]}"><i class="sym ${k === 'k' ? 'dot' : 'bar'}" aria-hidden="true"></i>${KIND[k]}er Vokal</span>`;
  $('#ask').innerHTML = 'Schritt 2: Was gehört in die Lücke?';
  options = shuffle(cur.pair.slice());
  $('#opts').innerHTML = options.map((l, i) =>
    `<button class="tile nt" data-l="${l}" aria-label="${l}"><b>${l}</b><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
  say(`Genau, das ${cur.sound} ist <b>${KIND[k]}</b>. Was schreibst du nach einem ${KIND[k]}en ${cur.vowelGap ? 'i' : 'Vokal'}?`.replace('nach einem langen i', 'für ein langes i').replace('nach einem kurzen i', 'für ein kurzes i'), 'happy');
}

function chooseLetters(l) {
  const tile = $(`.tile[data-l="${l}"]`);
  if (!tile) return;
  if (l !== cur.fill) {
    const hint = miss(cur, tile);
    if (hint) $(`.tile[data-l="${cur.fill}"]`).classList.add('hint');
    say(hint ? `Tipp: ${ruleText(cur)} Der leuchtende Baustein passt.`
      : `Das passt noch nicht. Denk daran: Der Vokal ist <b>${KIND[cur.kind]}</b>. Was folgt daraus?`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#word').innerHTML = cur.solved;
  solve(cur, $('#slot'), cur.kind === 'k' ? 'die' : 'der');
  say(`${pick(PRAISE, 'praise')} <span class="pill ${CLS[cur.kind]}">${cur.marked}</span> ${ruleText(cur)}`, 'happy');
  later(next, 2300);
}

$('#opts').addEventListener('click', e => {
  if (state.busy || !cur) return;
  const t = e.target.closest('.tile');
  if (!t) return;
  if (t.dataset.v) chooseLength(t.dataset.v); else chooseLetters(t.dataset.l);
});
$('#speak').addEventListener('click', () => { if (cur) speak(cur.plain); });
document.addEventListener('keydown', e => {
  if (!gameOn() || state.busy || !cur || !/^[12]$/.test(e.key)) return;
  if (step === 1) chooseLength(e.key === '1' ? 'k' : 'l'); else chooseLetters(options[+e.key - 1]);
});
