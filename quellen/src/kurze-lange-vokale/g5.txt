@@tipp
Das <b>Dehnungs-h</b> hörst du nicht. Es steht nach einem <b>langen Vokal</b> und nur vor <b>l, m, n</b> oder <b>r</b>: Za<b>h</b>n, Stu<b>h</b>l. Aber Achtung: Viele Wörter mit langem Vokal haben kein h, zum Beispiel Schule. Wörter mit Dehnungs-h sind <b>Merkwörter</b>.

/* ---------- Spiel: Das verschwundene h ---------- */
let all = [], queue = [], cur = null, placed = 0;   /* placed: hinter welchem Baustein gerade ein h steht (0 = keins) */

function startGame() {
  all = WORDS.map(w => {
    const t = w[0].split('-'), pos = w[1], h = w[2];
    const full = t.map((x, i) => x + (h && i === pos - 1 ? 'h' : '')).join('');
    return { tiles: t, pos, h, full, vowel: t[pos - 1], next: t[pos], misses: 0,
      marked: t.map((x, i) => i === pos - 1 ? `<span class="vl">${x}</span>${h ? '<b>h</b>' : ''}` : x).join('') };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Stimmt das Wort so? Oder fehlt ein h? Tippe den Buchstaben an, hinter den es gehört.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; placed = 0;
  render();
}

function render(locked) {
  const box = $('#letters');
  box.classList.toggle('locked', !!locked);
  box.dataset.w = cur.tiles.join('');
  box.style.setProperty('--n', cur.tiles.join('').length + 1);
  box.innerHTML = cur.tiles.map((x, i) =>
    `<button type="button" class="lt${x.length > 1 ? ' wide' : ''}${locked && i === cur.pos - 1 ? ' v' : ''}" data-i="${i + 1}" aria-label="${placed === i + 1 ? `${x}, dahinter steht jetzt ein h` : `${x}. Antippen setzt ein h dahinter`}">${locked && i === cur.pos - 1 ? `<span class="vl">${x}</span>` : x}</button>` +
    (placed === i + 1 ? `<button type="button" class="lt h" data-i="${i + 1}" aria-label="h wieder wegnehmen">h</button>` : '')).join('');
  $('#check').textContent = placed ? 'Mit h – prüfen' : 'Kein h – prüfen';
}

$('#letters').addEventListener('click', e => {
  const b = e.target.closest('.lt');
  if (!b || state.busy || !cur) return;
  const i = +b.dataset.i;
  if (i === cur.tiles.length && !b.classList.contains('h') && placed !== i) { say('Am Wortende steht kein Dehnungs-h. Es steht mitten im Wort hinter dem langen Vokal.', 'think'); wobble(b); return; }
  placed = placed === i ? 0 : i;
  sfx.pick();
  render();
  const again = $(`.lt[data-i="${placed || i}"]`);
  if (again) again.focus({ preventScroll: true });
});

function check() {
  if (state.busy || !cur) return;
  const want = cur.h ? cur.pos : 0;
  if (placed !== want) {
    const hint = miss(cur, $('#letters'));
    if (hint) {
      if (cur.h) {
        placed = 0; render();
        $(`.lt[data-i="${cur.pos}"]`).classList.add('hint');
        say(`Tipp: Dieses Wort ist ein Merkwort mit Dehnungs-h. Das h steht hinter dem langen <b>${cur.vowel}</b> und vor dem <b>${cur.next}</b>. Tippe den leuchtenden Buchstaben an.`, 'think');
      } else {
        placed = 0; render();
        say(`Tipp: <b>${cur.full}</b> hat ein langes ${cur.vowel}, aber kein Dehnungs-h. Das Wort stimmt schon so. Tippe auf „Prüfen“.`, 'think');
      }
    } else if (!placed) say('Hmm, hier fehlt etwas. Sprich das Wort laut: Wo ist der lange Vokal? Dahinter fehlt ein h.', 'think');
    else if (!cur.h) say('Das passt noch nicht. Nicht jedes Wort mit langem Vokal hat ein h. Schau noch einmal genau hin.', 'think');
    else say('Fast! Das Dehnungs-h steht direkt hinter dem langen Vokal. Probier eine andere Stelle.', 'think');
    return;
  }
  state.busy = true;
  render(true);
  solve(cur, $('#letters'), 'der');
  say(`${pick(PRAISE, 'praise')} <span class="pill la">${cur.marked.replace(/<\/?b>/g, '')}</span> ` + (cur.h
    ? `Das ist ein Merkwort mit Dehnungs-h: langes ${cur.vowel}, dann <b>h</b> vor dem ${cur.next}.`
    : `Das ${cur.vowel} ist lang, aber hier steht kein h.`), 'happy');
  later(next, 2300);
}
$('#check').addEventListener('click', check);
