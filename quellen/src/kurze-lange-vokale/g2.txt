/* ---------- Spiel: Doppel-Werkstatt ---------- */
const KIND = { k: 'kurz', l: 'lang' };
const CLS = { k: 'ku', l: 'la' };
let all = [], queue = [], cur = null;

/* Vorlesen als Zusatz: Das Spiel geht auch ohne Sprachausgabe. */
const canSpeak = (() => { try { return 'speechSynthesis' in window && typeof SpeechSynthesisUtterance === 'function'; } catch (e) { return false; } })();
function speak(text) {
  if (!canSpeak) return;
  try {
    const voices = speechSynthesis.getVoices();
    const de = voices.find(v => /^de([-_]|$)/i.test(v.lang));
    if (voices.length && !de) { say('Vorlesen geht auf diesem Gerät leider nicht. Sprich das Wort selbst laut.'); return; }
    const u = new SpeechSynthesisUtterance(text);
    u.lang = 'de-DE'; u.rate = .8; if (de) u.voice = de;
    speechSynthesis.cancel(); speechSynthesis.speak(u);
  } catch (e) { /* ohne Sprachausgabe weiterspielen */ }
}

function startGame() {
  all = WORDS.map(w => {
    const vowel = w[0].match(/\[(.+?)\]/)[1].toLowerCase(), fill = w[0].match(/\((.+?)\)/)[1];
    const mark = `<span class="${w[1] === 'k' ? 'vk' : 'vl'}">$1</span>`;
    return { kind: w[1], emoji: w[2], vowel, fill, single: fill[0], double: fill[0] + fill[0], misses: 0,
      plain: w[0].replace(/[\[\]()]/g, ''),
      marked: w[0].replace(/\[(.+?)\]/, mark).replace(/\((.+?)\)/, '<b>$1</b>'),
      gapped: w[0].replace(/\(.+?\)/, '<span class="slot" id="slot">?</span>').replace(/[\[\]]/g, ''),
      solved: w[0].replace(/\[(.+?)\]/, mark).replace(/\((.+?)\)/, `<span class="slot filled ${CLS[w[1]]}" id="slot">$1</span>`) };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  $('#speak').hidden = !canSpeak;
  say('Sprich das Wort zum Bild laut. Ist der Vokal vor der Lücke kurz oder lang?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#pic').textContent = cur.emoji;
  $('#word').innerHTML = cur.gapped;
  $('#word').setAttribute('aria-label', 'Wort mit Lücke');
  /* Immer gleich: links der einfache Mitlaut (blau, lang), rechts der doppelte (pink, kurz) */
  $('#tiles').innerHTML =
    `<button class="tile la" data-l="${cur.single}" aria-label="ein ${cur.single}, nach langem Vokal"><b>${cur.single}</b><small>nach langem Vokal</small><span class="kbd" aria-hidden="true">1</span></button>
     <button class="tile ku" data-l="${cur.double}" aria-label="doppeltes ${cur.single}, nach kurzem Vokal"><b>${cur.double}</b><small>nach kurzem Vokal</small><span class="kbd" aria-hidden="true">2</span></button>`;
}

function choose(l) {
  if (state.busy || !cur) return;
  const tile = $(`.tile[data-l="${l}"]`);
  if (!tile) return;
  if (l !== cur.fill) {
    const hint = miss(cur, tile);
    if (hint) $(`.tile[data-l="${cur.fill}"]`).classList.add('hint');
    say(hint
      ? (cur.kind === 'k'
        ? `Tipp: Das <b>${cur.vowel}</b> klingt kurz und knapp. Nach einem kurzen Vokal wird der Mitlaut verdoppelt. Der leuchtende Baustein passt.`
        : `Tipp: Das <b>${cur.vowel}</b> kannst du lang ziehen. Nach einem langen Vokal steht nur ein Mitlaut. Der leuchtende Baustein passt.`)
      : `Hmm, das passt noch nicht. Sprich das Wort noch einmal laut: Ist das <b>${cur.vowel}</b> kurz oder lang?`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#word').innerHTML = cur.solved;
  solve(cur, $('#slot'), cur.kind === 'k' ? 'die' : 'der');
  say(`${pick(PRAISE, 'praise')} <span class="pill ${CLS[cur.kind]}">${cur.marked.replace(/<\/?b>/g, '')}</span> ` + (cur.kind === 'k'
    ? `Das ${cur.vowel} ist kurz. Darum schreibst du <b>${cur.fill}</b>.`
    : `Das ${cur.vowel} ist lang. Darum schreibst du nur ein <b>${cur.fill}</b>.`), 'happy');
  later(next, 2000);
}

$('#tiles').addEventListener('click', e => {
  if (suppressClick) return;
  const t = e.target.closest('.tile');
  if (t) choose(t.dataset.l);
});
$('#speak').addEventListener('click', () => { if (cur) speak(cur.plain); });
document.addEventListener('keydown', e => {
  if (!gameOn() || state.busy || !cur || !/^[12]$/.test(e.key)) return;
  choose(e.key === '1' ? cur.single : cur.double);
});
enableDrag('.tile', '#bench', (tile) => choose(tile.dataset.l));
