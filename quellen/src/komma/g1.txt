/* ---------- Spiel: Komma-Klick ---------- */
let all = [], queue = [], cur = null;
const word = w => w.item ? `<span class="w auf">${w.t}</span>` : w.t;
const full = s => s.words.map((w, i) => word(w) + (s.commas.includes(i) ? '<b class="cm">,</b>' : '')).join(' ');
const gapEl = i => $(`.cgap[data-i="${i}"]`);

/* Zerlegt den Satz in Wörter, merkt sich die Komma-Stellen und die Teile der Aufzählung */
function parse(text) {
  const raw = text.replace(/\{([^}]+)\}/g, (m, g) => g.split(' ').map(w => '§' + w).join(' ')).split(' ');
  const words = [], commas = [];
  raw.forEach((w, i) => {
    const item = w[0] === '§';
    if (item) w = w.slice(1);
    if (w.endsWith(',')) { commas.push(i); w = w.slice(0, -1); }
    words.push({ t: w, item });
  });
  return { words, commas };
}

function startGame() {
  all = SENTENCES.map(([text, emoji]) => ({ ...parse(text), emoji, misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies den Satz. Tippe auf jede Stelle, an der ein Komma fehlt. Oder tippe auf „Kein Komma nötig“.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  cur.set = [];
  $('#none').classList.remove('hint');
  $('#pic').textContent = cur.emoji;
  const box = $('#sentence');
  box.classList.remove('solved');
  box.innerHTML = cur.words.map((w, i) => `<span class="wd">${word(w)}` + (i < cur.words.length - 1
    ? `<button class="cgap" data-i="${i}" aria-label="Komma nach „${w.t}“ setzen"><i></i></button>` : '') + '</span>').join('');
}

function hint() {
  const open = cur.commas.filter(i => !cur.set.includes(i));
  if (open.length) {
    open.forEach(i => gapEl(i).classList.add('hint'));
    return 'Tipp: Zwischen den <b class="w auf">blauen Teilen</b> der Aufzählung steht ein Komma. Nur vor <b>und</b> oder <b>oder</b> steht keins.';
  }
  $('#none').classList.add('hint');
  return 'Tipp: Hier werden nur zwei Teile aufgezählt. Dazwischen steht schon <b>und</b> oder <b>oder</b>. Der Satz braucht kein Komma.';
}

function done(target) {
  state.busy = true;
  $('#none').classList.remove('hint');
  solve(cur, target, cur.commas.length ? 'der' : 'teal');
  $('#sentence').classList.add('solved');
  const n = cur.commas.length;
  say(`${pick(PRAISE, 'praise')} ${n === 0 ? 'Nur zwei Teile mit „und“ oder „oder“. Der Satz braucht kein Komma:'
    : n === 1 ? 'Ein Komma trennt die Teile der Aufzählung:' : 'Zwei Kommas trennen die Teile der Aufzählung:'} ${full(cur)}`, 'happy');
  later(next, 2600);
}

function choose(i, el) {
  if (state.busy || !cur) return;
  if (i < 0) {
    if (!cur.commas.length) { done(el); return; }
    say(miss(cur, el) ? hint() : 'Schau genau hin: Hier werden mehr als zwei Teile aufgezählt. Was steht dazwischen?', 'think');
    return;
  }
  if (cur.commas.includes(i)) {
    cur.set.push(i);
    el.classList.remove('hint'); el.classList.add('set'); el.disabled = true;
    el.setAttribute('aria-label', 'Komma gesetzt');
    if (cur.set.length === cur.commas.length) { done(el); return; }
    sfx.pick();
    say('Richtig! Hier steht ein Komma. In diesem Satz fehlt noch eins.', 'happy');
    return;
  }
  const after = cur.words[i + 1].t.toLowerCase();
  const tip = miss(cur, el);
  say(tip ? hint()
    : !cur.commas.length ? 'Hmm. Zähl mal nach: Wie viele blaue Teile hat diese Aufzählung?'
    : after === 'und' || after === 'oder' ? `Fast! Vor <b>${after}</b> steht in der Aufzählung kein Komma.`
    : 'Fast! Das Komma steht zwischen zwei blauen Teilen der Aufzählung.', 'think');
}

$('#sentence').addEventListener('click', e => {
  const g = e.target.closest('.cgap');
  if (g) choose(+g.dataset.i, g);
});
$('#none').addEventListener('click', () => choose(-1, $('#none')));
document.addEventListener('keydown', e => {
  if (gameOn() && e.key === '0') choose(-1, $('#none'));
});
