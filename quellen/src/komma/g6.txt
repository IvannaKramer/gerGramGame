@@tipp
Ein Nebensatz kann <b>vorn</b>, <b>hinten</b> oder <b>mitten im Satz</b> stehen. Steht er mittendrin, braucht er zwei Kommas: eins davor und eins dahinter. Vorsicht bei <b>als</b> und <b>wie</b>: Folgt kein Nebensatz mit Verb am Ende, steht kein Komma: Ich bin größer als du.

/* ---------- Spiel: Komma-Meister ---------- */
let all = [], queue = [], cur = null;
const gapEl = i => $(`.cgap[data-i="${i}"]`);
/* Ein Bindewort ist lila, wenn es einen Nebensatz einleitet: nach einem Komma oder am Anfang eines Satzes mit Komma */
const isBind = (s, i) => BINDS.includes(s.words[i].toLowerCase()) && s.commas.length > 0 && (i === 0 || s.commas.includes(i - 1));
const word = (s, i) => isBind(s, i) ? `<b class="w neb">${s.words[i]}</b>` : s.words[i];
const full = s => s.words.map((w, i) => word(s, i) + (s.commas.includes(i) ? '<b class="cm">,</b>' : '')).join(' ');
const NUM = ['kein Komma', 'ein Komma', 'zwei Kommas', 'drei Kommas', 'vier Kommas'];

function startGame() {
  all = SENTENCES.map(([text, tip]) => {
    const raw = text.split(' '), commas = [];
    const words = raw.map((w, i) => { if (w.endsWith(',')) { commas.push(i); return w.slice(0, -1); } return w; });
    return { words, commas, tip, misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Setze alle Kommas im Satz. Tippe dann auf „Prüfen“. Vorsicht: Manche Sätze brauchen gar kein Komma!');
  show('game');
  next();
}

function showCount() {
  const n = cur.set.length;
  $('#count').textContent = n === 0 ? 'Du hast noch kein Komma gesetzt.' : `Du hast ${NUM[n] || n + ' Kommas'} gesetzt.`;
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  cur.set = [];
  $('#check').classList.remove('hint');
  const box = $('#sentence');
  box.classList.remove('solved');
  box.innerHTML = cur.words.map((w, i) => `<span class="wd">${w}` + (i < cur.words.length - 1
    ? `<button class="cgap" data-i="${i}" aria-pressed="false" aria-label="Komma nach „${w}“"><i></i></button>` : '') + '</span>').join('');
  showCount();
}

function toggle(i, el) {
  if (state.busy || !cur) return;
  const on = !cur.set.includes(i);
  cur.set = on ? cur.set.concat(i) : cur.set.filter(x => x !== i);
  el.classList.toggle('set', on);
  el.classList.remove('hint');
  el.setAttribute('aria-pressed', String(on));
  sfx.pick();
  showCount();
}

function check() {
  if (state.busy || !cur) return;
  const btn = $('#check');
  const missing = cur.commas.filter(i => !cur.set.includes(i)), extra = cur.set.filter(i => !cur.commas.includes(i));
  if (missing.length || extra.length) {
    const tip = miss(cur, btn);
    if (!tip) {
      say(extra.length && !missing.length ? 'Fast! Ein Komma ist hier zu viel. Welches gehört nicht hin?'
        : !extra.length ? 'Fast! Es fehlt noch mindestens ein Komma. Such Aufzählungen und Bindewörter.'
        : 'Noch nicht ganz. Such Aufzählungen und die Bindewörter <b>weil, dass, wenn, als, ob</b>. Wo endet der Nebensatz?', 'think');
      return;
    }
    /* Tipp: falsche Kommas verschwinden, die fehlenden Stellen leuchten */
    extra.forEach(i => { const el = gapEl(i); el.classList.remove('set'); el.setAttribute('aria-pressed', 'false'); wobble(el); });
    cur.set = cur.set.filter(i => cur.commas.includes(i));
    missing.forEach(i => gapEl(i).classList.add('hint'));
    if (!missing.length) btn.classList.add('hint');
    showCount();
    say(`Tipp: ${cur.tip} ${missing.length ? 'Tippe auf die leuchtenden Kästchen und dann auf „Prüfen“.' : 'Ich habe die Kommas weggenommen. Tippe jetzt auf „Prüfen“.'}`, 'think');
    return;
  }
  state.busy = true;
  btn.classList.remove('hint');
  const last = cur.commas.length ? gapEl(cur.commas[cur.commas.length - 1]) : btn;
  solve(cur, last, cur.commas.length ? 'sun' : 'teal');
  $('#sentence').classList.add('solved');
  $('#count').textContent = '';
  const n = cur.commas.length;
  say(`${pick(PRAISE, 'praise')} ${n ? `Hier stehen ${NUM[n]}:`.replace('stehen ein', 'steht ein') : 'Dieser Satz braucht kein Komma:'} ${full(cur)}`, 'happy');
  later(next, 3000);
}

$('#sentence').addEventListener('click', e => {
  const g = e.target.closest('.cgap');
  if (g) toggle(+g.dataset.i, g);
});
$('#check').addEventListener('click', check);
