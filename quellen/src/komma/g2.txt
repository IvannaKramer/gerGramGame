@@tipp
Ein <b>Nebensatz</b> beginnt mit einem <b>Bindewort</b>: weil, dass, wenn, als, ob. Vor dem Bindewort steht ein <b>Komma</b>. Das <b>Verb</b> steht im Nebensatz ganz am Ende: Ich friere<b>,</b> weil es kalt <u>ist</u>.

const BINDS = ['weil', 'dass', 'wenn', 'als', 'ob'];

const GAME = {
  id: 'jagd',
  unit: 'Sätze',
  next: { href: '3-komma-ampel.html', title: 'Komma-Ampel' },

/* ---------- Spiel: Bindewort-Jagd ---------- */
let all = [], queue = [], cur = null;
const full = s => s.words.map((w, i) =>
  (i === s.at + 1 ? `<b class="w neb">${w}</b>` : i === s.words.length - 1 ? `<span class="vb">${s.verb}</span>${w.slice(s.verb.length)}` : w) +
  (i === s.at ? '<b class="cm">,</b>' : '')).join(' ');

function startGame() {
  all = SENTENCES.map(([text, emoji]) => {
    const raw = text.split(' ');
    const at = raw.findIndex(w => w.endsWith(','));
    const words = raw.map(w => w.replace(/,$/, ''));
    return { words, at, bind: words[at + 1], verb: words[words.length - 1].replace(/[.!?]$/, ''), emoji, misses: 0 };
  });
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies den Satz. Mit welchem Wort beginnt der Nebensatz? Tippe es an.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#pic').textContent = cur.emoji;
  const box = $('#sentence');
  box.classList.remove('solved');
  box.innerHTML = cur.words.map((w, i) => `<button class="wbtn" data-i="${i}">${w}</button>`).join('');
}

function choose(i, el) {
  if (state.busy || !cur) return;
  const target = cur.at + 1;
  if (i !== target) {
    const tip = miss(cur, el);
    if (tip) $(`.wbtn[data-i="${target}"]`).classList.add('hint');
    say(tip ? `Tipp: Der Nebensatz beginnt mit <b class="w neb">${cur.bind}</b>. Sein Verb <b>${cur.verb}</b> steht ganz am Ende.`
      : i === cur.words.length - 1 ? 'Das ist das Verb am Ende des Nebensatzes. Such das Wort, mit dem der Nebensatz anfängt.'
      : 'Fast! Such eines dieser Wörter: <b>weil, dass, wenn, als, ob</b>.', 'think');
    return;
  }
  state.busy = true;
  solve(cur, el, 'bonus');
  const box = $('#sentence');
  box.classList.add('solved');
  box.innerHTML = full(cur);
  say(`${pick(PRAISE, 'praise')} Vor <span class="pill neb">${cur.bind}</span> steht ein Komma. Das Verb <b>${cur.verb}</b> steht am Ende: ${full(cur)}`, 'happy');
  later(next, 3000);
}

$('#sentence').addEventListener('click', e => {
  const b = e.target.closest('.wbtn');
  if (b) choose(+b.dataset.i, b);
});
