@@tipp
Ein <b>Komma</b> steht zwischen den Teilen einer Aufzählung und vor einem Bindewort wie <b>weil, dass, wenn, als, ob</b>. <b>Kein Komma</b> steht vor <b>und</b> oder <b>oder</b> in einer Aufzählung.

/* ---------- Spiel: Komma-Ampel ---------- */
const SPOT = / ?\[(,?)\] ?/;
let all = [], queue = [], cur = null;
const tileEl = k => $(`.tile[data-k="${k}"]`);
/* Zuerst die Bindewörter <…> färben, dann die Aufzählung {…}, zuletzt die Stelle einsetzen */
const paint = (s, spot) => s.text.replace(/<([a-zäöüß]+)>/g, '<b class="w neb">$1</b>').replace(/\{([^}]+)\}/g, '<span class="w auf">$1</span>').replace(SPOT, spot);
const full = s => paint(s, s.k === 'komma' ? '<b class="cm">,</b> ' : ' ');

function why(s) {
  return s.type === 'A' ? 'Zwischen den Teilen einer Aufzählung steht ein Komma.'
    : s.type === 'N' ? `Mit <b class="w neb">${s.next}</b> beginnt ein Nebensatz. Davor steht ein Komma.`
    : s.type === 'U' ? `Vor <b>${s.next}</b> steht in der Aufzählung kein Komma.`
    : 'Hier gibt es keine Aufzählung und keinen Nebensatz. Also steht kein Komma.';
}

function startGame() {
  all = SENTENCES.map(([text, type, emoji]) => ({ text, type, emoji, misses: 0,
    k: SPOT.exec(text)[1] ? 'komma' : 'kein',
    next: (/\] [<{]?([A-Za-zäöüß]+)/.exec(text) || [])[1] }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Schau auf die markierte Stelle. Gehört dort ein Komma hin?');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#pic').textContent = cur.emoji;
  $('#sentence').innerHTML = paint(cur, ' <span class="spot" id="spot" role="img" aria-label="markierte Stelle">?</span> ');
}

function choose(k) {
  if (state.busy || !cur) return;
  if (k !== cur.k) {
    const tip = miss(cur, tileEl(k));
    if (tip) tileEl(cur.k).classList.add('hint');
    say(tip ? `Tipp: ${why(cur)}`
      : cur.type === 'O' ? 'Hmm. Siehst du blaue Teile oder ein lila Bindewort? Wenn nicht, braucht der Satz kein Komma.'
      : cur.type === 'N' ? 'Fast! Schau auf das lila Wort hinter der Stelle. Was beginnt dort?'
      : 'Fast! Schau genau: Welches Wort steht direkt hinter der Stelle?', 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const spot = $('#spot');
  solve(cur, spot, k === 'komma' ? 'sun' : 'teal');
  $('#sentence').innerHTML = full(cur);
  say(`${pick(PRAISE, 'praise')} ${why(cur)}<br>${full(cur)}`, 'happy');
  later(next, 2800);
}

$('#tiles').addEventListener('click', e => {
  const t = e.target.closest('.tile');
  if (t) choose(t.dataset.k);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[12]$/.test(e.key)) return;
  choose(e.key === '1' ? 'komma' : 'kein');
});
