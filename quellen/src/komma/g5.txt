@@tipp
Frag dich bei jedem Satz zwei Dinge. Werden <b>drei oder mehr Teile</b> aufgezählt? Dann stehen Kommas in der Aufzählung. Gibt es ein <b>Bindewort</b> (weil, dass, wenn, als, ob) und ein Verb am Ende? Dann trennt ein Komma Haupt- und Nebensatz. Sonst braucht der Satz <b>kein Komma</b>.

const CARDS_PER_ROUND = 5;
const KINDS = ['auf', 'neb', 'kein'];
const NAMES = { auf: 'Aufzählung', neb: 'Nebensatz', kein: 'Kein Komma' };
const BINDS = ['weil', 'dass', 'wenn', 'als', 'ob'];

const GAME = {
  id: 'sortierer',
  unit: 'Sätze',
  next: { href: '6-komma-meister.html', title: 'Komma-Meister' },
  review: () => `<div class="review">${KINDS.map(k =>
    `<div class="col ${k}"><h3>${NAMES[k]}</h3><ul>${all.filter(s => s.k === k).map(s => `<li>${full(s)}</li>`).join('')}</ul></div>`).join('')}</div>`
};

/* ---------- Spiel: Regel-Sortierer ---------- */
const COLOR = { auf: 'der', neb: 'bonus', kein: 'teal' };
let all = [], rounds = [], round = 0, left = 0, selCard = null, selKind = null;
const byId = id => all.find(s => s.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = k => $(`.basket[data-kind="${k}"]`);
const bare = s => s.text.replace(/,/g, '');
/* Satz mit gelben Kommas; bei Nebensätzen ist das Bindewort lila */
const full = s => s.text.split(' ').map((w, i) => {
  const t = w.replace(/,$/, '');
  return (s.k === 'neb' && i === s.bindAt ? `<b class="w neb">${t}</b>` : t) + (w.endsWith(',') ? '<b class="cm">,</b>' : '');
}).join(' ');

function startGame() {
  all = SENTENCES.map(([text, k], i) => {
    const w = text.split(' ');
    const after = w.findIndex(x => x.endsWith(',')) + 1;
    const bindAt = BINDS.includes(w[0].toLowerCase()) ? 0 : after;
    return { id: 's' + i, text, k, bindAt, bind: w[bindAt].toLowerCase(), misses: 0, done: false };
  });
  /* In jeder Runde liegen Karten für alle Körbe */
  const n = Math.ceil(all.length / CARDS_PER_ROUND);
  rounds = Array.from({ length: n }, () => []);
  let r = 0;
  KINDS.forEach(k => shuffle(all.filter(s => s.k === k)).forEach(s => { rounds[r % n].push(s); r++; }));
  rounds.forEach(shuffle);
  resetState(all.length);
  round = 0;
  startRound();
  say('Tippe eine Karte an. Wähle dann den Korb mit der passenden Regel.');
  show('game');
}

function startRound() {
  selCard = selKind = null; left = rounds[round].length;
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#words').innerHTML = rounds[round].map(s =>
    `<button class="word-card" data-id="${s.id}" aria-pressed="false" style="--tilt:${(Math.random() * 1.6 - .8).toFixed(2)}deg"><span class="word">${bare(s)}</span></button>`).join('');
  renderSelection();
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selCard;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.kind === selKind;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const s = selCard && byId(selCard);
  if (s && s.misses >= 2) basketEl(s.k).classList.add('hint');
}

function tapCard(id) {
  const s = byId(id);
  if (!s || s.done || state.busy) return;
  if (selKind) { attempt(id, selKind); return; }
  selCard = selCard === id ? null : id;
  sfx.pick(); renderSelection();
  if (selCard) say('Werden drei oder mehr Teile aufgezählt? Gibt es ein Bindewort? Oder fehlt gar kein Komma?');
}
function tapBasket(k) {
  if (state.busy) return;
  if (selCard) { attempt(selCard, k); return; }
  selKind = selKind === k ? null : k;
  sfx.pick(); renderSelection();
  if (selKind) say(`Welcher Satz gehört in den Korb <span class="pill ${k}">${NAMES[k]}</span>?`);
}

function tipFor(s) {
  return s.k === 'auf' ? 'Tipp: In diesem Satz werden drei oder mehr Teile aufgezählt. Dazwischen fehlen Kommas. Das ist eine <b>Aufzählung</b>.'
    : s.k === 'neb' ? `Tipp: Das Bindewort <b class="w neb">${s.bind}</b> leitet einen <b>Nebensatz</b> ein. Sein Verb steht am Ende.`
    : 'Tipp: Hier gibt es kein Bindewort. Es werden auch keine drei Teile aufgezählt. Der Satz braucht <b>kein Komma</b>.';
}
function praiseFor(s) {
  return s.k === 'auf' ? 'Das ist eine Aufzählung. Vor „und“ oder „oder“ steht kein Komma:'
    : s.k === 'neb' ? `Mit <b class="w neb">${s.bind}</b> beginnt ein Nebensatz. Ein Komma trennt ihn vom Hauptsatz:`
    : 'Keine Aufzählung mit drei Teilen, kein Nebensatz. Hier fehlt kein Komma:';
}

function attempt(id, k) {
  const s = byId(id);
  if (!s || s.done) return;
  if (s.k !== k) {
    miss(s, cardEl(id));
    wobble(basketEl(k));
    selCard = id; selKind = null;
    renderSelection();
    say(s.misses < 2 ? 'Fast! Lies den Satz noch einmal. Such ein Bindewort oder zähle die Teile, die mit „und“ oder „oder“ verbunden sind.' : tipFor(s), 'think');
    return;
  }
  const card = cardEl(id);
  selCard = selKind = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(s, basketEl(k), COLOR[k]);
  say(`${pick(PRAISE, 'praise')} ${praiseFor(s)} ${full(s)}`, 'happy');
  if (--left === 0) {
    state.busy = true;
    if (round < rounds.length - 1) later(() => { round++; startRound(); state.busy = false; say('Runde geschafft! Hier kommen neue Karten.', 'happy'); }, 2600);
    else later(finish, 2400);
  }
}

$('#words').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapCard(c.dataset.id);
});
$$('.basket').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapBasket(b.dataset.kind); }));
document.addEventListener('keydown', e => {
  if (gameOn() && /^[123]$/.test(e.key)) tapBasket(KINDS[+e.key - 1]);
});
enableDrag('.word-card:not(.done)', '.basket', (card, basket) => attempt(card.dataset.id, basket.dataset.kind));
