@@tipp
Zwischen Hauptsatz und Nebensatz steht ein <b>Komma</b>. Der Nebensatz beginnt mit dem <b>Bindewort</b> (weil, dass, wenn, als, ob). Sein <b>Verb</b> steht ganz am Ende. Der Nebensatz kann auch vorn stehen: <b>Weil</b> ich müde <u>bin</u><b>,</b> schlafe ich.

/* ---------- Spiel: Satz-Baumeister ---------- */
let all = [], queue = [], cur = null;
const kindOf = (t, last) => t === ',' ? 'cma' : BINDS.includes(t.toLowerCase()) ? 'neb' : last ? 'verb' : 'plain';
const paint = p => p.kind === 'cma' ? '<b class="cm">,</b>' : p.kind === 'neb' ? `<b class="w neb">${p.t}</b>` : p.kind === 'verb' ? `<span class="vb">${p.t}</span>` : p.t;
/* Fertiger Satz: Das Komma klebt am Wort davor */
const full = s => s.parts.map((p, i) => (i && p.kind !== 'cma' ? ' ' : '') + paint(p)).join('') + s.end;

function parse(text) {
  const raw = text.split('/');
  const end = (/[.!?]$/.exec(raw[raw.length - 1].replace(/\]$/, '')) || [''])[0];
  const parts = raw.map(r => {
    const fixed = /^\[.*\]$/.test(r);
    return { t: r.replace(/^\[|\]$/g, '').replace(/[.!?]$/, ''), fixed };
  });
  const cards = parts.filter(p => !p.fixed);
  const verb = cards.filter(p => p.t !== ',').pop();
  parts.forEach(p => { p.kind = p.fixed ? 'fixed' : kindOf(p.t, p === verb); });
  return { parts, cards, end, front: !parts[0].fixed, bind: cards.find(p => p.kind === 'neb').t.toLowerCase(), verb: verb.t };
}

function startGame() {
  all = SENTENCES.map(t => ({ ...parse(t), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies den Hauptsatz. Tippe dann die Kärtchen in der richtigen Reihenfolge an.');
  show('game');
  next();
}

function render() {
  const nextCard = cur.cards[cur.pos];
  $('#build').innerHTML = cur.parts.map(p => {
    if (p.fixed) return `<span class="fix">${p.t}${p === cur.parts[cur.parts.length - 1] ? cur.end : ''}</span>`;
    const i = cur.cards.indexOf(p);
    return i < cur.pos ? `<span class="slot filled ${p.kind}">${p.t}</span>`
      : `<span class="slot${p === nextCard ? ' cur' : ''}" role="img" aria-label="leerer Platz"></span>`;
  }).join('');
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  cur.pos = 0;
  $('#build').classList.remove('solved');
  /* Kärtchen mischen – aber nie schon in der richtigen Reihenfolge */
  let order;
  do { order = shuffle(cur.cards.slice()); } while (order.every((c, i) => c === cur.cards[i]));
  $('#chips').innerHTML = order.map((c, i) =>
    `<button class="chip${c.t === ',' ? ' cma' : ''}" data-t="${c.t}" style="--tilt:${(Math.random() * 3 - 1.5).toFixed(2)}deg"${c.t === ',' ? ' aria-label="Komma"' : ''}>${c.t}<span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
  render();
}

/* Freundlicher Hinweis, passend zu dem Kärtchen, das jetzt dran wäre */
function nudge(want, got) {
  if (want.kind === 'cma') return cur.front ? 'Der Nebensatz ist zu Ende. Was fehlt, bevor der Hauptsatz weitergeht?' : 'Halt! Der Hauptsatz ist zu Ende. Was trennt Hauptsatz und Nebensatz?';
  if (got.kind === 'cma') return 'Das Komma steht zwischen Hauptsatz und Nebensatz. Hier passt es noch nicht.';
  if (want.kind === 'neb') return 'Fast! Mit welchem Wort beginnt ein Nebensatz?';
  if (got.kind === 'verb') return 'Fast! Das Verb kommt im Nebensatz erst ganz am Ende.';
  return 'Fast! Lies den Satz leise vor. Welches Kärtchen klingt hier richtig?';
}
function rule(want) {
  return want.kind === 'cma' ? 'Tipp: Zwischen Hauptsatz und Nebensatz steht ein <b>Komma</b>.'
    : want.kind === 'neb' ? `Tipp: Der Nebensatz beginnt mit dem Bindewort <b class="w neb">${want.t}</b>.`
    : want.kind === 'verb' ? `Tipp: Im Nebensatz steht das Verb ganz am Ende. Hier ist es <b>${want.t}</b>.`
    : `Tipp: Nach dem Bindewort kommt zuerst, wer etwas tut. Das Verb <b>${cur.verb}</b> kommt ganz zum Schluss. Jetzt ist <b>${want.t}</b> dran.`;
}

function tap(chip) {
  if (state.busy || !cur || chip.classList.contains('used')) return;
  const want = cur.cards[cur.pos], got = cur.cards.find(c => c.t === chip.dataset.t);
  if (got !== want) {
    const tip = miss(cur, chip);
    if (tip) $$('.chip').find(c => c.dataset.t === want.t).classList.add('hint');
    say(tip ? rule(want) : nudge(want, got), 'think');
    return;
  }
  $$('.chip').forEach(c => c.classList.remove('hint'));
  chip.classList.add('used'); chip.disabled = true;
  cur.pos++;
  render();
  if (cur.pos < cur.cards.length) { sfx.pick(); return; }
  state.busy = true;
  const box = $('#build');
  solve(cur, box, 'bonus');
  box.classList.add('solved');
  box.innerHTML = full(cur);
  say(`${pick(PRAISE, 'praise')} Komma zwischen Haupt- und Nebensatz, das Verb <b>${cur.verb}</b> am Ende:<br>${full(cur)}`, 'happy');
  later(next, 3000);
}

$('#chips').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.chip');
  if (c) tap(c);
});
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-9]$/.test(e.key)) return;
  const c = $$('.chip')[+e.key - 1];
  if (c) tap(c);
});
enableDrag('.chip:not(.used)', '#build', chip => tap(chip));
