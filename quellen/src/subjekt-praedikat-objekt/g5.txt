@@tipp
Frag bei jeder Karte: Ist das ein Verb? Dann ist es das <b>Prädikat</b>. Wer oder was tut etwas? Das ist das <b>Subjekt</b>. <b>Wem?</b> → Dativobjekt. <b>Wen oder was?</b> → Akkusativobjekt. Achte auf die kleinen Wörter: <b>der</b> Igel, <b>dem</b> Igel, <b>den</b> Igel.

/* ---------- Gemeinsam: Satzglieder, Farben und Fragen ---------- */
const ROLE = {
  P: { name: 'Prädikat', art: 'das', short: 'Prädikat', q: 'Was tut jemand?', col: 'die' },
  S: { name: 'Subjekt', art: 'das', short: 'Subjekt', q: 'Wer oder was?', col: 'der' },
  D: { name: 'Dativobjekt', art: 'das', short: 'Dativobjekt', q: 'Wem?', col: 'das' },
  A: { name: 'Akkusativobjekt', art: 'das', short: 'Akkusativobjekt', q: 'Wen oder was?', col: 'bonus' },
  Z: { name: 'Bestimmung der Zeit', art: 'die', short: 'Zeit', q: 'Wann?', col: 'teal' },
  O: { name: 'Bestimmung des Ortes', art: 'die', short: 'Ort', q: 'Wo?', col: 'sun' }
};
const PRON = ['ihm', 'ihn', 'ihr', 'mir', 'mich', 'dir', 'dich'];
const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
const STOP = '<span class="stop" aria-hidden="true">.</span>';
/* Liest eine Zeile der Liste: Einträge mit | sind Satzglieder, alles andere ist Zusatz (Bild, gesuchtes Satzglied). */
function read(row, id) {
  const s = { id, info: row.filter(x => !x.includes('|')), misses: 0, done: false };
  s.parts = row.filter(x => x.includes('|')).map((x, i) => { const [t, r, shown] = x.split('|'); return { t, r, shown, i, text: i === 0 ? cap(t) : t }; });
  return s;
}
const twoPart = s => s.parts.filter(p => p.r === 'P').length > 1;
const answer = (s, r) => s.parts.filter(p => p.r === r).map(p => p.t).join(' … ');
const the = r => `${ROLE[r].art} <b>${ROLE[r].name}</b>`;
const rolePill = r => `<span class="pill k${r}">${ROLE[r].name}</span>`;
const colored = (s, roles) => s.parts.map(p => roles.includes(p.r) ? `<span class="pill k${p.r}">${p.text}</span>` : p.text).join(' ') + '.';
/* Baut die Frage nach einem Satzglied, z. B. „Wem schenkt Oma einen Roller?“ */
function ask(s, r) {
  if (r === 'P') return ROLE.P.q;
  const P = s.parts.filter(p => p.r === 'P'), rest = s.parts.filter(p => p.r !== 'P' && p.r !== r);
  const rank = p => p.r === 'S' ? 0 : PRON.includes(p.t) ? 1 : 2 + 'ZDO-A'.indexOf(p.r);
  rest.sort((a, b) => rank(a) - rank(b));
  return [ROLE[r].q.slice(0, -1), P[0].t, ...rest.map(p => p.t), ...P.slice(1).map(p => p.t)].join(' ') + '?';
}
const blkHTML = (p, r) => `<button class="blk${r ? ' on k' + r : ''}" data-i="${p.i}">${p.text}<small>${r ? ROLE[r].short : ''}</small></button>`;
const blkEl = i => $(`#sent .blk[data-i="${i}"]`);
function paint(el, r) { el.className = 'blk on k' + r; el.querySelector('small').textContent = ROLE[r].short; }
const reviewAll = (list, roles) => `<div class="review c1"><div class="col plain"><h3>${roles.map(rolePill).join(' ')}</h3><ul>${list.map(s => `<li>${colored(s, roles)}</li>`).join('')}</ul></div></div>`;

/* ---------- Spiel: Satz-Bauplan ---------- */
const SHOW = ['P', 'S', 'D', 'A'];
let all = [], queue = [], cur = null, place = [], locked = [], sel = null, hintAt = null;
const slotEl = i => $(`.slot[data-i="${i}"]`);

function startGame() {
  all = SENTENCES.map((row, i) => read(row, i));
  queue = shuffle(all.slice());
  resetState(all.length);
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; sel = hintAt = null;
  place = cur.parts.map(() => null); locked = cur.parts.map(() => false);
  cur.order = shuffle(cur.parts.map(p => p.i));
  render();
  say('Tippe eine Karte an und dann das passende Feld im Bauplan.');
}

function render() {
  $('#plan').innerHTML = cur.parts.map((p, i) => {
    const c = place[i] === null ? null : cur.parts[place[i]];
    return `<button class="slot k${p.r}${locked[i] ? ' good' : ''}${hintAt === i ? ' hint' : ''}" data-i="${i}" aria-label="Feld ${i + 1}: ${ROLE[p.r].name}, ${c ? 'darin liegt ' + c.t : 'leer'}">` +
      `<span class="sl-h">${ROLE[p.r].name}</span><span class="sl-b">${c ? (i === 0 ? cap(c.t) : c.t) : ''}</span></button>`;
  }).join('') + STOP;
  const free = cur.order.filter(i => !place.includes(i));
  $('#pile').innerHTML = free.length ? free.map(i =>
    `<button class="blk${sel === i ? ' selected' : ''}${hintAt === i ? ' hint' : ''}" data-i="${i}" aria-pressed="${sel === i}">${cur.parts[i].t}</button>`).join('')
    : '<p class="empty">Alle Karten liegen. Lies den Satz und tippe auf „Prüfen“.</p>';
}

function tapCard(i) {
  if (state.busy || !cur) return;
  sel = sel === i ? null : i;
  sfx.pick(); render();
  if (sel !== null) say(`Was für ein Satzglied ist <b>${cur.parts[i].t}</b>? Tippe das passende Feld im Bauplan an.`);
}
function put(ci, si) {
  if (state.busy || !cur || locked[si]) return;
  place[si] = ci; sel = null; if (hintAt === ci) hintAt = null;
  sfx.pick(); render();
}
function tapSlot(i) {
  if (state.busy || !cur || locked[i]) return;
  if (sel !== null) { put(sel, i); return; }
  if (place[i] !== null) { place[i] = null; sfx.pick(); render(); }
  else say('Tippe zuerst eine Karte an. Dann legst du sie in dieses Feld.');
}

function check() {
  if (state.busy || !cur) return;
  if (place.includes(null)) { say('Lege zuerst alle Karten in den Bauplan.'); return; }
  const wrong = place.map((c, i) => c === i ? -1 : i).filter(i => i >= 0);
  if (wrong.length) {
    const w = cur.parts[wrong[0]], r = w.r, tip = miss(cur, slotEl(wrong[0]));
    const ok = place.length - wrong.length;
    place.forEach((c, i) => { if (c === i) locked[i] = true; else place[i] = null; });
    sel = null; hintAt = tip ? wrong[0] : null;
    render();
    const back = `${ok === 0 ? 'Noch liegt keine Karte richtig.' : ok === 1 ? 'Eine Karte liegt schon richtig.' : ok + ' Karten liegen schon richtig.'} Die anderen bekommst du zurück.`;
    say(tip ? (r === 'P' ? `Tipp: Das Prädikat ist das Verb. In das leuchtende Feld gehört <b>${w.t}</b>.`
        : `Tipp: ${ask(cur, r)} – <b>${w.t}</b>. Das ist ${the(r)}. Leg die leuchtende Karte in das leuchtende Feld.`)
      : `Fast! ${back} ` + (r === 'P' ? 'Welche Karte ist ein Verb?' : `Für das Feld ${rolePill(r)} fragst du: <b>${ROLE[r].q}</b>`), 'think');
    return;
  }
  state.busy = true;
  locked = locked.map(() => true); hintAt = null; render();
  solve(cur, $('#plan'), 'das');
  say(`${pick(PRAISE, 'praise')} ${colored(cur, SHOW)}`, 'happy');
  later(next, 2600);
}

$('#pile').addEventListener('click', e => {
  if (suppressClick) return;
  const b = e.target.closest('.blk');
  if (b) tapCard(+b.dataset.i);
});
$('#plan').addEventListener('click', e => {
  if (suppressClick) return;
  const s = e.target.closest('.slot');
  if (s) tapSlot(+s.dataset.i);
});
$('#check').addEventListener('click', check);
enableDrag('#pile .blk', '.slot:not(.good)', (card, slot) => put(+card.dataset.i, +slot.dataset.i));
