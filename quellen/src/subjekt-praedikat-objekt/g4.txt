/* ---------- Gemeinsam: Satzglieder, Farben und Fragen ---------- */
const ROLE = {
  P: { name: 'Prädikat', art: 'das', short: 'Prädikat', q: 'Was tut jemand?', col: 'die' },
  S: { name: 'Subjekt', art: 'das', short: 'Subjekt', q: 'Wer oder was?', col: 'der' },
  D: { name: 'Dativobjekt', art: 'das', short: 'Dativobjekt', q: 'Wem?', col: 'das' },
  A: { name: 'Akkusativobjekt', art: 'das', short: 'Akkusativobjekt', q: 'Wen oder was?', col: 'bonus' },
  Z: { name: 'Bestimmung der Zeit', art: 'die', short: 'Zeit', q: 'Wann?', col: 'teal' },
  O: { name: 'Bestimmung des Ortes', art: 'die', short: 'Ort', q: 'Wo?', col: 'sun' }
};
const PRON = ['ihm', 'ihn', 'ihr', 'mir', 'mich', 'dir', 'dich'];
const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
const STOP = '<span class="stop" aria-hidden="true">.</span>';
/* Liest eine Zeile der Liste: Einträge mit | sind Satzglieder, alles andere ist Zusatz (Bild, gesuchtes Satzglied). */
function read(row, id) {
  const s = { id, info: row.filter(x => !x.includes('|')), misses: 0, done: false };
  s.parts = row.filter(x => x.includes('|')).map((x, i) => { const [t, r, shown] = x.split('|'); return { t, r, shown, i, text: i === 0 ? cap(t) : t }; });
  return s;
}
const twoPart = s => s.parts.filter(p => p.r === 'P').length > 1;
const answer = (s, r) => s.parts.filter(p => p.r === r).map(p => p.t).join(' … ');
const the = r => `${ROLE[r].art} <b>${ROLE[r].name}</b>`;
const rolePill = r => `<span class="pill k${r}">${ROLE[r].name}</span>`;
const colored = (s, roles) => s.parts.map(p => roles.includes(p.r) ? `<span class="pill k${p.r}">${p.text}</span>` : p.text).join(' ') + '.';
/* Baut die Frage nach einem Satzglied, z. B. „Wem schenkt Oma einen Roller?“ */
function ask(s, r) {
  if (r === 'P') return ROLE.P.q;
  const P = s.parts.filter(p => p.r === 'P'), rest = s.parts.filter(p => p.r !== 'P' && p.r !== r);
  const rank = p => p.r === 'S' ? 0 : PRON.includes(p.t) ? 1 : 2 + 'ZDO-A'.indexOf(p.r);
  rest.sort((a, b) => rank(a) - rank(b));
  return [ROLE[r].q.slice(0, -1), P[0].t, ...rest.map(p => p.t), ...P.slice(1).map(p => p.t)].join(' ') + '?';
}
const blkHTML = (p, r) => `<button class="blk${r ? ' on k' + r : ''}" data-i="${p.i}">${p.text}<small>${r ? ROLE[r].short : ''}</small></button>`;
const blkEl = i => $(`#sent .blk[data-i="${i}"]`);
function paint(el, r) { el.className = 'blk on k' + r; el.querySelector('small').textContent = ROLE[r].short; }
const reviewAll = (list, roles) => `<div class="review c1"><div class="col plain"><h3>${roles.map(rolePill).join(' ')}</h3><ul>${list.map(s => `<li>${colored(s, roles)}</li>`).join('')}</ul></div></div>`;

/* ---------- Spiel: Vier Farben ---------- */
const SHOW = ['P', 'S', 'D', 'A'];
let all = [], queue = [], cur = null, selBlk = null, selPen = null;
const penEl = r => $(`.pen[data-r="${r}"]`);

function startGame() {
  all = SENTENCES.map((row, i) => read(row, i));
  queue = shuffle(all.slice());
  resetState(all.length);
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; cur.on = []; selBlk = selPen = null;
  $('#sent').innerHTML = cur.parts.map(p => blkHTML(p)).join('') + STOP;
  renderSel();
  say('Mein Rat: Such zuerst das <span class="pill kP">Prädikat</span>, dann das <span class="pill kS">Subjekt</span>, dann die Objekte.');
}

function renderSel() {
  $$('#sent .blk').forEach(b => b.classList.toggle('selected', +b.dataset.i === selBlk));
  $$('.pen').forEach(p => { const on = p.dataset.r === selPen; p.classList.toggle('selected', on); p.setAttribute('aria-pressed', String(on)); p.classList.remove('hint'); });
}

function tapBlk(i) {
  if (state.busy || !cur || cur.on.includes(i)) return;
  if (selPen) { attempt(i, selPen); return; }
  selBlk = selBlk === i ? null : i;
  sfx.pick(); renderSel();
  if (selBlk !== null) say(`Welche Farbe bekommt <b>${cur.parts[i].t}</b>? Tippe unten einen Stift an.`);
}
function tapPen(r) {
  if (state.busy || !cur) return;
  if (selBlk !== null) { attempt(selBlk, r); return; }
  selPen = selPen === r ? null : r;
  sfx.pick(); renderSel();
  if (selPen) say(`Du hast den Stift ${rolePill(r)}. Tippe das passende Satzglied an.`);
}

function attempt(i, r) {
  const p = cur.parts[i], el = blkEl(i);
  if (state.busy || !p || cur.on.includes(i)) return;
  if (p.r !== r) {
    const tip = miss(cur, el);
    selBlk = i; selPen = null; renderSel();
    if (tip) penEl(p.r).classList.add('hint');
    say(tip ? (p.r === 'P' ? `Tipp: <b>${p.t}</b> gehört zum Prädikat <span class="pill kP">${answer(cur, 'P')}</span>. Es sagt, was jemand tut.`
        : `Tipp: ${ask(cur, p.r)} – <b>${p.t}</b>. Mit <b>${ROLE[p.r].q}</b> fragst du nach dem ${ROLE[p.r].name}.`)
      : r === 'P' ? `Fast! Das Prädikat ist das Verb. <b>${p.t}</b> ist kein Verb.`
      : p.r === 'P' ? `Fast! <b>${p.t}</b> sagt, was jemand tut. Welcher Stift passt dazu?`
      : `Fast! Passt die Frage <b>${ROLE[r].q}</b> wirklich zu <b>${p.t}</b>? Probier die anderen Fragen aus.`, 'think');
    return;
  }
  paint(el, r); cur.on.push(i);
  selBlk = selPen = null; renderSel();
  if (cur.on.length < cur.parts.length) {
    sfx.pick();
    say(r === 'P' ? (twoPart(cur) && cur.parts.some(q => q.r === 'P' && !cur.on.includes(q.i))
        ? `Richtig! <span class="pill kP">${p.t}</span> gehört zum Prädikat. Es hat aber noch einen zweiten Teil!`
        : `Richtig! Das Prädikat ist <span class="pill kP">${answer(cur, 'P')}</span>.`)
      : `Richtig! ${ask(cur, r)} – <span class="pill k${r}">${p.t}</span> ist ${the(r)}.`, 'happy');
    return;
  }
  state.busy = true;
  solve(cur, el, ROLE[r].col);
  say(`${pick(PRAISE, 'praise')} ${colored(cur, SHOW)}`, 'happy');
  later(next, 2600);
}
