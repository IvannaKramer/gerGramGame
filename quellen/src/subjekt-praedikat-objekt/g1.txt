@@tipp
Das <b>Prädikat</b> ist das Verb im Satz. Es sagt, was jemand tut: Oma <b>kocht</b>. Das <b>Subjekt</b> sagt, wer es tut. Du findest es mit der Frage <b>Wer oder was?</b> Wer oder was kocht? → <b>Oma</b>.

/* ---------- Gemeinsam: Satzglieder, Farben und Fragen ---------- */
const ROLE = {
  P: { name: 'Prädikat', art: 'das', short: 'Prädikat', q: 'Was tut jemand?', col: 'die' },
  S: { name: 'Subjekt', art: 'das', short: 'Subjekt', q: 'Wer oder was?', col: 'der' },
  D: { name: 'Dativobjekt', art: 'das', short: 'Dativobjekt', q: 'Wem?', col: 'das' },
  A: { name: 'Akkusativobjekt', art: 'das', short: 'Akkusativobjekt', q: 'Wen oder was?', col: 'bonus' },
  Z: { name: 'Bestimmung der Zeit', art: 'die', short: 'Zeit', q: 'Wann?', col: 'teal' },
  O: { name: 'Bestimmung des Ortes', art: 'die', short: 'Ort', q: 'Wo?', col: 'sun' }
};
const PRON = ['ihm', 'ihn', 'ihr', 'mir', 'mich', 'dir', 'dich'];
const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
const STOP = '<span class="stop" aria-hidden="true">.</span>';
/* Liest eine Zeile der Liste: Einträge mit | sind Satzglieder, alles andere ist Zusatz (Bild, gesuchtes Satzglied). */
function read(row, id) {
  const s = { id, info: row.filter(x => !x.includes('|')), misses: 0, done: false };
  s.parts = row.filter(x => x.includes('|')).map((x, i) => { const [t, r, shown] = x.split('|'); return { t, r, shown, i, text: i === 0 ? cap(t) : t }; });
  return s;
}
const twoPart = s => s.parts.filter(p => p.r === 'P').length > 1;
const answer = (s, r) => s.parts.filter(p => p.r === r).map(p => p.t).join(' … ');
const the = r => `${ROLE[r].art} <b>${ROLE[r].name}</b>`;
const rolePill = r => `<span class="pill k${r}">${ROLE[r].name}</span>`;
const colored = (s, roles) => s.parts.map(p => roles.includes(p.r) ? `<span class="pill k${p.r}">${p.text}</span>` : p.text).join(' ') + '.';
/* Baut die Frage nach einem Satzglied, z. B. „Wem schenkt Oma einen Roller?“ */
function ask(s, r) {
  if (r === 'P') return ROLE.P.q;
  const P = s.parts.filter(p => p.r === 'P'), rest = s.parts.filter(p => p.r !== 'P' && p.r !== r);
  const rank = p => p.r === 'S' ? 0 : PRON.includes(p.t) ? 1 : 2 + 'ZDO-A'.indexOf(p.r);
  rest.sort((a, b) => rank(a) - rank(b));
  return [ROLE[r].q.slice(0, -1), P[0].t, ...rest.map(p => p.t), ...P.slice(1).map(p => p.t)].join(' ') + '?';
}
const blkHTML = (p, r) => `<button class="blk${r ? ' on k' + r : ''}" data-i="${p.i}">${p.text}<small>${r ? ROLE[r].short : ''}</small></button>`;
const blkEl = i => $(`#sent .blk[data-i="${i}"]`);
function paint(el, r) { el.className = 'blk on k' + r; el.querySelector('small').textContent = ROLE[r].short; }
const reviewAll = (list, roles) => `<div class="review c1"><div class="col plain"><h3>${roles.map(rolePill).join(' ')}</h3><ul>${list.map(s => `<li>${colored(s, roles)}</li>`).join('')}</ul></div></div>`;

/* ---------- Spiel: Subjekt oder Prädikat? ---------- */
let all = [], rounds = [], round = 0, left = 0, selCard = null, selKind = null;
const byId = id => all.find(s => s.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = k => $(`.basket[data-kind="${k}"]`);
const marked = s => s.parts.map(p => p.r === s.k ? `<mark>${p.text}</mark>` : p.text).join(' ') + '.';

function startGame() {
  all = SENTENCES.map((row, i) => { const s = read(row, 's' + i); s.k = s.info[1]; return s; });
  /* In jeder Runde liegen Karten für beide Körbe */
  const n = Math.ceil(all.length / CARDS_PER_ROUND);
  rounds = Array.from({ length: n }, () => []);
  let r = 0;
  ['P', 'S'].forEach(k => shuffle(all.filter(s => s.k === k)).forEach(s => { rounds[r % n].push(s); r++; }));
  rounds.forEach(shuffle);
  resetState(all.length);
  round = 0;
  startRound();
  say('Tippe eine Karte an. Schau dann genau auf die gelb markierten Wörter.');
  show('game');
}

function startRound() {
  selCard = selKind = null; left = rounds[round].length;
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#words').innerHTML = rounds[round].map(s =>
    `<button class="word-card" data-id="${s.id}" aria-pressed="false" style="--tilt:${(Math.random() * 2 - 1).toFixed(2)}deg">
      <span class="pic" aria-hidden="true">${s.info[0]}</span><span class="word">${marked(s)}</span></button>`).join('');
  renderSelection();
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selCard;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.kind === selKind;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const s = selCard && byId(selCard);
  if (s && s.misses >= 2) basketEl(s.k).classList.add('hint');
}

function tapCard(id) {
  const s = byId(id);
  if (!s || s.done || state.busy) return;
  if (selKind) { attempt(id, selKind); return; }
  selCard = selCard === id ? null : id;
  sfx.pick(); renderSelection();
  if (selCard) say(`Markiert ist <b>${answer(s, s.k)}</b>. Sagt das, <b>was jemand tut</b>? Oder sagt es, <b>wer</b> etwas tut?`);
}
function tapBasket(k) {
  if (state.busy) return;
  if (selCard) { attempt(selCard, k); return; }
  selKind = selKind === k ? null : k;
  sfx.pick(); renderSelection();
  if (selKind) say(`Auf welcher Karte ist ${the(k)} markiert? Tippe sie an.`);
}

function attempt(id, k) {
  const s = byId(id);
  if (!s || s.done) return;
  if (s.k !== k) {
    miss(s, cardEl(id));
    wobble(basketEl(k));
    selCard = id; selKind = null;
    renderSelection();
    say(s.misses < 2 ? `Fast! Frag einmal: <b>${ask(s, 'S')}</b> Ist die Antwort gelb markiert?`
      : s.k === 'S'
        ? `Tipp: ${ask(s, 'S')} – <span class="pill kS">${answer(s, 'S')}</span>. Die Antwort auf <b>Wer oder was?</b> ist das Subjekt.`
        : `Tipp: <span class="pill kP">${answer(s, 'P')}</span> sagt, was jemand tut. Das Verb im Satz ist das Prädikat.`, 'think');
    return;
  }
  const card = cardEl(id);
  selCard = selKind = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(s, basketEl(k), ROLE[k].col);
  say(`${pick(PRAISE, 'praise')} <span class="pill k${k}">${answer(s, k)}</span> ist ${the(k)}: ${colored(s, [k])}`, 'happy');
  if (--left === 0) {
    state.busy = true;
    if (round < rounds.length - 1) later(() => { round++; startRound(); state.busy = false; say('Runde geschafft! Hier kommen neue Karten.', 'happy'); }, 2000);
    else later(finish, 1800);
  }
}
