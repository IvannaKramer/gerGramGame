@@tipp
Prüfe jedes Satzglied mit seiner Frage: <b>Wer oder was?</b> → Subjekt. <b>Wem?</b> → Dativobjekt. <b>Wen oder was?</b> → Akkusativobjekt. Neu für Profis: <b>Wann?</b> → Bestimmung der Zeit. <b>Wo?</b> → Bestimmung des Ortes. Passt die Frage nicht zur Farbe, hast du den Fehler gefunden.

/* ---------- Gemeinsam: Satzglieder, Farben und Fragen ---------- */
const ROLE = {
  P: { name: 'Prädikat', art: 'das', short: 'Prädikat', q: 'Was tut jemand?', col: 'die' },
  S: { name: 'Subjekt', art: 'das', short: 'Subjekt', q: 'Wer oder was?', col: 'der' },
  D: { name: 'Dativobjekt', art: 'das', short: 'Dativobjekt', q: 'Wem?', col: 'das' },
  A: { name: 'Akkusativobjekt', art: 'das', short: 'Akkusativobjekt', q: 'Wen oder was?', col: 'bonus' },
  Z: { name: 'Bestimmung der Zeit', art: 'die', short: 'Zeit', q: 'Wann?', col: 'teal' },
  O: { name: 'Bestimmung des Ortes', art: 'die', short: 'Ort', q: 'Wo?', col: 'sun' }
};
const PRON = ['ihm', 'ihn', 'ihr', 'mir', 'mich', 'dir', 'dich'];
const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
const STOP = '<span class="stop" aria-hidden="true">.</span>';
/* Liest eine Zeile der Liste: Einträge mit | sind Satzglieder, alles andere ist Zusatz (Bild, gesuchtes Satzglied). */
function read(row, id) {
  const s = { id, info: row.filter(x => !x.includes('|')), misses: 0, done: false };
  s.parts = row.filter(x => x.includes('|')).map((x, i) => { const [t, r, shown] = x.split('|'); return { t, r, shown, i, text: i === 0 ? cap(t) : t }; });
  return s;
}
const twoPart = s => s.parts.filter(p => p.r === 'P').length > 1;
const answer = (s, r) => s.parts.filter(p => p.r === r).map(p => p.t).join(' … ');
const the = r => `${ROLE[r].art} <b>${ROLE[r].name}</b>`;
const rolePill = r => `<span class="pill k${r}">${ROLE[r].name}</span>`;
const colored = (s, roles) => s.parts.map(p => roles.includes(p.r) ? `<span class="pill k${p.r}">${p.text}</span>` : p.text).join(' ') + '.';
/* Baut die Frage nach einem Satzglied, z. B. „Wem schenkt Oma einen Roller?“ */
function ask(s, r) {
  if (r === 'P') return ROLE.P.q;
  const P = s.parts.filter(p => p.r === 'P'), rest = s.parts.filter(p => p.r !== 'P' && p.r !== r);
  const rank = p => p.r === 'S' ? 0 : PRON.includes(p.t) ? 1 : 2 + 'ZDO-A'.indexOf(p.r);
  rest.sort((a, b) => rank(a) - rank(b));
  return [ROLE[r].q.slice(0, -1), P[0].t, ...rest.map(p => p.t), ...P.slice(1).map(p => p.t)].join(' ') + '?';
}
const blkHTML = (p, r) => `<button class="blk${r ? ' on k' + r : ''}" data-i="${p.i}">${p.text}<small>${r ? ROLE[r].short : ''}</small></button>`;
const blkEl = i => $(`#sent .blk[data-i="${i}"]`);
function paint(el, r) { el.className = 'blk on k' + r; el.querySelector('small').textContent = ROLE[r].short; }
const reviewAll = (list, roles) => `<div class="review c1"><div class="col plain"><h3>${roles.map(rolePill).join(' ')}</h3><ul>${list.map(s => `<li>${colored(s, roles)}</li>`).join('')}</ul></div></div>`;

/* ---------- Spiel: Fehler-Detektiv ---------- */
const SHOW = ['P', 'S', 'D', 'A', 'Z', 'O'];
let all = [], queue = [], cur = null, step = 1, stepMiss = 0;
const fixEl = r => $(`#fix .pen[data-r="${r}"]`);

function startGame() {
  all = SENTENCES.map((row, i) => { const s = read(row, i); s.bad = s.parts.find(p => p.shown); return s; });
  queue = shuffle(all.slice());
  resetState(all.length);
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false; step = 1; stepMiss = 0;
  $('#fix').hidden = true;
  $('#task').textContent = 'Welches Satzglied hat die falsche Farbe?';
  $('#sent').classList.remove('locked');
  $('#sent').innerHTML = cur.parts.map(p => blkHTML(p, p.shown || p.r)).join('') + STOP;
  say('Prüfe jedes Satzglied mit seiner Frage. Tippe das Satzglied an, das die <b>falsche Farbe</b> hat.');
}

function tapBlk(i) {
  const p = cur && cur.parts[i], el = blkEl(i), bad = cur && cur.bad;
  if (!p || state.busy || step !== 1) return;
  if (p !== bad) {
    miss(cur, el);
    const tip = ++stepMiss >= 2;
    if (tip) blkEl(bad.i).classList.add('hint');
    say(tip ? `Tipp: Schau dir <b>${bad.t}</b> an. Passt dazu wirklich die Frage <b>${ROLE[bad.shown].q}</b>`
      : p.r === 'P' ? `Fast! <b>${p.t}</b> gehört wirklich zum Prädikat. Such weiter!`
      : `Fast! <b>${p.t}</b> stimmt: ${ask(cur, p.r)} – ${p.t}. Such weiter!`, 'think');
    return;
  }
  step = 2; stepMiss = 0; sfx.pick();
  $$('#sent .blk').forEach(b => b.classList.remove('hint'));
  el.className = 'blk ask'; el.querySelector('small').textContent = '';
  $('#sent').classList.add('locked');
  $('#task').textContent = 'Fehler gefunden! Was ist das Satzglied wirklich?';
  $$('#fix .pen').forEach(b => b.classList.remove('hint'));
  $('#fix').hidden = false;
  say(`Gut aufgepasst! <b>${p.t}</b> ist nicht ${the(p.shown)}. Was ist es wirklich? Wähle unten aus.`, 'happy');
}

function choose(r) {
  const bad = cur && cur.bad;
  if (!bad || state.busy || step !== 2) return;
  if (r !== bad.r) {
    miss(cur, fixEl(r));
    const tip = ++stepMiss >= 2;
    if (tip) fixEl(bad.r).classList.add('hint');
    say(tip ? `Tipp: ${ask(cur, bad.r)} – <b>${bad.t}</b>. Mit <b>${ROLE[bad.r].q}</b> fragst du nach ${bad.r === 'Z' || bad.r === 'O' ? 'der' : 'dem'} ${ROLE[bad.r].name}.`
      : r === bad.shown ? 'Diese Farbe hatte das Satzglied schon. Sie war falsch. Probier eine andere Frage aus.'
      : r === 'P' ? `Fast! <b>${bad.t}</b> ist kein Verb. Welche Frage passt?`
      : `Fast! Passt die Frage <b>${ROLE[r].q}</b> wirklich zu <b>${bad.t}</b>? Probier die anderen Fragen aus.`, 'think');
    return;
  }
  state.busy = true;
  const el = blkEl(bad.i);
  paint(el, r); el.classList.add('fixed');
  $('#fix').hidden = true;
  $('#task').textContent = 'Jetzt stimmen alle Farben:';
  solve(cur, el, ROLE[r].col);
  say(`${pick(PRAISE, 'praise')} ${ask(cur, r)} – <span class="pill k${r}">${bad.t}</span> ist ${the(r)}.`, 'happy');
  later(next, 2800);
}
