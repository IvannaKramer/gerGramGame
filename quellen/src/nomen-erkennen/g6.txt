@@tipp
Nicht vor jedem <b>Nomen</b> steht ein Artikel. Manchmal steht ein Adjektiv dazwischen: ein <b>schnelles Auto</b>. Manchmal fehlt der Artikel ganz: Ich habe <b>Hunger</b>. Mach trotzdem die Probe: <b>das</b> Auto, <b>der</b> Hunger.

   Das erste Wort darf kein Nomen sein (es ist im Spiel schon groß). Das Spiel schreibt alle anderen Wörter klein.
   Bitte keine Namen und keine Wörter, die man groß und klein schreiben kann (z. B. Morgen/morgen, Paar/paar). */

const GAME = {
  id: 'profi',
  unit: 'Sätze',
  next: null,
  review: () => {
    const li = s => `<li>${s.words.map(w => (w.noun ? `<b>${w.text}</b>` : w.text) + w.punct).join(' ')}</li>`;
    const half = Math.ceil(all.length / 2);

/* ---------- Spiel: Großschreib-Profi ---------- */
let all = [], queue = [], cur = null;
const cap = s => s.charAt(0).toUpperCase() + s.slice(1);
const low = s => s.charAt(0).toLowerCase() + s.slice(1);

function parse(text) {
  return text.split(' ').map(tok => {
    const m = tok.match(/^(\[?)([^\].!?,]+)\]?([.!?,]*)$/);
    return { text: m[2], punct: m[3], noun: m[1] === '[', up: false };
  });
}
const pretty = s => s.words.map(w => (w.noun ? `<span class="pill nom">${w.text}</span>` : w.text) + w.punct).join(' ');
const label = (w, i) => i === 0 ? w.text : (w.up ? cap(w.text) : low(w.text));

function startGame() {
  all = SENTENCES.map(s => ({ words: parse(s), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Tippe alle Nomen an, dann werden sie groß. Danach tippst du auf „Prüfen“.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  cur.words.forEach(w => { w.up = false; });
  $('#sent').classList.remove('solved');
  $('#sent').innerHTML = cur.words.map((w, i) => {
    const stop = w.punct ? `<span class="stop" aria-hidden="true">${w.punct}</span>` : '';
    return i === 0 ? `<span class="fix">${w.text}</span>${stop}`
      : `<button class="w" data-i="${i}" aria-pressed="false">${label(w, i)}</button>${stop}`;
  }).join('');
}

function toggle(el) {
  if (state.busy || !cur) return;
  const i = +el.dataset.i, w = cur.words[i];
  w.up = !w.up;
  sfx.pick();
  el.textContent = label(w, i);
  el.classList.toggle('up', w.up);
  el.classList.remove('hint');
  el.setAttribute('aria-pressed', String(w.up));
}

function check() {
  if (state.busy || !cur) return;
  const btns = $$('#sent .w');
  const wrongs = btns.filter(b => { const w = cur.words[+b.dataset.i]; return w.up !== w.noun; });
  if (wrongs.length) {
    const missing = wrongs.filter(b => cur.words[+b.dataset.i].noun).length, extra = wrongs.length - missing;
    const hint = miss(cur, $('#sent'));
    btns.forEach(b => b.classList.toggle('hint', hint && wrongs.includes(b)));
    if (hint) { say('Tipp: Schau dir die leuchtenden Wörter an. Mach die Artikelprobe: Passt <b>der</b>, <b>die</b> oder <b>das</b> davor? Nur dann ist es ein Nomen.', 'think'); return; }
    const parts = [];
    if (missing) parts.push(missing === 1 ? 'Ein Nomen ist noch klein.' : `${missing} Nomen sind noch klein.`);
    if (extra) parts.push(extra === 1 ? 'Ein Wort ist groß, obwohl es kein Nomen ist.' : `${extra} Wörter sind groß, obwohl sie keine Nomen sind.`);
    say(`Fast! ${parts.join(' ')} Mach bei jedem Wort die Artikelprobe.`, 'think');
    return;
  }
  state.busy = true;
  $('#sent').classList.add('solved');
  solve(cur, $('#sent'), 'der');
  say(`${pick(PRAISE, 'praise')} ${pretty(cur)}`, 'happy');
  later(next, 2800);
}

$('#sent').addEventListener('click', e => { const b = e.target.closest('.w'); if (b) toggle(b); });
$('#check').addEventListener('click', check);
