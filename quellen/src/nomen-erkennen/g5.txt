@@tipp
Wörter einer Familie sehen sich ähnlich. Aber nur vor das <b>Nomen</b> passt ein Artikel: <b>der</b> Fleiß (Nomen), aber nicht „der fleißig“. Viele Nomen kann man nicht anfassen: der Fleiß, die Kraft, das Lob.

/* ---------- Spiel: Wortfamilien-Detektiv ---------- */
const KINDS = { n: ['Nomen', 'nom'], v: ['Verb', 'vrb'], a: ['Adjektiv', 'adj'] };
let all = [], queue = [], cur = null;
const bare = f => f.n.split(' ')[1];
const text = (f, k) => k === 'n' ? bare(f) : f[k];

function startGame() {
  all = FAMILIES.map(f => ({ n: f[0], v: f[1], a: f[2], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Drei Wörter aus einer Familie. Welches ist das Nomen? Mach die Artikelprobe!');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $('#fam').innerHTML = shuffle(['n', 'v', 'a']).map((k, i) =>
    `<button class="fcard" data-k="${k}" style="--tilt:${[-1.2, .8, -.5][i]}deg"><span class="t">${text(cur, k).toUpperCase()}</span><span class="kbd" aria-hidden="true">${i + 1}</span></button>`).join('');
}

function choose(el) {
  if (!el || state.busy || el.disabled || !cur) return;
  const k = el.dataset.k, W = `<b>${text(cur, k).toUpperCase()}</b>`;
  if (k !== 'n') {
    const hint = miss(cur, el);
    el.disabled = true; el.classList.add('limp');
    if (hint) $('.fcard[data-k="n"]').classList.add('hint');
    say(hint ? `Tipp: Mach die Artikelprobe: <b>${cur.n}</b>. Nur hier passt der Artikel. Das leuchtende Wort ist das Nomen.`
      : k === 'v' ? `${W} sagt, was jemand tut. Das ist ein Verb. Vor welches Wort passt ein Artikel?`
      : `${W} sagt, wie jemand oder etwas ist. Das ist ein Adjektiv. Vor welches Wort passt ein Artikel?`, 'think');
    return;
  }
  state.busy = true;
  solve(cur, el, 'der');
  $$('.fcard').forEach(b => {
    const kk = b.dataset.k;
    b.disabled = true;
    b.className = 'fcard shown ' + KINDS[kk][1];
    b.innerHTML = `<span class="t">${kk === 'n' ? cur.n : cur[kk]}</span><small>${KINDS[kk][0]}</small>`;
  });
  say(`${pick(PRAISE, 'praise')} <span class="pill nom">${cur.n}</span> ist das Nomen. <span class="pill vrb">${cur.v}</span> ist ein Verb und <span class="pill adj">${cur.a}</span> ein Adjektiv.`, 'happy');
  later(next, 2800);
}

$('#fam').addEventListener('click', e => choose(e.target.closest('.fcard')));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-3]$/.test(e.key)) return;
  choose($$('.fcard')[+e.key - 1]);
});
