/* Aufbau: [Bild, Satz]
   Im Satz stehen die Nomen in eckigen Klammern [ ] und die Verben in geschweiften Klammern { }.
   Jeder Satz hat genau zwei Nomen. Das Spiel zeigt den Satz in GROSSBUCHSTABEN, deshalb bitte keine Wörter mit ß. */
const SENTENCES = [
  ['🐕', 'Der [Hund] {frisst} einen [Knochen].'],
  ['📰', 'Das [Mädchen] {liest} eine [Zeitung].'],
  ['🧀', 'Die [Maus] {riecht} den [Käse].'],
  ['🌞', 'Die [Sonne] {scheint} am [Himmel].'],
  ['🚌', 'Der [Bus] {hält} an der [Schule].'],
  ['🍎', 'Der [Apfel] {liegt} auf dem [Teller].'],
  ['👶', 'Das [Baby] {hat} [Hunger].'],
  ['🎂', '[Opa] {kauft} einen [Kuchen].'],
  ['🐸', 'Der [Frosch] {springt} in den [Teich].'],
  ['🌠', 'Meine [Schwester] {hat} einen [Wunsch].'],
  ['🐣', 'Das [Küken] {sitzt} im [Nest].'],
  ['👫', 'Die [Freunde] {spielen} im [Garten].'],
  ['🚗', 'Das [Auto] {steht} vor der [Garage].'],
  ['🌧️', 'Der [Regen] {tropft} auf das [Dach].'],
  ['🍀', 'Der [Lehrer] {wünscht} uns [Glück].']
];
/* Diese Wörter erkennt das Spiel als Artikel */
const ARTICLES = ['der', 'die', 'das', 'den', 'dem', 'ein', 'eine', 'einen', 'einem', 'einer'];

const GAME = {
  id: 'jagd',
  unit: 'Sätze',
  next: { href: '4-wortarten-koerbe.html', title: 'Wortarten-Körbe' },
  review: () => {
    const li = s => `<li>${s.words.map(w => (w.t === 'n' ? `<b>${w.text}</b>` : w.text) + w.punct).join(' ')}</li>`;
    const half = Math.ceil(all.length / 2);
    return `<div class="review c2"><div class="col nom"><ul>${all.slice(0, half).map(li).join('')}</ul></div>
      <div class="col nom"><ul>${all.slice(half).map(li).join('')}</ul></div></div>`;
  }
};

/* ---------- Spiel: Nomen-Jagd ---------- */
let all = [], queue = [], cur = null, left = 0;

/* Zerlegt einen Satz in Wörter: t = n (Nomen), v (Verb), a (Artikel) oder x (anderes Wort) */
function parse(text) {
  return text.split(' ').map(tok => {
    const m = tok.match(/^([\[{]?)([^\]}.!?,]+)[\]}]?([.!?,]*)$/);
    const word = m[2];
    const t = m[1] === '[' ? 'n' : m[1] === '{' ? 'v' : ARTICLES.includes(word.toLowerCase()) ? 'a' : 'x';
    return { text: word, punct: m[3], t, found: false };
  });
}
const pretty = s => s.words.map(w => (w.t === 'n' ? `<span class="pill nom">${w.text}</span>` : w.text) + w.punct).join(' ');

function startGame() {
  all = SENTENCES.map(s => ({ emoji: s[0], words: parse(s[1]), misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Tippe die zwei Nomen im Satz an. Achte auf die Artikel: Danach kommt oft ein Nomen!');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  cur.words.forEach(w => { w.found = false; });
  left = cur.words.filter(w => w.t === 'n').length;
  $('#pic').textContent = cur.emoji;
  $('#sent').innerHTML = cur.words.map((w, i) =>
    `<button class="w" data-i="${i}">${w.text.toUpperCase()}</button>${w.punct ? `<span class="stop" aria-hidden="true">${w.punct}</span>` : ''}`).join('');
  showCount();
}
function showCount() {
  $('#count').innerHTML = left === 0 ? 'Alle Nomen gefunden!' : left === 1 ? 'Noch <b>1 Nomen</b> versteckt sich.' : `Finde <b>${left} Nomen</b>.`;
}

function tap(el) {
  if (state.busy || !cur) return;
  const w = cur.words[+el.dataset.i], W = `<b>${w.text.toUpperCase()}</b>`;
  if (w.found) return;
  if (w.t !== 'n') {
    const hint = miss(cur, el);
    if (hint) {
      $$('.w').forEach(b => { if (cur.words[+b.dataset.i].t === 'n' && !cur.words[+b.dataset.i].found) b.classList.add('hint'); });
      say(`Tipp: Mach die Artikelprobe. ${W} ist kein Nomen. Die leuchtenden Wörter sind Nomen.`, 'think');
    } else say(w.t === 'v' ? `${W} sagt, was jemand tut. Das ist ein Verb, kein Nomen. Such weiter!`
      : w.t === 'a' ? `${W} ist ein Artikel. Er zeigt dir: Gleich danach kommt ein Nomen!`
      : `${W} ist kein Nomen. Davor passt kein Artikel. Such weiter!`, 'think');
    return;
  }
  w.found = true; left--;
  el.classList.remove('hint');
  el.classList.add('found', 'nom');
  el.textContent = w.text;
  el.disabled = true;
  showCount();
  if (left > 0) {
    sfx.pick();
    say(`Richtig! <span class="pill nom">${w.text}</span> ist ein Nomen. Such noch eins!`, 'happy');
    return;
  }
  state.busy = true;
  solve(cur, el, 'der');
  say(`${pick(PRAISE, 'praise')} So schreibst du den Satz: ${pretty(cur)}`, 'happy');
  later(next, 2800);
}

$('#sent').addEventListener('click', e => { const b = e.target.closest('.w'); if (b) tap(b); });
