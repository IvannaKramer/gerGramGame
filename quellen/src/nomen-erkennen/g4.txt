/* ---------- Spiel: Wortarten-Körbe ---------- */
let all = [], rounds = [], round = 0, left = 0, selCard = null, selKind = null;
const byId = id => all.find(w => w.id === id);
const cardEl = id => $(`.word-card[data-id="${id}"]`);
const basketEl = k => $(`.basket[data-kind="${k}"]`);
const full = w => w.k === 'n' ? `${w.extra} ${w.word}` : w.word;
const CAPS = w => w.word.toUpperCase();

function startGame() {
  all = WORDS.map((w, i) => ({ id: 'w' + i, word: w[0], k: w[1], extra: w[2], misses: 0, done: false }));
  /* In jeder Runde liegen Karten für alle drei Körbe */
  const n = Math.ceil(all.length / CARDS_PER_ROUND);
  rounds = Array.from({ length: n }, () => []);
  let r = 0;
  Object.keys(KINDS).forEach(k => shuffle(all.filter(w => w.k === k)).forEach(w => { rounds[r % n].push(w); r++; }));
  rounds.forEach(shuffle);
  resetState(all.length);
  round = 0;
  startRound();
  say('Tippe eine Karte an und dann den passenden Korb. Ist das Wort ein Nomen, ein Verb oder ein Adjektiv?');
  show('game');
}

function startRound() {
  selCard = selKind = null; left = rounds[round].length;
  $('#round').textContent = `Runde ${round + 1} von ${rounds.length}`;
  $('#words').innerHTML = rounds[round].map(w =>
    `<button class="word-card" data-id="${w.id}" aria-pressed="false" style="--tilt:${(Math.random() * 3 - 1.5).toFixed(2)}deg">
      <span class="word">${CAPS(w)}</span></button>`).join('');
  renderSelection();
}

function renderSelection() {
  $$('.word-card').forEach(el => {
    const on = el.dataset.id === selCard;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
  });
  $$('.basket').forEach(el => {
    const on = el.dataset.kind === selKind;
    el.classList.toggle('selected', on);
    el.setAttribute('aria-pressed', String(on));
    el.classList.remove('hint');
  });
  const w = selCard && byId(selCard);
  if (w && w.misses >= 2) basketEl(w.k).classList.add('hint');
}

function tapCard(id) {
  const w = byId(id);
  if (!w || w.done || state.busy) return;
  if (selKind) { attempt(id, selKind); return; }
  selCard = selCard === id ? null : id;
  sfx.pick(); renderSelection();
  if (selCard) say(`<b>${CAPS(w)}</b> – passt ein Artikel davor? Sagt es, was jemand tut? Oder wie etwas ist?`);
}
function tapBasket(k) {
  if (state.busy) return;
  if (selCard) { attempt(selCard, k); return; }
  selKind = selKind === k ? null : k;
  sfx.pick(); renderSelection();
  if (selKind) say(`Welche Karte gehört in den Korb <span class="pill ${KINDS[k][2]}">${KINDS[k][0]}</span>?`);
}

function hintText(w) {
  return w.k === 'n' ? `Tipp: Mach die Artikelprobe: <b>${full(w)}</b>. Der Artikel passt. Also ist es ein Nomen.`
    : w.k === 'v' ? `Tipp: <b>${CAPS(w)}</b> sagt, was jemand tut. Die Grundform heißt <b>${w.extra}</b>. Das ist ein Verb.`
    : `Tipp: <b>${CAPS(w)}</b> sagt, wie jemand oder etwas ist. Das ist ein Adjektiv.`;
}
function praiseText(w) {
  const pill = `<span class="pill ${KINDS[w.k][2]}">${full(w)}</span>`;
  return w.k === 'n' ? `${pill} ist ein Nomen. Du schreibst es groß.`
    : w.k === 'v' ? `${pill} ist ein Verb. Es kommt von <b>${w.extra}</b>. Du schreibst es klein.`
    : `${pill} ist ein Adjektiv. Du schreibst es klein.`;
}

function attempt(id, k) {
  const w = byId(id);
  if (!w || w.done) return;
  if (w.k !== k) {
    miss(w, cardEl(id));
    wobble(basketEl(k));
    selCard = id; selKind = null;
    renderSelection();
    say(w.misses >= 2 ? hintText(w)
      : k === 'n' ? `Fast! Mach die Probe: der, die oder das <b>${CAPS(w)}</b>? Das passt nicht.`
      : k === 'v' ? `Fast! Sagt <b>${CAPS(w)}</b>, was jemand tut? Nein. Probier einen anderen Korb.`
      : `Fast! Sagt <b>${CAPS(w)}</b>, wie etwas ist? Nein. Probier einen anderen Korb.`, 'think');
    return;
  }
  const card = cardEl(id);
  selCard = selKind = null;
  renderSelection();
  card.classList.add('vanish');
  later(() => { card.classList.remove('vanish'); card.classList.add('done'); card.disabled = true; }, 330);
  solve(w, basketEl(k), KINDS[k][3]);
  say(`${pick(PRAISE, 'praise')} ${praiseText(w)}`, 'happy');
  if (--left === 0) {
    state.busy = true;
    if (round < rounds.length - 1) later(() => { round++; startRound(); state.busy = false; say('Runde geschafft! Hier kommen neue Karten.', 'happy'); }, 2000);
    else later(finish, 1800);
  }
}

$('#words').addEventListener('click', e => {
  if (suppressClick) return;
  const c = e.target.closest('.word-card');
  if (c) tapCard(c.dataset.id);
});
$$('.basket').forEach(b => b.addEventListener('click', () => { if (!suppressClick) tapBasket(b.dataset.kind); }));
enableDrag('.word-card:not(.done)', '.basket', (card, basket) => attempt(card.dataset.id, basket.dataset.kind));
