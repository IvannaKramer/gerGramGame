/* Aufbau: Kürzel: [Name, Farbe, Satz nach der richtigen Antwort, Tipp nach zwei Fehlversuchen] */
const GROUPS = {
  leb: ['Lebewesen',            'teal',  'ist ein Lebewesen.',
        'Tipp: Lebewesen sind Menschen und Tiere. Sie leben, atmen und bewegen sich.'],
  dng: ['Dinge',                'bonus', 'ist ein Ding. Man kann es anfassen.',

  gef: ['Gefühle und Gedanken', 'org',   'gehört zu den Gefühlen und Gedanken. Man kann es nicht anfassen. Es ist trotzdem ein Nomen!',

  adj:  ['Adjektiv', 'adj', 'das', 'sagt, wie etwas ist']

const GAME = {
  id: 'sortierer',
  next: { href: '2-gross-oder-klein.html', title: 'Groß oder klein?' },

/* ---------- Spiel: Nomen oder kein Nomen? ---------- */
let all = [], queue = [], cur = null, step = 1;
const isNoun = w => w.k in GROUPS;
const full = w => isNoun(w) ? `${w.art} ${w.word}` : w.word;
const CAPS = w => w.word.toUpperCase();

function startGame() {
  all = WORDS.map(w => ({ word: w[0], art: w[1], k: w[2], emoji: w[3], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Lies das Wort. Passt <b>der</b>, <b>die</b> oder <b>das</b> davor? Dann ist es ein Nomen.');
  show('game');
  next();
}

function setStep(n) {
  step = n;
  $('#step1').hidden = n !== 1;
  $('#step2').hidden = n !== 2;
  $$('.tile').forEach(t => t.classList.remove('hint'));
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  setStep(1);
  $('#pic').textContent = cur.emoji;
  $('#word').className = 'wordbig';
  $('#word').textContent = CAPS(cur);
  $('#probe').textContent = 'Probe: Passt der, die oder das davor?';
}

/* Schritt 1: Nomen oder kein Nomen? */
function answer1(el) {
  if (state.busy || !cur || step !== 1) return;
  const noun = isNoun(cur);
  if ((el.dataset.a === 'n') !== noun) {
    const hint = miss(cur, el);
    if (hint) $(`#step1 .tile[data-a="${noun ? 'n' : 'k'}"]`).classList.add('hint');
    if (noun) say(hint
      ? `Tipp: Mach die Artikelprobe: <b>${full(cur)}</b>. Der Artikel passt. Also ist es ein Nomen.`
      : `Fast! Probier es mit einem Artikel: <b>${full(cur)}</b>. Passt das?`, 'think');
    else say(hint
      ? `Tipp: <b>${CAPS(cur)}</b> ${OTHER[cur.k][3]}. Das ist ein ${OTHER[cur.k][0]} und kein Nomen.`
      : `Hmm. <b>${CAPS(cur)}</b> ${OTHER[cur.k][3]}. Ist das ein Name für ein Lebewesen, ein Ding oder ein Gefühl?`, 'think');
    return;
  }
  if (noun) {
    sfx.pick();
    $('#word').className = 'wordbig nom';
    $('#word').textContent = full(cur);
    $('#probe').textContent = 'Der Artikel passt. Nomen schreibst du groß.';
    setStep(2);
    say(`Richtig! <span class="pill nom">${full(cur)}</span> ist ein Nomen. In welche Gruppe gehört es?`, 'happy');
    return;
  }
  const o = OTHER[cur.k];
  state.busy = true;
  $('#word').className = 'wordbig ' + o[1];
  $('#word').textContent = cur.word;
  $('#probe').textContent = 'Kein Nomen. Du schreibst es klein.';
  solve(cur, el, o[2]);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${o[1]}">${cur.word}</span> ist ein ${o[0]}. Es ${o[3]}. Du schreibst es klein.`, 'happy');
  later(next, 2300);
}

/* Schritt 2: Welche Gruppe? */
function answer2(el) {
  if (state.busy || !cur || step !== 2) return;
  const g = GROUPS[cur.k];
  if (el.dataset.g !== cur.k) {
    const hint = miss(cur, el);
    if (hint) $(`#step2 .tile[data-g="${cur.k}"]`).classList.add('hint');
    say(hint ? g[3] : `Fast! Überlege: Lebt es? Kann man es anfassen? Oder fühlt und denkt man es nur?`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  solve(cur, el, g[1]);
  say(`${pick(PRAISE, 'praise')} <span class="pill ${cur.k}">${full(cur)}</span> ${g[2]}`, 'happy');
  later(next, 2300);
}

$('#step1').addEventListener('click', e => { const t = e.target.closest('.tile'); if (t) answer1(t); });
$('#step2').addEventListener('click', e => { const t = e.target.closest('.tile'); if (t) answer2(t); });
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[1-3]$/.test(e.key)) return;
  const t = $$(`#step${step} .tile`)[+e.key - 1];
  if (t) (step === 1 ? answer1 : answer2)(t);
});
