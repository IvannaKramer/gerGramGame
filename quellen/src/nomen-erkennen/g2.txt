/* ---------- Spiel: Groß oder klein? ---------- */
let all = [], queue = [], cur = null;
const isNoun = w => w.k === 'nomen';
const full = w => isNoun(w) ? `${w.art} ${w.word}` : w.word;

function startGame() {
  all = WORDS.map(w => ({ word: w[0], art: w[1], k: w[2], emoji: w[3], misses: 0 }));
  queue = shuffle(all.slice());
  resetState(all.length);
  say('Welcher Buchstabe fehlt? Nomen schreibst du groß, alle anderen Wörter klein.');
  show('game');
  next();
}

function next() {
  cur = queue.shift();
  if (!cur) { finish(); return; }
  state.busy = false;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  const first = cur.word.charAt(0);
  $('#big').textContent = first.toUpperCase();
  $('#small').textContent = first.toLowerCase();
  $('#pic').textContent = cur.emoji;
  $('#word').className = 'wordbig';
  $('#word').innerHTML = `<span class="slot" role="img" aria-label="Lücke">?</span>${cur.word.slice(1)}`;
  $('#probe').textContent = 'Probe: Passt der, die oder das davor?';
}

function choose(el) {
  if (state.busy || !cur) return;
  const noun = isNoun(cur), k = KINDS[cur.k];
  if ((el.dataset.a === 'g') !== noun) {
    const hint = miss(cur, el);
    if (hint) $(`.tile[data-a="${noun ? 'g' : 'k'}"]`).classList.add('hint');
    if (noun) say(hint
      ? `Tipp: Mach die Artikelprobe: <b>${full(cur)}</b>. Der Artikel passt. Also ist es ein Nomen und wird groß geschrieben.`
      : `Fast! Probier es mit einem Artikel: <b>${full(cur)}</b>. Passt das?`, 'think');
    else say(hint
      ? `Tipp: <b>${cur.word}</b> ${k[3]}. Das ist ein ${k[0]}. Nur Nomen schreibst du groß.`
      : `Hmm. Dieses Wort ${k[3]}. Ist das ein Nomen?`, 'think');
    return;
  }
  state.busy = true;
  $$('.tile').forEach(t => t.classList.remove('hint'));
  $('#word').className = 'wordbig ' + k[1];
  $('#word').textContent = full(cur);
  $('#probe').textContent = noun ? 'Der Artikel passt. Nomen schreibst du groß.' : 'Kein Nomen. Du schreibst es klein.';
  solve(cur, el, k[2]);
  say(noun
    ? `${pick(PRAISE, 'praise')} <span class="pill nom">${full(cur)}</span> ist ein Nomen. Nomen schreibst du groß.`
    : `${pick(PRAISE, 'praise')} <span class="pill ${k[1]}">${cur.word}</span> ist ein ${k[0]}. Es ${k[3]}. Du schreibst es klein.`, 'happy');
  later(next, 2100);
}

$$('.tile').forEach(t => t.addEventListener('click', () => choose(t)));
document.addEventListener('keydown', e => {
  if (!gameOn() || !/^[12]$/.test(e.key)) return;
  choose($$('.tile')[+e.key - 1]);
});
